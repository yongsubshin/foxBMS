92static float_t ADC_ConvertVoltage(uint16_t adcCounts) {
93 /* AXIVION Routine Generic-MissingParameterAssert: adcValue_mV: parameter accepts whole range */
94
95 /** docref: For details to equation see Equation 28 in Technical Reference Manual SPNU563A - March 2018 page 852 */
96 float_t result_mV =
97 (((adcCounts + ADC_CONVERSION_OFFSET) * (ADC_VREFHIGH_mV - ADC_VREFLOW_mV)) / ADC_CONVERSION_FACTOR_12BIT) +
98 ADC_VREFLOW_mV;
99
100 return result_mV;
101}

105extern void ADC_Control(void) {
106 bool conversionFinished = true;
107
108 switch (adc_conversionState) {
109 case ADC_START_CONVERSION:
110 adcStartConversion(adcREG1, adcGROUP1);
111 adc_conversionState = ADC_WAIT_CONVERSION_FINISHED;
112 break;
113
114 case ADC_WAIT_CONVERSION_FINISHED:
115 conversionFinished = true;
116 if (ADC_CONVERSION_ENDDBIT != adcIsConversionComplete(adcREG1, adcGROUP1)) {
117 conversionFinished = false;
118 }
119 if (conversionFinished == true) {
120 adc_conversionState = ADC_CONVERSION_FINISHED;
121 }
122 break;
123
124 /* Start initialization procedure, data sheet figure 106 page 79 */
125 case ADC_CONVERSION_FINISHED:
126 adcGetData(adcREG1, adcGROUP1, &adc_adc1RawVoltages[0]);
127 for (uint8_t i = 0u; i < MCU_ADC1_MAX_NR_CHANNELS; i++) {
128 adc_adc1Voltages.adc1ConvertedVoltages_mV[i] = ADC_ConvertVoltage(adc_adc1RawVoltages[i].value);
129 }
130 (void)DATA_WRITE_DATA(&adc_adc1Voltages);
131 adc_conversionState = ADC_START_CONVERSION;
132 break;
133
134 default: /* invalid state */ /* LCOV_EXCL_LINE */
135 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
136 break; /* LCOV_EXCL_LINE */
137 }
138}
