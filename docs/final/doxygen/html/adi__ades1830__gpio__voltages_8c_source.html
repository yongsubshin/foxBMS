86/**
87 * @brief Saves the GPIO voltage values read from the daisy-chain.
88 * @details There are up to 5 registers to read _(A, B, C, D, E) to get all
89 * GPIO voltages.
90 * This function is called to store the result from the transmission
91 * buffer to the appropriate location in the driver.
92 * @param pAdiState state of the ADI driver
93 * @param pData receive buffer
94 * @param registerSet auxiliary register that was read (voltage register
95 * A, B, C or D).
96 * @param storeLocation location where read data has to be stored
97 */
98static void ADI_SaveRxToGpioVoltageBuffer(
99 ADI_STATE_s *pAdiState,
100 uint8_t *pData,

102 ADI_AUXILIARY_STORE_LOCATION_e storeLocation);

104/*========== Static Function Implementations ================================*/
105/* RequirementId: D7.1 V1R0 FUN-2.10.01.02 */
106static void ADI_SaveRxToGpioVoltageBuffer(
107 ADI_STATE_s *pAdiState,
108 uint8_t *pData,
109 uint8_t registerSet,
110 ADI_AUXILIARY_STORE_LOCATION_e storeLocation) {
111 FAS_ASSERT(pAdiState != NULL_PTR);
112 FAS_ASSERT(pData != NULL_PTR);
113 FAS_ASSERT(
114 (registerSet == ADI_AUXILIARY_RESULT_REGISTER_SET_A) || (registerSet == ADI_AUXILIARY_RESULT_REGISTER_SET_B) ||
115 (registerSet == ADI_AUXILIARY_RESULT_REGISTER_SET_C) || (registerSet == ADI_AUXILIARY_RESULT_REGISTER_SET_D) ||
116 (registerSet == ADI_AUXILIARY_RESULT_REGISTER_SET_E));
117 FAS_ASSERT(
118 (storeLocation == ADI_AUXILIARY_VOLTAGE) || (storeLocation == ADI_REDUNDANT_AUXILIARY_VOLTAGE) ||
119 (storeLocation == ADI_AUXILIARY_VOLTAGE_OPEN_WIRE));
120
121 uint16_t cellOffset = 0u;
122 uint16_t voltageIndex = 0u;
123 uint16_t rawValue = 0u;
124 int16_t signedValue = 0;
125 float_t floatVoltage = 0.0f;
126 int16_t voltage = 0;
127 uint16_t bufferLSB = 0u;
128 uint16_t bufferMSB = 0u;
129 uint8_t numberOfVoltagesInRegister = 0u;
130 DATA_BLOCK_ALL_GPIO_VOLTAGES_s *pGpioVoltageTable = NULL_PTR;
131
132 switch (storeLocation) {
133 case ADI_AUXILIARY_VOLTAGE:
134 pGpioVoltageTable = pAdiState->data.allGpioVoltages;
135 break;
136 case ADI_REDUNDANT_AUXILIARY_VOLTAGE:
137 pGpioVoltageTable = pAdiState->data.allGpioVoltagesRedundant;
138 break;
139 case ADI_AUXILIARY_VOLTAGE_OPEN_WIRE:
140 pGpioVoltageTable = pAdiState->data.allGpioVoltageOpenWire;
141 break;
142 default: /* LCOV_EXCL_LINE */
143 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
144 break; /* LCOV_EXCL_LINE */
145 }
146
147 /* Default: 3 GPIO voltages in AUX registers */
148 numberOfVoltagesInRegister = ADI_MAX_NUMBER_OF_GPIO_VOLTAGES_IN_REGISTER;
149
150 switch (registerSet) {
151 case ADI_AUXILIARY_RESULT_REGISTER_SET_A:
152 /* RDAUXA command -> auxiliary register group A */
153 cellOffset = ADI_VOLTAGE_00_02_OFFSET;
154 break;
155 case ADI_AUXILIARY_RESULT_REGISTER_SET_B:
156 /* RDAUXB command -> auxiliary register group B */
157 cellOffset = ADI_VOLTAGE_03_05_OFFSET;
158 break;
159 case ADI_AUXILIARY_RESULT_REGISTER_SET_C:
160 /* RDAUXC command -> auxiliary register group C */
161 cellOffset = ADI_VOLTAGE_06_08_OFFSET;
162 break;
163 case ADI_AUXILIARY_RESULT_REGISTER_SET_D:
164 /* RDAUXD command -> auxiliary register group D */
165 cellOffset = ADI_VOLTAGE_09_11_OFFSET;
166 numberOfVoltagesInRegister = ADI_NUMBER_OF_GPIO_VOLTAGES_IN_REGISTER_D;
167 break;
168 default: /* LCOV_EXCL_LINE */
169 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
170 break; /* LCOV_EXCL_LINE */
171 }
172
173 for (uint16_t m = 0u; m < ADI_N_ADI; m++) {
174 /* parse voltages contained in one register */
175 for (uint16_t gpio = 0u; gpio < numberOfVoltagesInRegister; gpio++) {
176 voltageIndex = gpio + cellOffset;
177 if (voltageIndex < SLV_NR_OF_GPIOS_PER_MODULE) {
178 bufferMSB = (uint16_t)(pData
179 [(ADI_RAW_VOLTAGE_SIZE_IN_BYTES * gpio) +
180 (m * ADI_MAX_REGISTER_SIZE_IN_BYTES) + 1u]);
181 bufferLSB =
182 (uint16_t)(pData[(ADI_RAW_VOLTAGE_SIZE_IN_BYTES * gpio) + (m * ADI_MAX_REGISTER_SIZE_IN_BYTES)]);
183 rawValue = bufferLSB | (bufferMSB << ADI_BYTE_SHIFT);
184 signedValue = (int16_t)rawValue;
185 floatVoltage = ((float_t)signedValue * ADI_VOLTAGE_CONVERSION_FACTOR * ADI_VOLTAGE_CONVERSION_UNIT) +
186 ADI_VOLTAGE_CONVERSION_OFFSET;
187 voltage = (int16_t)floatVoltage; /* Unit mV */
188
189 /* RequirementId: D7.1 V1R0 SIF-4.40.02.01 */
190 /* Check that register does not contain cleared value */
191 if (rawValue != ADI_REGISTER_CLEARED_VALUE) {
192 pAdiState->data.errorTable->auxiliaryRegisterContentIsNotStuck[pAdiState->currentString][m] = true;
193 /* Check PEC for every IC in the daisy-chain */
194 if (ADI_EvaluateDiagnosticGpioVoltages(pAdiState, m) == true) {
195 pGpioVoltageTable->gpioVoltages_mV[pAdiState->currentString]
196 [voltageIndex + (m * SLV_NR_OF_GPIOS_PER_MODULE)] = voltage;
197 pGpioVoltageTable->invalidGpioVoltages[pAdiState->currentString][m] &= (~(1u << voltageIndex));
198 } else {
199 pGpioVoltageTable->invalidGpioVoltages[pAdiState->currentString][m] |= (1u << voltageIndex);
200 }
201 } else {
202 pAdiState->data.errorTable->auxiliaryRegisterContentIsNotStuck[pAdiState->currentString][m] = false;
203 }
204 }
205 }
206 }
207}

209/*========== Extern Function Implementations ================================*/
210extern void ADI_GetGpioVoltages(
211 ADI_STATE_s *pAdiState,
212 ADI_AUXILIARY_REGISTER_TYPE_e registerType,
213 ADI_AUXILIARY_STORE_LOCATION_e storeLocation) {
214 FAS_ASSERT(pAdiState != NULL_PTR);
215 FAS_ASSERT((registerType == ADI_AUXILIARY_REGISTER) || (registerType == ADI_REDUNDANT_AUXILIARY_REGISTER));
216 FAS_ASSERT(
217 (storeLocation == ADI_AUXILIARY_VOLTAGE) || (storeLocation == ADI_REDUNDANT_AUXILIARY_VOLTAGE) ||
218 (storeLocation == ADI_AUXILIARY_VOLTAGE_OPEN_WIRE));
219 uint16_t registerA[ADI_COMMAND_DEFINITION_LENGTH] = {0};
220 uint16_t registerB[ADI_COMMAND_DEFINITION_LENGTH] = {0};
221 uint16_t registerC[ADI_COMMAND_DEFINITION_LENGTH] = {0};
222 uint16_t registerD[ADI_COMMAND_DEFINITION_LENGTH] = {0};
223 uint16_t registerE[ADI_COMMAND_DEFINITION_LENGTH] = {0};
224
225 switch (registerType) {
226 case ADI_AUXILIARY_REGISTER:
227 ADI_CopyCommandBytes(adi_cmdRdauxa, registerA);
228 ADI_CopyCommandBytes(adi_cmdRdauxb, registerB);
229 ADI_CopyCommandBytes(adi_cmdRdauxc, registerC);
230 ADI_CopyCommandBytes(adi_cmdRdauxd, registerD);
231 ADI_CopyCommandBytes(adi_cmdRdauxe, registerE);
232 break;
233 case ADI_REDUNDANT_AUXILIARY_REGISTER:
234 ADI_CopyCommandBytes(adi_cmdRdraxa, registerA);
235 ADI_CopyCommandBytes(adi_cmdRdraxb, registerB);
236 ADI_CopyCommandBytes(adi_cmdRdraxc, registerC);
237 ADI_CopyCommandBytes(adi_cmdRdraxd, registerD);
238 break;
239 default: /* LCOV_EXCL_LINE */
240 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
241 break; /* LCOV_EXCL_LINE */
242 }
243
244 ADI_CopyCommandBytes(registerA, adi_command);
245 ADI_ReadRegister(adi_command, adi_dataReceive, pAdiState);
246 ADI_SaveRxToGpioVoltageBuffer(pAdiState, adi_dataReceive, ADI_AUXILIARY_RESULT_REGISTER_SET_A, storeLocation);
247
248 ADI_CopyCommandBytes(registerB, adi_command);
249 ADI_ReadRegister(adi_command, adi_dataReceive, pAdiState);
250 ADI_SaveRxToGpioVoltageBuffer(pAdiState, adi_dataReceive, ADI_AUXILIARY_RESULT_REGISTER_SET_B, storeLocation);
251
252 ADI_CopyCommandBytes(registerC, adi_command);
253 ADI_ReadRegister(adi_command, adi_dataReceive, pAdiState);
254 ADI_SaveRxToGpioVoltageBuffer(pAdiState, adi_dataReceive, ADI_AUXILIARY_RESULT_REGISTER_SET_C, storeLocation);
255
256 ADI_CopyCommandBytes(registerD, adi_command);
257 ADI_ReadRegister(adi_command, adi_dataReceive, pAdiState);
258 ADI_SaveRxToGpioVoltageBuffer(pAdiState, adi_dataReceive, ADI_AUXILIARY_RESULT_REGISTER_SET_D, storeLocation);
259}
