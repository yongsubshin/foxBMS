193static void ADI_AccessToDatabase(ADI_STATE_s *pAdiState) {
194 FAS_ASSERT(pAdiState != NULL_PTR);
195
196 /* Increment state variable each time new values are written into database */
197 (void)DATA_WRITE_DATA(
198 pAdiState->data.cellVoltage, pAdiState->data.allGpioVoltages, pAdiState->data.cellTemperature);
199 ADI_Wait(2u); /* Block task to leave CPU time for the other tasks */
200
201 (void)DATA_READ_DATA(pAdiState->data.balancingControl);
202}

204/* RequirementId: D7.1 V1R0 FUN-6.10.01.02 */
205static void ADI_BalanceControl(ADI_STATE_s *pAdiState) {
206 FAS_ASSERT(pAdiState != NULL_PTR);
207
208 /* Unmute balancing, cell voltage measurements must have been stopped before */
209 ADI_CopyCommandBytes(adi_cmdUnmute, adi_command);
210 ADI_TransmitCommand(adi_command, pAdiState);
211
212 /* Write the balancing registers of the ades183x */
213 ADI_DetermineBalancingRegisterConfiguration(pAdiState);
214
215 /* Wait ADI_BALANCING_TIME_ms milliseconds in order to balance
216 Measurements are stopped during this time */
217 ADI_Wait(ADI_BALANCING_TIME_ms);
218
219 /* Mute balancing, so that cell voltage measurements can be restarted */
220 ADI_CopyCommandBytes(adi_cmdMute, adi_command);
221 ADI_TransmitCommand(adi_command, pAdiState);
222}

224static STD_RETURN_TYPE_e ADI_GetRequest(AFE_REQUEST_e *request) {
225 FAS_ASSERT(request != NULL_PTR);
226 STD_RETURN_TYPE_e requestReceived = STD_NOT_OK;
227 OS_STD_RETURN_e receivedFromQueue =
228 OS_ReceiveFromQueue(ftsk_afeRequestQueue, (void *)request, ADI_QUEUE_TIMEOUT_MS);
229 if (receivedFromQueue == OS_SUCCESS) {
230 /* request queue was full, request retrieved */
231 requestReceived = STD_OK;
232 }
233 return requestReceived;
234}

236static bool ADI_ProcessMeasurementNotStartedState(ADI_STATE_s *pAdiState, AFE_REQUEST_e *request) {
237 FAS_ASSERT(pAdiState != NULL_PTR);
238 FAS_ASSERT(request != NULL_PTR);
239
240 bool measurementStarted = false;
241 STD_RETURN_TYPE_e requestReceived = ADI_GetRequest(request);
242 if (requestReceived == STD_OK) { /* request queue was not empty */
243 if (*request == AFE_START_REQUEST) {
244 ADI_SanityConfigurationCheck(pAdiState);
245 ADI_InitializeMeasurement(pAdiState);
246 measurementStarted = true;
247 } else { /* Until requested to start, block task to leave CPU time for the other tasks */
248 ADI_Wait(1u);
249 }
250 } else { /* Until requested to start, block task to leave CPU time for the other tasks */
251 ADI_Wait(1u);
252 }
253 return measurementStarted;
254}

256static void ADI_RunCurrentStringMeasurement(ADI_STATE_s *pAdiState) {
257 FAS_ASSERT(pAdiState != NULL_PTR);
258
259 /* Start auxiliary voltage measurement, all channels */
260 ADI_CopyCommandBytes(adi_cmdAdax, adi_command);
261 ADI_WriteCommandConfigurationBits(adi_command, ADI_ADAX_OW_POS, ADI_ADAX_OW_LEN, 0u);
262 ADI_WriteCommandConfigurationBits(adi_command, ADI_ADAX_PUP_POS, ADI_ADAX_PUP_LEN, 0u);
263 ADI_WriteCommandConfigurationBits(adi_command, ADI_ADAX_CH4_POS, ADI_ADAX_CH4_LEN, 0u);
264 ADI_WriteCommandConfigurationBits(adi_command, ADI_ADAX_CH03_POS, ADI_ADAX_CH03_LEN, 0u);
265 ADI_TransmitCommand(adi_command, pAdiState);
266 /* Start redundant auxiliary voltage measurement, one channel */
267 ADI_CopyCommandBytes(adi_cmdAdax2, adi_command);
268 ADI_WriteCommandConfigurationBits(
269 adi_command,
270 ADI_ADAX2_CH03_POS,
271 ADI_ADAX2_CH03_LEN,
272 pAdiState->redundantAuxiliaryChannel[pAdiState->currentString]);
273 ADI_TransmitCommand(adi_command, pAdiState);
274
275 ADI_Wait(ADI_WAIT_TIME_1_FOR_ADAX_FULL_CYCLE);
276
277 /* Snapshot to freeze cell voltage measurement result registers */
278 ADI_CopyCommandBytes(adi_cmdSnap, adi_command);
279 ADI_TransmitCommand(adi_command, pAdiState);
280 /* Retrieve filtered cell voltages */
281 ADI_GetVoltages(pAdiState, ADI_CELL_VOLTAGE_REGISTER, ADI_CELL_VOLTAGE);
282 ADI_GetVoltages(pAdiState, ADI_FILTERED_CELL_VOLTAGE_REGISTER, ADI_FILTERED_CELL_VOLTAGE);
283 /* Release snapshot to refresh cell voltage measurement result registers again */
284 ADI_CopyCommandBytes(adi_cmdUnsnap, adi_command);
285 ADI_TransmitCommand(adi_command, pAdiState);
286
287 /* Wait until auxiliary measurement cycle is finished */
288 ADI_Wait(ADI_WAIT_TIME_2_FOR_ADAX_FULL_CYCLE);
289
290 /* Retrieve string and module voltages */
291 ADI_GetStringAndModuleVoltage(pAdiState);
292 /* Retrieve GPIO voltages (all channels) */
293 ADI_GetGpioVoltages(pAdiState, ADI_AUXILIARY_REGISTER, ADI_AUXILIARY_VOLTAGE);
294 /* Get temperatures via GPIO voltages */
295 ADI_GetTemperatures(pAdiState);
296
297 ADI_AccessToDatabase(pAdiState);
298
299 /* If at least one cell is balanced, measurements are stopped before activating balancing */
300 if (pAdiState->data.balancingControl->nrBalancedCells[pAdiState->currentString] > 0u) {
301 /* Stop cell voltage measurements before activating balancing */
302 ADI_StopContinuousCellVoltageMeasurements(pAdiState);
303 /* Activate balancing (if necessary) */
304 ADI_BalanceControl(pAdiState);
305 /* Wait ADI_BALANCING_TIME_ms milliseconds in order to balance
306 Measurements are stopped during this time */
307 ADI_Wait(ADI_BALANCING_TIME_ms);
308 /* Restart cell voltage measurements before activating balancing */
309 ADI_RestartContinuousCellVoltageMeasurements(pAdiState);
310 }
311
312 ADI_Diagnostic(pAdiState);
313
314 /* Cycle finished for string, clear values to check that they are not stuck during next reading */
315 ADI_CopyCommandBytes(adi_cmdClrcell, adi_command);
316 ADI_TransmitCommand(adi_command, pAdiState);
317}

326static void ADI_SanityConfigurationCheck(ADI_STATE_s *pAdiState) {
327 FAS_ASSERT(pAdiState != NULL_PTR);
328
329 /* Check configuration for temperature sensors */
330 uint8_t configuredTemperatureSensorInputs = 0u;
331 for (uint16_t gpioIndex = 0u; gpioIndex < SLV_NR_OF_GPIOS_PER_MODULE; gpioIndex++) {
332 if (adi_temperatureInputsUsed[gpioIndex] != 0u) {
333 configuredTemperatureSensorInputs++;
334 }
335 }
336 if (configuredTemperatureSensorInputs != BS_NR_OF_TEMP_SENSORS_PER_MODULE) {
337 /* According to SW configuration number of configured temperature sensors
338 * configured in AFE driver then in overall battery system configuration */
339 FAS_ASSERT(FAS_TRAP);
340 }
341}

386/* RequirementId: D7.1 V1R0 FUN-1.10.01.03 */
387/* RequirementId: D7.1 V1R0 FUN-2.10.01.03 */
388/* RequirementId: D7.1 V1R0 FUN-4.10.01.03 */
389extern void ADI_MeasurementCycle(ADI_STATE_s *pAdiState) {
390 FAS_ASSERT(pAdiState != NULL_PTR);
391 STD_RETURN_TYPE_e requestReceived = STD_OK;
392 AFE_REQUEST_e request = AFE_NO_REQUEST;
393
394 /* AXIVION Next Line Style MisraC2012-2.2 FaultDetection-DeadBranches: non-blocking driver requires an infinite
395 * loop for the driver implementation */
396 while (FOREVER()) {
397 if (pAdiState->measurementStarted == false) { /* Wait until requested to start */
398 pAdiState->measurementStarted = ADI_ProcessMeasurementNotStartedState(pAdiState, &request);
399 } else {
400 while (pAdiState->currentString < pAdiState->spiNumberInterfaces) {
401 ADI_RunCurrentStringMeasurement(pAdiState);
402 ++pAdiState->currentString;
403 }
404 pAdiState->currentString = 0u;
405 if (ADI_IsFirstMeasurementCycleFinished(pAdiState) == false) {
406 ADI_SetFirstMeasurementCycleFinished(pAdiState);
407 }
408 requestReceived = ADI_GetRequest(&request);
409 if (requestReceived == STD_OK) {
410 /* request queue was not empty */
411 if (request == AFE_STOP_REQUEST) {
412 pAdiState->measurementStarted = false;
413 }
414 }
415 }
416
417#ifdef UNITY_UNIT_TEST /* break out of loop in unit test*/
418 break;
419#endif
420 }
421}
