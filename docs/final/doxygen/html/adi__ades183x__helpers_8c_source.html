109static void ADI_StoredConfigurationFillRegisterData(
110 uint8_t module,
111 ADI_CFG_REGISTER_SET_e registerSet,
112 uint8_t registerOffset,
113 uint8_t data,
114 uint8_t position,
115 uint8_t mask,
116 ADI_STATE_s *adiState) {
117 /* AXIVION Routine Generic-MissingParameterAssert: module: parameter accepts whole range */
118 /* AXIVION Routine Generic-MissingParameterAssert: data: parameter accepts whole range */
119 FAS_ASSERT(adiState != NULL_PTR);
120 FAS_ASSERT(registerSet <= ADI_CFG_REGISTER_SET_B);
121 /* Only 8 bit in a register part */
122 FAS_ASSERT(position <= ADI_MAX_BIT_POSITION_IN_BYTE);
123 /* All 0s mask would mean do nothing */
124 FAS_ASSERT(mask > 0u);
125 FAS_ASSERT(registerOffset <= ADI_REGISTER_OFFSET5);
126
127 if (registerSet == ADI_CFG_REGISTER_SET_A) {
128 ADI_WriteDataBits(
129 &adi_configurationRegisterAgroup[adiState->currentString][(module * ADI_WRCFGA_LEN) + registerOffset],
130 data,
131 position,
132 mask);
133 } else if (registerSet == ADI_CFG_REGISTER_SET_B) {
134 ADI_WriteDataBits(
135 &adi_configurationRegisterBgroup[adiState->currentString][(module * ADI_WRCFGB_LEN) + registerOffset],
136 data,
137 position,
138 mask);
139 } else { /* invalid register set */
140 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
141 }

144static void ADI_IncrementCommandCounter(ADI_STATE_s *adiState) {
145 FAS_ASSERT(adiState != NULL_PTR);
146
147 /**
148 * SM_SPI_CNT: SPI Frame Counter
149 * Increment driver stored value of command counter if command causes increase.
150 */
151 for (uint16_t i = 0u; i < ADI_N_ADI; i++) {
152 if (adiState->data.commandCounter[adiState->currentString][i] < ADI_COMMAND_COUNTER_MAX_VALUE) {
153 adiState->data.commandCounter[adiState->currentString][i]++;
154 } else {
155 adiState->data.commandCounter[adiState->currentString][i] = ADI_COMMAND_COUNTER_RESTART_VALUE;
156 }
157 }

161extern STD_RETURN_TYPE_e ADI_CheckConfigurationRegister(ADI_CFG_REGISTER_SET_e registerSet, ADI_STATE_s *adiState) {
162 FAS_ASSERT(adiState != NULL_PTR);
163 FAS_ASSERT(registerSet <= ADI_CFG_REGISTER_SET_B);
164
165 STD_RETURN_TYPE_e returnValue = STD_OK;
166 bool configurationAIsOk = true;
167 bool configurationBIsOk = true;
168
169 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
170 /* Test configuration register A */
171 for (uint8_t byte = 0; byte < ADI_WRCFGA_LEN; byte++) {
172 if (adi_configurationRegisterAgroup[adiState->currentString][byte + (m * ADI_WRCFGA_LEN)] !=
173 adi_readConfigurationRegisterAgroup[adiState->currentString][byte + (m * ADI_WRCFGA_LEN)]) {
174 configurationAIsOk = false;
175 returnValue = STD_NOT_OK;
176 break;
177 }
178 }
179 if (configurationAIsOk == false) {
180 adiState->data.errorTable->configurationAIsOk[adiState->currentString][m] = false;
181 } else {
182 adiState->data.errorTable->configurationAIsOk[adiState->currentString][m] = true;
183 }
184 configurationAIsOk = true;
185 /* Test configuration register B */
186 for (uint8_t byte = 0; byte < ADI_WRCFGA_LEN; byte++) {
187 if (adi_configurationRegisterBgroup[adiState->currentString][byte + (m * ADI_WRCFGB_LEN)] !=
188 adi_readConfigurationRegisterBgroup[adiState->currentString][byte + (m * ADI_WRCFGB_LEN)]) {
189 configurationBIsOk = false;
190 returnValue = STD_NOT_OK;
191 break;
192 }
193 }
194 if (configurationBIsOk == false) {
195 adiState->data.errorTable->configurationBIsOk[adiState->currentString][m] = false;
196 } else {
197 adiState->data.errorTable->configurationBIsOk[adiState->currentString][m] = true;
198 }
199 configurationBIsOk = true;
200 }
201 return returnValue;
202}

204extern void ADI_WriteCommandConfigurationBits(
205 uint16_t *configuredCommand,
206 uint8_t position,
207 uint8_t length,
208 uint16_t configuration) {
209 FAS_ASSERT(configuredCommand != NULL_PTR);
210 FAS_ASSERT(length > 0u);
211 FAS_ASSERT(length <= ADI_COMMAND_CODE_LENGTH);
212 FAS_ASSERT(position <= (ADI_COMMAND_CODE_LENGTH - 1u));
213 FAS_ASSERT(configuration <= (~(ADI_DATA_MASK_SEED << length)));
214
215 uint16_t command = (((uint16_t)configuredCommand[ADI_COMMAND_FIRST_BYTE_POSITION]) << ADI_BYTE_SHIFT) |
216 configuredCommand[ADI_COMMAND_SECOND_BYTE_POSITION];
217 uint16_t mask = (~(ADI_COMMAND_MASK_SEED << length)) << position;
218 /* Set bits to modify to 0 */
219 command &= ~mask;
220 /* Now set bits, works if 0 or not */
221 command |= (configuration << position) & mask;
222 configuredCommand[ADI_COMMAND_FIRST_BYTE_POSITION] = (command >> ADI_BYTE_SHIFT) & ADI_ONE_BYTE_MASK;
223 configuredCommand[ADI_COMMAND_SECOND_BYTE_POSITION] = command & ADI_ONE_BYTE_MASK;
224}

226extern STD_RETURN_TYPE_e ADI_StoredConfigurationWriteToAfe(ADI_CFG_REGISTER_SET_e registerSet, ADI_STATE_s *adiState) {
227 FAS_ASSERT(adiState != NULL_PTR);
228 FAS_ASSERT(registerSet <= ADI_CFG_REGISTER_SET_B);
229 STD_RETURN_TYPE_e returnValueA = STD_OK;
230 STD_RETURN_TYPE_e returnValueB = STD_OK;
231 STD_RETURN_TYPE_e returnValue = STD_OK;
232
233 if (registerSet == ADI_CFG_REGISTER_SET_A) {
234 ADI_WriteRegister(
235 adi_cmdWrcfga,
236 adi_configurationRegisterAgroup[adiState->currentString],
237 ADI_PEC_NO_FAULT_INJECTION,
238 adiState);
239 ADI_CopyCommandBytes(adi_cmdRdcfga, adi_command);
240 ADI_ReadRegister(adi_command, adi_readConfigurationRegisterAgroup[adiState->currentString], adiState);
241 returnValueA = ADI_CheckConfigurationRegister(ADI_CFG_REGISTER_SET_A, adiState);
242 } else if (registerSet == ADI_CFG_REGISTER_SET_B) {
243 ADI_WriteRegister(
244 adi_cmdWrcfgb,
245 adi_configurationRegisterBgroup[adiState->currentString],
246 ADI_PEC_NO_FAULT_INJECTION,
247 adiState);
248 ADI_CopyCommandBytes(adi_cmdRdcfgb, adi_command);
249 ADI_ReadRegister(adi_command, adi_readConfigurationRegisterBgroup[adiState->currentString], adiState);
250 returnValueB = ADI_CheckConfigurationRegister(ADI_CFG_REGISTER_SET_B, adiState);
251 } else { /* invalid register set */
252 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
253 }
254 if ((returnValueA == STD_NOT_OK) || (returnValueB == STD_NOT_OK)) {
255 returnValue = STD_NOT_OK;
256 }
257 return returnValue;
258}

268extern void ADI_ReadDataBits(uint8_t receivedData, uint8_t *pDataToRead, uint8_t position, uint8_t mask) {
269 /* AXIVION Routine Generic-MissingParameterAssert: receivedData: parameter accepts whole range */
270 FAS_ASSERT(pDataToRead != NULL_PTR);
271 /* Only 8 bit in a register part */
272 FAS_ASSERT(position <= ADI_MAX_BIT_POSITION_IN_BYTE);
273 /* All 0s mask would mean do nothing */
274 FAS_ASSERT(mask > 0u);
275
276 *pDataToRead = (receivedData & mask) >> position;

279/* RequirementId: D7.1 V1R0 FUN-0.0.01.01 */
280/* RequirementId: D7.1 V1R0 SIF-4.10.02.02 */
281/* RequirementId: D7.1 V1R0 SIF-4.10.04.02 */
282/* RequirementId: D7.1 V1R0 SIF-4.10.04.03 */
283/* RequirementId: D7.1 V1R0 SIF-4.10.05.01 */
284extern void ADI_ReadRegister(uint16_t *registerToRead, uint8_t *data, ADI_STATE_s *adiState) {
285 FAS_ASSERT(registerToRead != NULL_PTR);
286 FAS_ASSERT(data != NULL_PTR);
287 FAS_ASSERT(adiState != NULL_PTR);
288
289 uint8_t receivedDataBytes[ADI_SIZE_OF_DATA_FOR_PEC_COMPUTATION_WITH_COUNTER] = {0};
290 uint16_t calculatedPec = 0u;
291 uint8_t pecAsByteArray[ADI_PEC_SIZE_IN_BYTES] = {0};
292 uint8_t afeCommandCounter = 0u;
293 uint16_t registerLengthInBytes = registerToRead[ADI_COMMAND_DATA_LENGTH_POSITION];
294 uint16_t byte = 0u; /* variable to parse data bytes */
295
296 /* One frame = data + 2 bytes PEC */
297 /* + 2u: the two additional bytes corresponding to the PEC */
298 uint16_t spiFrameLength = registerLengthInBytes + ADI_PEC_SIZE_IN_BYTES;
299 uint16_t dataLength = registerLengthInBytes;
300
301 for (uint16_t i = 0; i < ADI_N_BYTES_FOR_DATA_TRANSMISSION; i++) {
302 adiState->data.txBuffer[i] = 0x0;
303 }
304
305 /**
306 * SM_SPI_PEC: SPI Packet Error Code
307 * Calculate PEC for command.
308 */
309 /* Compute PEC of the two command bytes */
310 receivedDataBytes[ADI_COMMAND_FIRST_BYTE_POSITION] = (uint8_t)registerToRead[ADI_COMMAND_FIRST_BYTE_POSITION];
311 receivedDataBytes[ADI_COMMAND_SECOND_BYTE_POSITION] = (uint8_t)registerToRead[ADI_COMMAND_SECOND_BYTE_POSITION];
312 calculatedPec = ADI_Pec15(ADI_COMMAND_SIZE_IN_BYTES, receivedDataBytes);
313
314 adiState->data.txBuffer[ADI_COMMAND_FIRST_BYTE_POSITION] = registerToRead[ADI_COMMAND_FIRST_BYTE_POSITION];
315 adiState->data.txBuffer[ADI_COMMAND_SECOND_BYTE_POSITION] = registerToRead[ADI_COMMAND_SECOND_BYTE_POSITION];
316 adiState->data.txBuffer[ADI_COMMAND_PEC_FIRST_BYTE_POSITION] =
317 (uint8_t)((calculatedPec >> ADI_BYTE_SHIFT) & ADI_ONE_BYTE_MASK);
318 adiState->data.txBuffer[ADI_COMMAND_PEC_SECOND_BYTE_POSITION] = (uint8_t)(calculatedPec & ADI_ONE_BYTE_MASK);
319
320 /* 4u: two bytes command + two bytes command PEC */
321 /* Register length + 2u: The two additional bytes correspond to the PEC */
322 uint16_t frameLength = ADI_COMMAND_AND_PEC_SIZE_IN_BYTES +
323 ((registerLengthInBytes + ADI_PEC_SIZE_IN_BYTES) * ADI_N_ADI);
324 ADI_SpiTransmitReceiveData(adiState, adiState->data.txBuffer, adiState->data.rxBuffer, (uint32_t)frameLength);
325
326 for (uint16_t i = 0; i < ADI_N_ADI; i++) {
327 /* Put received messages in data buffer, used outside function to access data */
328 for (byte = 0u; byte < registerLengthInBytes; byte++) {
329 data[byte + (i * dataLength)] =
330 (uint8_t)(adiState->data.rxBuffer
331 [(ADI_FIRST_DATA_BYTE_POSITION_IN_TRANSMISSION_FRAME + byte) + (i * spiFrameLength)]);
332 }
333 /* receivedDataBytes is a local variable holding read data and used to compute the PEC */
334 for (byte = 0u; byte < registerLengthInBytes; byte++) {
335 receivedDataBytes[byte] =
336 (uint8_t)(adiState->data.rxBuffer
337 [(ADI_FIRST_DATA_BYTE_POSITION_IN_TRANSMISSION_FRAME + byte) + (i * spiFrameLength)]);
338 }
339 /* Data PEC is also computed on command counter, so one byte with command counter is added */
340 receivedDataBytes[byte] =
341 (uint8_t)(adiState->data.rxBuffer
342 [(ADI_FIRST_DATA_BYTE_POSITION_IN_TRANSMISSION_FRAME + byte) + (i * spiFrameLength)]) &
343 ADI_COMMAND_COUNTER_MASK;
344
345 calculatedPec = ADI_Pec10(
346 (uint8_t)(registerToRead[ADI_COMMAND_DATA_LENGTH_POSITION] & ADI_ONE_BYTE_MASK), receivedDataBytes, true);
347 pecAsByteArray[ADI_DATA_PEC_FIRST_BYTE_POSITION] =
348 (uint8_t)((calculatedPec >> ADI_BYTE_SHIFT) & ADI_ONE_BYTE_MASK);
349 pecAsByteArray[ADI_DATA_PEC_SECOND_BYTE_POSITION] = (uint8_t)(calculatedPec & ADI_ONE_BYTE_MASK);
350
351 /* Position of the PEC (two bytes) in the first read frame */
352 /* 4u: two bytes command + two bytes command PEC, followed by number of bytes in register */
353 uint16_t crcByte0Position = registerLengthInBytes + ADI_COMMAND_AND_PEC_SIZE_IN_BYTES;
354 uint16_t crcByte1Position = registerLengthInBytes + ADI_COMMAND_AND_PEC_SIZE_IN_BYTES + 1u;
355
356 /**
357 * SM_SPI_PEC: SPI Packet Error Code
358 * PEC check on read value.
359 */
360 /* if calculated PEC not equal to received PEC */
361 const uint8_t pecLowByte = adiState->data.rxBuffer[crcByte0Position + (i * spiFrameLength)] &
362 ADI_PEC10_MSB_EXCLUDE_COMMAND_COUNTER;
363 const uint8_t pecHighByte =
364 (adiState->data.rxBuffer[crcByte1Position + (i * spiFrameLength)] & ADI_ONE_BYTE_MASK);
365
366 const bool pecLowByteOk = (bool)(pecAsByteArray[ADI_DATA_PEC_FIRST_BYTE_POSITION] == pecLowByte);
367 const bool pecHighByteOk = (bool)(pecAsByteArray[ADI_DATA_PEC_SECOND_BYTE_POSITION] == pecHighByte);
368 /* update error table of the corresponding ades183x */
369 if (pecLowByteOk && pecHighByteOk) {
370 adiState->data.errorTable->crcIsOk[adiState->currentString][i] = true;
371 } else {
372 adiState->data.errorTable->crcIsOk[adiState->currentString][i] = false;
373 }
374
375 /* CRC is placed after the data bytes */
376 uint16_t crcFirstBytePosition = ADI_FIRST_DATA_BYTE_POSITION_IN_TRANSMISSION_FRAME + registerLengthInBytes;
377 /**
378 * SM_SPI_CNT: SPI Frame Counter
379 * Compare driver stored value of command counter with values sent by the AFEs.
380 */
381 /* Retrieve command counter value sent by the AFE, included in first CRC byte of answer frame */
382 uint16_t commandCounterData = adiState->data.rxBuffer[crcFirstBytePosition + (i * spiFrameLength)] &
383 ADI_COMMAND_COUNTER_MASK;
384 afeCommandCounter = (uint8_t)((commandCounterData >> ADI_COMMAND_COUNTER_POSITION) & ADI_ONE_BYTE_MASK);
385
386 if (afeCommandCounter != adiState->data.commandCounter[adiState->currentString][i]) {
387 /**
388 * command counter flag stays set as soon as an error was detected
389 * it can only be reset within the reset function
390 */
391 adiState->data.errorTable->commandCounterIsOk[adiState->currentString][i] = false;
392 }
393 }
394}

397extern void ADI_SpiTransmitReceiveData(
398 ADI_STATE_s *adiState,
399 uint16_t *pTxBuff,
400 uint16_t *pRxBuff,
401 uint32_t frameLength) {
402 /* AXIVION Routine Generic-MissingParameterAssert: frameLength: parameter accepts whole range */
403 /* AXIVION Routine Generic-MissingParameterAssert: pRxBuff: parameter accepts whole range */
404 FAS_ASSERT(adiState != NULL_PTR);
405 FAS_ASSERT(pTxBuff != NULL_PTR);
406
407 STD_RETURN_TYPE_e retValDummy = STD_OK;
408 STD_RETURN_TYPE_e retVal = STD_OK;
409 if (frameLength == 0u) {
410 /* Length = 0: only transmit dummy byte */
411 /* START SPI function to adapt for different environment */
412 /* No DMA, blocking */
413 retVal = SPI_TransmitDummyByte(&spi_adiInterface[adiState->currentString], 0u);
414 /* END SPI function to adapt for different environment */
415 } else {
416 if (pRxBuff == NULL_PTR) {
417 /* Transmit only (receive discarded) */
418 /* START SPI function to adapt for different environment */
419 /* No DMA, blocking */
420 retValDummy =
421 SPI_TransmitDummyByte(&spi_adiInterface[adiState->currentString], ADI_SPI_WAKEUP_WAIT_TIME_US);
422 retVal = SPI_TransmitData(&spi_adiInterface[adiState->currentString], pTxBuff, frameLength);
423 /* END SPI function to adapt for different environment */
424 } else {
425 /* Transmit and receive */
426 /* START SPI function to adapt for different environment */
427 /* Uses DMA and notifications */
428 retValDummy =
429 SPI_TransmitDummyByte(&spi_adiInterface[adiState->currentString], ADI_SPI_WAKEUP_WAIT_TIME_US);
430 retVal =
431 SPI_TransmitReceiveDataDma(&spi_adiInterface[adiState->currentString], pTxBuff, pRxBuff, frameLength);
432 /* Now wait for notification in DMA Rx finished interrupt */
433 uint32_t timeout = ADI_TRANSMISSION_TIMEOUT;
434 FAS_ASSERT(timeout > 0u);
435 uint32_t ulNotifiedValue = 0u;
436 /* Suspend task and wait for notification, clear notification value on entry and exit */
437 (void)OS_WaitForNotification(&ulNotifiedValue, timeout);
438 if (ulNotifiedValue != ADI_DMA_SPI_FINISHED_NOTIFICATION_VALUE) {
439 /* if ulNotifiedValue not written by xTaskNotify() in DMA interrupt: transmission timeout */
440 retVal = STD_NOT_OK;
441 }
442 /* END SPI function to adapt for different environment */
443 }
444 }
445
446 if ((retVal == STD_NOT_OK) || (retValDummy == STD_NOT_OK)) {
447 adiState->data.errorTable->spiIsOk[adiState->currentString] = false;
448 } else if ((retVal == STD_OK) && (retValDummy == STD_OK)) {
449 adiState->data.errorTable->spiIsOk[adiState->currentString] = true;
450 } else {
451 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
452 }
453}

479extern void ADI_TransmitCommand(uint16_t *command, ADI_STATE_s *adiState) {
480 FAS_ASSERT(command != NULL_PTR);
481 FAS_ASSERT(adiState != NULL_PTR);
482
483 uint8_t PEC_Check[ADI_MAX_REGISTER_SIZE_IN_BYTES] = {0};
484 uint16_t PEC_result = 0u;
485
486 /**
487 * SM_SPI_PEC: SPI Packet Error Code
488 * Calculate PEC for command.
489 */
490 PEC_Check[ADI_COMMAND_FIRST_BYTE_POSITION] = (uint8_t)command[ADI_COMMAND_BYTE0_POSITION];
491 PEC_Check[ADI_COMMAND_SECOND_BYTE_POSITION] = (uint8_t)command[ADI_COMMAND_BYTE1_POSITION];
492 PEC_result = ADI_Pec15(ADI_COMMAND_SIZE_IN_BYTES, PEC_Check);
493
494 adiState->data.txBuffer[ADI_COMMAND_FIRST_BYTE_POSITION] = command[ADI_COMMAND_BYTE0_POSITION];
495 adiState->data.txBuffer[ADI_COMMAND_SECOND_BYTE_POSITION] = command[ADI_COMMAND_BYTE1_POSITION];
496 adiState->data.txBuffer[ADI_COMMAND_PEC_FIRST_BYTE_POSITION] =
497 (uint8_t)((PEC_result >> ADI_BYTE_SHIFT) & ADI_ONE_BYTE_MASK);
498 adiState->data.txBuffer[ADI_COMMAND_PEC_SECOND_BYTE_POSITION] = (uint8_t)(PEC_result & ADI_ONE_BYTE_MASK);
499
500 ADI_SpiTransmitReceiveData(adiState, adiState->data.txBuffer, NULL_PTR, ADI_COMMAND_AND_PEC_SIZE_IN_BYTES);
501
502 /**
503 * SM_SPI_CNT: SPI Frame Counter
504 * Increment driver stored value of command counter if command causes increase.
505 */
506 /** If command increments AFE command counter, increment driver command counter */
507 if (command[ADI_COMMAND_INC_POSITION] == 1u) {
508 ADI_IncrementCommandCounter(adiState);
509 }
510}

584/* RequirementId: D7.1 V1R0 FUN-0.0.01.02 */
585/* RequirementId: D7.1 V1R0 SIF-4.10.02.02 */
586/* RequirementId: D7.1 V0R11 SIF-4.10.03.03 */
587/* RequirementId: D7.1 V1R0 SIF-4.10.04.01 */
588extern void ADI_WriteRegister(
589 const uint16_t *registerToWrite,
590 uint8_t *data,
591 ADI_PEC_FAULT_INJECTION_e pecFaultInjection,
592 ADI_STATE_s *adiState) {
593 FAS_ASSERT(registerToWrite != NULL_PTR);
594 FAS_ASSERT(data != NULL_PTR);
595 FAS_ASSERT(adiState != NULL_PTR);
596
597 uint16_t PEC_result = 0;
598 uint8_t PEC_Check[ADI_SIZE_OF_DATA_FOR_PEC_COMPUTATION] = {0x00, 0x00, 0x00, 0x00, 0x00, 0x00};
599 uint16_t registerLengthInBytes = registerToWrite[ADI_COMMAND_DATA_LENGTH_POSITION];
600 uint16_t spiFrameLength = registerLengthInBytes + 2u;
601 uint16_t dataLength = registerLengthInBytes;
602
603 /**
604 * SM_SPI_PEC: SPI Packet Error Code
605 * Calculate PEC for command.
606 */
607 /* Compute PEC of the two command bytes */
608 PEC_Check[ADI_COMMAND_FIRST_BYTE_POSITION] = (uint8_t)registerToWrite[ADI_COMMAND_FIRST_BYTE_POSITION];
609 PEC_Check[ADI_COMMAND_SECOND_BYTE_POSITION] = (uint8_t)registerToWrite[ADI_COMMAND_SECOND_BYTE_POSITION];
610 PEC_result = ADI_Pec15(ADI_COMMAND_SIZE_IN_BYTES, PEC_Check);
611
612 adiState->data.txBuffer[ADI_COMMAND_FIRST_BYTE_POSITION] = registerToWrite[ADI_COMMAND_FIRST_BYTE_POSITION];
613 adiState->data.txBuffer[ADI_COMMAND_SECOND_BYTE_POSITION] = registerToWrite[ADI_COMMAND_SECOND_BYTE_POSITION];
614 if (pecFaultInjection == ADI_COMMAND_PEC_FAULT_INJECTION) {
615 adiState->data.txBuffer[ADI_COMMAND_PEC_FIRST_BYTE_POSITION] = (PEC_result >> ADI_BYTE_SHIFT) &
616 ADI_ONE_BYTE_MASK;
617 adiState->data.txBuffer[ADI_COMMAND_PEC_SECOND_BYTE_POSITION] = (PEC_result & ADI_ONE_BYTE_MASK) + 1u;
618 } else {
619 adiState->data.txBuffer[ADI_COMMAND_PEC_FIRST_BYTE_POSITION] = (PEC_result >> ADI_BYTE_SHIFT) &
620 ADI_ONE_BYTE_MASK;
621 adiState->data.txBuffer[ADI_COMMAND_PEC_SECOND_BYTE_POSITION] = PEC_result & ADI_ONE_BYTE_MASK;
622 }
623
624 for (uint16_t i = 0u; i < ADI_N_ADI; i++) {
625 /* data[] contains the data to write to the daisy-chain, it is used outside of the function */
626 for (uint16_t byte = 0; byte < registerLengthInBytes; byte++) {
627 adiState->data
628 .txBuffer[(ADI_FIRST_DATA_BYTE_POSITION_IN_TRANSMISSION_FRAME + byte) + (i * spiFrameLength)] =
629 data[byte + (i * dataLength)];
630 }
631 /**
632 * SM_SPI_PEC: SPI Packet Error Code
633 * Calculate PEC for values to be sent to the daisy-chain.
634 */
635 /* Calculate PEC of all data (1 PEC value for 6 bytes) */
636 /* PEC_Check is a local variable holding data to write and used to compute the PEC */
637 for (uint16_t byte = 0; byte < registerLengthInBytes; byte++) {
638 uint16_t bufferData =
639 adiState->data
640 .txBuffer[(ADI_FIRST_DATA_BYTE_POSITION_IN_TRANSMISSION_FRAME + byte) + (i * spiFrameLength)];
641 PEC_Check[byte] = (uint8_t)((bufferData)&ADI_ONE_BYTE_MASK);
642 }
643
644 PEC_result = ADI_Pec10((uint8_t)(registerLengthInBytes & ADI_ONE_BYTE_MASK), PEC_Check, false);
645
646 /* CRC is placed after the data bytes */
647 uint16_t crcFirstBytePosition = ADI_FIRST_DATA_BYTE_POSITION_IN_TRANSMISSION_FRAME + registerLengthInBytes;
648 if (pecFaultInjection == ADI_DATA_PEC_FAULT_INJECTION) {
649 adiState->data.txBuffer[crcFirstBytePosition + (i * spiFrameLength)] =
650 (PEC_result >> ADI_BYTE_SHIFT) & ADI_PEC10_FULL_EXCLUDE_COMMAND_COUNTER;
651 adiState->data.txBuffer[crcFirstBytePosition + 1u + (i * spiFrameLength)] =
652 (PEC_result & ADI_ONE_BYTE_MASK) + 1u;
653 } else {
654 adiState->data.txBuffer[crcFirstBytePosition + (i * spiFrameLength)] =
655 (PEC_result >> ADI_BYTE_SHIFT) & ADI_PEC10_FULL_EXCLUDE_COMMAND_COUNTER;
656 adiState->data.txBuffer[crcFirstBytePosition + 1u + (i * spiFrameLength)] = PEC_result & ADI_ONE_BYTE_MASK;
657 }
658 }
659
660 uint16_t frameLength = ADI_COMMAND_AND_PEC_SIZE_IN_BYTES +
661 ((registerLengthInBytes + ADI_PEC_SIZE_IN_BYTES) * ADI_N_ADI);
662 ADI_SpiTransmitReceiveData(adiState, adiState->data.txBuffer, adiState->data.rxBuffer, (uint32_t)frameLength);
663
664 /**
665 * SM_SPI_CNT: SPI Frame Counter
666 * Increment driver stored value of command counter if command causes increase.
667 */
668 /* If command increments AFE command counter, increment driver command counter */
669 /* Do not increment if a PEC error was injected */
670 if ((registerToWrite[ADI_COMMAND_INC_POSITION] == 1u) && (pecFaultInjection == ADI_PEC_NO_FAULT_INJECTION)) {
671 ADI_IncrementCommandCounter(adiState);
672 }
673}
