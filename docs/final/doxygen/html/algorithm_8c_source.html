82/*========== Static Function Implementations ================================*/
83static void ALGO_Initialization(void) {
84 /* iterate over all algorithms */
85 for (uint16_t i = 0u; i < algo_length; i++) {
86 /* check if the cycle time is valid */
87 FAS_ASSERT((algo_algorithms[i].cycleTime_ms % ALGO_TICK_ms) == 0u);
88
89 /* check only uninitialized algorithms */
90 if (algo_algorithms[i].state == ALGO_UNINITIALIZED) {
91 /* directly make ready when init function is a null pointer otherwise run init */
92 if (algo_algorithms[i].fpInitialization == NULL_PTR) {
93 algo_algorithms[i].state = ALGO_READY;
94 } else {
95 const STD_RETURN_TYPE_e result = algo_algorithms[i].fpInitialization();
96 FAS_ASSERT((STD_OK == result) || (STD_NOT_OK == result));
97 if (STD_OK == result) {
98 algo_algorithms[i].state = ALGO_READY;
99 } else {
100 algo_algorithms[i].state = ALGO_FAILED_INIT;
101 }
102 }
103 }
104 }
105}

115extern void ALGO_MainFunction(void) {
116 OS_EnterTaskCritical();
117 const bool initializationRequested = algo_initializationRequested;
118 OS_ExitTaskCritical();
119 if (initializationRequested == true) {
120 ALGO_Initialization();
121 OS_EnterTaskCritical();
122 algo_initializationRequested = false;
123 OS_ExitTaskCritical();
124 }
125
126 static uint32_t counter_ticks = 0u;
127
128 for (uint16_t i = 0u; i < algo_length; i++) {
129 const bool runAlgorithmAsap = (algo_algorithms[i].cycleTime_ms == 0u);
130 const bool runAlgorithmCycleElapsed =
131 ((algo_algorithms[i].cycleTime_ms != 0u) && ((counter_ticks % algo_algorithms[i].cycleTime_ms) == 0u));
132 if ((runAlgorithmAsap != false) || (runAlgorithmCycleElapsed != false)) {
133 /* Cycle time elapsed -> call function */
134 if (algo_algorithms[i].state == ALGO_READY) {
135 /* Set state to running -> reset to READY before leaving algorithm function */
136 algo_algorithms[i].state = ALGO_RUNNING;
137 algo_algorithms[i].startTime = OS_GetTickCount();
138 algo_algorithms[i].fpAlgorithm();
139 ALGO_MarkAsDone(i);
140 }
141 /* check if we need to reinitialize */
142 if (algo_algorithms[i].state == ALGO_REINIT_REQUESTED) {
143 /* set to uninitialized so that the algorithm can be reinitialized */
144 algo_algorithms[i].state = ALGO_UNINITIALIZED;
145
146 ALGO_UnlockInitialization();
147 }
148 }
149 }
150
151 counter_ticks += ALGO_TICK_ms;
152}

154extern void ALGO_MonitorExecutionTime(void) {
155 const uint32_t timestamp = OS_GetTickCount();
156
157 for (uint16_t i = 0u; i < algo_length; i++) {
158 if ((algo_algorithms[i].startTime != 0u) && (algo_algorithms[i].state == ALGO_RUNNING) &&
159 ((algo_algorithms[i].startTime + algo_algorithms[i].maxCalculationDuration_ms) < timestamp)) {
160 /* Block task from further execution because of runtime violation, but task will finish its execution */
161 algo_algorithms[i].state = ALGO_BLOCKED;
162
163 /* TODO: Add diag call to notify error in algorithm module */
164 }
165 }
166}
