126static void BAL_ActivateBalancing(void) {
127 float_t cellBalancingCurrent = 0.0f;
128 uint32_t difference = 0;
129
130 (void)DATA_READ_DATA(&bal_balancing, &bal_cellVoltage);
131
132 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
133 uint16_t nrBalancedCells = 0u;
134 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
135 for (uint16_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
136 if (bal_state.balancingAllowed == false) {
137 bal_balancing.activateBalancing[s][m][cb] = false;
138 } else {
139 if (bal_balancing.deltaCharge_mAs[s][m][cb] > 0u) {
140 bal_balancing.activateBalancing[s][m][cb] = true;
141 nrBalancedCells++;
142 cellBalancingCurrent = ((float_t)(bal_cellVoltage.cellVoltage_mV[s][m][cb])) /
143 SLV_BALANCING_RESISTANCE_ohm;
144 difference = (BAL_STATEMACH_BALANCING_TIME_100ms / 10u) * (uint32_t)(cellBalancingCurrent);
145 bal_state.active = true;
146 bal_balancing.enableBalancing = true;
147 /* we are working with unsigned integers */
148 if (difference > bal_balancing.deltaCharge_mAs[s][m][cb]) {
149 bal_balancing.deltaCharge_mAs[s][m][cb] = 0u;
150 } else {
151 bal_balancing.deltaCharge_mAs[s][m][cb] -= difference;
152 }
153 } else {
154 bal_balancing.activateBalancing[s][m][cb] = false;
155 }
156 }
157 }
158 }
159 bal_balancing.nrBalancedCells[s] = nrBalancedCells;
160 }
161
162 (void)DATA_WRITE_DATA(&bal_balancing);
163}

181static void BAL_ProcessStateCheckBalancing(void) {
182 if (bal_state.substate == BAL_ENTRY) {
183 if (bal_state.balancingGlobalAllowed == false) {
184 if (bal_state.active == true) {
185 BAL_Deactivate();
186 }
187 bal_state.active = false;
188 bal_state.substate = BAL_ENTRY;
189 } else {
190 bal_state.substate = BAL_CHECK_IMBALANCES;
191 }
192
193 bal_state.timer = BAL_STATEMACH_SHORTTIME_100ms;
194 return;
195 } else if (bal_state.substate == BAL_CHECK_IMBALANCES) {
196 if (bal_state.active == true) {
197 BAL_Deactivate();
198 }
199 if (BAL_CheckImbalances() == true) {
200 bal_state.state = BAL_STATEMACH_BALANCE;
201 bal_state.substate = BAL_ENTRY;
202 } else {
203 bal_state.substate = BAL_COMPUTE_IMBALANCES;
204 }
205 bal_state.timer = BAL_STATEMACH_SHORTTIME_100ms;
206 return;
207 } else if (bal_state.substate == BAL_COMPUTE_IMBALANCES) {
208 if (BMS_GetBatterySystemState() == BMS_AT_REST) {
209 BAL_ComputeImbalances();
210 bal_state.state = BAL_STATEMACH_BALANCE;
211 bal_state.substate = BAL_ENTRY;
212 } else {
213 bal_state.substate = BAL_CHECK_IMBALANCES;
214 }
215 bal_state.timer = BAL_STATEMACH_SHORTTIME_100ms;
216 return;
217 }
218}

220static void BAL_ProcessStateBalancing(void) {
221 bool activateBalancing = true;
222
223 if (bal_state.substate == BAL_ENTRY) {
224 if (bal_state.balancingGlobalAllowed == false) {
225 if (bal_state.active == true) {
226 BAL_Deactivate();
227 }
228 bal_state.active = false;
229 bal_state.substate = (BAL_STATEMACH_SUB_e)BAL_STATEMACH_CHECK_BALANCING;
230 } else {
231 bal_state.substate = BAL_ACTIVATE_BALANCING;
232 }
233 bal_state.timer = BAL_STATEMACH_SHORTTIME_100ms;
234 return;
235 } else if (bal_state.substate == BAL_ACTIVATE_BALANCING) {
236 DATA_BLOCK_MIN_MAX_s bal_minMax = {.header.uniqueId = DATA_BLOCK_ID_MIN_MAX};
237 (void)DATA_READ_DATA(&bal_minMax);
238 bal_state.timer = BAL_STATEMACH_BALANCING_TIME_100ms;
239 /* do not balance under a certain voltage level */
240 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
241 if ((bal_minMax.minimumCellVoltage_mV[s] <= BAL_LOWER_VOLTAGE_LIMIT_mV) ||
242 (bal_minMax.maximumTemperature_ddegC[s] >= BAL_UPPER_TEMPERATURE_LIMIT_ddegC) ||
243 (BAL_CheckImbalances() == false) || (bal_state.balancingGlobalAllowed == false)) {
244 activateBalancing = false;
245 if (bal_state.active == true) {
246 BAL_Deactivate();
247 }
248 bal_state.state = BAL_STATEMACH_CHECK_BALANCING;
249 bal_state.substate = BAL_ENTRY;
250 return;
251 }
252 }
253
254 if (activateBalancing == true) {
255 BAL_ActivateBalancing();
256 }
257 return;
258 }

261 bool returnValue = false;
262
263 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
264 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
265 for (uint16_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
266 if (bal_balancing.deltaCharge_mAs[s][m][cb] > 0) {
267 returnValue = true;
268 }
269 }
270 }
271 }
272 return returnValue;

275static void BAL_ComputeImbalances(void) {
276 int16_t voltageMin_mV = 0;
277 uint16_t minVoltageModuleIndex = 0u;
278 uint16_t minVoltageModuleCellBlockIndex = 0u;
279 float_t SOC = 0.0f;
280 uint32_t DOD = 0u;
281 uint32_t maxDOD = 0u;
282
283 (void)DATA_READ_DATA(&bal_balancing, &bal_cellVoltage);
284
285 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
286 /* Assign first cell voltage to*/
287 voltageMin_mV = INT16_MAX;
288 minVoltageModuleIndex = 0u;
289 minVoltageModuleCellBlockIndex = 0u;
290 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
291 for (uint16_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
292 if (bal_cellVoltage.cellVoltage_mV[s][m][cb] <= voltageMin_mV) {
293 voltageMin_mV = bal_cellVoltage.cellVoltage_mV[s][m][cb];
294 minVoltageModuleIndex = m;
295 minVoltageModuleCellBlockIndex = cb;
296 }
297 }
298 }
299
300 SOC = SE_GetStateOfChargeFromVoltage(
301 ((float_t)(bal_cellVoltage.cellVoltage_mV[s][minVoltageModuleIndex][minVoltageModuleCellBlockIndex])) /
302 1000.0f);
303 maxDOD = BC_CAPACITY_mAh * (uint32_t)((1.0f - SOC) * 3600.0f);
304 bal_balancing.deltaCharge_mAs[s][minVoltageModuleIndex][minVoltageModuleCellBlockIndex] = 0u;
305
306 /* update balancing threshold */
307 bal_state.balancingThreshold = BAL_GetBalancingThreshold_mV() + BAL_HYSTERESIS_mV;
308
309 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
310 for (uint16_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
311 if ((m != minVoltageModuleIndex) || (cb != minVoltageModuleCellBlockIndex)) {
312 if (bal_cellVoltage.cellVoltage_mV[s][m][cb] >= (voltageMin_mV + bal_state.balancingThreshold)) {
313 SOC = SE_GetStateOfChargeFromVoltage(
314 ((float_t)(bal_cellVoltage.cellVoltage_mV[s][m][cb])) / 1000.0f);
315 DOD = BC_CAPACITY_mAh * (uint32_t)((1.0f - SOC) * 3600.0f);
316 bal_balancing.deltaCharge_mAs[s][m][cb] = (maxDOD - DOD);
317 }
318 }
319 }
320 }
321 }
322
323 (void)DATA_WRITE_DATA(&bal_balancing);
324}

345extern void BAL_Trigger(void) {
346 BAL_STATE_REQUEST_e stateRequest = BAL_STATE_NO_REQUEST;
347
348 /* Check re-entrance of function */
349 if (BAL_CheckReEntrance(&bal_state) > 0u) {
350 return;
351 }
352
353 if (bal_state.timer > 0u) {
354 if ((--bal_state.timer) > 0) {
355 bal_state.triggerEntry--;
356 return; /* handle state machine only if timer has elapsed */
357 }
358 }
359
360 switch (bal_state.state) {
361 case BAL_STATEMACH_UNINITIALIZED:
362 BAL_SaveLastStates(&bal_state);
363 stateRequest = BAL_TransferStateRequest(&bal_state);
364 BAL_ProcessStateUninitialized(&bal_state, stateRequest);
365 break;
366 case BAL_STATEMACH_INITIALIZATION:
367 BAL_SaveLastStates(&bal_state);
368 BAL_Init(&bal_balancing);
369 BAL_ProcessStateInitialization(&bal_state);
370 break;
371 case BAL_STATEMACH_INITIALIZED:
372 BAL_SaveLastStates(&bal_state);
373 BAL_ProcessStateInitialized(&bal_state);
374 break;
375 case BAL_STATEMACH_CHECK_BALANCING:
376 BAL_SaveLastStates(&bal_state);
377 BAL_ProcessStateCheckBalancing();
378 break;
379 case BAL_STATEMACH_BALANCE:
380 BAL_SaveLastStates(&bal_state);
381 BAL_ProcessStateBalancing();
382 break;
383 default:
384 /* invalid state */
385 FAS_ASSERT(FAS_TRAP);
386 break;
387 }
388 bal_state.triggerEntry--;
389}
