182/**
183 * @brief Maximum break current of main contactors.
184 * @details The contacts of the main contactors can be welded, when attempting
185 * to interrupt the current flow while a current higher than the
186 * maximum break current of the contactor is flowing.
187 *
188 * Thus, the contactors will not be opened if the floating current is
189 * above this value. The fuse should trigger to interrupt current
190 * flows above this value.
191 */
192#define BS_MAIN_CONTACTORS_MAXIMUM_BREAK_CURRENT_mA (3500)
193
194/**
195 * @brief Maximum fuse trigger duration
196 * @details If the current is above #BS_MAIN_CONTACTORS_MAXIMUM_BREAK_CURRENT_mA,
197 * the BMS state machine will wait this time until the fuse triggers so
198 * that the current will be interrupted by the fuse and not the
199 * contactors. After this time, the BMS will nevertheless try to open
200 * the contactors.
201 */
202#define BS_MAIN_FUSE_MAXIMUM_TRIGGER_DURATION_ms (3000u)
203
204/**
205 * @brief Maximum string current limit in mA that is used in the SOA module
206 * to check for string overcurrent
207 * @details When maximum safety limit (MSL) is violated, error state is
208 * requested and contactors will open.
209 */
210#define BS_MAXIMUM_STRING_CURRENT_mA (2400u)

212/**

276/**
277 * @brief Maximum voltage drop over fuse.
278 * @details If the measured voltage difference between battery voltage
279 * voltage after fuse is larger than this value. It can be concluded,
280 * that the fuse has tripped. The voltage difference can be estimated
281 * by the maximum current and the resistance.
282 * For a Cooper Bussmann 1000A fuse the voltage drop can be estimated
283 * to: I_max = 1000A and P_loss = 206W
284 * -> voltage drop at 1000A roughly 206mV
285 * -> select 500mV because of measurement inaccuracies
286 */
287#define BS_MAX_VOLTAGE_DROP_OVER_FUSE_mV (500)
