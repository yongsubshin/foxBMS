114static IMD_FSM_STATES_e IR155_MeasureInsulation(DATA_BLOCK_INSULATION_MONITORING_s *pTableInsulationMonitoring) {
115 FAS_ASSERT(pTableInsulationMonitoring != NULL_PTR);
116 pTableInsulationMonitoring->isInsulationMeasurementValid = true;
117 pTableInsulationMonitoring->insulationResistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
118 pTableInsulationMonitoring->areDeviceFlagsValid = true;
119 pTableInsulationMonitoring->dfIsCriticalResistanceDetected = false;
120 pTableInsulationMonitoring->dfIsChassisFaultDetected = false;
121 pTableInsulationMonitoring->dfIsDeviceErrorDetected = false;
122 /* Bender IR155 does not support a digital warning threshold -> check measured insulation resistance */
123 pTableInsulationMonitoring->dfIsWarnableResistanceDetected = false;
124
125 /* Evaluate input pwm signal */
126 ir155_state.measurement = IR155_GetMeasurementValues();
127
128 /* Measurement is not valid, either because of startup delay or detected ground error before startup */
129 if (ir155_state.timeUntilValidMeasurement_ms > 0u) {
130 if (ir155_state.timeUntilValidMeasurement_ms >= ir155_state.periodTriggerTime_ms) {
131 ir155_state.timeUntilValidMeasurement_ms -= ir155_state.periodTriggerTime_ms;
132 ir155_state.measurement.measurementState = IR155_MEASUREMENT_NOT_VALID;
133 } else {
134 /* 0 < hysteresisCounter < measCycleTime
135 * Measurement is valid from that moment on */
136 ir155_state.timeUntilValidMeasurement_ms = 0u;
137 }
138
139 if (ir155_state.timeUntilValidMeasurement_ms == 0u) {
140 /* If hysteresis is over, reset diag flag and reset ground error flag */
141 /* TODO: How to handle the ground error fault after reset? Do this in diag callback?*/
142 /* Reset non-volatile ground error flag - Do this in diag callback? */
143 fram_insulationFlags.groundErrorDetected = false;
144 }
145 }
146
147 /* Evaluate if resistance measurement is valid or not */
148 if (ir155_state.measurement.measurementState == IR155_MEASUREMENT_NOT_VALID) {
149 /* Measurement result is not valid */
150 pTableInsulationMonitoring->isInsulationMeasurementValid = false;
151 } else {
152 pTableInsulationMonitoring->isInsulationMeasurementValid = true;
153 }
154
155 /* Set measured resistance value */
156 pTableInsulationMonitoring->insulationResistance_kOhm = ir155_state.measurement.resistance_kOhm;
157
158 /* Evaluate device state */
159 if ((ir155_state.measurement.measurementState == IR155_IMD_ERROR_MEASUREMENT) ||
160 (ir155_state.measurement.measurementState == IR155_IMD_ERROR_MEASUREMENT_UNKNOWN)) {
161 /* Mark device flags and insulation measurement as invalid if a device error is detected */
162 pTableInsulationMonitoring->areDeviceFlagsValid = false;
163 pTableInsulationMonitoring->isInsulationMeasurementValid = false;
164 } else {
165 pTableInsulationMonitoring->areDeviceFlagsValid = true;
166 }
167
168 /* Set error flag if Pin OHKS detects an error
169 * Pin state high: No fault; Insulation resistance > response value
170 * Pin state low: Insulation resistance <= response value detected;
171 * Device error; Fault in the earth connection
172 * Undervoltage detected or device switched off
173 */
174 if (ir155_state.measurement.digitalStatusPin == STD_PIN_LOW) {
175 pTableInsulationMonitoring->dfIsCriticalResistanceDetected = true;
176 } else {
177 pTableInsulationMonitoring->dfIsCriticalResistanceDetected = false;
178 }
179
180 /* Check for device error: invalid measurement period or duty-cycle detected */
181 if ((ir155_state.measurement.measurementState == IR155_RESISTANCE_MEASUREMENT_UNKNOWN) ||
182 (ir155_state.measurement.measurementState == IR155_RESISTANCE_ESTIMATION_UNKNOWN) ||
183 (ir155_state.measurement.measurementState == IR155_UNDERVOLTAGE_MEASUREMENT_UNKNOWN) ||
184 (ir155_state.measurement.measurementState == IR155_GROUND_ERROR_STATE_UNKNOWN)) {
185 pTableInsulationMonitoring->dfIsDeviceErrorDetected = true;
186 } else {
187 pTableInsulationMonitoring->dfIsDeviceErrorDetected = false;
188 }
189
190 /* Check for chassis fault */
191 pTableInsulationMonitoring->dfIsChassisShortToHvPlus = false; /* This feature is not supported by the device */
192 pTableInsulationMonitoring->dfIsChassisShortToHvMinus = false; /* This feature is not supported by the device */
193 if (ir155_state.measurement.measurementState == IR155_GROUND_ERROR_STATE) {
194 pTableInsulationMonitoring->dfIsChassisFaultDetected = true;
195 } else {
196 pTableInsulationMonitoring->dfIsDeviceErrorDetected = false;
197 }
198
199 /* No digital signal for warning threshold available for this device -> use measured insulation resistance */
200 if (pTableInsulationMonitoring->insulationResistance_kOhm <= IMD_WARNING_THRESHOLD_INSULATION_RESISTANCE_kOhm) {
201 pTableInsulationMonitoring->dfIsWarnableResistanceDetected = true;
202 } else {
203 pTableInsulationMonitoring->dfIsWarnableResistanceDetected = false;
204 }
205
206 /* TODO: bool dfIsMeasurementUpToDate; Check if measurement result is up to data */
207 return IMD_FSM_STATE_RUNNING;
208}

215extern IMD_FSM_STATES_e IMD_ProcessEnableState(void) {
216 /* Enable supply for Bender */
217 IO_PinSet(&IR155_SUPPLY_ENABLE_PORT->DOUT, IR155_SUPPLY_ENABLE_PIN);
218 return IMD_FSM_STATE_RUNNING;

221extern IMD_FSM_STATES_e IMD_ProcessRunningState(DATA_BLOCK_INSULATION_MONITORING_s *pTableInsulationMonitoring) {
222 FAS_ASSERT(pTableInsulationMonitoring != NULL_PTR);
223 return IR155_MeasureInsulation(pTableInsulationMonitoring);
224}
225
226extern IMD_FSM_STATES_e IMD_ProcessShutdownState(void) {
227 /* Disable supply for Bender */
228 IO_PinReset(&IR155_SUPPLY_ENABLE_PORT->DOUT, IR155_SUPPLY_ENABLE_PIN);
229 return IMD_FSM_STATE_IMD_ENABLE;
230}
