67/** Maximum measurable resistance according to formula:
68 * ((90 * 1200kOhm) / (dc - 5) ) - 1200kOhm
69 * with dc ]5%,95%[ */
70#define IR155_MAXIMUM_INSULATION_RESISTANCE_kOhm (106800u)
71#define IR155_MINIMUM_INSULATION_RESISTANCE_kOhm (0u)

75#define IR155_NORMAL_MODE_UPPER_DUTY_CYCLE_LIMIT_perc (96.0f)

141/*========== Static Function Implementations ================================*/
142static IR155_MEASUREMENT_MODE_e IR155_GetMeasurementMode(float_t frequency_Hz) {
143 FAS_ASSERT(frequency_Hz >= 0.0f);
144 IR155_MEASUREMENT_MODE_e retVal = IR155_UNKNOWN;
145
146 if ((frequency_Hz >= IR155_NORMAL_CONDITION_LOWER_FREQUENCY_Hz) &&
147 (frequency_Hz < IR155_NORMAL_CONDITION_UPPER_FREQUENCY_Hz)) {
148 retVal = IR155_NORMAL_MODE;
149 } else if (
150 (frequency_Hz >= IR155_UNDERVOLTAGE_LOWER_FREQUENCY_Hz) &&
151 (frequency_Hz < IR155_UNDERVOLTAGE_UPPER_FREQUENCY_Hz)) {
152 retVal = IR155_UNDERVOLTAGE_MODE; /* should not be detected as default threshold 0V, EOL Bender configurable! */
153 } else if (
154 (frequency_Hz >= IR155_SPEED_START_LOWER_FREQUENCY_Hz) &&
155 (frequency_Hz < IR155_SPEED_START_UPPER_FREQUENCY_Hz)) {
156 retVal = IR155_SPEED_START_MODE;
157 } else if (
158 (frequency_Hz >= IR155_IMD_DEVICE_ERROR_LOWER_FREQUENCY_Hz) &&
159 (frequency_Hz < IR155_IMD_DEVICE_ERROR_UPPER_FREQUENCY_Hz)) {
160 retVal = IR155_IMD_ERROR_MODE;
161 } else if (
162 (frequency_Hz >= IR155_GROUND_ERROR_LOWER_FREQUENCY_Hz) &&
163 (frequency_Hz < IR155_GROUND_ERROR_UPPER_FREQUENCY_Hz)) {
164 retVal = IR155_GROUND_ERROR_MODE;
165 } else if (frequency_Hz <= IR155_MINIMUM_FREQUENCY_Hz) {
166 retVal = IR155_SHORT_CLAMP;
167 } else {
168 retVal = IR155_UNDEFINED_FREQUENCY;
169 }
170 return retVal;
171}

173static uint32_t IR155_CalculateResistance(float_t dutyCycle_perc) {
174 FAS_ASSERT(dutyCycle_perc <= 100.0f);
175 FAS_ASSERT(dutyCycle_perc > 0.0f);
176
177 float_t insulationResistance_kOhm = 0.0f;
178 if (dutyCycle_perc <= 5.0f) {
179 insulationResistance_kOhm = (float_t)IR155_MAXIMUM_INSULATION_RESISTANCE_kOhm;
180 } else if (dutyCycle_perc > 95.0f) {
181 insulationResistance_kOhm = (float_t)IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
182 } else {
183 insulationResistance_kOhm = ((90.0f * 1200.0f) / (dutyCycle_perc - 5.0f)) - 1200.0f;
184 }
185 return (uint32_t)insulationResistance_kOhm;
186}

188static bool IR155_IsDutyCycleWithinInterval(float_t dutyCycle_perc, float_t lowerLimit_perc, float_t upperLimit_perc) {
189 bool retval = false;
190 if ((lowerLimit_perc < dutyCycle_perc) && (upperLimit_perc > dutyCycle_perc)) {
191 retval = true;
192 } else {
193 retval = false;
194 }
195 return retval;
196}

200void IR155_Initialize(uint8_t triggerTime_ms) {
201 /* Timer peripheral initialization if not already done. */
202 if (false == PWM_IsEcapModuleInitialized()) {
203 PWM_Initialize(); /* TODO: split PWM_Init function in a dedicated function for ecap and etpwm module */
204 }
205
206 /* Read non-volatile FRAM */
207 (void)FRAM_ReadData(FRAM_BLOCK_ID_INSULATION_FLAG);
208
209 /* Check ground error flag */
210 if (fram_insulationFlags.groundErrorDetected == true) {
211 /* GROUND ERROR occurred before shutting down */
212 ir155_state.timeUntilValidMeasurement_ms = IR155_WAIT_TIME_AFTER_GROUND_ERROR_ms;
213 } else {
214 /* Normal startup delay -> wait 2.2s until first measurement is trustworthy */
215 ir155_state.timeUntilValidMeasurement_ms = IR155_STARTUP_TIME_ms;
216 }
217
218 ir155_state.periodTriggerTime_ms = triggerTime_ms;
219 ir155_state.ir155Initialized = true;
220}

236IR155_MEASUREMENT_s IR155_GetMeasurementValues(void) {
237 /* Initialize struct */
238 IR155_MEASUREMENT_s measurementResult = {
239 .isMeasurementValid = false,
240 .isUndervoltageDetected = true,
241 .pwmSignal.dutyCycle_perc = 0.0f,
242 .pwmSignal.frequency_Hz = 0.0f,
243 .resistance_kOhm = 0u,
244 .digitalStatusPin = STD_PIN_UNDEFINED,
245 .measurementMode = IR155_UNKNOWN,
246 .measurementState = IR155_UNINITIALIZED,
247 };
248
249 /* read value of Bender IR155 digital status pin (OHKS) */
250 measurementResult.digitalStatusPin = IR155_GET_DIGITAL_STATUS_PIN_STATE();
251
252 /* get duty-cycle and frequency from PWM input measurement */
253 measurementResult.pwmSignal = PWM_GetPwmData();
254
255 /* TODO: How-to check valid data? */
256 bool isPwmMeasurementValid = true;
257 if (isPwmMeasurementValid == true) {
258 /* TODO: Throw error? */
259 isPwmMeasurementValid = true;
260 } else {
261 /* Invalid values measurement */
262
263 /* Invalidate data? */
264 isPwmMeasurementValid = false;
265 }
266
267 /* Get measurement mode */
268 if (isPwmMeasurementValid == true) {
269 measurementResult.measurementMode = IR155_GetMeasurementMode(measurementResult.pwmSignal.frequency_Hz);
270 } else {
271 measurementResult.measurementMode = IR155_UNKNOWN;
272 /* TODO: Throw what error? */
273 }
274
275 switch (measurementResult.measurementMode) {
276 case IR155_NORMAL_MODE:
277 ir155_state.measurement.isUndervoltageDetected = false;
278 if (true == IR155_IsDutyCycleWithinInterval(
279 measurementResult.pwmSignal.dutyCycle_perc,
280 IR155_NORMAL_MODE_LOWER_DUTY_CYCLE_LIMIT_perc,
281 IR155_NORMAL_MODE_UPPER_DUTY_CYCLE_LIMIT_perc)) {
282 measurementResult.resistance_kOhm =
283 IR155_CalculateResistance(measurementResult.pwmSignal.dutyCycle_perc);
284 measurementResult.isMeasurementValid = true;
285 measurementResult.measurementState = IR155_RESISTANCE_MEASUREMENT;
286 } else {
287 measurementResult.measurementState = IR155_RESISTANCE_MEASUREMENT_UNKNOWN;
288 measurementResult.isMeasurementValid = false;
289 measurementResult.resistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
290 }
291 break;
292
293 case IR155_SPEED_START_MODE:
294 measurementResult.isUndervoltageDetected = false;
295 if (true == IR155_IsDutyCycleWithinInterval(
296 measurementResult.pwmSignal.dutyCycle_perc,
297 IR155_SPEED_START_ESTIMATION_GOOD_LOWER_DUTY_CYCLE_LIMIT_perc,
298 IR155_SPEED_START_ESTIMATION_GOOD_UPPER_DUTY_CYCLE_LIMIT_perc)) {
299 measurementResult.resistance_kOhm = IR155_MAXIMUM_INSULATION_RESISTANCE_kOhm;
300 measurementResult.isMeasurementValid = true;
301 measurementResult.measurementState = IR155_RESISTANCE_ESTIMATION;
302 } else if (
303 true == IR155_IsDutyCycleWithinInterval(
304 measurementResult.pwmSignal.dutyCycle_perc,
305 IR155_SPEED_START_ESTIMATION_BAD_LOWER_DUTY_CYCLE_LIMIT_perc,
306 IR155_SPEED_START_ESTIMATION_BAD_UPPER_DUTY_CYCLE_LIMIT_perc)) {
307 measurementResult.resistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
308 measurementResult.isMeasurementValid = true;
309 measurementResult.measurementState = IR155_RESISTANCE_ESTIMATION;
310 } else {
311 measurementResult.resistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
312 measurementResult.isMeasurementValid = false;
313 measurementResult.measurementState = IR155_RESISTANCE_ESTIMATION_UNKNOWN;
314 }
315 break;
316
317 case IR155_UNDERVOLTAGE_MODE:
318 measurementResult.isUndervoltageDetected = true;
319 if (true == IR155_IsDutyCycleWithinInterval(
320 measurementResult.pwmSignal.dutyCycle_perc,
321 IR155_UNDERVOLTAGE_MODE_LOWER_DUTY_CYCLE_LIMIT_perc,
322 IR155_UNDERVOLTAGE_MODE_UPPER_DUTY_CYCLE_LIMIT_perc)) {
323 measurementResult.resistance_kOhm =
324 IR155_CalculateResistance(measurementResult.pwmSignal.dutyCycle_perc);
325 measurementResult.isMeasurementValid = true;
326 measurementResult.measurementState = IR155_UNDERVOLTAGE_MEASUREMENT;
327 } else {
328 measurementResult.isMeasurementValid = false;
329 measurementResult.resistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
330 measurementResult.measurementState = IR155_UNDERVOLTAGE_MEASUREMENT_UNKNOWN;
331 }
332 break;
333
334 case IR155_IMD_ERROR_MODE:
335 measurementResult.isUndervoltageDetected = false;
336 measurementResult.resistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
337 if (true == IR155_IsDutyCycleWithinInterval(
338 measurementResult.pwmSignal.dutyCycle_perc,
339 IR155_DEVICE_ERROR_LOWER_DUTY_CYCLE_LIMIT_perc,
340 IR155_DEVICE_ERROR_UPPER_DUTY_CYCLE_LIMIT_perc)) {
341 /* Error detected and verified with duty cycle */
342 measurementResult.isMeasurementValid = true;
343 measurementResult.measurementState = IR155_IMD_ERROR_MEASUREMENT;
344 } else {
345 /* Error detected but invalid duty cycle */
346 measurementResult.measurementState = IR155_IMD_ERROR_MEASUREMENT_UNKNOWN;
347 measurementResult.isMeasurementValid = false;
348 }
349 break;
350 case IR155_GROUND_ERROR_MODE:
351 measurementResult.isUndervoltageDetected = false;
352 measurementResult.resistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
353 if (true == IR155_IsDutyCycleWithinInterval(
354 measurementResult.pwmSignal.dutyCycle_perc,
355 IR155_GROUND_ERROR_LOWER_DUTY_CYCLE_LIMIT_perc,
356 IR155_GROUND_ERROR_UPPER_DUTY_CYCLE_LIMIT_perc)) {
357 /* Error detected and verified with duty cycle */
358 measurementResult.isMeasurementValid = true;
359 measurementResult.measurementState = IR155_GROUND_ERROR_STATE;
360 } else {
361 /* Error detected but invalid duty cycle */
362 measurementResult.measurementState = IR155_GROUND_ERROR_STATE_UNKNOWN;
363 measurementResult.isMeasurementValid = false;
364 }
365 break;
366
367 case IR155_SHORT_CLAMP:
368 measurementResult.isUndervoltageDetected = false;
369 measurementResult.resistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
370 measurementResult.measurementState = IR155_SIGNAL_SHORT;
371 measurementResult.isMeasurementValid = true;
372 break;
373
374 default:
375 measurementResult.resistance_kOhm = IR155_MINIMUM_INSULATION_RESISTANCE_kOhm;
376 measurementResult.measurementState = IR155_MEASUREMENT_NOT_VALID;
377 measurementResult.isMeasurementValid = false;
378 break;
379 }
380
381 return measurementResult;
382}
