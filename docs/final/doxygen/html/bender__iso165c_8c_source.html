431static I165C_RESPONSE_RETURN_VALUE_e I165C_CheckResponse(uint8_t command, CAN_BUFFER_ELEMENT_s *pCanMessage) {
432 FAS_ASSERT(pCanMessage != NULL_PTR);
433 I165C_RESPONSE_RETURN_VALUE_e messageReceived = I165C_RESPONSE_NO_RESPONSE;
434 uint8_t numberItems = 0u;
435 uint8_t queueReadTries = I165C_MAX_QUEUE_READS;
436
437 /* Use loop on queue because IMD_info message could come meanwhile */
438 do {
439 numberItems = OS_GetNumberOfStoredMessagesInQueue(ftsk_imdCanDataQueue);
440 if (numberItems > 0u) {
441 if (OS_ReceiveFromQueue(ftsk_imdCanDataQueue, (void *)pCanMessage, 0u) == OS_SUCCESS) {
442 /* data queue was not empty */
443 if ((command == pCanMessage->data[CAN_BYTE_0_POSITION]) && (pCanMessage->id == CANRX_IMD_RESPONSE_ID) &&
444 (pCanMessage->idType == CANRX_IMD_RESPONSE_ID_TYPE)) {
445 /* Response with expected multiplexer was received */
446 messageReceived = I165C_RESPONSE_SUCCESS;
447 break;
448 } else {
449 if ((pCanMessage->data[CAN_BYTE_0_POSITION] == 0xFF) &&
450 (pCanMessage->id == CANRX_IMD_RESPONSE_ID) &&
451 (pCanMessage->idType == CANRX_IMD_RESPONSE_ID_TYPE)) {
452 /* Response with error multiplexer was received */
453 messageReceived = I165C_RESPONSE_ERROR;
454 }
455 }
456 }
457 }
458 queueReadTries--;
459 } while ((numberItems > 0u) && (queueReadTries > 0u));
460
461 return messageReceived;
462}

464static bool I165C_GetImdInfo(CAN_BUFFER_ELEMENT_s *pCanMessage) {
465 FAS_ASSERT(pCanMessage != NULL_PTR);
466 bool imdInfoReceived = false;
467 uint8_t numberItems = 0u;
468 uint8_t queueReadTries = I165C_MAX_QUEUE_READS;
469
470 /* Use loop on queue because other messages could come meanwhile */
471 do {
472 numberItems = OS_GetNumberOfStoredMessagesInQueue(ftsk_imdCanDataQueue);
473 if (numberItems > 0u) {
474 if (OS_ReceiveFromQueue(ftsk_imdCanDataQueue, (void *)pCanMessage, 0u) == OS_SUCCESS) {
475 /* data queue was not empty */
476 if (pCanMessage->id == I165C_MESSAGE_TYPE_IMD_INFO) {
477 imdInfoReceived = true;
478 break;
479 }
480 }
481 }
482 queueReadTries--;
483 } while ((numberItems > 0u) && (queueReadTries > 0u));
484
485 return imdInfoReceived;
486}

488static void I165C_SetRelayState(uint8_t relay, uint8_t relayState) {
489 FAS_ASSERT((relay == I165C_D_VIFC_HV_RELAIS_NEGATIVE) || (relay == I165C_D_VIFC_HV_RELAIS_POSITIVE));
490 FAS_ASSERT((relayState == I165C_RELAY_STATE_OPEN) || (relayState == I165C_RELAY_STATE_CLOSED));
491
492 if (relay == I165C_D_VIFC_HV_RELAIS_NEGATIVE) {
493 if (relayState == I165C_RELAY_STATE_OPEN) {
494 CANTX_ImdRequest(CANTX_IMD_REQUEST_OPEN_NEGATIVE_RELAY);
495 } else {
496 CANTX_ImdRequest(CANTX_IMD_REQUEST_CLOSE_NEGATIVE_RELAY);
497 }
498 } else {
499 if (relayState == I165C_RELAY_STATE_OPEN) {
500 CANTX_ImdRequest(CANTX_IMD_REQUEST_OPEN_POSITIVE_RELAY);
501 } else {
502 CANTX_ImdRequest(CANTX_IMD_REQUEST_CLOSE_POSITIVE_RELAY);
503 }
504 }
505}

507static void I165C_RequestRelayState(uint8_t relay) {
508 FAS_ASSERT((relay == I165C_D_VIFC_HV_RELAIS_NEGATIVE) || (relay == I165C_D_VIFC_HV_RELAIS_POSITIVE));
509
510 switch (relay) {
511 case I165C_D_VIFC_HV_RELAIS_NEGATIVE:
512 CANTX_ImdRequest(CANTX_IMD_REQUEST_NEGATIVE_RELAY_STATE);
513 break;
514 case I165C_D_VIFC_HV_RELAIS_POSITIVE:
515 CANTX_ImdRequest(CANTX_IMD_REQUEST_POSITIVE_RELAY_STATE);
516 break;
517 default:
518 FAS_ASSERT(FAS_TRAP);
519 break;
520 }
521}

544static I165C_RESPONSE_RETURN_VALUE_e I165C_CheckAcknowledgeArrived(
545 uint8_t command,
546 uint8_t *pTries,
547 CAN_BUFFER_ELEMENT_s *pCanMessage) {
548 FAS_ASSERT(pTries != NULL_PTR);
549 FAS_ASSERT(pCanMessage != NULL_PTR);
550 /* AXIVION Routine Generic-MissingParameterAssert: command: parameter accepts whole range */
551
552 I165C_RESPONSE_RETURN_VALUE_e acknowledgeReceived = I165C_CheckResponse(command, pCanMessage);
553 if (acknowledgeReceived == I165C_RESPONSE_NO_RESPONSE) {
554 (*pTries)++;
555 } else {

1040 I165C_SetEnableState(&i165c_enableState, i165c_enableState.previousState, I165C_FSM_SHORT_TIME);
1041 }
1042 /* Issue: 621 */
1043 }
1044 break;
1045 }
1046 }
1047 return nextState;
1048}
