119/**
120 * frames have up to two data words
121 * (1 data word length = 16 bits)
122 * or up to 4 data bytes
123 * (1 data byte length = 8 bits)
124 *
125 * CAUTION: data word order is reversed !!!!!!!!!
126 * (see data sheet page 15)
127 */

129/** data word 1 is in first position */
130#define I165C_DW1 (0u)
131/** data word 2 is in second position */
132#define I165C_DW2 (1u)
133/** data word 3 is in third position */
134#define I165C_DW3 (2u)
135/** data byte 1 is in first position */
136#define I165C_DB1 (3u)
137/** data byte 2 is in second position */
138#define I165C_DB2 (4u)
139/** data byte 3 is in third position */
140#define I165C_DB3 (5u)
141/** data byte 4 is in fourth position */
142#define I165C_DB4 (6u)
