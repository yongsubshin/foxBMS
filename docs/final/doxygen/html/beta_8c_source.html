73/** Defines for calculating the ADC voltage on the ends of the operating range.
74 * The ADC voltage is calculated with the following formula:
75 *
76 * V_adc = ( ( V_supply * R_ntc ) / ( R + R_ntc ) )
77 *
78 * Depending on the position of the NTC in the voltage resistor (R_1/R_2),
79 * different R_ntc values are used for the calculation.
80 */
81/**@{*/
82#if defined(BETA_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && (BETA_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
83#define BETA_ADC_VOLTAGE_V_MAX_V \
84 (float_t)( \
85 (BETA_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * BETA_ResistanceFromTemperature(1400)) / \
86 (BETA_ResistanceFromTemperature(1400) + BETA_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm))
87#define BETA_ADC_VOLTAGE_V_MIN_V \
88 (float_t)( \
89 (BETA_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * BETA_ResistanceFromTemperature(-400)) / \
90 (BETA_ResistanceFromTemperature(-400) + BETA_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm))
91#else /* BETA_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == false */
92#define BETA_ADC_VOLTAGE_V_MIN_V \
93 ((float_t)((BETA_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * BETA_ResistanceFromTemperature(1400)) / \
94 (BETA_ResistanceFromTemperature(1400) + BETA_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))
95#define BETA_ADC_VOLTAGE_V_MAX_V \
96 ((float_t)((BETA_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * BETA_ResistanceFromTemperature(-400)) / \
97 (BETA_ResistanceFromTemperature(-400) + BETA_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))
98#endif
99/**@}*/

107extern int16_t BETA_GetTemperatureFromBeta(uint16_t adcVoltage_mV) {
108 int16_t temperature_ddegC = 0;
109 float_t resistance_Ohm = 0.0f;
110 float_t adcVoltage_V = (float_t)adcVoltage_mV / 1000.0f; /* Convert mV to V */
111
112 /* Check for valid ADC measurements to prevent undefined behavior */
113 if (adcVoltage_V > BETA_ADC_VOLTAGE_V_MAX_V) {
114 /* Invalid measured ADC voltage -> sensor out of operating range or disconnected/shorted */
115 temperature_ddegC = INT16_MIN;
116 } else if (adcVoltage_V < BETA_ADC_VOLTAGE_V_MIN_V) {
117 /* Invalid measured ADC voltage -> sensor out of operating range or shorted/disconnected */
118 temperature_ddegC = INT16_MAX;
119 } else {
120 /* Calculate NTC resistance based on measured ADC voltage */
121#if defined(BETA_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && (BETA_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
122 /* R_1 = R_2 * ( ( V_supply / V_adc ) - 1 ) */
123 resistance_Ohm = BETA_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
124 ((BETA_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V / adcVoltage_V) - 1);
125#else /* BETA_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == false */
126 /* R_2 = R_1 * ( V_2 / (V_supply - V_adc ) ) */
127 resistance_Ohm = BETA_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
128 (adcVoltage_V / (BETA_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V - adcVoltage_V));
129#endif /* BETA_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 */
130 /* Use BETA formula to compute temperature with resistance*/
131 temperature_ddegC = BETA_TemperatureFromResistance(resistance_Ohm);
132 }
133
134 /* Return temperature based on measured NTC resistance */
135 return temperature_ddegC;
136}

138extern int16_t BETA_TemperatureFromResistance(float_t resistance_Ohm) {
139 int16_t temperature_ddegC = 0;
140 if (resistance_Ohm > 0.0f) {
141 float_t temperature_degC = (1.0f / ((log(resistance_Ohm / BETA_R_REF_Ohm) / BETA_BETACOEFFICIENT) +
142 (1.0f / (BETA_T_REF_C + BETA_KELVIN)))) -
143 BETA_KELVIN;
144 temperature_ddegC = (int16_t)(10.0f * temperature_degC); /* Convert to deci &deg;C */
145 } else {
146 /* Invalid value if as resistance can not be negative */
147 temperature_ddegC = INT16_MIN;
148 }
149 return temperature_ddegC;
150}
151
152extern float_t BETA_ResistanceFromTemperature(int16_t temperature_ddegC) {
153 float_t resistance_Ohm = 0.0f;
154 resistance_Ohm = BETA_R_REF_Ohm *
155 exp(BETA_BETACOEFFICIENT * ((1.0f / (((float_t)temperature_ddegC / 10.0f) + BETA_KELVIN)) -
156 (1.0f / (BETA_T_REF_C + BETA_KELVIN))));
157 return resistance_Ohm;
158}
