317static BMS_RETURN_TYPE_e BMS_CheckStateRequest(BMS_STATE_REQUEST_e statereq) {
318 if (statereq == BMS_STATE_ERROR_REQUEST) {
319 return BMS_OK;
320 }
321
322 if (bms_state.stateRequest == BMS_STATE_NO_REQUEST) {
323 /* init only allowed from the uninitialized state */
324 if (statereq == BMS_STATE_INIT_REQUEST) {
325 if (bms_state.state == BMS_STATEMACH_UNINITIALIZED) {
326 return BMS_OK;
327 } else {
328 return BMS_ALREADY_INITIALIZED;
329 }
330 } else {
331 return BMS_ILLEGAL_REQUEST;
332 }
333 } else {
334 return BMS_REQUEST_PENDING;
335 }
336}

386 uint8_t openWireDetected = 0;
387

389 /* Iterate over all modules */
390 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
391 /* Iterate over all voltage sense wires: cells per module + 1 */
392 for (uint8_t wire = 0u; wire < (BS_NR_OF_CELL_BLOCKS_PER_MODULE + 1); wire++) {
393 /* open wire detected */
394 if (bms_tableOpenWire.openWire[s][(wire + (m * (BS_NR_OF_CELL_BLOCKS_PER_MODULE + 1))) == 1] > 0u) {
395 openWireDetected++;
396
397 /* Add additional error handling here */
398 }
399 }
400 }
401 /* Set error if open wire detected */
402 if (openWireDetected == 0u) {
403 (void)DIAG_Handler(DIAG_ID_AFE_OPEN_WIRE, DIAG_EVENT_OK, DIAG_STRING, s);
404 } else {
405 (void)DIAG_Handler(DIAG_ID_AFE_OPEN_WIRE, DIAG_EVENT_NOT_OK, DIAG_STRING, s);
406 }
407 }

410static STD_RETURN_TYPE_e BMS_CheckPrecharge(uint8_t stringNumber, const DATA_BLOCK_PACK_VALUES_s *pPackValues) {
411 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
412 /* make sure that we do not access the arrays in the database
413 tables out of bounds */
414 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
415 FAS_ASSERT(pPackValues != NULL_PTR);
416
417 /* Only check precharging if current value and voltages are valid */
418 if ((pPackValues->invalidStringCurrent[stringNumber] == 0u) &&
419 (pPackValues->invalidStringVoltage[stringNumber] == 0u) && (pPackValues->invalidHvBusVoltage == 0u)) {
420 /* Only current not current direction is checked */
421 const int32_t current_mA = MATH_AbsInt32_t(pPackValues->stringCurrent_mA[stringNumber]);
422 const int64_t cont_prechargeVoltDiff_mV = MATH_AbsInt64_t(
423 (int64_t)pPackValues->stringVoltage_mV[stringNumber] - (int64_t)pPackValues->highVoltageBusVoltage_mV);
424
425 /* Check if precharging has been successful */
426 if ((cont_prechargeVoltDiff_mV < BMS_PRECHARGE_VOLTAGE_THRESHOLD_mV) &&
427 (current_mA < BMS_PRECHARGE_CURRENT_THRESHOLD_mA)) {
428 retVal = STD_OK;
429 (void)DIAG_Handler(DIAG_ID_PRECHARGE_ABORT_REASON_VOLTAGE, DIAG_EVENT_OK, DIAG_STRING, stringNumber);
430 (void)DIAG_Handler(DIAG_ID_PRECHARGE_ABORT_REASON_CURRENT, DIAG_EVENT_OK, DIAG_STRING, stringNumber);
431 } else {
432 if (cont_prechargeVoltDiff_mV >= BMS_PRECHARGE_VOLTAGE_THRESHOLD_mV) {
433 /* Voltage difference too large */
434 (void)DIAG_Handler(
435 DIAG_ID_PRECHARGE_ABORT_REASON_VOLTAGE, DIAG_EVENT_NOT_OK, DIAG_STRING, stringNumber);
436 (void)DIAG_Handler(DIAG_ID_PRECHARGE_ABORT_REASON_CURRENT, DIAG_EVENT_OK, DIAG_STRING, stringNumber);
437 }
438 if (current_mA >= BMS_PRECHARGE_CURRENT_THRESHOLD_mA) {
439 /* Current flow too high */
440 (void)DIAG_Handler(
441 DIAG_ID_PRECHARGE_ABORT_REASON_CURRENT, DIAG_EVENT_NOT_OK, DIAG_STRING, stringNumber);
442 (void)DIAG_Handler(DIAG_ID_PRECHARGE_ABORT_REASON_VOLTAGE, DIAG_EVENT_OK, DIAG_STRING, stringNumber);
443 }
444 retVal = STD_NOT_OK;
445 }
446 }
447 return retVal;
448}

450static bool BMS_IsAnyFatalErrorFlagSet(void) {
451 bool fatalErrorActive = false;
452
453 for (uint16_t entry = 0u; entry < diag_device.numberOfFatalErrors; entry++) {
454 const STD_RETURN_TYPE_e diagnosisState =
455 DIAG_GetDiagnosisEntryState(diag_device.pFatalErrorLinkTable[entry]->id);
456 if (STD_NOT_OK == diagnosisState) {
457 /* Fatal error detected -> get delay of this error until contactors shall be opened */
458 const uint32_t kDelay_ms = DIAG_GetDelay(diag_device.pFatalErrorLinkTable[entry]->id);
459 /* Check if delay of detected failure is smaller than the delay of a previously detected failure */
460 if (bms_state.minimumActiveDelay_ms > kDelay_ms) {
461 bms_state.minimumActiveDelay_ms = kDelay_ms;
462 }
463 fatalErrorActive = true;
464 }
465 }
466 return fatalErrorActive;
467}

469static STD_RETURN_TYPE_e BMS_IsBatterySystemStateOkay(void) {
470 STD_RETURN_TYPE_e retVal = STD_OK; /* is set to STD_NOT_OK if error detected */
471 static uint32_t previousTimestamp = 0u;
472 uint32_t timestamp = OS_GetTickCount();
473
474 /* Check if any fatal error is detected */
475 const bool isErrorActive = BMS_IsAnyFatalErrorFlagSet();
476
477 /** Check if a fatal error has been detected previously. If yes, check delay */
478 if (bms_state.transitionToErrorState == true) {
479 /* Decrease active delay since last call */
480 const uint32_t timeSinceLastCall_ms = timestamp - previousTimestamp;
481 if (timeSinceLastCall_ms <= bms_state.remainingDelay_ms) {
482 bms_state.remainingDelay_ms -= timeSinceLastCall_ms;
483 } else {
484 bms_state.remainingDelay_ms = 0u;
485 }
486
487 /* Check if delay from a new error is shorter then active delay from
488 * previously detected error in BMS state machine */
489 if (bms_state.remainingDelay_ms >= bms_state.minimumActiveDelay_ms) {
490 bms_state.remainingDelay_ms = bms_state.minimumActiveDelay_ms;
491 }
492 } else {
493 /* Delay is not active, check if it should be activated */
494 if (isErrorActive == true) {
495 bms_state.transitionToErrorState = true;
496 bms_state.remainingDelay_ms = bms_state.minimumActiveDelay_ms;
497 }
498 }
499
500 /** Set previous timestamp for next call */
501 previousTimestamp = timestamp;
502
503 /* Check if bms state machine should switch to error state. This is the case
504 * if the delay is activated and the remaining delay is down to 0 */
505 if ((bms_state.transitionToErrorState == true) && (bms_state.remainingDelay_ms == 0u)) {
506 retVal = STD_NOT_OK;
507 }
508
509 return retVal;
510}

512static bool BMS_IsContactorFeedbackValid(uint8_t stringNumber, CONT_TYPE_e contactorType) {
513 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
514 FAS_ASSERT(contactorType != CONT_UNDEFINED);
515 bool feedbackValid = false;
516 /* Read latest error flags from database */
517 DATA_BLOCK_ERROR_STATE_s tableErrorFlags = {.header.uniqueId = DATA_BLOCK_ID_ERROR_STATE};
518 (void)DATA_READ_DATA(&tableErrorFlags);
519 /* Check if contactor feedback is valid */
520 switch (contactorType) {
521 case CONT_PLUS:
522 if (tableErrorFlags.contactorInPositivePathOfStringFeedbackError[stringNumber] == false) {
523 feedbackValid = true;
524 }
525 break;
526 case CONT_MINUS:
527 if (tableErrorFlags.contactorInNegativePathOfStringFeedbackError[stringNumber] == false) {
528 feedbackValid = true;
529 }
530 break;
531 case CONT_PRECHARGE:
532 if (tableErrorFlags.prechargeContactorFeedbackError[stringNumber] == false) {
533 feedbackValid = true;
534 }
535 break;
536 default:
537 /* CONT_UNDEFINED already prevent via assert */
538 break;
539 }
540 return feedbackValid;
541}

568static uint8_t BMS_GetClosestString(BMS_CONSIDER_PRECHARGE_e precharge, DATA_BLOCK_PACK_VALUES_s *pPackValues) {
569 FAS_ASSERT(pPackValues != NULL_PTR);
570 uint8_t closestStringIndex = BMS_NO_STRING_AVAILABLE;
571 int32_t closedStringVoltage_mV = 0;
572 bool searchString = false;
573
574 /* Get voltage of first closed string */
575 if (pPackValues->invalidStringVoltage[bms_state.firstClosedString] == 0u) {
576 closedStringVoltage_mV = pPackValues->stringVoltage_mV[bms_state.firstClosedString];
577 searchString = true;
578 } else if (pPackValues->invalidHvBusVoltage == 0u) {
579 /* Use high voltage bus voltage if measured string voltage is invalid */
580 closedStringVoltage_mV = pPackValues->highVoltageBusVoltage_mV;
581 searchString = true;
582 } else {
583 /* Do not search for next string if no valid voltages could be measured */
584 searchString = false;
585 }

590 const uint8_t isStringVoltageValid = pPackValues->invalidStringVoltage[s];

593 int32_t minimumVoltageDifference_mV = INT32_MAX;
594 int32_t voltageDifference_mV = labs(closedStringVoltage_mV - pPackValues->stringVoltage_mV[s]);
595 if (voltageDifference_mV <= minimumVoltageDifference_mV) {
596 if (bms_state.deactivatedStrings[s] == 0u) {
597 if (precharge == BMS_TAKE_PRECHARGE_INTO_ACCOUNT) {
598 if (bs_stringsWithPrecharge[s] == BS_STRING_WITH_PRECHARGE) {
599 minimumVoltageDifference_mV = voltageDifference_mV;
600 closestStringIndex = s;
601 }
602 } else {
603 minimumVoltageDifference_mV = voltageDifference_mV;
604 closestStringIndex = s;
605 }
606 }
607 }
608 }
609 }
610 }
611 return closestStringIndex;

669
670 /* Only update system state if current value is valid */

674 if (pPackValues->packCurrent_mA >= BS_REST_CURRENT_mA) { /* TODO: string use pack current */

678 bms_state.currentFlowState = BMS_CHARGING;
679 bms_state.restTimer_10ms = BS_RELAXATION_PERIOD_10ms;

686 } else {
687 bms_state.restTimer_10ms--;
688 bms_state.currentFlowState = BMS_RELAXATION;
689 }
690 }
691 } else {
692 /* Negative current values equal a discharge of the battery system */
693 if (pPackValues->packCurrent_mA <= -BS_REST_CURRENT_mA) {
694 bms_state.currentFlowState = BMS_DISCHARGING;
695 bms_state.restTimer_10ms = BS_RELAXATION_PERIOD_10ms;
696 } else if (pPackValues->packCurrent_mA >= BS_REST_CURRENT_mA) {
697 bms_state.currentFlowState = BMS_CHARGING;
698 bms_state.restTimer_10ms = BS_RELAXATION_PERIOD_10ms;
699 } else {
700 /* Current below rest current: either battery system is at rest
701 * or the relaxation process is still ongoing */
702 if (bms_state.restTimer_10ms == 0u) {
703 /* Rest timer elapsed -> battery system at rest */
704 bms_state.currentFlowState = BMS_AT_REST;
705 } else {
706 bms_state.restTimer_10ms--;
707 bms_state.currentFlowState = BMS_RELAXATION;
708 }
709 }
710 }

714static CONT_TYPE_e BMS_GetFirstContactorToBeOpened(uint8_t stringNumber, BMS_CURRENT_FLOW_STATE_e flowDirection) {
715 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
716 /* AXIVION Routine Generic-MissingParameterAssert: flowDirection: parameter accepts all defined enums */
717 CONT_TYPE_e contactorToBeOpened = CONT_UNDEFINED;
718 CONT_CURRENT_BREAKING_DIRECTION_e breakingDirection = CONT_BIDIRECTIONAL;
719 /* Required preferred opening direction dependent on the current direction */
720 if (flowDirection == BMS_CHARGING) {
721 breakingDirection = CONT_CHARGING_DIRECTION;
722 } else {
723 breakingDirection = CONT_DISCHARGING_DIRECTION;
724 }
725 /* Iterate over contactor array and search for wanted contactor */
726 uint8_t contactor = 0u;
727 for (; contactor < BS_NR_OF_CONTACTORS; contactor++) {
728 /* Search for:
729 * 1. contactor from requested string
730 * 2. contactor mounted in preferred opening direction or is bidirectional
731 * 3. is no precharge contactor */
732 bool correctString = (bool)(stringNumber == cont_contactorStates[contactor].stringIndex);
733 bool inPreferredDirection = (bool)(breakingDirection == cont_contactorStates[contactor].breakingDirection);
734 bool hasNoPreferredDirection = (bool)(cont_contactorStates[contactor].breakingDirection == CONT_BIDIRECTIONAL);
735 bool noPrechargeContactor = (bool)(cont_contactorStates[contactor].type != CONT_PRECHARGE);
736 if (correctString && noPrechargeContactor && (inPreferredDirection || hasNoPreferredDirection)) {
737 contactorToBeOpened = cont_contactorStates[contactor].type;
738 break;
739 }
740 }
741 if (contactor == BS_NR_OF_CONTACTORS) {
742 /* No contactor mounted in preferred current direction found. Select
743 * the PLUS contactor found in array cont_contactorStates from the
744 * passed string */
745 for (contactor = 0u; contactor < BS_NR_OF_CONTACTORS; contactor++) {
746 /* Search for:
747 * 1. contactor from requested string
748 * 2. is PLUS contactor */
749 if ((stringNumber == cont_contactorStates[contactor].stringIndex) &&
750 (cont_contactorStates[contactor].type == CONT_PLUS)) {
751 contactorToBeOpened = cont_contactorStates[contactor].type;
752 break;
753 }
754 }
755 }
756 if (contactor == BS_NR_OF_CONTACTORS) {
757 /* No PLUS contactor found. Select MINUS contactor found in array
758 * cont_contactorStates from the passed string */
759 for (contactor = 0u; contactor < BS_NR_OF_CONTACTORS; contactor++) {
760 /* Search for:
761 * 1. contactor from requested string
762 * 2. is PLUS contactor */
763 if ((stringNumber == cont_contactorStates[contactor].stringIndex) &&
764 (cont_contactorStates[contactor].type == CONT_MINUS)) {
765 contactorToBeOpened = cont_contactorStates[contactor].type;
766 break;
767 }
768 }
769 }
770 if (contactor == BS_NR_OF_CONTACTORS) {
771 /* No PLUS or MAIN_MINUS contactor found in requested string. */
772 FAS_ASSERT(FAS_TRAP);
773 }
774 return contactorToBeOpened;
775}

777static CONT_TYPE_e BMS_GetSecondContactorToBeOpened(uint8_t stringNumber, CONT_TYPE_e firstOpenedContactorType) {
778 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
779 FAS_ASSERT((firstOpenedContactorType != CONT_UNDEFINED) && (firstOpenedContactorType != CONT_PRECHARGE));
780 CONT_TYPE_e contactorToBeOpened = CONT_UNDEFINED;
781 /* Check what contactor has already been opened and select the other one */
782 if (firstOpenedContactorType == CONT_PLUS) {
783 contactorToBeOpened = CONT_MINUS;
784 } else {
785 contactorToBeOpened = CONT_PLUS;
786 }
787 /* Iterate over contactor array and search for wanted contactor */
788 uint8_t contactor = 0u;
789 for (; contactor < BS_NR_OF_CONTACTORS; contactor++) {
790 /* Search for specific contactor from requested string */

793 contactorToBeOpened = cont_contactorStates[contactor].type;

795 }
796 }
797 if (contactor == BS_NR_OF_CONTACTORS) {
798 /* No PLUS or MAIN_MINUS contactor found in requested string.
799 * Apparently, only one contactor has been defined for this string */
800 FAS_ASSERT(FAS_TRAP);
801 }
802 return contactorToBeOpened;

833void BMS_Trigger(void) {
834 BMS_STATE_REQUEST_e statereq = BMS_STATE_NO_REQUEST;
835 DATA_BLOCK_SYSTEM_STATE_s systemState = {.header.uniqueId = DATA_BLOCK_ID_SYSTEM_STATE};
836 uint32_t timestamp = OS_GetTickCount();
837 static uint32_t nextOpenWireCheck = 0;
838 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
839 static uint8_t stringNumber = 0u;
840 static uint8_t nextStringNumber = 0u;
841 CONT_ELECTRICAL_STATE_TYPE_e contactorState = CONT_SWITCH_UNDEFINED;
842 bool contactorFeedbackValid = false;
843 STD_RETURN_TYPE_e contRetVal = STD_NOT_OK;
844
845 if (bms_state.state != BMS_STATEMACH_UNINITIALIZED) {
846 BMS_GetMeasurementValues();
847 BMS_UpdateBatterySystemState(&bms_tablePackValues);
848 SOA_CheckVoltages(&bms_tableMinMax);
849 SOA_CheckTemperatures(&bms_tableMinMax, &bms_tablePackValues);
850 SOA_CheckCurrent(&bms_tablePackValues);
851 SOA_CheckSlaveTemperatures();
852 BMS_CheckOpenSenseWire();
853 CONT_CheckFeedback();
854 }
855 /* Check re-entrance of function */
856 if (BMS_CheckReEntrance() > 0u) {
857 return;
858 }
859
860 if (bms_state.nextStringClosedTimer > 0u) {
861 bms_state.nextStringClosedTimer--;
862 }
863 if (bms_state.stringOpenTimeout > 0u) {
864 bms_state.stringOpenTimeout--;
865 }
866
867 if (bms_state.stringCloseTimeout > 0u) {
868 bms_state.stringCloseTimeout--;
869 }
870
871 if (bms_state.OscillationTimeout > 0u) {
872 bms_state.OscillationTimeout--;
873 }
874
875 if (bms_state.timer > 0u) {
876 if ((--bms_state.timer) > 0u) {
877 bms_state.triggerentry--;
878 return; /* handle state machine only if timer has elapsed */
879 }
880 }
881
882 /****Happens every time the state machine is triggered**************/
883 switch (bms_state.state) {
884 /****************************UNINITIALIZED****************************/
885 case BMS_STATEMACH_UNINITIALIZED:
886 /* waiting for Initialization Request */
887 statereq = BMS_TransferStateRequest();
888 if (statereq == BMS_STATE_INIT_REQUEST) {
889 BMS_SAVE_LAST_STATES();
890 bms_state.timer = BMS_STATEMACH_SHORTTIME;
891 bms_state.state = BMS_STATEMACH_INITIALIZATION;
892 bms_state.substate = BMS_ENTRY;
893 } else if (statereq == BMS_STATE_NO_REQUEST) {
894 /* no actual request pending */
895 } else {
896 bms_state.ErrRequestCounter++; /* illegal request pending */
897 }
898 break;
899
900 /****************************INITIALIZATION***************************/
901 case BMS_STATEMACH_INITIALIZATION:
902 BMS_SAVE_LAST_STATES();
903 /* Reset ALERT mode flag */
904 (void)DIAG_Handler(DIAG_ID_ALERT_MODE, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
905 bms_state.initFinished = STD_OK;
906 bms_state.timer = BMS_STATEMACH_LONGTIME;
907 bms_state.state = BMS_STATEMACH_INITIALIZED;
908 bms_state.substate = BMS_ENTRY;
909 break;
910
911 /****************************INITIALIZED******************************/
912 case BMS_STATEMACH_INITIALIZED:
913 BMS_SAVE_LAST_STATES();
914 if (IMD_RequestInsulationMeasurement() == IMD_ILLEGAL_REQUEST) {
915 /* Initialization of IMD device not finished yet -> wait until this is finished before moving on */
916 bms_state.timer = BMS_STATEMACH_LONGTIME;
917 } else {
918 bms_state.timer = BMS_STATEMACH_SHORTTIME;
919 bms_state.state = BMS_STATEMACH_IDLE;
920 bms_state.substate = BMS_ENTRY;
921 }
922 break;
923
924 /****************************IDLE*************************************/
925 case BMS_STATEMACH_IDLE:
926 BMS_SAVE_LAST_STATES();
927
928 if (bms_state.substate == BMS_ENTRY) {
929 (void)DATA_READ_DATA(&systemState);
930 systemState.bmsCanState = BMS_CAN_STATE_IDLE;
931 (void)DATA_WRITE_DATA(&systemState);
932 bms_state.timer = BMS_STATEMACH_SHORTTIME;
933 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
934 break;
935 } else if (bms_state.substate == BMS_CHECK_ERROR_FLAGS) {
936 if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
937 bms_state.timer = BMS_STATEMACH_SHORTTIME;
938 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
939 bms_state.nextState = BMS_STATEMACH_ERROR;
940 bms_state.substate = BMS_ENTRY;
941 break;
942 } else {
943 bms_state.timer = BMS_STATEMACH_SHORTTIME;
944 bms_state.substate = BMS_CHECK_STATE_REQUESTS;
945 break;
946 }
947 } else if (bms_state.substate == BMS_CHECK_STATE_REQUESTS) {
948 if (BMS_CheckCanRequests() == BMS_REQ_ID_STANDBY) {
949 bms_state.timer = BMS_STATEMACH_SHORTTIME;
950 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
951 bms_state.nextState = BMS_STATEMACH_STANDBY;
952 bms_state.substate = BMS_ENTRY;
953 break;
954 } else {
955 bms_state.timer = BMS_STATEMACH_SHORTTIME;
956 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
957 }
958 break;
959 }
960 break;
961
962 /****************************OPEN CONTACTORS**************************/
963 case BMS_STATEMACH_OPEN_CONTACTORS:
964 BMS_SAVE_LAST_STATES();
965
966 if (bms_state.substate == BMS_ENTRY) {
967 BAL_SetStateRequest(BAL_STATE_NO_BALANCING_REQUEST);
968 /* Check if the error reason is the loss of supply voltage clamp 30C */
969 if (DIAG_GetDiagnosisEntryState(DIAG_ID_SUPPLY_VOLTAGE_CLAMP_30C_LOST) == STD_NOT_OK) {
970 bms_state.timer = BMS_STATEMACH_SHORTTIME;
971 bms_state.substate = BMS_HANDLE_SUPPLY_VOLTAGE_30C_LOSS;
972 } else {
973 bms_state.timer = BMS_STATEMACH_SHORTTIME;
974 bms_state.substate = BMS_OPEN_ALL_PRECHARGE_CONTACTORS;
975 }
976 break;
977 } else if (bms_state.substate == BMS_OPEN_ALL_PRECHARGE_CONTACTORS) {
978 /* Precharge contactors can always be opened as the precharge
979 * resistor limits the maximum current */
980 CONT_OpenAllPrechargeContactors();
981
982 /* Regular string opening - Open one string after another,
983 * starting with highest string index */
984 stringNumber = BS_NR_OF_STRINGS - 1u;
985 bms_state.substate = BMS_OPEN_FIRST_STRING_CONTACTOR;
986 bms_state.timer = BMS_TIME_WAIT_AFTER_OPENING_PRECHARGE;
987
988 } else if (bms_state.substate == BMS_OPEN_FIRST_STRING_CONTACTOR) {
989 /* Precharge contactors have been opened -> start opening first string contactor */
990 /* TODO: Check if precharge contactors have been opened? */
991 if ((bms_tablePackValues.invalidStringCurrent[stringNumber] == 0u) &&
992 (MATH_AbsInt32_t(bms_tablePackValues.stringCurrent_mA[stringNumber]) <
993 BS_MAIN_CONTACTORS_MAXIMUM_BREAK_CURRENT_mA)) {
994 /* Current is below maximum break current -> open first contactor
995 * Check the mounting direction of the contactors and open the contactor that is mounted in the
996 * preferred current flow direction. Open the plus contactor first if, there is no contactor
997 * in preferred direction to the current flow to open available. This may be either because both
998 * contactors are installed in the same direction or because the contactors are bidirectional.
999 */
1000 const BMS_CURRENT_FLOW_STATE_e flowDirection =
1001 BMS_GetCurrentFlowDirection(bms_tablePackValues.stringCurrent_mA[stringNumber]);
1002 bms_state.contactorToBeOpened = BMS_GetFirstContactorToBeOpened(stringNumber, flowDirection);
1003 bms_state.stringToBeOpened = stringNumber;
1004 /* Open first contactor */
1005 CONT_OpenContactor(stringNumber, bms_state.contactorToBeOpened);
1006 bms_state.timer = BMS_WAIT_TIME_AFTER_OPENING_STRING_CONTACTOR;
1007 bms_state.substate = BMS_OPEN_SECOND_STRING_CONTACTOR;
1008 bms_state.stringOpenTimeout = BMS_STRING_OPEN_TIMEOUT;
1009 } else {
1010 /* Current is above maximum contactor break current -> contactor can not be opened */
1011 bms_state.timeAboveContactorBreakCurrent_ms += BMS_STATEMACHINE_TASK_CYCLE_CONTEXT_MS;
1012 if (bms_state.timeAboveContactorBreakCurrent_ms > BS_MAIN_FUSE_MAXIMUM_TRIGGER_DURATION_ms) {
1013 /* Fuse should have been triggered by now but apparently has not yet. Do not wait any
1014 * longer. Activate ALERT mode and nevertheless start opening the contactors */
1015 (void)DIAG_Handler(DIAG_ID_ALERT_MODE, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
1016 const BMS_CURRENT_FLOW_STATE_e flowDirection =
1017 BMS_GetCurrentFlowDirection(bms_tablePackValues.stringCurrent_mA[stringNumber]);
1018 bms_state.contactorToBeOpened = BMS_GetFirstContactorToBeOpened(stringNumber, flowDirection);
1019 bms_state.stringToBeOpened = stringNumber;
1020 /* Open first contactor */
1021 CONT_OpenContactor(bms_state.stringToBeOpened, bms_state.contactorToBeOpened);
1022 bms_state.timer = BMS_WAIT_TIME_AFTER_OPENING_STRING_CONTACTOR;
1023 bms_state.substate = BMS_OPEN_SECOND_STRING_CONTACTOR;
1024 }
1025 }
1026 break;
1027 } else if (bms_state.substate == BMS_OPEN_SECOND_STRING_CONTACTOR) {
1028 /* Check if first contactor has been opened correctly */
1029 contactorState = CONT_GetContactorState(bms_state.stringToBeOpened, bms_state.contactorToBeOpened);
1030 contactorFeedbackValid =
1031 BMS_IsContactorFeedbackValid(bms_state.stringToBeOpened, bms_state.contactorToBeOpened);
1032 /* If we want to open the contactors because of a feedback
1033 * error for this contactor, the statement will never be true.
1034 * Thus, also continue if a feedback error for this contactor
1035 * is detected as we are not able to get a valid feedback
1036 * information at this point */
1037 if ((contactorState == CONT_SWITCH_OFF) || (contactorFeedbackValid == false)) {
1038 /* First contactor opened correctly.
1039 * Open second contactor. Pass first opened contactor into function */
1040 bms_state.contactorToBeOpened =
1041 BMS_GetSecondContactorToBeOpened(stringNumber, bms_state.contactorToBeOpened);
1042 /* Open second contactor */
1043 CONT_OpenContactor(bms_state.stringToBeOpened, bms_state.contactorToBeOpened);
1044 bms_state.timer = BMS_WAIT_TIME_AFTER_OPENING_STRING_CONTACTOR;
1045 bms_state.substate = BMS_CHECK_SECOND_STRING_CONTACTOR;
1046 } else {
1047 /* String not opened, re-issue closing request */
1048 CONT_OpenContactor(bms_state.stringToBeOpened, bms_state.contactorToBeOpened);
1049 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1050 /* TODO: add timeout */
1051 }
1052 } else if (bms_state.substate == BMS_CHECK_SECOND_STRING_CONTACTOR) {
1053 /* Check if second contactor has been opened correctly */
1054 contactorState = CONT_GetContactorState(bms_state.stringToBeOpened, bms_state.contactorToBeOpened);
1055 contactorFeedbackValid =
1056 BMS_IsContactorFeedbackValid(bms_state.stringToBeOpened, bms_state.contactorToBeOpened);
1057 /* If we want to open the contactors because of a feedback
1058 * error for this contactor, the statement will never be true.
1059 * Thus, also continue if a feedback error for this contactor
1060 * is detected as we are not able to get a valid feedback
1061 * information at this point */
1062 if ((contactorState == CONT_SWITCH_OFF) || (contactorFeedbackValid == false)) {
1063 /* Opening for this string finished. Reset state variables used for opening */
1064 bms_state.contactorToBeOpened = CONT_UNDEFINED;
1065 bms_state.stringToBeOpened = 0u;
1066 /* String opened. Decrement string counter */
1067 if (bms_state.numberOfClosedStrings > 0u) {
1068 bms_state.numberOfClosedStrings--;
1069 }
1070 bms_state.closedStrings[stringNumber] = 0u;
1071 if (stringNumber > 0u) {
1072 /* Not all strings opened yet -> open next string */
1073 stringNumber--;
1074 bms_state.substate = BMS_OPEN_FIRST_STRING_CONTACTOR;
1075 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1076 break;
1077 } else {
1078 /* All strings opened -> prepare to leave state BMS_STATEMACH_OPEN_CONTACTORS */
1079 bms_state.substate = BMS_OPEN_STRINGS_EXIT;
1080 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1081 }
1082 break;
1083 } else if (bms_state.stringOpenTimeout == 0u) {
1084 /* String takes too long to open, go to next string */
1085 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1086 bms_state.substate = BMS_OPEN_FIRST_STRING_CONTACTOR;
1087 break;
1088 } else {
1089 /* String not opened, re-issue closing request */
1090 CONT_OpenContactor(bms_state.stringToBeOpened, bms_state.contactorToBeOpened);
1091 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1092 break;
1093 }
1094 } else if (bms_state.substate == BMS_HANDLE_SUPPLY_VOLTAGE_30C_LOSS) {
1095 CONT_OpenAllContactors();
1096 SPS_SwitchOffAllGeneralIoChannels();
1097 bms_state.substate = BMS_OPEN_STRINGS_EXIT;
1098 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1099 break;
1100 } else if (bms_state.substate == BMS_OPEN_STRINGS_EXIT) {
1101 if (bms_state.nextState == BMS_STATEMACH_STANDBY) {
1102 /* Opening due to STANDBY request -> switch to BMS_STATEMACH_STANDBY */
1103 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1104 bms_state.state = BMS_STATEMACH_STANDBY;
1105 bms_state.substate = BMS_ENTRY;
1106 break;
1107 } else {
1108 /* Opening due to detected error -> switch to BMS_STATEMACH_ERROR */
1109 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1110 bms_state.state = BMS_STATEMACH_ERROR;
1111 bms_state.substate = BMS_ENTRY;
1112 }
1113 } else {
1114 FAS_ASSERT(FAS_TRAP);
1115 }
1116 break;
1117
1118 /****************************STANDBY**********************************/
1119 case BMS_STATEMACH_STANDBY:
1120 BMS_SAVE_LAST_STATES();
1121 if (bms_state.substate == BMS_ENTRY) {
1122 BAL_SetStateRequest(BAL_STATE_ALLOW_BALANCING_REQUEST);
1123#if BS_STANDBY_PERIODIC_OPEN_WIRE_CHECK == TRUE
1124 nextOpenWireCheck = timestamp + BS_STANDBY_OPEN_WIRE_PERIOD_ms;
1125#endif /* BS_STANDBY_PERIODIC_OPEN_WIRE_CHECK == TRUE */
1126 bms_state.timer = BMS_STATEMACH_MEDIUM_TIME;
1127 bms_state.substate = BMS_CHECK_ERROR_FLAGS_INTERLOCK;
1128 (void)DATA_READ_DATA(&systemState);
1129 systemState.bmsCanState = BMS_CAN_STATE_STANDBY;
1130 (void)DATA_WRITE_DATA(&systemState);
1131 break;
1132 } else if (bms_state.substate == BMS_CHECK_ERROR_FLAGS_INTERLOCK) {
1133 if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
1134 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1135 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1136 bms_state.nextState = BMS_STATEMACH_ERROR;
1137 bms_state.substate = BMS_ENTRY;
1138 break;
1139 } else {
1140 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1141 bms_state.substate = BMS_INTERLOCK_CHECKED;
1142 break;
1143 }
1144 } else if (bms_state.substate == BMS_INTERLOCK_CHECKED) {
1145 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1146 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
1147 break;
1148 } else if (bms_state.substate == BMS_CHECK_ERROR_FLAGS) {
1149 if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
1150 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1151 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1152 bms_state.nextState = BMS_STATEMACH_ERROR;
1153 bms_state.substate = BMS_ENTRY;
1154 break;
1155 } else {
1156 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1157 bms_state.substate = BMS_CHECK_STATE_REQUESTS;
1158 break;
1159 }
1160 } else if (bms_state.substate == BMS_CHECK_STATE_REQUESTS) {
1161 if (BMS_CheckCanRequests() == BMS_REQ_ID_NORMAL) {
1162 bms_state.powerPath = BMS_POWER_PATH_0;
1163 bms_state.nextState = BMS_STATEMACH_DISCHARGE;
1164 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1165 bms_state.state = BMS_STATEMACH_PRECHARGE;
1166 bms_state.substate = BMS_ENTRY;
1167 break;
1168 }
1169 if (BMS_CheckCanRequests() == BMS_REQ_ID_CHARGE) {
1170 bms_state.powerPath = BMS_POWER_PATH_1;
1171 bms_state.nextState = BMS_STATEMACH_CHARGE;
1172 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1173 bms_state.state = BMS_STATEMACH_PRECHARGE;
1174 bms_state.substate = BMS_ENTRY;
1175 break;
1176 } else {
1177#if BS_STANDBY_PERIODIC_OPEN_WIRE_CHECK == TRUE
1178 if (nextOpenWireCheck <= timestamp) {
1179 MEAS_RequestOpenWireCheck();
1180 nextOpenWireCheck = timestamp + BS_STANDBY_OPEN_WIRE_PERIOD_ms;
1181 }
1182#endif /* BS_STANDBY_PERIODIC_OPEN_WIRE_CHECK == TRUE */
1183 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1184 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
1185 break;
1186 }
1187 } else {
1188 FAS_ASSERT(FAS_TRAP);
1189 }
1190 break;
1191
1192 /****************************PRECHARGE********************************/
1193 case BMS_STATEMACH_PRECHARGE:
1194 BMS_SAVE_LAST_STATES();
1195
1196 if (bms_state.substate == BMS_ENTRY) {
1197 (void)DATA_READ_DATA(&systemState);
1198 systemState.bmsCanState = BMS_CAN_STATE_PRECHARGE;
1199 (void)DATA_WRITE_DATA(&systemState);
1200 if (bms_state.nextState == BMS_STATEMACH_CHARGE) {
1201 stringNumber = BMS_GetLowestString(BMS_TAKE_PRECHARGE_INTO_ACCOUNT, &bms_tablePackValues);
1202 } else {
1203 stringNumber = BMS_GetHighestString(BMS_TAKE_PRECHARGE_INTO_ACCOUNT, &bms_tablePackValues);
1204 }
1205 if (stringNumber == BMS_NO_STRING_AVAILABLE) {
1206 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1207 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1208 bms_state.nextState = BMS_STATEMACH_ERROR;
1209 bms_state.substate = BMS_ENTRY;
1210 break;
1211 }
1212 bms_state.prechargeTryCounter = 0u;
1213 bms_state.firstClosedString = stringNumber;
1214 if (bms_state.OscillationTimeout == 0u) {
1215 /* Close MINUS string contactor */
1216 if (CONT_CloseContactor(bms_state.firstClosedString, CONT_MINUS) == STD_OK) {
1217 bms_state.stringCloseTimeout = BMS_STRING_CLOSE_TIMEOUT;
1218 bms_state.timer = BMS_WAIT_TIME_AFTER_CLOSING_STRING_CONTACTOR;
1219 bms_state.substate = BMS_PRECHARGE_CLOSE_PRECHARGE;
1220 } else {
1221 /* Invalid contactor requested */
1222 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1223 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1224 bms_state.nextState = BMS_STATEMACH_ERROR;
1225 bms_state.substate = BMS_ENTRY;
1226 }
1227 } else if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
1228 /* If precharge re-enter timeout not elapsed, wait (and check errors while waiting) */
1229 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1230 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1231 bms_state.nextState = BMS_STATEMACH_ERROR;
1232 bms_state.substate = BMS_ENTRY;
1233 break;
1234 }
1235 break;
1236 } else if (bms_state.substate == BMS_PRECHARGE_CLOSE_PRECHARGE) {
1237 /* Check if MINUS contactor has been successfully closed */
1238 contactorState = CONT_GetContactorState(bms_state.firstClosedString, CONT_MINUS);
1239 if (contactorState == CONT_SWITCH_ON) {
1240 bms_state.OscillationTimeout = BMS_OSCILLATION_TIMEOUT;
1241 contRetVal = CONT_ClosePrecharge(bms_state.firstClosedString);
1242 bms_state.closedPrechargeContactors[stringNumber] = 1u;
1243 if (contRetVal == STD_OK) {
1244 bms_state.timer = BMS_TIME_WAIT_AFTER_CLOSING_PRECHARGE;
1245 bms_state.substate = BMS_CHECK_ERROR_FLAGS_CLOSING_PRECHARGE;
1246 } else {
1247 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1248 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1249 bms_state.nextState = BMS_STATEMACH_ERROR;
1250 bms_state.substate = BMS_ENTRY;
1251 }
1252 } else if (bms_state.stringCloseTimeout == 0u) {
1253 /* String takes too long to close */
1254 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1255 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1256 bms_state.nextState = BMS_STATEMACH_ERROR;
1257 bms_state.substate = BMS_ENTRY;
1258 } else {
1259 /* String not closed, re-issue closing request */
1260 CONT_CloseContactor(bms_state.firstClosedString, CONT_MINUS);
1261 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1262 }
1263 break;
1264 } else if (bms_state.substate == BMS_CHECK_ERROR_FLAGS_CLOSING_PRECHARGE) {
1265 if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
1266 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1267 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1268 bms_state.nextState = BMS_STATEMACH_ERROR;
1269 bms_state.substate = BMS_ENTRY;
1270 break;
1271 } else {
1272 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1273 bms_state.substate = BMS_CHECK_STATE_REQUESTS;
1274 break;
1275 }
1276 } else if (bms_state.substate == BMS_CHECK_STATE_REQUESTS) {
1277 if (BMS_CheckCanRequests() == BMS_REQ_ID_STANDBY) {
1278 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1279 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1280 bms_state.nextState = BMS_STATEMACH_STANDBY;
1281 bms_state.substate = BMS_ENTRY;
1282 break;
1283 } else {
1284 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1285 bms_state.substate = BMS_PRECHARGE_CHECK_VOLTAGES;
1286 }
1287 } else if (bms_state.substate == BMS_PRECHARGE_CHECK_VOLTAGES) {
1288 contactorState = CONT_GetContactorState(bms_state.firstClosedString, CONT_PRECHARGE);
1289 retVal = BMS_CheckPrecharge(bms_state.firstClosedString, &bms_tablePackValues);
1290 /* Check if precharge contactor is closed and precharge is finished */
1291 if ((contactorState == CONT_SWITCH_ON) && (retVal == STD_OK)) {
1292 /* Successfully precharged. Close string PLUS contactor */
1293 CONT_CloseContactor(bms_state.firstClosedString, CONT_PLUS);
1294 bms_state.stringCloseTimeout = BMS_STRING_CLOSE_TIMEOUT;
1295 bms_state.timer = BMS_WAIT_TIME_AFTER_CLOSING_STRING_CONTACTOR;
1296 bms_state.substate = BMS_CHECK_CLOSE_SECOND_STRING_CONTACTOR_PRECHARGE_STATE;
1297 break;
1298 } else {
1299 /* Precharging failed. Open precharge contactor. */
1300 contRetVal = CONT_OpenPrecharge(bms_state.firstClosedString);
1301 /* Check if retry limit has been reached */
1302 if (bms_state.prechargeTryCounter < (BMS_PRECHARGE_TRIES - 1u)) {
1303 bms_state.closedPrechargeContactors[stringNumber] = 0u;
1304 if (contRetVal == STD_OK) {
1305 bms_state.timer = BMS_TIME_WAIT_AFTER_PRECHARGE_FAIL;
1306 bms_state.substate = BMS_PRECHARGE_CLOSE_PRECHARGE;
1307 bms_state.prechargeTryCounter++;
1308 } else {
1309 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1310 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1311 bms_state.nextState = BMS_STATEMACH_ERROR;
1312 bms_state.substate = BMS_ENTRY;
1313 }
1314 break;
1315 } else {
1316 bms_state.closedPrechargeContactors[stringNumber] = 0u;
1317 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1318 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1319 bms_state.nextState = BMS_STATEMACH_ERROR;
1320 bms_state.substate = BMS_ENTRY;
1321 break;
1322 }
1323 }
1324 } else if (bms_state.substate == BMS_CHECK_CLOSE_SECOND_STRING_CONTACTOR_PRECHARGE_STATE) {
1325 contactorState = CONT_GetContactorState(bms_state.firstClosedString, CONT_PLUS);
1326 if (contactorState == CONT_SWITCH_ON) {
1327 bms_state.closedStrings[bms_state.firstClosedString] = 1u;
1328 bms_state.numberOfClosedStrings++;
1329 bms_state.timer = BMS_WAIT_TIME_AFTER_CLOSING_STRING_CONTACTOR;
1330 bms_state.substate = BMS_CHECK_ERROR_FLAGS_PRECHARGE_CLOSING_STRINGS;
1331 break;
1332 } else if (bms_state.stringCloseTimeout == 0u) {
1333 /* String takes too long to close */
1334 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1335 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1336 bms_state.nextState = BMS_STATEMACH_ERROR;
1337 bms_state.substate = BMS_ENTRY;
1338 break;
1339 } else {
1340 /* String not closed, re-issue closing request */
1341 CONT_CloseContactor(bms_state.firstClosedString, CONT_PLUS);
1342 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1343 bms_state.substate = BMS_CHECK_ERROR_FLAGS_PRECHARGE_FIRST_STRING;
1344 break;
1345 }
1346 } else if (bms_state.substate == BMS_CHECK_ERROR_FLAGS_PRECHARGE_FIRST_STRING) {
1347 if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
1348 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1349 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1350 bms_state.nextState = BMS_STATEMACH_ERROR;
1351 bms_state.substate = BMS_ENTRY;
1352 break;
1353 } else {
1354 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1355 bms_state.substate = BMS_CHECK_CLOSE_SECOND_STRING_CONTACTOR_PRECHARGE_STATE;
1356 break;
1357 }
1358 } else if (bms_state.substate == BMS_CHECK_ERROR_FLAGS_PRECHARGE_CLOSING_STRINGS) {
1359 /* Always make one error check after the first string was closed successfully */
1360 if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
1361 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1362 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1363 bms_state.nextState = BMS_STATEMACH_ERROR;
1364 bms_state.substate = BMS_ENTRY;
1365 break;
1366 } else {
1367 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1368 bms_state.substate = BMS_PRECHARGE_OPEN_PRECHARGE;
1369 break;
1370 }
1371 } else if (bms_state.substate == BMS_PRECHARGE_OPEN_PRECHARGE) {
1372 contRetVal = CONT_OpenPrecharge(bms_state.firstClosedString);
1373 if (contRetVal == STD_OK) {
1374 bms_state.closedPrechargeContactors[stringNumber] = 0u;
1375 bms_state.timer = BMS_TIME_WAIT_AFTER_OPENING_PRECHARGE;
1376 bms_state.substate = BMS_PRECHARGE_CHECK_OPEN_PRECHARGE;
1377 } else {
1378 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1379 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1380 bms_state.nextState = BMS_STATEMACH_ERROR;
1381 bms_state.substate = BMS_ENTRY;
1382 }
1383 break;
1384 } else if (bms_state.substate == BMS_PRECHARGE_CHECK_OPEN_PRECHARGE) {
1385 contactorState = CONT_GetContactorState(bms_state.firstClosedString, CONT_PRECHARGE);
1386 if (contactorState == CONT_SWITCH_OFF) {
1387 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1388 bms_state.state = BMS_STATEMACH_NORMAL;
1389 bms_state.substate = BMS_ENTRY;
1390 break;
1391 } else if (bms_state.stringCloseTimeout == 0u) {
1392 /* Precharge contactor takes too long to open */
1393 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1394 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1395 bms_state.nextState = BMS_STATEMACH_ERROR;
1396 bms_state.substate = BMS_ENTRY;
1397 break;
1398 } else {
1399 /* Precharge contactor not opened, re-issue open request */
1400 CONT_OpenPrecharge(bms_state.firstClosedString);
1401 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1402 bms_state.substate = BMS_CHECK_ERROR_FLAGS_PRECHARGE_FIRST_STRING;
1403 break;
1404 }
1405 } else {
1406 FAS_ASSERT(FAS_TRAP);
1407 }
1408 break;
1409
1410 /****************************NORMAL**************************************/
1411 case BMS_STATEMACH_NORMAL:
1412 BMS_SAVE_LAST_STATES();
1413
1414 if (bms_state.substate == BMS_ENTRY) {
1415#if BS_NORMAL_PERIODIC_OPEN_WIRE_CHECK == TRUE
1416 nextOpenWireCheck = timestamp + BS_NORMAL_OPEN_WIRE_PERIOD_ms;
1417#endif /* BS_NORMAL_PERIODIC_OPEN_WIRE_CHECK == TRUE */
1418 (void)DATA_READ_DATA(&systemState);
1419 if (bms_state.nextState == BMS_STATEMACH_CHARGE) {
1420 systemState.bmsCanState = BMS_CAN_STATE_CHARGE;
1421 } else {
1422 systemState.bmsCanState = BMS_CAN_STATE_NORMAL;
1423 }
1424 (void)DATA_WRITE_DATA(&systemState);
1425 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1426 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
1427 bms_state.nextStringClosedTimer = 0u;
1428 break;
1429 } else if (bms_state.substate == BMS_CHECK_ERROR_FLAGS) {
1430 if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
1431 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1432 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1433 bms_state.nextState = BMS_STATEMACH_ERROR;
1434 bms_state.substate = BMS_ENTRY;
1435 break;
1436 } else {
1437 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1438 bms_state.substate = BMS_CHECK_STATE_REQUESTS;
1439 break;
1440 }
1441 } else if (bms_state.substate == BMS_CHECK_STATE_REQUESTS) {
1442 if (BMS_CheckCanRequests() == BMS_REQ_ID_STANDBY) {
1443 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1444 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1445 bms_state.nextState = BMS_STATEMACH_STANDBY;
1446 bms_state.substate = BMS_ENTRY;
1447 break;
1448 } else {
1449#if BS_NORMAL_PERIODIC_OPEN_WIRE_CHECK == TRUE
1450 if (nextOpenWireCheck <= timestamp) {
1451 MEAS_RequestOpenWireCheck();
1452 nextOpenWireCheck = timestamp + BS_NORMAL_OPEN_WIRE_PERIOD_ms;
1453 }
1454#endif /* BS_NORMAL_PERIODIC_OPEN_WIRE_CHECK == TRUE */
1455 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1456 bms_state.substate = BMS_NORMAL_CLOSE_NEXT_STRING;
1457 break;
1458 }
1459 } else if (bms_state.substate == BMS_NORMAL_CLOSE_NEXT_STRING) {
1460 if (bms_state.nextStringClosedTimer == 0u) {
1461 nextStringNumber =
1462 BMS_GetClosestString(BMS_DO_NOT_TAKE_PRECHARGE_INTO_ACCOUNT, &bms_tablePackValues);
1463 if (nextStringNumber == BMS_NO_STRING_AVAILABLE) {
1464 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1465 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
1466 break;
1467 } else if (
1468 (BMS_GetStringVoltageDifference(nextStringNumber, &bms_tablePackValues) <=
1469 BMS_NEXT_STRING_VOLTAGE_LIMIT_MV) &&
1470 (BMS_GetAverageStringCurrent(&bms_tablePackValues) <= BMS_AVERAGE_STRING_CURRENT_LIMIT_MA)) {
1471 /* Voltage/current conditions suitable to close a further string. Close first string contactor
1472 */
1473 CONT_CloseContactor(nextStringNumber, CONT_MINUS);
1474 bms_state.nextStringClosedTimer = BMS_STRING_CLOSE_TIMEOUT;
1475 bms_state.timer = BMS_WAIT_TIME_AFTER_CLOSING_STRING_CONTACTOR;
1476 bms_state.substate = BMS_NORMAL_CLOSE_SECOND_STRING_CONTACTOR;
1477 break;
1478 }
1479 } else {
1480 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1481 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
1482 break;
1483 }
1484 } else if (bms_state.substate == BMS_NORMAL_CLOSE_SECOND_STRING_CONTACTOR) {
1485 contactorState = CONT_GetContactorState(nextStringNumber, CONT_MINUS);
1486 if (contactorState == CONT_SWITCH_ON) {
1487 /* First string contactor closed. Close second string contactor */
1488 CONT_CloseContactor(nextStringNumber, CONT_PLUS);
1489 bms_state.timer = BMS_WAIT_TIME_AFTER_CLOSING_STRING_CONTACTOR;
1490 bms_state.substate = BMS_NORMAL_CLOSE_SECOND_STRING_CONTACTOR;
1491 } else if (bms_state.stringCloseTimeout == 0u) {
1492 /* String takes too long to close */
1493 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1494 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1495 bms_state.nextState = BMS_STATEMACH_ERROR;
1496 bms_state.substate = BMS_ENTRY;
1497 break;
1498 } else {
1499 /* String minus contactor has not been closed successfully. Re-trigger closing */
1500 CONT_CloseContactor(nextStringNumber, CONT_MINUS);
1501 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1502 }
1503 break;
1504 } else if (bms_state.substate == BMS_CHECK_STRING_CLOSED) {
1505 contactorState = CONT_GetContactorState(nextStringNumber, CONT_PLUS);
1506 if (contactorState == CONT_SWITCH_ON) {
1507 bms_state.numberOfClosedStrings++;
1508 bms_state.closedStrings[nextStringNumber] = 1u;
1509 bms_state.nextStringClosedTimer = BMS_WAIT_TIME_BETWEEN_CLOSING_STRINGS;
1510 /* Go to begin of NORMAL case to redo the full procedure with error check and request check */
1511 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
1512 break;
1513 } else if (bms_state.stringCloseTimeout == 0u) {
1514 /* String takes too long to close */
1515 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1516 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1517 bms_state.nextState = BMS_STATEMACH_ERROR;
1518 bms_state.substate = BMS_ENTRY;
1519 break;
1520 } else if (BMS_IsBatterySystemStateOkay() == STD_NOT_OK) {
1521 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1522 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1523 bms_state.nextState = BMS_STATEMACH_ERROR;
1524 bms_state.substate = BMS_ENTRY;
1525 break;
1526 } else if (BMS_CheckCanRequests() == BMS_REQ_ID_STANDBY) {
1527 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1528 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1529 bms_state.nextState = BMS_STATEMACH_STANDBY;
1530 bms_state.substate = BMS_ENTRY;
1531 break;
1532 } else {
1533 /* String not closed, re-issue closing request */
1534 CONT_CloseContactor(nextStringNumber, CONT_PLUS);
1535 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1536 break;
1537 }
1538 } else {
1539 FAS_ASSERT(FAS_TRAP);
1540 }
1541 break;
1542
1543 /****************************ERROR*************************************/
1544 case BMS_STATEMACH_ERROR:
1545 BMS_SAVE_LAST_STATES();
1546
1547 if (bms_state.substate == BMS_ENTRY) {
1548 /* Set BMS System state to error */
1549 (void)DATA_READ_DATA(&systemState);
1550 systemState.bmsCanState = BMS_CAN_STATE_ERROR;
1551 (void)DATA_WRITE_DATA(&systemState);
1552 /* Deactivate balancing */
1553 BAL_SetStateRequest(BAL_STATE_NO_BALANCING_REQUEST);
1554 /* Change LED toggle frequency to indicate an error */

1556 /* Set timer for next open wire check */

1558 /* Switch to next substate */
1559 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1560 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
1561 break;
1562 } else if (bms_state.substate == BMS_CHECK_ERROR_FLAGS) {
1563 if (DIAG_IsAnyFatalErrorSet() == true) {
1564 /* we stay already in requested state */
1565 if (nextOpenWireCheck <= timestamp) {
1566 /* Perform open-wire check periodically */
1567 /* MEAS_RequestOpenWireCheck(); */ /*TODO: check with strings */
1568 nextOpenWireCheck = timestamp + AFE_ERROR_OPEN_WIRE_PERIOD_ms;
1569 }
1570 } else {
1571 /* No error detected anymore - reset fatal error related variables */
1572 bms_state.minimumActiveDelay_ms = BMS_NO_ACTIVE_DELAY_TIME_ms;
1573 bms_state.minimumActiveDelay_ms = BMS_NO_ACTIVE_DELAY_TIME_ms;

1575 /* Check for STANDBY request */
1576 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1577 bms_state.substate = BMS_CHECK_STATE_REQUESTS;
1578 break;
1579 }
1580 } else if (bms_state.substate == BMS_CHECK_STATE_REQUESTS) {

1582 /* Activate balancing again */
1583 BAL_SetStateRequest(BAL_STATE_ALLOW_BALANCING_REQUEST);
1584 /* Set LED frequency to normal operation as we leave error
1585 state subsequently */
1586 LED_SetToggleTime(LED_NORMAL_OPERATION_ON_OFF_TIME_ms);
1587
1588 /* Verify that all contactors are opened and switch to
1589 * STANDBY state afterwards */
1590 bms_state.state = BMS_STATEMACH_OPEN_CONTACTORS;
1591 bms_state.nextState = BMS_STATEMACH_STANDBY;
1592 bms_state.substate = BMS_ENTRY;
1593 break;
1594 } else {
1595 bms_state.timer = BMS_STATEMACH_SHORTTIME;
1596 bms_state.substate = BMS_CHECK_ERROR_FLAGS;
1597 break;
1598 }
1599 } else {
1600 /* invalid state -> this should never be reached */
1601 FAS_ASSERT(FAS_TRAP);
1602 }
1603 break;
1604 default:
1605 /* invalid state */
1606 FAS_ASSERT(FAS_TRAP);
1607 break;
1608 } /* end switch (bms_state.state) */
1609
1610 /* Send an asynchronous bms state message if the state or substate changed*/
1611 if ((bms_state.state != bms_state.lastState) || (bms_state.substate != bms_state.lastSubstate)) {
1612 CANTX_TransmitBmsState();
1613 }
1614
1615 bms_state.triggerentry--;
1616 bms_state.counter++;
