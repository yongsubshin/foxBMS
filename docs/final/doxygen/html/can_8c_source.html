86/**
87 * IF2ARB register configuration
88 *
89 * Bits 28-0 ID: Message identifier
90 * ID[28:0] 29-bit Identifier ("Extended Frame").
91 * ID[28:18] 11-bit Identifier ("Standard Frame").
92 */
93/** ID shift for standard identifier */
94#define CAN_IF2ARB_STANDARD_IDENTIFIER_SHIFT (18u)
95/** ID shift for extended identifier */
96#define CAN_IF2ARB_EXTENDED_IDENTIFIER_SHIFT (0u)
97
98/* Bit 29 Dir: Message direction
99 * 0 Direction = Receive: On Tx Request, a Remote Frame with the identifier of this message object is
100 * transmitted. On receiving a Data Frame with a matching identifier, this message is stored in this
101 * message object.
102 * 1 Direction = Transmit: On TxRequest, the respective message object is transmitted as a Data
103 * Frame. On receiving a Remote Frame with a matching identifier, the TxRequest bit of this message
104 * object is set (if RemoteEntry = 1).
105 */
106/** IF2ARB set TX direction */
107#define CAN_IF2ARB_SET_TX_DIRECTION ((uint32)1u << 29u)
108/* Bit 30 - Xtd: Extended identifier
109 * 0 The 11-bit ("standard") identifier is used for this message object
110 * 1 The 29-bit ("extended") identifier is used for this message object
111 */
112/** IF2ARB use standard identifier */
113#define CAN_IF2ARB_USE_STANDARD_IDENTIFIER ((uint32)0u << 30u)
114/** IF2ARB use extended identifier */
115#define CAN_IF2ARB_USE_EXTENDED_IDENTIFIER ((uint32)1u << 30u)
116

118 * messages with extended identifiers. */ /**@{*/

120#define CAN_HIGHEST_MAILBOX_FOR_EXTENDED_IDENTIFIERS (64u)

638 /**
639 * Read even if queues are not created, otherwise message boxes get full.
640 * Possible return values:
641 * - 0: no new data
642 * - 1: no data lost
643 * - 3: data lost */
644 uint32_t retval = canGetData(pNode, messageBox, (uint8 *)&messageData[0]); /* copy to RAM */
645
646 /* Check that CAN RX queue is started before using it and data is valid */
647 if ((ftsk_allQueuesCreated == true) && (retval == CAN_HAL_RETVAL_NO_DATA_LOST)) {

654 (messageBox <= CAN_HIGHEST_MAILBOX_FOR_EXTENDED_IDENTIFIERS))) {
655 /* Extract standard identifier from IF2ARB register*/

657 can_rxBuffer.idType = CAN_STANDARD_IDENTIFIER_11_BIT;
658 } else {

684static STD_RETURN_TYPE_e CAN_PeriodicTransmit(void) {
685 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
686 static uint32_t counterTicks = 0;
687 uint8_t data[CAN_MAX_DLC] = {0};
688
689 CAN_SendMessagesFromQueue();
690
691 for (uint16_t i = 0u; i < can_txMessagesLength; i++) {
692 if (CAN_IsMessagePeriodElapsed(counterTicks, i) == true) {
693 if (can_txMessages[i].callbackFunction != NULL_PTR) {
694 can_txMessages[i].callbackFunction(
695 can_txMessages[i].message, data, can_txMessages[i].pMuxId, &can_kShim);
696 if (CAN_DataSend(
697 can_txMessages[i].canNode,
698 can_txMessages[i].message.id,
699 can_txMessages[i].message.idType,
700 data) != STD_OK) {
701 /* message was not sent */
702 /* store the message */
703 CAN_BUFFER_ELEMENT_s unsentMessage = {
704 .canNode = can_txMessages[i].canNode,
705 .id = can_txMessages[i].message.id,
706 .idType = can_txMessages[i].message.idType,
707 .data = {0u, 0u, 0u, 0u, 0u, 0u, 0u, 0u}};
708
709 for (uint8_t j = 0; j < can_txMessages[i].message.dlc; j++) {
710 unsentMessage.data[j] = data[j];
711 }
712
713 /* add message to queue */
714 if (OS_SendToBackOfQueue(ftsk_canTxUnsentMessagesQueue, (void *)&unsentMessage, 0u) == OS_SUCCESS) {
715 /* Queue is not full */
716 (void)DIAG_Handler(DIAG_ID_CAN_TX_QUEUE_FULL, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
717 } else {
718 /* Queue is full */
719 (void)DIAG_Handler(DIAG_ID_CAN_TX_QUEUE_FULL, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
720 }
721 }
722 retVal = STD_OK;
723 }
724 }
725 }
726
727 /* check if counter needs to be reset */
728 counterTicks++;
729 if (counterTicks == can_counterResetValue) {
730 counterTicks = 0;
731 }
732
733 return retVal;
734}

736static uint32_t CAN_CalculateCounterResetValue(void) {
737 /* initialize with first array entry for first iteration */
738 uint32_t resetValue = can_txMessages[0].timing.period;
739
740 /* calculate least common multiple (LCM) of the first and second can_TxMessage period.
741 After that, calculate LCM of the next can_TxMessage period and last iterations LCM
742 until no more can_TxMessages remain. */
743 for (uint8_t i = 1u; i < can_txMessagesLength; i++) {
744 const uint32_t value1 = resetValue;
745 const uint32_t value2 = can_txMessages[i].timing.period;
746
747 /* calculate greatest common divisor using Euclid's algorithm */
748 /* the algorithm is designed for starting with bigger value, but also
749 works when starting with smaller value, so just start with value1 */
750
751 uint32_t result = value1;
752 uint32_t divisor = value2;
753 uint32_t tempValue = 0u;
754
755 while (divisor != 0u) {
756 tempValue = result % divisor;
757 result = divisor;
758 divisor = tempValue;
759 }
760
761 /* calculate least common multiple using the greatest common divisor */
762 resetValue = (value1 / result) * value2;
763 /* AXIVION Routine FaultDetection-DivisionByZero: /: message periods cannot be 0, so result is never 0 */
764 /* AXIVION Routine MisraC2012Directive-4.1: /: message periods cannot be 0, so result is never 0 */
765 /* AXIVION Routine MisraC2012Directive-4.1: *: only wraps around if periods are too big */
766 }
767
768 /* scale down to match internal counter of CAN_PeriodicTransmit */
769 resetValue /= CAN_TICK_ms;
770 return resetValue;
771}

773static bool CAN_IsMessagePeriodElapsed(uint32_t ticksSinceStart, uint16_t messageIndex) {
774 /* AXIVION Routine Generic-MissingParameterAssert: ticksSinceStart: parameter accepts whole range */
775 FAS_ASSERT(messageIndex < can_txMessagesLength);
776
777 bool retVal = false;
778 if (((ticksSinceStart * CAN_TICK_ms) % (can_txMessages[messageIndex].timing.period)) ==
779 can_txMessages[messageIndex].timing.phase) {
780 /* AXIVION Routine MisraC2012Directive-4.1: *: counter gets reset in periodicTransmit, no wrap around */
781 /* AXIVION Routine FaultDetection-DivisionByZero: %: message period is never zero, checked by config */
782 /* AXIVION Routine MisraC2012Directive-4.1: %: message period is never zero, checked by config */
783 retVal = true;
784 }
785 return retVal;

924extern STD_RETURN_TYPE_e CAN_DataSend(CAN_NODE_s *pNode, uint32_t id, CAN_IDENTIFIER_TYPE_e idType, uint8 *pData) {
925 FAS_ASSERT(pNode != NULL_PTR);
926 /* AXIVION Routine Generic-MissingParameterAssert: id: parameter accepts whole range */
927 FAS_ASSERT((idType == CAN_STANDARD_IDENTIFIER_11_BIT) || (idType == CAN_EXTENDED_IDENTIFIER_29_BIT));
928 FAS_ASSERT(pData != NULL_PTR);
929 FAS_ASSERT((pNode == CAN_NODE_1) || (pNode == CAN_NODE_2));
930
931 STD_RETURN_TYPE_e result = STD_NOT_OK;
932
933 /**
934 * Parse all TX message boxes until we find a free one,
935 * then use it to send the CAN message.
936 * In the HAL, message box numbers start from 1, not 0.
937 */
938 for (uint8_t messageBox = 1u; messageBox <= CAN_NR_OF_TX_MESSAGE_BOX; messageBox++) {
939 if (canIsTxMessagePending(pNode->canNodeRegister, messageBox) == 0u) {
940 /* id shifted by 18 to use standard frame */
941 /* standard frame: bits [28:18] */
942 /* extended frame: bits [28:0] */
943 /* bit 29 set to 1: to set direction Tx in IF2ARB register */
944 /* bit 30 set to 1: 29-bit ("extended") identifier is used for this message object */
945 if (idType == CAN_STANDARD_IDENTIFIER_11_BIT) {
946 canUpdateID(
947 pNode->canNodeRegister,
948 messageBox,
949 ((id << CAN_IF2ARB_STANDARD_IDENTIFIER_SHIFT) | CAN_IF2ARB_SET_TX_DIRECTION |
950 CAN_IF2ARB_USE_STANDARD_IDENTIFIER));
951 } else {
952 /* Extended 29-bit identifier is used for this node */
953 canUpdateID(
954 pNode->canNodeRegister,
955 messageBox,
956 ((id << CAN_IF2ARB_EXTENDED_IDENTIFIER_SHIFT) | CAN_IF2ARB_SET_TX_DIRECTION |
957 CAN_IF2ARB_USE_EXTENDED_IDENTIFIER));
958 }
959 canTransmit(pNode->canNodeRegister, messageBox, pData);
960 result = STD_OK;
961 break;
962 }
963 }
964 return result;
965}
