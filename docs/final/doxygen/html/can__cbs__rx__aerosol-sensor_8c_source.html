119/*========== Static Function Implementations ================================*/
120static void CANRX_HandleAerosolSensorErrors(const CAN_SHIM_s *const kpkCanShim, uint16_t signalData) {
121 FAS_ASSERT(kpkCanShim != NULL_PTR);
122 FAS_ASSERT(
123 (signalData == CANRX_AEROSOL_SENSOR_NO_ERROR) || (signalData == CANRX_AEROSOL_SENSOR_PHOTOELECTRIC_ERROR) ||
124 (signalData == CANRX_AEROSOL_SENSOR_VOLTAGE_SUPPLY_OVER_VOLTAGE_ERROR) ||
125 (signalData == CANRX_AEROSOL_SENSOR_VOLTAGE_SUPPLY_UNDER_VOLTAGE_ERROR));
126 switch (signalData) {
127 case CANRX_AEROSOL_SENSOR_NO_ERROR: /* resets error flags */
128 kpkCanShim->pTableAerosolSensor->photoelectricError = false;
129 kpkCanShim->pTableAerosolSensor->supplyOvervoltageError = false;
130 kpkCanShim->pTableAerosolSensor->supplyUndervoltageError = false;
131 break;
132 case CANRX_AEROSOL_SENSOR_PHOTOELECTRIC_ERROR: /* sets flag for photoelectric device fault */
133 kpkCanShim->pTableAerosolSensor->photoelectricError = true;
134 break;
135 case CANRX_AEROSOL_SENSOR_VOLTAGE_SUPPLY_OVER_VOLTAGE_ERROR: /* sets flag for voltage supply over voltage */
136 kpkCanShim->pTableAerosolSensor->supplyOvervoltageError = true;
137 break;
138 case CANRX_AEROSOL_SENSOR_VOLTAGE_SUPPLY_UNDER_VOLTAGE_ERROR: /* sets flag for voltage supply under voltage */
139 kpkCanShim->pTableAerosolSensor->supplyUndervoltageError = true;
140 break;
141 default: /* LCOV_EXCL_LINE */
142 /* undefined value */ /* LCOV_EXCL_LINE */
143 break; /* LCOV_EXCL_LINE */
144 }
145}

147static void CANRX_HandleAerosolSensorStatus(const CAN_SHIM_s *const kpkCanShim, uint16_t signalData) {
148 FAS_ASSERT(kpkCanShim != NULL_PTR);
149 FAS_ASSERT((signalData == CANRX_AEROSOL_SENSOR_STATUS_NORMAL) || (signalData == CANRX_AEROSOL_SENSOR_STATUS_ALARM));
150
151 switch (signalData) {
152 case CANRX_AEROSOL_SENSOR_STATUS_NORMAL: /* set status to normal */
153 kpkCanShim->pTableAerosolSensor->sensorStatus = 0u;
154 break;
155 case CANRX_AEROSOL_SENSOR_STATUS_ALARM: /* set status to alarm, update diag */
156 kpkCanShim->pTableAerosolSensor->sensorStatus = 1u;
157 (void)DIAG_Handler(DIAG_ID_AEROSOL_ALERT, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
158 break;
159 default: /* LCOV_EXCL_LINE */
160 /* undefined value */ /* LCOV_EXCL_LINE */
161 break; /* LCOV_EXCL_LINE */
162 }
163}

180/*========== Extern Function Implementations ================================*/
181extern uint32_t CANRX_AerosolSensor(
182 CAN_MESSAGE_PROPERTIES_s message,
183 const uint8_t *const kpkCanData,
184 const CAN_SHIM_s *const kpkCanShim) {
185 FAS_ASSERT(message.id == CANRX_BAS_AEROSOL_SENSOR_ID);
186 FAS_ASSERT(message.idType == CANRX_BAS_AEROSOL_SENSOR_ID_TYPE);
187 FAS_ASSERT(message.dlc == CAN_DEFAULT_DLC);
188 FAS_ASSERT(kpkCanData != NULL_PTR);
189 FAS_ASSERT(kpkCanShim != NULL_PTR);
190
191 uint64_t messageData = 0u;
192 uint64_t canSignal = 0u;
193
194 CAN_RxGetMessageDataFromCanData(&messageData, kpkCanData, message.endianness);
195
196 /* Handle error flags */
197 CAN_RxGetSignalDataFromMessageData(
198 messageData,
199 CANRX_AEROSOL_SENSOR_FAULTS_START_BIT,
200 CANRX_AEROSOL_SENSOR_FAULTS_LENGTH,
201 &canSignal,
202 message.endianness);
203 uint16_t signalData = (uint16_t)canSignal;
204 CANRX_HandleAerosolSensorErrors(kpkCanShim, signalData);
205
206 /* Set sensor status */
207 CAN_RxGetSignalDataFromMessageData(
208 messageData,
209 CANRX_AEROSOL_SENSOR_STATUS_START_BIT,
210 CANRX_AEROSOL_SENSOR_STATUS_LENGTH,
211 &canSignal,
212 message.endianness);
213 signalData = (uint16_t)canSignal;
214 CANRX_HandleAerosolSensorStatus(kpkCanShim, signalData);
215
216 /* Set data */
217 CAN_RxGetSignalDataFromMessageData(
218 messageData,
219 CANRX_PARTICULAR_MATTER_CONCENTRATION_START_BIT,
220 CANRX_PARTICULAR_MATTER_CONCENTRATION_LENGTH,
221 &canSignal,
222 message.endianness);
223 signalData = (uint16_t)canSignal;
224 CANRX_SetParticulateMatterConcentration(kpkCanShim, signalData);
225
226 /* Set CRC check code*/
227 CAN_RxGetSignalDataFromMessageData(
228 messageData,
229 CANRX_AEROSOL_SENSOR_CRC_CHECK_CODE_START_BIT,
230 CANRX_AEROSOL_SENSOR_CRC_CHECK_CODE_LENGTH,
231 &canSignal,
232 message.endianness);
233 signalData = (uint16_t)canSignal;
234 CANRX_SetAerosolSensorCrcCheckCode(kpkCanShim, signalData);
235
236 (void)DATA_WRITE_DATA(kpkCanShim->pTableAerosolSensor);
237 return 0u;
238}
