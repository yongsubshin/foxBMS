136static void CANRX_ClearAllPersistentFlags(uint64_t messageData) {
137 /* AXIVION Routine Generic-MissingParameterAssert: messageData: parameter accepts whole range */
138 uint64_t signalData = 0u;
139 CAN_RxGetSignalDataFromMessageData(
140 messageData,
141 CANRX_STATE_REQUEST_RESET_PERSISTENT_FLAGS_START_BIT,
142 CANRX_STATE_REQUEST_RESET_PERSISTENT_FLAGS_LENGTH,
143 &signalData,
144 CANRX_BMS_STATE_REQUEST_ENDIANNESS);
145
146 if (signalData == 1u) {
147 /* clear deep discharge */
148 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
149 (void)DIAG_Handler(DIAG_ID_DEEP_DISCHARGE_DETECTED, DIAG_EVENT_OK, DIAG_STRING, s);
150 }
151 /* clear sys mon */
152 SYSM_ClearAllTimingViolations();
153 }
154}

156static void CANRX_HandleModeRequest(uint64_t messageData, const CAN_SHIM_s *const kpkCanShim) {
157 FAS_ASSERT(kpkCanShim != NULL_PTR);
158 /* AXIVION Routine Generic-MissingParameterAssert: messageData: parameter accepts whole range */
159 uint64_t signalData = 0u;
160 CAN_RxGetSignalDataFromMessageData(
161 messageData,
162 CANRX_STATE_REQUEST_REQUEST_BMS_MODE_START_BIT,
163 CANRX_STATE_REQUEST_REQUEST_BMS_MODE_LENGTH,
164 &signalData,
165 CANRX_BMS_STATE_REQUEST_ENDIANNESS);
166
167 /** 0x00: Disconnect strings from HV bus
168 * 0x01: Connect strings to HV bus to start discharge
169 * 0x02: Connect strings to HV bus to start charging
170 */
171 uint8_t stateRequest = BMS_REQ_ID_NOREQ;
172
173 switch (signalData) {
174 case 0u:
175 stateRequest = BMS_REQ_ID_STANDBY;
176 break;
177 case 1u:
178 stateRequest = BMS_REQ_ID_NORMAL;
179 break;
180 case 2u:
181 stateRequest = BMS_REQ_ID_CHARGE;
182 break;
183 default:
184 /* default value already set in initialization */
185 break;
186 }
187 kpkCanShim->pTableStateRequest->previousStateRequestViaCan = kpkCanShim->pTableStateRequest->stateRequestViaCan;
188 kpkCanShim->pTableStateRequest->stateRequestViaCan = stateRequest;
189 if ((kpkCanShim->pTableStateRequest->stateRequestViaCan !=
190 kpkCanShim->pTableStateRequest->previousStateRequestViaCan) ||
191 (OS_CheckTimeHasPassed(kpkCanShim->pTableStateRequest->header.timestamp, CANRX_CAN_REQUEST_UPDATE_TIME_ms))) {
192 kpkCanShim->pTableStateRequest->stateRequestViaCanPending = stateRequest;
193 }
194 if (kpkCanShim->pTableStateRequest->stateCounter == (uint8_t)UINT8_MAX) {
195 /* overflow of state counter */
196 kpkCanShim->pTableStateRequest->stateCounter = 0u;
197 } else {
198 kpkCanShim->pTableStateRequest->stateCounter++;
199 }
200}

202static void CANRX_HandleBalancingRequest(uint64_t messageData) {
203 /* AXIVION Routine Generic-MissingParameterAssert: messageData: parameter accepts whole range */
204 uint64_t signalData = 0u;
205 CAN_RxGetSignalDataFromMessageData(
206 messageData,
207 CANRX_STATE_REQUEST_ACTIVATE_BALANCING_START_BIT,
208 CANRX_STATE_REQUEST_ACTIVATE_BALANCING_LENGTH,
209 &signalData,
210 CANRX_BMS_STATE_REQUEST_ENDIANNESS);
211
212 /* AXIVION Next Codeline Style MisraC2012-2.2 MisraC2012-14.3: Depending on implementation STD_NOT_OK might be
213 * returned. */
214 if (BAL_GetInitializationState() == STD_OK) {
215 if (signalData == 0u) {
216 BAL_SetStateRequest(BAL_STATE_GLOBAL_DISABLE_REQUEST);
217 } else {
218 BAL_SetStateRequest(BAL_STATE_GLOBAL_ENABLE_REQUEST);
219 }
220 }

223static void CANRX_SetBalancingThreshold(uint64_t messageData) {
224 /* AXIVION Routine Generic-MissingParameterAssert: messageData: parameter accepts whole range */
225 uint64_t signalData = 0u;
226 CAN_RxGetSignalDataFromMessageData(
227 messageData,
228 CANRX_STATE_REQUEST_SET_BALANCING_THRESHOLD_START_BIT,
229 CANRX_STATE_REQUEST_SET_BALANCING_THRESHOLD_LENGTH,
230 &signalData,
231 CANRX_BMS_STATE_REQUEST_ENDIANNESS);
232
233 /* cap signal data to UINT16_MAX */
234 int32_t cappedSignalData = (int32_t)signalData;
235 if (signalData > (uint64_t)UINT16_MAX) {
236 cappedSignalData = (uint64_t)UINT16_MAX;
237 }
238 BAL_SetBalancingThreshold(cappedSignalData);
239}

241/*========== Extern Function Implementations ================================*/
242extern uint32_t CANRX_BmsStateRequest(
243 CAN_MESSAGE_PROPERTIES_s message,
244 const uint8_t *const kpkCanData,
245 const CAN_SHIM_s *const kpkCanShim) {
246 FAS_ASSERT(message.id == CANRX_BMS_STATE_REQUEST_ID);
247 FAS_ASSERT(message.idType == CANRX_BMS_STATE_REQUEST_ID_TYPE);
248 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
249 FAS_ASSERT(message.endianness == CANRX_BMS_STATE_REQUEST_ENDIANNESS);
250 FAS_ASSERT(kpkCanData != NULL_PTR);
251 FAS_ASSERT(kpkCanShim != NULL_PTR);
252 (void)DATA_READ_DATA(kpkCanShim->pTableStateRequest);
253
254 uint64_t messageData = 0u;
255 CAN_RxGetMessageDataFromCanData(&messageData, kpkCanData, CANRX_BMS_STATE_REQUEST_ENDIANNESS);
256
257 /* Get mode request */
258 CANRX_HandleModeRequest(messageData, kpkCanShim);
259
260 /* check for reset flag */
261 CANRX_ClearAllPersistentFlags(messageData);
262
263 /* Get balancing request */
264 CANRX_HandleBalancingRequest(messageData);
265
266 /* Get balancing threshold */
267 CANRX_SetBalancingThreshold(messageData);
268
269 /* TODO: Implement missing signals */
270
271 (void)DATA_WRITE_DATA(kpkCanShim->pTableStateRequest);
272
273 return 0u;
274}
