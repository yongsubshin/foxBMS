95/*========== Static Function Implementations ================================*/
96static void CANRX_GetCanAfeCellVoltagesFromMessage(
97 CAN_CAN2AFE_CELL_VOLTAGES_QUEUE_s *pCellVoltages,
98 uint64_t messageData) {
99 /* CAN signal parameters for can cell voltages */
100 static const uint8_t canrx_kCanCellVoltageMuxBitStart = CANRX_CAN_CELL_VOLTAGE_MUX_BIT_START;
101 static const uint8_t canrx_kCanCellVoltageMuxLength = CANRX_CAN_CELL_VOLTAGE_MUX_LENGTH;
102 static const uint8_t canrx_kCanCellVoltageInvalidFlagBitStart[CAN_NUM_OF_VOLTAGES_IN_CAN_CELL_VOLTAGES_MSG] = {
103 CANRX_CAN_CELL_VOLTAGE0_INVALID_FLAG_BIT_START,
104 CANRX_CAN_CELL_VOLTAGE1_INVALID_FLAG_BIT_START,
105 CANRX_CAN_CELL_VOLTAGE2_INVALID_FLAG_BIT_START,
106 CANRX_CAN_CELL_VOLTAGE3_INVALID_FLAG_BIT_START};
107 static const uint8_t canrx_kCanCellVoltageInvalidFlagLength = CANRX_CAN_CELL_VOLTAGE_INVALID_FLAG_LENGTH;
108 static const uint8_t canrx_kCanCellVoltageBitStart[CAN_NUM_OF_VOLTAGES_IN_CAN_CELL_VOLTAGES_MSG] = {
109 CANRX_CAN_CELL_VOLTAGE0_BIT_START,
110 CANRX_CAN_CELL_VOLTAGE1_BIT_START,
111 CANRX_CAN_CELL_VOLTAGE2_BIT_START,
112 CANRX_CAN_CELL_VOLTAGE3_BIT_START};
113 static const uint8_t canrx_kCanCellVoltageLength = CANRX_CAN_CELL_VOLTAGE_LENGTH;
114
115 /* Declare and initialize the extracted signal from message */
116 uint64_t pCanSignalMuxValue = 0u;
117 uint64_t pCanSignalInvalidFlag = 0u;
118 uint64_t pCanSignalVoltage = 0u;
119
120 /* Get the mux value */
121 CAN_RxGetSignalDataFromMessageData(
122 messageData,
123 canrx_kCanCellVoltageMuxBitStart,
124 canrx_kCanCellVoltageMuxLength,
125 &pCanSignalMuxValue,
126 CANRX_AFE_CELL_VOLTAGES_ENDIANNESS);
127 pCellVoltages->muxValue = (uint8_t)pCanSignalMuxValue;
128
129 /* Get the invalid flag */
130 for (uint8_t i = 0; i < CAN_NUM_OF_VOLTAGES_IN_CAN_CELL_VOLTAGES_MSG; i++) {
131 pCanSignalInvalidFlag = 0u;
132 CAN_RxGetSignalDataFromMessageData(
133 messageData,
134 canrx_kCanCellVoltageInvalidFlagBitStart[i],
135 canrx_kCanCellVoltageInvalidFlagLength,
136 &pCanSignalInvalidFlag,
137 CANRX_AFE_CELL_VOLTAGES_ENDIANNESS);
138 pCellVoltages->invalidFlag[i] = ((uint8_t)pCanSignalInvalidFlag > 0u);
139 }
140
141 /* Get the voltages */
142 for (uint8_t i = 0; i < CAN_NUM_OF_VOLTAGES_IN_CAN_CELL_VOLTAGES_MSG; i++) {
143 pCanSignalVoltage = 0u;
144 CAN_RxGetSignalDataFromMessageData(
145 messageData,
146 canrx_kCanCellVoltageBitStart[i],
147 canrx_kCanCellVoltageLength,
148 &pCanSignalVoltage,
149 CANRX_AFE_CELL_VOLTAGES_ENDIANNESS);
150 pCellVoltages->cellVoltage[i] = (uint16_t)pCanSignalVoltage;
151 }
152}

154/*========== Extern Function Implementations ================================*/
155extern uint32_t CANRX_CellVoltages(
156 CAN_MESSAGE_PROPERTIES_s message,
157 const uint8_t *const kpkCanData,
158 const CAN_SHIM_s *const kpkCanShim) {
159 /* Check the information of this can message */
160 FAS_ASSERT(message.id == CANRX_AFE_CELL_VOLTAGES_ID);
161 FAS_ASSERT(message.idType == CANRX_AFE_CELL_VOLTAGES_ID_TYPE);
162 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
163 FAS_ASSERT(kpkCanData != NULL_PTR);
164 FAS_ASSERT(kpkCanShim != NULL_PTR);
165
166 /* Get the message data from can data and pack it into CAN_CAN2AFE_CELL_VOLTAGES_QUEUE_s */
167 uint64_t messageData = 0u;
168 CAN_CAN2AFE_CELL_VOLTAGES_QUEUE_s canrx_canCellVoltages = {0};
169 CAN_RxGetMessageDataFromCanData(&messageData, kpkCanData, message.endianness);
170 CANRX_GetCanAfeCellVoltagesFromMessage(&canrx_canCellVoltages, messageData);
171
172 /* Write the whole temperature message to the queue*/
173 if (OS_SendToBackOfQueue(ftsk_canToAfeCellVoltagesQueue, (void *)&canrx_canCellVoltages, 0u) == OS_SUCCESS) {
174 /* queue is not full */
175 (void)DIAG_Handler(DIAG_ID_CAN_RX_QUEUE_FULL, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
176 } else {
177 /* queue is full */
178 (void)DIAG_Handler(DIAG_ID_CAN_RX_QUEUE_FULL, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
179 }
180
181 return 0u;
182}
