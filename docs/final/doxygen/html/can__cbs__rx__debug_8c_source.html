556static void CANRX_ProcessRtcMux(uint64_t messageData, CAN_ENDIANNESS_e endianness) {

558 FAS_ASSERT(endianness == CAN_BIG_ENDIAN);

560 RTC_TIME_DATA_s timeData = {0};
561

940extern uint32_t CANRX_Debug(
941 CAN_MESSAGE_PROPERTIES_s message,
942 const uint8_t *const kpkCanData,
943 const CAN_SHIM_s *const kpkCanShim) {
944 FAS_ASSERT(message.id == CANRX_DEBUG_ID);
945 FAS_ASSERT(message.idType == CANRX_DEBUG_ID_TYPE);
946 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
947 FAS_ASSERT(kpkCanData != NULL_PTR);
948 FAS_ASSERT(kpkCanShim != NULL_PTR);
949
950 uint64_t messageData = 0u;
951 uint64_t muxValue = 0u;
952
953 uint32_t retVal = 1u;
954 /* Used to ensure that we can actually send our answer and not clog the TX message boxes. */
955 if (SYS_GetSystemState(&sys_state) >= SYS_FSM_STATE_PRE_RUNNING) {
956 /* Get the message data and retrieve the multiplexer value from it */
957 CAN_RxGetMessageDataFromCanData(&messageData, kpkCanData, message.endianness);
958 CAN_RxGetSignalDataFromMessageData(
959 messageData,
960 CANRX_DEBUG_MESSAGE_MUX_START_BIT,
961 CANRX_DEBUG_MESSAGE_MUX_LENGTH,
962 &muxValue,
963 message.endianness);
964
965 /* process the respective handler function per supported multiplexer
966 value; in case of an unsupported multiplexer value, transmit that
967 information on the CAN bus. */
968 switch (muxValue) {
969 case CANRX_DEBUG_MESSAGE_MUX_VALUE_VERSION_INFORMATION:
970 CANRX_ProcessVersionInformationMux(messageData, message.endianness);
971 break;
972 case CANRX_DEBUG_MESSAGE_MUX_VALUE_RTC:
973 CANRX_ProcessRtcMux(messageData, message.endianness);
974 break;
975 case CANRX_DEBUG_MESSAGE_MUX_VALUE_SOFTWARE_RESET:
976 CANRX_ProcessSoftwareResetMux(messageData, message.endianness);
977 break;
978 case CANRX_DEBUG_MESSAGE_MUX_VALUE_FRAM_INITIALIZATION:
979 CANRX_ProcessFramInitializationMux(messageData, message.endianness);
980 break;
981 case CANRX_DEBUG_MESSAGE_MUX_VALUE_TIME_INFO:
982 CANRX_ProcessTimeInfoMux(messageData, message.endianness);
983 break;
984 case CANRX_DEBUG_MESSAGE_MUX_VALUE_UPTIME_INFO:
985 CANRX_ProcessUptimeInfoMux(messageData, message.endianness);
986 break;
987 default:
988 CANTX_DebugUnsupportedMultiplexerVal(message.id, (uint32_t)muxValue);
989 break;
990 }
991 retVal = 0u;
992 }
