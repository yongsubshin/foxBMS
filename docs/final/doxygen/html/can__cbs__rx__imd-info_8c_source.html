159/*========== Static Function Implementations ================================*/
160static void CANRX_TransferImdInfoMessageToCanBuffer(
161 uint8_t messageDlc,
162 const uint8_t *const kpkCanData,
163 CAN_BUFFER_ELEMENT_s *canBuffer) {
164 FAS_ASSERT(0u < messageDlc);
165 FAS_ASSERT(messageDlc <= CAN_MAX_DLC);
166 FAS_ASSERT(kpkCanData != NULL_PTR);
167 FAS_ASSERT(canBuffer != NULL_PTR);
168
169 /* determine dlc size of the message */
170 const uint8_t boundedDlc = MATH_MinimumOfTwoUint8_t(messageDlc, CAN_MAX_DLC);
171
172 /* transfer can data to buffer element*/
173 for (uint8_t i = 0; i < boundedDlc; i++) {
174 canBuffer->data[i] = kpkCanData[i];
175 }
176}

356/*========== Extern Function Implementations ================================*/
357extern uint32_t CANRX_ImdInfo(
358 CAN_MESSAGE_PROPERTIES_s message,
359 const uint8_t *const kpkCanData,
360 const CAN_SHIM_s *const kpkCanShim) {
361 /* This handler is only implemented for little endian */
362 FAS_ASSERT(message.endianness == CAN_LITTLE_ENDIAN);
363 FAS_ASSERT(message.id == CANRX_IMD_INFO_ID);
364 FAS_ASSERT(message.idType == CANRX_IMD_INFO_ID_TYPE);
365 FAS_ASSERT(message.dlc <= CAN_MAX_DLC); /* Currently max 8 bytes in a CAN frame */
366 FAS_ASSERT(kpkCanData != NULL_PTR);
367 FAS_ASSERT(kpkCanShim != NULL_PTR);
368 /* set up can buffer element */
369 CAN_BUFFER_ELEMENT_s canBuffer = {
370 .canNode = I165C_CAN_NODE, .id = message.id, .idType = message.idType, .data = {0u}};
371 uint32_t retVal = 1u;
372
373 /* copy message data to buffer element */
374 CANRX_TransferImdInfoMessageToCanBuffer(message.dlc, kpkCanData, &canBuffer);
375
376 /*send buffer element to the back of queue*/
377 if (OS_SendToBackOfQueue(*(kpkCanShim->pQueueImd), (void *)&canBuffer, 0u) == OS_SUCCESS) {
378 retVal = 0u;
379 }
380 return retVal;
381}

400extern bool CANRX_ImdInfoCheckMeasurementMode(const uint8_t *const kpkCanData, uint8_t mode) {
401 FAS_ASSERT(kpkCanData != NULL_PTR);
402 FAS_ASSERT((mode == I165C_ENABLE_MEASUREMENT) || (mode == I165C_DISABLE_MEASUREMENT));
403
404 uint64_t messageData = 0u;
405 uint64_t signalData = 0u;
406 bool returnValue = false;
407
408 /* get message data from can data */
409 CAN_RxGetMessageDataFromCanData(&messageData, kpkCanData, CAN_LITTLE_ENDIAN);
410 /* get current measurement mode from message data */
411 CAN_RxGetSignalDataFromMessageData(
412 messageData,
413 CANRX_IMD_INFO_VIFC_INSULATION_MEASUREMENT_STATUS_START_BIT,
414 CANRX_IMD_INFO_VIFC_INSULATION_MEASUREMENT_STATUS_LENGTH,
415 &signalData,
416 CAN_LITTLE_ENDIAN);
417
418 /* Compare input mode with actual mode from IMD */
419 uint8_t actualMeasurementMode = (uint8_t)signalData;
420 if (mode == actualMeasurementMode) {
421 returnValue = true;
422 }
423 return returnValue;
424}

427 FAS_ASSERT(kpkCanData != NULL_PTR);

459extern bool CANRX_ImdInfoIsSelfTestFinished(const uint8_t *const kpkCanData) {
460 FAS_ASSERT(kpkCanData != NULL_PTR);
461
462 uint64_t messageData = 0u;
463 bool selfTestFinished = true;
464
465 /* Get message data from can data */
466 CAN_RxGetMessageDataFromCanData(&messageData, kpkCanData, CAN_LITTLE_ENDIAN);
467
468 if (CANRX_SelfTestRunning(messageData)) {
469 selfTestFinished = false;
470 } else {
471 if (CANRX_InsulationMeasurementActive(messageData)) {
472 selfTestFinished = false;
473 } else {
474 selfTestFinished = CANRX_SelfTestExecuted(messageData);
475 }
476 }
477
478 return selfTestFinished;
479}
