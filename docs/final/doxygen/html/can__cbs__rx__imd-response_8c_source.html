93/*========== Static Function Implementations ================================*/
94static void CANRX_TransferImdResponseMessageToCanBuffer(
95 uint8_t messageDlc,
96 const uint8_t *const kpkCanData,
97 CAN_BUFFER_ELEMENT_s *canBuffer) {
98 FAS_ASSERT(0u < messageDlc);
99 FAS_ASSERT(messageDlc <= CAN_MAX_DLC);
100 FAS_ASSERT(kpkCanData != NULL_PTR);
101 FAS_ASSERT(canBuffer != NULL_PTR);
102
103 /* determine dlc size of the message */
104 const uint8_t boundedDlc = MATH_MinimumOfTwoUint8_t(messageDlc, CAN_MAX_DLC);
105
106 /* transfer can data to buffer element*/
107 for (uint8_t i = 0; i < boundedDlc; i++) {
108 canBuffer->data[i] = kpkCanData[i];
109 }
110}

112/*========== Extern Function Implementations ================================*/
113extern uint32_t CANRX_ImdResponse(
114 CAN_MESSAGE_PROPERTIES_s message,
115 const uint8_t *const kpkCanData,
116 const CAN_SHIM_s *const kpkCanShim) {
117 /* This handler is only implemented for little endian */
118 FAS_ASSERT(message.endianness == CAN_LITTLE_ENDIAN);
119 FAS_ASSERT(message.id == CANRX_IMD_RESPONSE_ID);
120 FAS_ASSERT(message.idType == CANRX_IMD_RESPONSE_ID_TYPE);
121 FAS_ASSERT(message.dlc <= CAN_MAX_DLC); /* Currently max 8 bytes in a CAN frame */
122 FAS_ASSERT(kpkCanData != NULL_PTR);
123 FAS_ASSERT(kpkCanShim != NULL_PTR);
124 /* set up can buffer element */
125 CAN_BUFFER_ELEMENT_s canBuffer = {
126 .canNode = I165C_CAN_NODE, .id = message.id, .idType = message.idType, .data = {0u}};
127 uint32_t retVal = 1u;
128
129 /* copy message data to buffer element */
130 CANRX_TransferImdResponseMessageToCanBuffer(message.dlc, kpkCanData, &canBuffer);
131
132 /*send buffer element to the back of queue*/
133 if (OS_SendToBackOfQueue(*(kpkCanShim->pQueueImd), (void *)&canBuffer, 0u) == OS_SUCCESS) {
134 retVal = 0u;
135 }
136 return retVal;
137}

160extern void CANRX_ImdResponseCheckInsulationFaultTendency(
161 const uint8_t *const kpkCanData,
162 DATA_BLOCK_INSULATION_MONITORING_s *pTableInsulationMonitoring) {
163 FAS_ASSERT(kpkCanData != NULL_PTR);
164 FAS_ASSERT(pTableInsulationMonitoring != NULL_PTR);
165
166 uint64_t messageData = 0u;
167 uint64_t signalData = 0u;
168
169 /* Get insulation fault bias */
170 CAN_RxGetMessageDataFromCanData(&messageData, kpkCanData, CAN_LITTLE_ENDIAN);
171 CAN_RxGetSignalDataFromMessageData(
172 messageData,
173 CANRX_IMD_RESPONSE_INSULATION_FAULT_BIAS_START_BIT,
174 CANRX_IMD_RESPONSE_INSULATION_FAULT_BIAS_LENGTH,
175 &signalData,
176 CAN_LITTLE_ENDIAN);
177
178 if (signalData == I165C_BIAS_TO_HV_PLUS) {
179 pTableInsulationMonitoring->dfIsChassisShortToHvPlus = true;
180 pTableInsulationMonitoring->dfIsChassisShortToHvMinus = false;
181 } else if (signalData == I165C_BIAS_TO_HV_MINUS) {
182 pTableInsulationMonitoring->dfIsChassisShortToHvMinus = true;
183 pTableInsulationMonitoring->dfIsChassisShortToHvPlus = false;
184 } else {
185 pTableInsulationMonitoring->dfIsChassisShortToHvPlus = false;
186 pTableInsulationMonitoring->dfIsChassisShortToHvMinus = false;
187 }
188}

190extern bool CANRX_ImdResponseCheckRelayState(const uint8_t *const kpkCanData, uint8_t relay, uint8_t relayState) {
191 FAS_ASSERT(kpkCanData != NULL_PTR);
192 FAS_ASSERT((relay == I165C_D_VIFC_HV_RELAIS_NEGATIVE) || (relay == I165C_D_VIFC_HV_RELAIS_POSITIVE));
193 FAS_ASSERT((relayState == I165C_RELAY_STATE_OPEN) || (relayState == I165C_RELAY_STATE_CLOSED));
194
195 uint64_t messageData = 0u;
196 uint64_t signalData = 0u;
197 bool checkSuccess = true;
198
199 /* Get relay state from response message */
200 CAN_RxGetMessageDataFromCanData(&messageData, kpkCanData, CAN_LITTLE_ENDIAN);
201
202 /* Check if relay matches */
203 CAN_RxGetSignalDataFromMessageData(
204 messageData,
205 CANRX_IMD_RESPONSE_HV_RELAIS_START_BIT,
206 CANRX_IMD_RESPONSE_HV_RELAIS_LENGTH,
207 &signalData,
208 CAN_LITTLE_ENDIAN);
209 if (relay != signalData) {
210 checkSuccess = false;
211 }
212
213 /* Check if relay state matches */
214 CAN_RxGetSignalDataFromMessageData(
215 messageData,
216 CANRX_IMD_RESPONSE_HV_RELAIS_STATE_START_BIT,
217 CANRX_IMD_RESPONSE_HV_RELAIS_STATE_LENGTH,
218 &signalData,
219 CAN_LITTLE_ENDIAN);
220 if (relayState != signalData) {
221 checkSuccess = false;
222 }
223
224 return checkSuccess;
225}
