172/**
173 * @brief Helper function for CANTX_CellVoltages()
174 * @details Used in the CANTX_CellVoltages() callback to set invalid flag data
175 * and voltage data in the CAN frame.
176 * @param[in] cellId ID of th cell
177 * @param[in] pMessage pointer to CAN frame data
178 * @param[in] cellVoltageSignal signal characteristics for voltage
179 * data
180 * @param[in] cellVoltageInvalidFlagSignal signal characteristics for invalid
181 * flag data
182 * @param[in] endianness big or little endianness of data
183 * @param[in] kpkCanShim shim to the database entries
184 */
185static void CANTX_VoltageSetData(
186 uint16_t cellId,
187 uint64_t *pMessage,
188 CAN_SIGNAL_TYPE_s cellVoltageSignal,
189 CAN_SIGNAL_TYPE_s cellVoltageInvalidFlagSignal,
190 CAN_ENDIANNESS_e endianness,
191 const CAN_SHIM_s *const kpkCanShim);

193/*========== Static Function Implementations ================================*/
194static void CANTX_VoltageSetData(
195 uint16_t cellId,
196 uint64_t *pMessage,
197 CAN_SIGNAL_TYPE_s cellVoltageSignal,
198 CAN_SIGNAL_TYPE_s cellVoltageInvalidFlagSignal,
199 CAN_ENDIANNESS_e endianness,
200 const CAN_SHIM_s *const kpkCanShim) {
201 FAS_ASSERT(cellId < BS_NR_OF_CELL_BLOCKS);
202 FAS_ASSERT(pMessage != NULL_PTR);
203 /* AXIVION Routine Generic-MissingParameterAssert: cellVoltageSignal: parameter accepts whole range */
204 /* AXIVION Routine Generic-MissingParameterAssert: cellVoltageInvalidFlagSignal: parameter accepts whole range */
205 FAS_ASSERT(endianness == CAN_BIG_ENDIAN);
206 FAS_ASSERT(kpkCanShim != NULL_PTR);
207 /* Get string, module and cell number */
208 const uint8_t stringNumber = DATA_GetStringNumberFromVoltageIndex(cellId);
209 const uint8_t moduleNumber = DATA_GetModuleNumberFromVoltageIndex(cellId);
210 const uint8_t cellBlockNumber = DATA_GetCellNumberFromVoltageIndex(cellId);
211
212 uint32_t signalDataIsValid = 0u;
213 /* Valid bits data */
214 if (kpkCanShim->pTableCellVoltage->invalidCellVoltage[stringNumber][moduleNumber][cellBlockNumber] == false) {
215 signalDataIsValid = 1u;
216 } else {
217 signalDataIsValid = 0u;
218 }
219 /* Set valid bit data in CAN frame */
220 CAN_TxSetMessageDataWithSignalData(
221 pMessage,
222 cellVoltageInvalidFlagSignal.bitStart,
223 cellVoltageInvalidFlagSignal.bitLength,
224 signalDataIsValid,
225 endianness);
226
227 /*Voltage data */
228 float_t signalData_mV =
229 (float_t)(kpkCanShim->pTableCellVoltage->cellVoltage_mV[stringNumber][moduleNumber][cellBlockNumber]);
230 /* Apply offset and factor */
231 CAN_TxPrepareSignalData(&signalData_mV, cellVoltageSignal);
232 /* Set voltage data in CAN frame */
233 CAN_TxSetMessageDataWithSignalData(
234 pMessage, cellVoltageSignal.bitStart, cellVoltageSignal.bitLength, (uint32_t)signalData_mV, endianness);
235}

237/*========== Extern Function Implementations ================================*/
238extern uint32_t CANTX_CellVoltages(
239 CAN_MESSAGE_PROPERTIES_s message,
240 uint8_t *pCanData,
241 uint8_t *pMuxId,
242 const CAN_SHIM_s *const kpkCanShim) {
243 FAS_ASSERT(message.id == CANTX_CELL_VOLTAGES_ID);
244 FAS_ASSERT(message.idType == CANTX_CELL_VOLTAGES_ID_TYPE);
245 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
246 FAS_ASSERT(message.endianness == CAN_BIG_ENDIAN);
247 FAS_ASSERT(pCanData != NULL_PTR);
248 FAS_ASSERT(pMuxId != NULL_PTR);
249 FAS_ASSERT(kpkCanShim != NULL_PTR);
250 uint64_t messageData = 0u;
251
252 /* Reset mux if maximum was reached */
253 if (*pMuxId >= CANTX_NUMBER_OF_CAN_MESSAGES_FOR_CELL_VOLTAGES) {
254 *pMuxId = 0u;
255 /* First signal to transmit cell voltages: get database values */
256 (void)DATA_READ_DATA(kpkCanShim->pTableCellVoltage);
257 }
258
259 /* Set mux signal in CAN frame */
260 CAN_TxSetMessageDataWithSignalData(
261 &messageData,
262 cantx_cellVoltageMultiplexer.bitStart,
263 cantx_cellVoltageMultiplexer.bitLength,
264 (uint64_t)*pMuxId,
265 message.endianness);
266
267 /* Set other signals in CAN frame */
268 /* Calculate the global cell ID based on the multiplexer value for the first cell */
269 uint16_t cellId = (*pMuxId * CANTX_NUMBER_OF_MUX_VOLTAGES_PER_MESSAGE);
270 CANTX_VoltageSetData(
271 cellId, &messageData, cantx_cellVoltage0_mV, cantx_cellVoltage0InvalidFlag, message.endianness, kpkCanShim);
272 cellId++; /* Increment global cell ID */
273 if (cellId < BS_NR_OF_CELL_BLOCKS) {
274 CANTX_VoltageSetData(
275 cellId, &messageData, cantx_cellVoltage1_mV, cantx_cellVoltage1InvalidFlag, message.endianness, kpkCanShim);
276 cellId++; /* Increment global cell ID */
277 if (cellId < BS_NR_OF_CELL_BLOCKS) {
278 CANTX_VoltageSetData(
279 cellId,
280 &messageData,
281 cantx_cellVoltage2_mV,
282 cantx_cellVoltage2InvalidFlag,
283 message.endianness,
284 kpkCanShim);
285 cellId++; /* Increment global cell ID */
286 if (cellId < BS_NR_OF_CELL_BLOCKS) {
287 CANTX_VoltageSetData(
288 cellId,
289 &messageData,
290 cantx_cellVoltage3_mV,
291 cantx_cellVoltage3InvalidFlag,
292 message.endianness,
293 kpkCanShim);
294 }
295 }
296 }
297
298 /* Increment multiplexer for next cell */
299 (*pMuxId)++;
300
301 /* All signal data copied in CAN frame, now copy data in the buffer that will be use to send the frame */
302 CAN_TxSetCanDataWithMessageData(messageData, pCanData, message.endianness);
303
304 return 0u;
305}
