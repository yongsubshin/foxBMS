88extern void CANTX_CrashDump(CANTX_FATAL_ERRORS_ACTIONS_e action) {
89 FAS_ASSERT(action < CANTX_FATAL_ERRORS_ACTIONS_MAX_E);
90 uint8_t data[CANTX_CRASH_DUMP_DLC] = {GEN_REPEAT_U(0u, GEN_STRIP(CAN_MAX_DLC))};
91 uint64_t messageData = 0u;
92
93 switch (action) {
94 case CANTX_FATAL_ERRORS_ACTIONS_STACK_OVERFLOW:
95 CAN_TxSetMessageDataWithSignalData(
96 &messageData,
97 CANTX_CRASH_DUMP_MESSAGE_MUX_START_BIT,
98 CANTX_CRASH_DUMP_MESSAGE_MUX_LENGTH,
99 CANTX_CRASH_DUMP_MESSAGE_MUX_VALUE_STACK_OVERFLOW,
100 CANTX_CRASH_DUMP_ENDIANNESS);
101 break;
102 default: /* LCOV_EXCL_LINE */
103 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
104 break; /* LCOV_EXCL_LINE */
105 }
106
107 /* set can data */
108 CAN_TxSetCanDataWithMessageData(messageData, data, CANTX_CRASH_DUMP_ENDIANNESS);
109 /* send message */
110 (void)CAN_DataSend(CAN_NODE_DEBUG_MESSAGE, CANTX_CRASH_DUMP_ID, CANTX_CRASH_DUMP_ID_TYPE, data);
111}
