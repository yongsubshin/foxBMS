150static STD_RETURN_TYPE_e CANTX_RequestRelayOpen(uint8_t relay) {
151 FAS_ASSERT((relay == I165C_D_VIFC_HV_RELAIS_POSITIVE) || (relay == I165C_D_VIFC_HV_RELAIS_NEGATIVE));
152 uint8_t data[] = {GEN_REPEAT_U(0u, GEN_STRIP(CAN_MAX_DLC))};
153
154 uint64_t message = 0u;
155 CAN_TxSetMessageDataWithSignalData(
156 &message,
157 CANTX_IMD_REQUEST_MUX_START_BIT,
158 CANTX_IMD_REQUEST_MUX_LENGTH,
159 I165C_CMD_S_VIFC_SET_HV_RELAIS,
160 CAN_LITTLE_ENDIAN);
161 CAN_TxSetMessageDataWithSignalData(
162 &message,
163 CANTX_IMD_REQUEST_DATA_WORD_1_START_BIT,
164 CANTX_IMD_REQUEST_DATA_WORD_1_LENGTH,
165 relay,
166 CAN_LITTLE_ENDIAN);
167 CAN_TxSetMessageDataWithSignalData(
168 &message,
169 CANTX_IMD_REQUEST_DATA_WORD_2_START_BIT,
170 CANTX_IMD_REQUEST_DATA_WORD_2_LENGTH,
171 I165C_RELAY_STATE_OPEN,
172 CAN_LITTLE_ENDIAN);
173
174 CAN_TxSetCanDataWithMessageData(message, &data[0], CAN_LITTLE_ENDIAN);
175 return CAN_DataSend(I165C_CAN_NODE, I165C_MESSAGE_TYPE_IMD_REQUEST, I165C_TX_MESSAGE_IDENTIFIER_TYPE, &data[0]);
176}

406/*========== Extern Function Implementations ================================*/
407extern STD_RETURN_TYPE_e CANTX_ImdRequest(CANTX_IMD_REQUEST_ACTIONS_e action) {
408 FAS_ASSERT(action < CANTX_IMD_REQUEST_LAST_ACTION);
409 STD_RETURN_TYPE_e successfullyQueued = STD_NOT_OK;
410 switch (action) {
411 case CANTX_IMD_REQUEST_OPEN_POSITIVE_RELAY:
412 successfullyQueued = CANTX_RequestRelayOpen(I165C_D_VIFC_HV_RELAIS_POSITIVE);
413 break;
414 case CANTX_IMD_REQUEST_OPEN_NEGATIVE_RELAY:
415 successfullyQueued = CANTX_RequestRelayOpen(I165C_D_VIFC_HV_RELAIS_NEGATIVE);
416 break;
417 case CANTX_IMD_REQUEST_CLOSE_POSITIVE_RELAY:
418 successfullyQueued = CANTX_RequestRelayClose(I165C_D_VIFC_HV_RELAIS_POSITIVE);
419 break;
420 case CANTX_IMD_REQUEST_CLOSE_NEGATIVE_RELAY:
421 successfullyQueued = CANTX_RequestRelayClose(I165C_D_VIFC_HV_RELAIS_NEGATIVE);
422 break;
423 case CANTX_IMD_REQUEST_POSITIVE_RELAY_STATE:
424 successfullyQueued = CANTX_RequestRelayState(I165C_D_VIFC_HV_RELAIS_POSITIVE);
425 break;
426 case CANTX_IMD_REQUEST_NEGATIVE_RELAY_STATE:
427 successfullyQueued = CANTX_RequestRelayState(I165C_D_VIFC_HV_RELAIS_NEGATIVE);
428 break;
429 case CANTX_IMD_REQUEST_ENABLE_MEASUREMENT:
430 successfullyQueued = CANTX_RequestEnableMeasurement();
431 break;
432 case CANTX_IMD_REQUEST_DISABLE_MEASUREMENT:
433 successfullyQueued = CANTX_RequestDisableMeasurement();
434 break;
435 case CANTX_IMD_REQUEST_SET_AVERAGING_FACTOR:
436 successfullyQueued = CANTX_SetAveragingFactor();
437 break;
438 case CANTX_IMD_REQUEST_READ_RESISTANCE:
439 successfullyQueued = CANTX_RequestReadResistance();
440 break;
441 case CANTX_IMD_REQUEST_INITIALIZATION_UNLOCK:
442 successfullyQueued = CANTX_RequestUnlock();
443 break;
444 case CANTX_IMD_REQUEST_INITIALIZATION_SELF_TEST:
445 successfullyQueued = CANTX_RequestSelfTest();
446 break;
447 case CANTX_IMD_REQUEST_INITIALIZATION_SET_ERROR_THRESHOLD:
448 successfullyQueued = CANTX_SetErrorThreshold();
449 break;
450 case CANTX_IMD_REQUEST_INITIALIZATION_SET_WARNING_THRESHOLD:
451 successfullyQueued = CANTX_SetWarningThreshold();
452 break;
453 default:
454 FAS_ASSERT(FAS_TRAP);
455 break; /* LCOV_EXCL_LINE */
456 }
457 return successfullyQueued;
458}
