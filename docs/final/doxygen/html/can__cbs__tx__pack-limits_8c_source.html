252static uint64_t CANTX_CalculateMaximumDischargeCurrent(const CAN_SHIM_s *const kpkCanShim) {
253 FAS_ASSERT(kpkCanShim != NULL_PTR);
254
255 /* maximum discharge current */
256 float_t signalData = (float_t)kpkCanShim->pTableSof->recommendedContinuousPackDischargeCurrent_mA;
257 CAN_TxPrepareSignalData(&signalData, cantx_signalMaximumDischargeCurrent);
258 uint64_t data = (uint64_t)signalData;
259 return data;
260}

272static uint64_t CANTX_CalculateMaximumDischargePower(const CAN_SHIM_s *const kpkCanShim) {
273 FAS_ASSERT(kpkCanShim != NULL_PTR);
274
275 /* maximum charge power = discharge_current_A * battery_voltage_V */
276 float_t signalData =
277 ((float_t)kpkCanShim->pTableSof->recommendedContinuousPackDischargeCurrent_mA *
278 UNIT_CONVERSION_FACTOR_1_1000_TH_FLOAT) *
279 ((float_t)kpkCanShim->pTablePackValues->batteryVoltage_mV * UNIT_CONVERSION_FACTOR_1_1000_TH_FLOAT);
280 CAN_TxPrepareSignalData(&signalData, cantx_signalMaximumDischargePower);
281 uint64_t data = (uint64_t)signalData;
282 return data;
283}

314static void CANTX_BuildPackLimitsMessage(uint64_t *pMessageData, const CAN_SHIM_s *const kpkCanShim) {
315 FAS_ASSERT(pMessageData != NULL_PTR);
316 FAS_ASSERT(kpkCanShim != NULL_PTR);
317
318 /* Maximum discharge current */
319 uint64_t data = CANTX_CalculateMaximumDischargeCurrent(kpkCanShim);
320 CAN_TxSetMessageDataWithSignalData(
321 pMessageData,
322 cantx_signalMaximumDischargeCurrent.bitStart,
323 cantx_signalMaximumDischargeCurrent.bitLength,
324 data,
325 CAN_BIG_ENDIAN);
326 /* Maximum charge current */
327 data = CANTX_CalculateMaximumChargeCurrent(kpkCanShim);
328 CAN_TxSetMessageDataWithSignalData(
329 pMessageData,
330 cantx_signalMaximumChargeCurrent.bitStart,
331 cantx_signalMaximumChargeCurrent.bitLength,
332 data,
333 CAN_BIG_ENDIAN);
334 /* Maximum discharge power */
335 data = CANTX_CalculateMaximumDischargePower(kpkCanShim);
336 CAN_TxSetMessageDataWithSignalData(
337 pMessageData,
338 cantx_signalMaximumDischargePower.bitStart,
339 cantx_signalMaximumDischargePower.bitLength,
340 data,
341 CAN_BIG_ENDIAN);
342 /* Maximum charge power */
343 data = CANTX_CalculateMaximumChargePower(kpkCanShim);
344 CAN_TxSetMessageDataWithSignalData(
345 pMessageData,
346 cantx_signalMaximumChargePower.bitStart,
347 cantx_signalMaximumChargePower.bitLength,
348 data,
349 CAN_BIG_ENDIAN);
350 /* Minimum battery voltage */
351 data = CANTX_CalculateMinimumBatteryVoltage();
352 CAN_TxSetMessageDataWithSignalData(
353 pMessageData,
354 cantx_signalMinimumBatteryVoltage.bitStart,
355 cantx_signalMinimumBatteryVoltage.bitLength,
356 data,
357 CAN_BIG_ENDIAN);
358 /* Maximum battery voltage */
359 data = CANTX_CalculateMaximumBatteryVoltage();
360 CAN_TxSetMessageDataWithSignalData(
361 pMessageData,
362 cantx_signalMaximumBatteryVoltage.bitStart,
363 cantx_signalMaximumBatteryVoltage.bitLength,
364 data,
365 CAN_BIG_ENDIAN);
366}

368/*========== Extern Function Implementations ================================*/
369extern uint32_t CANTX_PackLimits(
370 CAN_MESSAGE_PROPERTIES_s message,
371 uint8_t *pCanData,
372 uint8_t *pMuxId,
373 const CAN_SHIM_s *const kpkCanShim) {
374 /* pMuxId is not used here, therefore has to be NULL_PTR */
375 FAS_ASSERT(pMuxId == NULL_PTR);
376 FAS_ASSERT(message.id == CANTX_PACK_LIMITS_ID);
377 FAS_ASSERT(message.idType == CANTX_PACK_LIMITS_ID_TYPE);
378 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
379 FAS_ASSERT(message.endianness == CANTX_PACK_LIMITS_ENDIANNESS);
380 FAS_ASSERT(pCanData != NULL_PTR);
381 FAS_ASSERT(kpkCanShim != NULL_PTR);
382 uint64_t messageData = 0u;
383
384 (void)DATA_READ_DATA(kpkCanShim->pTableSof, can_kShim.pTablePackValues);
385
386 CANTX_BuildPackLimitsMessage(&messageData, kpkCanShim);
387
388 /* now copy data in the buffer that will be used to send data */
389 CAN_TxSetCanDataWithMessageData(messageData, pCanData, message.endianness);
390
391 return 0u;
392}
