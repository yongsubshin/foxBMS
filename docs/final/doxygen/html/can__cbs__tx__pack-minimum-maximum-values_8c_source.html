176/*========== Static Function Implementations ================================*/
177static int16_t CANTX_GetPackMaximumVoltage(const CAN_SHIM_s *const kpkCanShim) {
178 FAS_ASSERT(kpkCanShim != NULL_PTR);
179
180 int16_t packMaximumVoltage_mV = INT16_MIN;
181 if (0u == BMS_GetNumberOfConnectedStrings()) {
182 /* Calculate min/max values of complete pack if all slice switches are open */
183 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
184 if (kpkCanShim->pTableMinMax->maximumCellVoltage_mV[s] >= packMaximumVoltage_mV) {
185 packMaximumVoltage_mV = kpkCanShim->pTableMinMax->maximumCellVoltage_mV[s];
186 }
187 }
188 } else {
189 /* Calculate min/max values of connected slices */
190 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
191 if (BMS_IsStringClosed(s) == true) {
192 if (kpkCanShim->pTableMinMax->maximumCellVoltage_mV[s] >= packMaximumVoltage_mV) {
193 packMaximumVoltage_mV = kpkCanShim->pTableMinMax->maximumCellVoltage_mV[s];
194 }
195 }
196 }
197 }
198 return packMaximumVoltage_mV;
199}

293static void CANTX_BuildPackMinimumMaximumValuesMessage(const CAN_SHIM_s *const kpkCanShim, uint64_t *pMessageData) {
294 FAS_ASSERT(kpkCanShim != NULL_PTR);
295 FAS_ASSERT(pMessageData != NULL_PTR);
296
297 /* maximum cell voltage */
298 uint64_t signalData = (int64_t)CANTX_GetPackMaximumVoltage(kpkCanShim);
299 CAN_TxSetMessageDataWithSignalData(
300 pMessageData,
301 CANTX_SIGNAL_MAXIMUM_CELL_VOLTAGE_START_BIT,
302 CANTX_SIGNAL_MAXIMUM_CELL_VOLTAGE_LENGTH,
303 signalData,
304 CAN_BIG_ENDIAN);
305 /* minimum cell voltage */
306 signalData = (int64_t)CANTX_GetPackMinimumVoltage(kpkCanShim);
307 CAN_TxSetMessageDataWithSignalData(
308 pMessageData,
309 CANTX_SIGNAL_MINIMUM_CELL_VOLTAGE_START_BIT,
310 CANTX_SIGNAL_MINIMUM_CELL_VOLTAGE_LENGTH,
311 signalData,
312 CAN_BIG_ENDIAN);
313
314 /* TODO: cooling inlet temperature*/
315 /* TODO: cooling outlet temperature */
316
317 /* maximum temperature*/
318 signalData = CANTX_CalculatePackMaximumTemperature(kpkCanShim);
319 CAN_TxSetMessageDataWithSignalData(
320 pMessageData,
321 cantx_signalMaximumCellTemperature.bitStart,
322 cantx_signalMaximumCellTemperature.bitLength,
323 signalData,
324 CAN_BIG_ENDIAN);
325
326 /* minimum temperature*/
327 signalData = CANTX_CalculatePackMinimumTemperature(kpkCanShim);
328 CAN_TxSetMessageDataWithSignalData(
329 pMessageData,
330 cantx_signalMinimumCellTemperature.bitStart,
331 cantx_signalMinimumCellTemperature.bitLength,
332 signalData,
333 CAN_BIG_ENDIAN);
334}

336/*========== Extern Function Implementations ================================*/
337extern uint32_t CANTX_PackMinimumMaximumValues(
338 CAN_MESSAGE_PROPERTIES_s message,
339 uint8_t *pCanData,
340 uint8_t *pMuxId,
341 const CAN_SHIM_s *const kpkCanShim) {
342 FAS_ASSERT(message.id == CANTX_PACK_MINIMUM_MAXIMUM_VALUES_ID);
343 FAS_ASSERT(message.idType == CANTX_PACK_MINIMUM_MAXIMUM_VALUES_ID_TYPE);
344 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
345 FAS_ASSERT(message.endianness == CANTX_PACK_MINIMUM_MAXIMUM_VALUES_ENDIANNESS);
346 FAS_ASSERT(pCanData != NULL_PTR);
347 FAS_ASSERT(pMuxId == NULL_PTR); /* pMuxId is not used here, therefore has to be NULL_PTR */
348 FAS_ASSERT(kpkCanShim != NULL_PTR);
349 uint64_t messageData = 0u;
350
351 (void)DATA_READ_DATA(kpkCanShim->pTableMinMax);
352
353 CANTX_BuildPackMinimumMaximumValuesMessage(kpkCanShim, &messageData);
354
355 /* now copy data in the buffer that will be used to send data */
356 CAN_TxSetCanDataWithMessageData(messageData, pCanData, message.endianness);
357
358 return 0u;
359}
