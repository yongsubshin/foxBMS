191/*========== Static Function Implementations ================================*/
192static uint64_t CANTX_CalculateBatteryVoltage(const CAN_SHIM_s *const kpkCanShim) {
193 FAS_ASSERT(kpkCanShim != NULL_PTR);
194
195 /* Battery voltage */
196 float_t signalData = kpkCanShim->pTablePackValues->batteryVoltage_mV;
197 CAN_TxPrepareSignalData(&signalData, cantx_signalBatteryVoltage);
198 uint64_t data = (uint64_t)signalData;
199 return data;
200}

232static void CANTX_BuildP0Message(const CAN_SHIM_s *const kpkCanShim, uint64_t *pMessageData) {
233 FAS_ASSERT(kpkCanShim != NULL_PTR);
234 FAS_ASSERT(pMessageData != NULL_PTR);
235
236 /* Battery voltage */
237 uint64_t data = CANTX_CalculateBatteryVoltage(kpkCanShim);
238 CAN_TxSetMessageDataWithSignalData(
239 pMessageData, cantx_signalBatteryVoltage.bitStart, cantx_signalBatteryVoltage.bitLength, data, CAN_BIG_ENDIAN);
240
241 /* Bus voltage */
242 data = CANTX_CalculateBusVoltage(kpkCanShim);
243 CAN_TxSetMessageDataWithSignalData(
244 pMessageData, cantx_signalBusVoltage.bitStart, cantx_signalBusVoltage.bitLength, data, CAN_BIG_ENDIAN);
245
246 /* System Power */
247 data = CANTX_CalculatePower(kpkCanShim);
248 CAN_TxSetMessageDataWithSignalData(
249 pMessageData, cantx_signalPower.bitStart, cantx_signalPower.bitLength, data, CAN_BIG_ENDIAN);
250
251 /* System current */
252 data = CANTX_CalculateCurrent(kpkCanShim);
253 CAN_TxSetMessageDataWithSignalData(
254 pMessageData, cantx_signalCurrent.bitStart, cantx_signalCurrent.bitLength, data, CAN_BIG_ENDIAN);
255}

257/*========== Extern Function Implementations ================================*/
258extern uint32_t CANTX_PackValuesP0(
259 CAN_MESSAGE_PROPERTIES_s message,
260 uint8_t *pCanData,
261 uint8_t *pMuxId,
262 const CAN_SHIM_s *const kpkCanShim) {
263 /* pMuxId is not used here, therefore has to be NULL_PTR */
264 FAS_ASSERT(pMuxId == NULL_PTR);
265
266 FAS_ASSERT(message.id == CANTX_PACK_VALUES_P0_ID);
267 FAS_ASSERT(message.idType == CANTX_PACK_VALUES_P0_ID_TYPE);
268 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
269 FAS_ASSERT(message.endianness == CAN_BIG_ENDIAN);
270 FAS_ASSERT(pCanData != NULL_PTR);
271 FAS_ASSERT(kpkCanShim != NULL_PTR);
272 uint64_t messageData = 0u;
273
274 /* Read database entry */
275 (void)DATA_READ_DATA(kpkCanShim->pTablePackValues);
276
277 /* build message from data */
278 CANTX_BuildP0Message(kpkCanShim, &messageData);
279
280 /* now copy data in the buffer that will be used to send data */
281 CAN_TxSetCanDataWithMessageData(messageData, pCanData, message.endianness);
282
283 return 0u;
284}
