107/*========== Static Function Implementations ================================*/
108static uint64_t CANTX_GetInsulationResistance(const CAN_SHIM_s *const kpkCanShim) {
109 FAS_ASSERT(kpkCanShim != NULL_PTR);
110
111 /* insulation resistance */
112 float_t signalData = (float_t)kpkCanShim->pTableInsulation->insulationResistance_kOhm;
113 CAN_TxPrepareSignalData(&signalData, cantx_insulationResistance);
114 uint64_t data = (uint64_t)signalData;
115 return data;
116}
117
118static void CANTX_SetInsulationResistance(uint64_t *pMessageData, const CAN_SHIM_s *const kpkCanShim) {
119 FAS_ASSERT(pMessageData != NULL_PTR);
120 FAS_ASSERT(kpkCanShim != NULL_PTR);
121
122 uint64_t signalData = CANTX_GetInsulationResistance(kpkCanShim);
123 CAN_TxSetMessageDataWithSignalData(
124 pMessageData,
125 cantx_insulationResistance.bitStart,
126 cantx_insulationResistance.bitLength,
127 signalData,
128 CAN_BIG_ENDIAN);
129}

131/*========== Extern Function Implementations ================================*/
132extern uint32_t CANTX_PackValuesP1(
133 CAN_MESSAGE_PROPERTIES_s message,
134 uint8_t *pCanData,
135 uint8_t *pMuxId,
136 const CAN_SHIM_s *const kpkCanShim) {
137 /* pMuxId is not used here, therefore has to be NULL_PTR */
138 FAS_ASSERT(pMuxId == NULL_PTR);
139 FAS_ASSERT(message.id == CANTX_PACK_VALUES_P1_ID);
140 FAS_ASSERT(message.idType == CANTX_PACK_VALUES_P1_ID_TYPE);
141 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
142 FAS_ASSERT(message.endianness == CANTX_PACK_VALUES_P1_ENDIANNESS);
143 FAS_ASSERT(pCanData != NULL_PTR);
144 FAS_ASSERT(kpkCanShim != NULL_PTR);
145 uint64_t messageData = 0u;
146
147 /* read database entry */
148 (void)DATA_READ_DATA(kpkCanShim->pTableInsulation);
149
150 /* build message from data */
151 CANTX_SetInsulationResistance(&messageData, kpkCanShim);
152
153 /* now copy data in the buffer that will be used to send data */
154 CAN_TxSetCanDataWithMessageData(messageData, pCanData, CAN_BIG_ENDIAN);
155
156 return 0u;
157}
