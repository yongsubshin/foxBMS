143static uint64_t CANTX_CalculateStringMaximumTemperature(uint8_t stringNumber, const CAN_SHIM_s *const kpkCanShim) {
144 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
145 FAS_ASSERT(kpkCanShim != NULL_PTR);
146
147 /* Maximum cell temperature */
148 float_t signalData = (float_t)kpkCanShim->pTableMinMax->maximumTemperature_ddegC[stringNumber];
149 CAN_TxPrepareSignalData(&signalData, cantx_signalMaximumStringCellTemperature);
150 uint64_t data = (int64_t)signalData;
151 return data;
152}

165static void CANTX_BuildStringMessage(uint64_t *pMessageData, uint8_t stringNumber, const CAN_SHIM_s *const kpkCanShim) {
166 FAS_ASSERT(pMessageData != NULL_PTR);
167 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
168 FAS_ASSERT(kpkCanShim != NULL_PTR);
169
170 /* set data in CAN frame */
171 CAN_TxSetMessageDataWithSignalData(
172 pMessageData, CANTX_STRING_MUX_START_BIT, CANTX_STRING_MUX_LENGTH, (uint64_t)stringNumber, CAN_BIG_ENDIAN);
173
174 /* maximum cell voltage */
175 uint64_t signalData = (uint64_t)kpkCanShim->pTableMinMax->maximumCellVoltage_mV[stringNumber];
176 CAN_TxSetMessageDataWithSignalData(
177 pMessageData,
178 CANTX_SIGNAL_STRING_MAXIMUM_CELL_VOLTAGE_START_BIT,
179 CANTX_SIGNAL_STRING_MAXIMUM_CELL_VOLTAGE_LENGTH,
180 signalData,
181 CAN_BIG_ENDIAN);
182
183 /* minimum cell voltage */
184 signalData = (uint64_t)kpkCanShim->pTableMinMax->minimumCellVoltage_mV[stringNumber];
185 CAN_TxSetMessageDataWithSignalData(
186 pMessageData,
187 CANTX_SIGNAL_STRING_MINIMUM_CELL_VOLTAGE_START_BIT,
188 CANTX_SIGNAL_STRING_MINIMUM_CELL_VOLTAGE_LENGTH,
189 signalData,
190 CAN_BIG_ENDIAN);
191
192 /* maximum temperature */
193 signalData = CANTX_CalculateStringMaximumTemperature(stringNumber, kpkCanShim);
194 CAN_TxSetMessageDataWithSignalData(
195 pMessageData,
196 cantx_signalMaximumStringCellTemperature.bitStart,
197 cantx_signalMaximumStringCellTemperature.bitLength,
198 signalData,
199 CAN_BIG_ENDIAN);
200
201 /* minimum temperature */
202 signalData = CANTX_CalculateStringMinimumTemperature(stringNumber, kpkCanShim);
203 CAN_TxSetMessageDataWithSignalData(
204 pMessageData,
205 cantx_signalMinimumStringCellTemperature.bitStart,
206 cantx_signalMinimumStringCellTemperature.bitLength,
207 signalData,
208 CAN_BIG_ENDIAN);
209}

211/*========== Extern Function Implementations ================================*/
212extern uint32_t CANTX_StringMinimumMaximumValues(
213 CAN_MESSAGE_PROPERTIES_s message,
214 uint8_t *pCanData,
215 uint8_t *pMuxId,
216 const CAN_SHIM_s *const kpkCanShim) {
217 FAS_ASSERT(message.id == CANTX_STRING_MINIMUM_MAXIMUM_VALUES_ID);
218 FAS_ASSERT(message.idType == CANTX_STRING_MINIMUM_MAXIMUM_VALUES_ID_TYPE);
219 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
220 FAS_ASSERT(message.endianness == CANTX_STRING_MINIMUM_MAXIMUM_VALUES_ENDIANNESS);
221 FAS_ASSERT(pCanData != NULL_PTR);
222 FAS_ASSERT(pMuxId != NULL_PTR);
223 FAS_ASSERT(*pMuxId < BS_NR_OF_STRINGS);
224 FAS_ASSERT(kpkCanShim != NULL_PTR);
225 uint64_t messageData = 0u;
226
227 /** Database entry with minimum and maximum values does not need to be read
228 * within this callback as it is already read by function
229 * #CANTX_StringMinimumMaximumValues */
230 const uint8_t stringNumber = *pMuxId;
231
232 /* adding data to the message */
233 CANTX_BuildStringMessage(&messageData, stringNumber, kpkCanShim);
234
235 /* now copy data in the buffer that will be use to send data */
236 CAN_TxSetCanDataWithMessageData(messageData, pCanData, message.endianness);
237
238 /* Increment multiplexer for next cell */
239 (*pMuxId)++;
240
241 /* Check mux value */
242 if (*pMuxId >= BS_NR_OF_STRINGS) {
243 *pMuxId = 0u;
244 }
245
246 return 0u;
247}
