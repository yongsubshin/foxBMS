264 data,
265 CANTX_STRING_STATE_ENDIANNESS);

770extern uint32_t CANTX_StringState(
771 CAN_MESSAGE_PROPERTIES_s message,
772 uint8_t *pCanData,
773 uint8_t *pMuxId,
774 const CAN_SHIM_s *const kpkCanShim) {
775 FAS_ASSERT(message.id == CANTX_STRING_STATE_ID);
776 FAS_ASSERT(message.idType == CANTX_STRING_STATE_ID_TYPE);
777 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
778 FAS_ASSERT(pCanData != NULL_PTR);
779 FAS_ASSERT(pMuxId != NULL_PTR);
780 FAS_ASSERT(*pMuxId < BS_NR_OF_STRINGS);
781 FAS_ASSERT(kpkCanShim != NULL_PTR);
782 uint64_t messageData = 0u;
783
784 const uint8_t stringNumber = *pMuxId;
785
786 /* First signal to transmit cell voltages: get database values */
787 if (stringNumber == 0u) {
788 /* Do not read pTableMsl and pTableErrorState as they already are read
789 * with a higher frequency from CAN_TxState callback */
790 (void)DATA_READ_DATA(kpkCanShim->pTableRsl, kpkCanShim->pTableMol);
791 }
792
793 /* Set mux value */
794 CAN_TxSetMessageDataWithSignalData(
795 &messageData,
796 CANTX_SIGNAL_MUX_STRING_START_BIT,
797 CANTX_SIGNAL_MUX_STRING_LENGTH,
798 stringNumber,
799 CANTX_STRING_STATE_ENDIANNESS);
800
801 /* String connected */
802 uint64_t data = CAN_ConvertBooleanToInteger(BMS_IsStringClosed(stringNumber));
803 CAN_TxSetMessageDataWithSignalData(
804 &messageData,
805 CANTX_SIGNAL_STRING_IS_STRING_CONNECTED_START_BIT,
806 CANTX_SIGNAL_STRING_IS_STRING_CONNECTED_LENGTH,
807 data,
808 CANTX_STRING_STATE_ENDIANNESS);
809
810 /* call function to set the message data */
811 CANTX_BuildStringStateMessage(kpkCanShim, &messageData, stringNumber);
812
813 /* now copy data in the buffer that will be used to send data */
814 CAN_TxSetCanDataWithMessageData(messageData, pCanData, CANTX_STRING_STATE_ENDIANNESS);
815
816 /* Increment multiplexer for next cell */
817 (*pMuxId)++;
818
819 /* Check mux value */
820 if (*pMuxId >= BS_NR_OF_STRINGS) {
821 *pMuxId = 0u;
822 }
823
824 return 0u;
825}
