161static uint64_t CANTX_CalculateStringVoltage(uint8_t stringNumber, const CAN_SHIM_s *const kpkCanShim) {
162 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
163 FAS_ASSERT(kpkCanShim != NULL_PTR);
164
165 /* String voltage */
166 float_t signalData = (float_t)kpkCanShim->pTablePackValues->stringVoltage_mV[stringNumber];
167 CAN_TxPrepareSignalData(&signalData, cantx_signalStringVoltage);
168 uint64_t data = (uint64_t)signalData;
169 return data;
170}

194static void CANTX_BuildString0Message(
195 uint64_t *pMessageData,
196 uint8_t stringNumber,
197 const CAN_SHIM_s *const kpkCanShim) {
198 FAS_ASSERT(pMessageData != NULL_PTR);
199 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
200 FAS_ASSERT(kpkCanShim != NULL_PTR);
201
202 /* mux value */
203 uint64_t signalData = stringNumber;
204 CAN_TxSetMessageDataWithSignalData(
205 pMessageData, CANTX_STRING_P0_MUX_START_BIT, CANTX_STRING_P0_MUX_LENGTH, signalData, CAN_BIG_ENDIAN);
206
207 /* String voltage */
208 signalData = CANTX_CalculateStringVoltage(stringNumber, kpkCanShim);
209 CAN_TxSetMessageDataWithSignalData(
210 pMessageData,
211 cantx_signalStringVoltage.bitStart,
212 cantx_signalStringVoltage.bitLength,
213 signalData,
214 CAN_BIG_ENDIAN);
215
216 /* String Current */
217 signalData = CANTX_CalculateStringCurrent(stringNumber, kpkCanShim);
218 CAN_TxSetMessageDataWithSignalData(
219 pMessageData,
220 cantx_signalStringCurrent.bitStart,
221 cantx_signalStringCurrent.bitLength,
222 signalData,
223 CAN_BIG_ENDIAN);
224
225 /* String Power */
226 signalData = CANTX_CalculateStringPower(stringNumber, kpkCanShim);
227 CAN_TxSetMessageDataWithSignalData(
228 pMessageData, cantx_signalStringPower.bitStart, cantx_signalStringPower.bitLength, signalData, CAN_BIG_ENDIAN);
229}

231/*========== Extern Function Implementations ================================*/
232extern uint32_t CANTX_StringValuesP0(
233 CAN_MESSAGE_PROPERTIES_s message,
234 uint8_t *pCanData,
235 uint8_t *pMuxId,
236 const CAN_SHIM_s *const kpkCanShim) {
237 FAS_ASSERT(message.id == CANTX_STRING_VALUES_P0_ID);
238 FAS_ASSERT(message.idType == CANTX_STRING_VALUES_P0_ID_TYPE);
239 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
240 FAS_ASSERT(message.endianness == CANTX_STRING_VALUES_P0_ENDIANNESS);
241 FAS_ASSERT(pCanData != NULL_PTR);
242 FAS_ASSERT(pMuxId != NULL_PTR);
243 FAS_ASSERT(*pMuxId < BS_NR_OF_STRINGS);
244 FAS_ASSERT(kpkCanShim != NULL_PTR);
245 uint64_t messageData = 0u;
246
247 const uint8_t stringNumber = *pMuxId;
248
249 /* First signal to transmit cell voltages: get database values */
250 if (stringNumber == 0u) {
251 /* Do not read pTableMsl and pTableErrorState as they already are read
252 * with a higher frequency from CANTX_BmsState callback */
253 (void)DATA_READ_DATA(kpkCanShim->pTablePackValues);
254 }
255
256 /* set message data */
257 CANTX_BuildString0Message(&messageData, stringNumber, kpkCanShim);
258
259 /* now copy data in the buffer that will be used to send data */
260 CAN_TxSetCanDataWithMessageData(messageData, pCanData, message.endianness);
261
262 /* Increment multiplexer for next cell */
263 (*pMuxId)++;
264
265 /* Check mux value */
266 if (*pMuxId >= BS_NR_OF_STRINGS) {
267 *pMuxId = 0u;
268 }
269
270 return 0u;
271}
