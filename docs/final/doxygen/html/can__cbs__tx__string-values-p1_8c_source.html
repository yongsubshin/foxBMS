102static void CANTX_SetStringEnergyCounter(
103 uint64_t *pMessageData,
104 uint8_t stringNumber,
105 const CAN_SHIM_s *const kpkCanShim) {
106 FAS_ASSERT(pMessageData != NULL_PTR);
107 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
108 FAS_ASSERT(kpkCanShim != NULL_PTR);
109
110 /* String voltage */
111 OS_EnterTaskCritical(); /* this access has to be protected as it conflicts with the 1ms task */
112 uint64_t signalData = (uint64_t)kpkCanShim->pTableCurrentSensor->energyCounter_Wh[stringNumber];
113 OS_ExitTaskCritical();
114
115 /* set data in CAN frame */
116 CAN_TxSetMessageDataWithSignalData(
117 pMessageData,
118 CANTX_STRING_P1_ENERGY_COUNTER_START_BIT,
119 CANTX_STRING_P1_ENERGY_COUNTER_LENGTH,
120 signalData,
121 CAN_BIG_ENDIAN);
122}

124/*========== Extern Function Implementations ================================*/
125extern uint32_t CANTX_StringValuesP1(
126 CAN_MESSAGE_PROPERTIES_s message,
127 uint8_t *pCanData,
128 uint8_t *pMuxId,
129 const CAN_SHIM_s *const kpkCanShim) {
130 FAS_ASSERT(message.id == CANTX_STRING_VALUES_P1_ID);
131 FAS_ASSERT(message.idType == CANTX_STRING_VALUES_P1_ID_TYPE);
132 FAS_ASSERT(message.dlc == CAN_FOXBMS_MESSAGES_DEFAULT_DLC);
133 FAS_ASSERT(pCanData != NULL_PTR);
134 FAS_ASSERT(pMuxId != NULL_PTR);
135 FAS_ASSERT(*pMuxId < BS_NR_OF_STRINGS);
136 FAS_ASSERT(kpkCanShim != NULL_PTR);
137 uint64_t messageData = 0u;
138 uint64_t signalData = 0u;
139
140 const uint8_t stringNumber = *pMuxId;
141
142 /* First signal to transmit cell voltages: get database values */
143 if (stringNumber == 0u) {
144 /* Do not read pTableMsl and pTableErrorState as they already are read
145 * with a higher frequency from CANTX_BmsState callback */
146 (void)DATA_READ_DATA(kpkCanShim->pTableCurrentSensor);
147 }
148
149 signalData = (uint64_t)stringNumber;
150 CANTX_SetStringP1Mux(&messageData, signalData);
151 CANTX_SetStringEnergyCounter(&messageData, stringNumber, kpkCanShim);
152
153 /* now copy data in the buffer that will be used to send data */
154 CAN_TxSetCanDataWithMessageData(messageData, pCanData, message.endianness);
155
156 /* Increment multiplexer for next cell */
157 (*pMuxId)++;
158
159 /* Check mux value */
160 if (*pMuxId >= BS_NR_OF_STRINGS) {
161 *pMuxId = 0u;
162 }
163
164 return 0u;
165}
