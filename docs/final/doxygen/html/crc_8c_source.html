75extern STD_RETURN_TYPE_e CRC_CalculateCrc(uint64_t *pCrc, uint8_t *pData, uint32_t lengthInBytes) {
76 FAS_ASSERT(pCrc != NULL_PTR);
77 FAS_ASSERT(pData != NULL_PTR);
78
79#ifndef UNITY_UNIT_TEST
80 static uint16_t crcCalls = 0u;
81#endif
82
83 CRC_REGISTER_SIDE_e registerSide = CRC_REGISTER_LOW;
84 uint32_t dataBufferLow = 0u;
85 uint32_t dataBufferHigh = 0u;
86 uint32_t remainingBytes = lengthInBytes;
87 uint32_t remainingData = 0u;
88 STD_RETURN_TYPE_e retVal = STD_OK;
89
90 uint8_t *pRead = pData;
91 if (crcCalls == 0u) {
92 crcCalls++;
93
94 /* Set mode to Data Capture Mode, otherwise writing the seed
95 starts the computation */
96 crcREG1->CTRL2 &= CRC_DATA_CAPTURE_MODE_CLEAR_MASK;
97 /* Set seed*/
98 crcREG1->PSA_SIGREGH1 = CRC_SEED_HIGH;
99 crcREG1->PSA_SIGREGL1 = CRC_SEED_LOW;
100 /* Set mode to Full-CPU Mode to start the computation when writing the data*/
101 crcREG1->CTRL2 |= CRC_FULL_CPU_MODE_SET_MASK;
102
103 /* AXIVION Next Codeline Style MisraC2012-11.3: 64 bit access needed, partial 32 bit access starts computation
104 */
105 /* Pointer to access the two signature registers, where input data will be written */
106 volatile uint64_t *pCrcRegister = (volatile uint64 *)(&crcREG1->PSA_SIGREGL1);
107
108 /* Treat packets of 64 bit data */
109 while (remainingBytes >= CRC_REGISTER_SIZE_IN_BYTES) {
110 /* Invert two 32 bit chunks before 64 bit write, due to big endian */
111 if (registerSide == CRC_REGISTER_LOW) {
112 dataBufferLow = 0u;
113 for (uint8_t i = 0u; i < CRC_REGISTER_SIZE_IN_BYTES; i++) {
114 uint8_t dataBuffer = *(pRead + i);
115 dataBufferLow |= ((uint32_t)dataBuffer) << ((CRC_REVERSE_BYTES_ORDER - i) * CRC_BYTE_SIZE_IN_BITS);
116 }
117 registerSide = CRC_REGISTER_HIGH;
118 } else { /* registerSide is CRC_REGISTER_HIGH */
119 dataBufferHigh = 0u;
120 for (uint8_t i = 0u; i < CRC_REGISTER_SIZE_IN_BYTES; i++) {
121 uint8_t dataBuffer = *(pRead + i);
122 dataBufferHigh |= ((uint32_t)dataBuffer) << ((CRC_REVERSE_BYTES_ORDER - i) * CRC_BYTE_SIZE_IN_BITS);
123 }
124 /* Signature low and high available, write to hardware register */
125 uint64_t crcData = (((uint64_t)dataBufferHigh) << CRC_REGISTER_SIZE_IN_BITS) | dataBufferLow;
126 *pCrcRegister = crcData;
127 registerSide = CRC_REGISTER_LOW;
128 }
129 pRead = (pRead + CRC_REGISTER_SIZE_IN_BYTES);
130 remainingBytes -= CRC_REGISTER_SIZE_IN_BYTES;
131 }
132
133 if (remainingBytes > 0u) {
134 /* Now treat last packet that is less than 32 bits if existing */
135 /* Get data in a 32 bit variable, pad with 0 */
136 while (remainingBytes > 0u) {
137 uint8_t dataBuffer = *pRead;
138 remainingData |= ((uint32_t)(dataBuffer)) << (CRC_BYTE_SIZE_IN_BITS * remainingBytes);
139 pRead++;
140 remainingBytes--;
141 }
142 if (registerSide == CRC_REGISTER_LOW) {
143 dataBufferLow = remainingData;
144 registerSide = CRC_REGISTER_HIGH;
145 } else { /* registerSide is CRC_REGISTER_HIGH */
146 dataBufferHigh = remainingData;
147 /* Signature low and high available, write to hardware register */
148 uint64_t crcData = (((uint64_t)dataBufferHigh) << CRC_REGISTER_SIZE_IN_BITS) | dataBufferLow;
149 *pCrcRegister = crcData;
150 registerSide = CRC_REGISTER_LOW;
151 }
152 }
153
154 /* No data remaining but only low register data available: compute CRC */
155 if (registerSide == CRC_REGISTER_HIGH) {
156 crcREG1->PSA_SIGREGL1 = dataBufferLow;
157 }
158
159 *pCrc = crcREG1->PSA_SIGREGL1;
160 *pCrc |= ((uint64_t)crcREG1->PSA_SIGREGH1) << CRC_REGISTER_SIZE_IN_BITS;
161 crcCalls--;
162 } else {
163 *pCrc = 0u;
164 retVal = STD_NOT_OK;
165 }
166
167 return retVal;
168}
