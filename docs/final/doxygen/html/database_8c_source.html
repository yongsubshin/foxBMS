42/**
43 * @file database.c
44 * @author foxBMS Team
45 * @date 2015-08-18 (date of creation)
46 * @updated 2025-08-07 (date of last update)
47 * @version v1.10.0
48 * @ingroup ENGINE
49 * @prefix DATA
50 *
51 * @brief Database module implementation
52 * @details The database read/write functions put references to the database
53 * entries to be read/written in the database queue. The function
54 * DATA_Task() reads the queue to get the database entries to be
55 * read/written. Up to DATA_MAX_ENTRIES_PER_ACCESS entries can be
56 * read/written with the same function call. To avoid code
57 * duplication, the functions to read/write 1 to
58 * DATA_MAX_ENTRIES_PER_ACCESS-1 entries call the function to
59 * read/write DATA_MAX_ENTRIES_PER_ACCESS entries and use NULL_PTR
60 * for the entries that are not to be read/written. DATA_Task()
61 * checks that the first entry is not a NULL_PTR and asserts if it
62 * is not the case. If subsequent entries are found in the database
63 * queue, they are simply ignored if they are NULL_PTR.
64 */

129/*========== Static Function Implementations ================================*/
130static STD_RETURN_TYPE_e DATA_AccessDatabaseEntries(
131 DATA_BLOCK_ACCESS_TYPE_e accessType,
132 void *pData0,
133 void *pData1,
134 void *pData2,
135 void *pData3) {
136 FAS_ASSERT((accessType == DATA_WRITE_ACCESS) || (accessType == DATA_READ_ACCESS));
137 FAS_ASSERT(pData0 != NULL_PTR);
138 /* AXIVION Routine Generic-MissingParameterAssert: pData1: pointer might be NULL_PTR (i.e., if the caller
139 * is DATA_Read1DataBlock/DATA_Write1DataBlock). The DATA_Task function checks the pointer being not NULL_PTR prior
140 * to usage. */
141 /* AXIVION Routine Generic-MissingParameterAssert: pData2: pointer might be NULL_PTR (i.e., if the caller
142 * is DATA_Read1DataBlock/DATA_Write1DataBlock or DATA_Read2DataBlocks/DATA_Write2DataBlocks). The DATA_Task
143 * function checks the pointer being not NULL_PTR prior to usage. */
144 /* AXIVION Routine Generic-MissingParameterAssert: pData3: pointer might be NULL_PTR (i.e., if the caller
145 * is DATA_Read1DataBlock/DATA_Write1DataBlock, DATA_Read2DataBlocks/DATA_Write2DataBlocks or
146 * DATA_Read3DataBlocks/DATA_Write3DataBlocks). The DATA_Task function checks the pointer being not NULL_PTR prior
147 * to usage. */
148 STD_RETURN_TYPE_e retval = STD_NOT_OK;
149 /* prepare send message with attributes of data block */
150 DATA_QUEUE_MESSAGE_s data_sendMessage = {
151 .pDatabaseEntry[DATA_ENTRY_0] = pData0,
152 .pDatabaseEntry[DATA_ENTRY_1] = pData1,
153 .pDatabaseEntry[DATA_ENTRY_2] = pData2,
154 .pDatabaseEntry[DATA_ENTRY_3] = pData3,
155 .accessType = accessType,
156 };
157 /* Send a pointer to a message object and maximum block time: DATA_QUEUE_TIMEOUT_MS */
158 if (OS_SendToBackOfQueue(ftsk_databaseQueue, (void *)&data_sendMessage, DATA_QUEUE_TIMEOUT_MS) == OS_SUCCESS) {
159 retval = STD_OK;
160 }
161 return retval;
162}

164static void DATA_CopyData(
165 DATA_BLOCK_ACCESS_TYPE_e accessType,
166 uint32_t dataLength,
167 void *pDatabaseStruct,
168 void *pPassedDataStruct) {
169 FAS_ASSERT((accessType == DATA_WRITE_ACCESS) || (accessType == DATA_READ_ACCESS));
170 /* AXIVION Routine Generic-MissingParameterAssert: dataLength: parameter accepts whole range */
171 /* Copy data either into database or passed database struct */
172 FAS_ASSERT(pDatabaseStruct != NULL_PTR);
173 FAS_ASSERT(pPassedDataStruct != NULL_PTR);
174
175 if (accessType == DATA_WRITE_ACCESS) {
176 /* Pointer on data block header of passed struct */
177 /* AXIVION Next Codeline Style MisraC2012-11.5 this casted is required in order to have a generic interface
178 * for all database entries. */
179 DATA_BLOCK_HEADER_s *pHeader = (DATA_BLOCK_HEADER_s *)pPassedDataStruct;
180 /* Update timestamps in passed database struct and then copy this struct into database */
181 pHeader->previousTimestamp = pHeader->timestamp;
182 pHeader->timestamp = OS_GetTickCount();
183 /* Copy passed struct in database struct */
184 /* memcpy has no return value therefore there is nothing to check: casting to void */
185 /* AXIVION Next Codeline Style MisraC2012-21.18
186 * */
187 (void)memcpy(pDatabaseStruct, pPassedDataStruct, dataLength);
188 } else if (accessType == DATA_READ_ACCESS) {
189 /* Copy database entry in passed struct */
190 /* memcpy has no return value therefore there is nothing to check: casting to void */
191 (void)memcpy(pPassedDataStruct, pDatabaseStruct, dataLength);
192 } else { /* LCOV_EXCL_LINE */
193 /* invalid database operation */ /* LCOV_EXCL_LINE */
194 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
195 }
196}

198static void DATA_IterateOverDatabaseEntries(const DATA_QUEUE_MESSAGE_s *kpReceiveMessage) {
199 FAS_ASSERT(kpReceiveMessage != NULL_PTR);
200 for (uint8_t queueEntry = 0u; queueEntry < DATA_MAX_ENTRIES_PER_ACCESS; queueEntry++) {
201 /* Iterate over pointer array and handle all access operations if pointer != NULL_PTR
202 * All pointers in the array, expect the first one, might be NULL_PTR, which is valid.
203 * Again, to understand explanation see the comments in DATA_Read1DataBlock, DATA_Read2DataBlocks,
204 * DATA_Read3DataBlocks and DATA_Read4DataBlocks as well as DATA_Write1DataBlock,
205 * DATA_Write2DataBlocks, DATA_Write3DataBlocks and DATA_Write4DataBlocks */
206 if (kpReceiveMessage->pDatabaseEntry[queueEntry] != NULL_PTR) {
207 /* pointer to passed database struct */
208 void *pPassedDataStruct = kpReceiveMessage->pDatabaseEntry[queueEntry];
209 /* Get access type (read or write) of passed data struct */
210 DATA_BLOCK_ACCESS_TYPE_e accessType = kpReceiveMessage->accessType;
211
212 /* Get pointer to database header entry */
213 const DATA_BLOCK_HEADER_s *kpHeader = (DATA_BLOCK_HEADER_s *)kpReceiveMessage->pDatabaseEntry[queueEntry];
214 uint8_t uniqueId = (uint8_t)(kpHeader->uniqueId);
215 FAS_ASSERT(uniqueId < (uint8_t)DATA_BLOCK_ID_MAX);
216 uint8_t entryIndex = data_uniqueIdToDatabaseEntry[uniqueId];
217 /* Pointer to database struct representation of passed struct */
218 void *pDatabaseStruct = (void *)data_baseHeader.pDatabase[entryIndex].pDatabaseEntry;
219 /* Get dataLength of database entry */
220 uint32_t dataLength = data_baseHeader.pDatabase[entryIndex].dataLength;
221
222 DATA_CopyData(accessType, dataLength, pDatabaseStruct, pPassedDataStruct);
223 }
224 }

228STD_RETURN_TYPE_e DATA_Initialize(void) {
229 STD_RETURN_TYPE_e retval = STD_OK;
230 /* Check that database queue has been created */
231 const bool allQueuesCreatedCopyForAssert = ftsk_allQueuesCreated;
232 FAS_ASSERT(allQueuesCreatedCopyForAssert == true);
233
234 FAS_STATIC_ASSERT((sizeof(data_database) != 0u), "No database defined");
235 /* make sure that no basic assumptions are broken -- since data_database is
236 declared with length DATA_BLOCK_ID_MAX, this assert should never fail */
237 FAS_STATIC_ASSERT(
238 ((sizeof(data_database) / sizeof(DATA_BASE_s)) == (uint8_t)(DATA_BLOCK_ID_MAX)), "Database array length error");
239
240 /* Iterate over database and set respective read/write pointer for each database entry */
241 for (uint16_t i = 0u; i < data_baseHeader.nrDatabaseEntries; i++) {
242 /* Initialize database entry with 0, set write pointer to start of database entry */
243 uint8_t *pStartDatabaseEntryWR = (uint8_t *)data_baseHeader.pDatabase[i].pDatabaseEntry;
244
245 /* Start after uniqueId entry. Set j'th byte to zero in database entry */
246 for (uint32_t j = 0u; j < (data_baseHeader.pDatabase + i)->dataLength; j++) {
247 if (j >= sizeof(DATA_BLOCK_ID_e)) {
248 *pStartDatabaseEntryWR = 0;
249 }
250 pStartDatabaseEntryWR++;
251 }
252 }
253
254 /* Configure link between uniqueId and database entry array position */
255 for (uint8_t databaseEntry = 0u; databaseEntry < data_baseHeader.nrDatabaseEntries; databaseEntry++) {
256 /* Get pointer to database header entry */
257 DATA_BLOCK_HEADER_s *pHeader = (DATA_BLOCK_HEADER_s *)data_baseHeader.pDatabase[databaseEntry].pDatabaseEntry;
258 /* make sure that the database entry is not a null pointer (which would happen if an entry is missing
259 despite the ID existing) */
260 FAS_ASSERT(pHeader != NULL_PTR);
261 /* Get uniqueId */
262 DATA_BLOCK_ID_e blockId = pHeader->uniqueId;
263 if ((blockId < DATA_BLOCK_ID_MAX) && (databaseEntry < DATA_BLOCK_ID_MAX)) {
264 /* e.g., data_uniqueIdToDatabaseEntry[<some id>] = configured database entry index */
265 data_uniqueIdToDatabaseEntry[blockId] = databaseEntry;
266 } else {
267 /* Configuration error -> set retval to #STD_NOT_OK */
268 retval = STD_NOT_OK;
269 }
270 }
271
272 if (ftsk_databaseQueue == NULL_PTR) {
273 retval = STD_NOT_OK; /* Failed to create the queue */
274 }
275 return retval;
276}

278void DATA_Task(void) {
279 if (ftsk_databaseQueue != NULL_PTR) {
280 DATA_QUEUE_MESSAGE_s receiveMessage = {
281 .accessType = DATA_READ_ACCESS,
282 .pDatabaseEntry = {GEN_REPEAT_U(NULL_PTR, GEN_STRIP(DATA_MAX_ENTRIES_PER_ACCESS))}};
283 /* scan queue and wait for a message up to a maximum amount of 1ms (block time) */
284 if (OS_ReceiveFromQueue(ftsk_databaseQueue, (&receiveMessage), 1u) == OS_SUCCESS) {
285 /* plausibility check, error whether the first pointer is a NULL_PTR, as this must not happen.
286 * See the comments in DATA_Read1DataBlock, DATA_Read2DataBlocks, DATA_Read3DataBlocks and
287 * DATA_Read4DataBlocks as well as DATA_Write1DataBlock, DATA_Write2DataBlocks,
288 * DATA_Write3DataBlocks and DATA_Write4DataBlocks */
289 FAS_ASSERT(receiveMessage.pDatabaseEntry[0] != NULL_PTR);
290 /* ready to start reading/writing database entries */
291 DATA_IterateOverDatabaseEntries(&receiveMessage);
292 }
293 }
294}

362extern void DATA_ExecuteDataBist(void) {
363 /* compile database entry */
364 DATA_BLOCK_DUMMY_FOR_SELF_TEST_s dummyWriteTable = {.header.uniqueId = DATA_BLOCK_ID_DUMMY_FOR_SELF_TEST};
365 dummyWriteTable.member1 = UINT8_MAX;
366 dummyWriteTable.member2 = DATA_DUMMY_VALUE_UINT8_T_ALTERNATING_BIT_PATTERN;
367
368 /* write entry */
369 STD_RETURN_TYPE_e writeReturnValue = (void)DATA_WRITE_DATA(&dummyWriteTable);
370 FAS_ASSERT(writeReturnValue == STD_OK);
371
372 DATA_BLOCK_DUMMY_FOR_SELF_TEST_s dummyReadTable = {.header.uniqueId = DATA_BLOCK_ID_DUMMY_FOR_SELF_TEST};
373
374 /* read entry into new variable */
375 STD_RETURN_TYPE_e readReturnValue = (void)DATA_READ_DATA(&dummyReadTable);
376 FAS_ASSERT(readReturnValue == STD_OK);
377
378 FAS_ASSERT(dummyReadTable.member1 == dummyWriteTable.member1);
379 FAS_ASSERT(dummyReadTable.member2 == dummyWriteTable.member2);
380}
