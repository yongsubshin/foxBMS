75extern bool DATA_DatabaseEntryUpdatedAtLeastOnce(DATA_BLOCK_HEADER_s dataBlockHeader) {
76 bool retval = false;
77 if (!((dataBlockHeader.timestamp == 0u) && (dataBlockHeader.previousTimestamp == 0u))) {
78 /* Only possibility for timestamp AND previous timestamp to be 0 is, if
79 the database entry has never been updated. Thus if this is not the
80 case the database entry must have been updated */
81 retval = true;
82 }
83 return retval;
84}

86extern bool DATA_EntryUpdatedWithinInterval(DATA_BLOCK_HEADER_s dataBlockHeader, uint32_t timeInterval) {
87 bool retval = false;
88 uint32_t currentTimestamp = OS_GetTickCount();
89
90 /* Unsigned integer arithmetic also works correctly if currentTimestamp is
91 larger than pHeader->timestamp (timer overflow), thus no need to use abs() */
92 const uint32_t timeDifferenceLastCall = currentTimestamp - dataBlockHeader.timestamp;
93 const bool updatedAtLeastOnce = DATA_DatabaseEntryUpdatedAtLeastOnce(dataBlockHeader);
94 if ((timeDifferenceLastCall <= timeInterval) && (updatedAtLeastOnce == true)) {
95 /* Difference between current timestamp and last update timestamp is
96 smaller than passed time interval */
97 retval = true;
98 }
99 return retval;
100}

102extern bool DATA_EntryUpdatedPeriodicallyWithinInterval(DATA_BLOCK_HEADER_s dataBlockHeader, uint32_t timeInterval) {
103 bool retval = false;
104 const uint32_t currentTimestamp = OS_GetTickCount();
105
106 /* Unsigned integer arithmetic also works correctly if currentTimestamp is
107 smaller than dataBlockHeader.timestamp (timer overflow), thus no need to use abs() */
108 const uint32_t timeDifferenceLastCall = currentTimestamp - dataBlockHeader.timestamp;
109 const uint32_t timeDifferenceBetweenCalls = dataBlockHeader.timestamp - dataBlockHeader.previousTimestamp;
110
111 if ((timeDifferenceLastCall <= timeInterval) && (timeDifferenceBetweenCalls <= timeInterval) &&
112 (DATA_DatabaseEntryUpdatedAtLeastOnce(dataBlockHeader) == true)) {
113 /* Difference between timestamps is smaller than passed time interval */
114 retval = true;
115 }
116 return retval;
117}

124extern uint8_t DATA_GetModuleNumberFromVoltageIndex(uint16_t cellIndex) {
125 FAS_ASSERT(cellIndex < (BS_NR_OF_CELL_BLOCKS_PER_STRING * BS_NR_OF_STRINGS));
126 uint8_t stringNumber = DATA_GetStringNumberFromVoltageIndex(cellIndex);
127 uint8_t moduleNumber =
128 (uint8_t)((cellIndex / BS_NR_OF_CELL_BLOCKS_PER_MODULE) - (stringNumber * BS_NR_OF_MODULES_PER_STRING));
129 return moduleNumber;
130}
