125static uint16_t DECAN_ModifiedModuloFunction(uint16_t a, uint16_t b) {
126 FAS_ASSERT(a > 0);
127 FAS_ASSERT(b > 0);
128
129 int16_t c = a % b;
130 if (c == 0) {
131 c = b;
132 }
133 return c;
134}

136static void DECAN_ConvertIndexForVoltage(uint16_t *s, uint16_t *m, uint16_t *cb, uint16_t oneNumIdxOfVoltage) {
137 /* oneNumIdxOfVoltage can not be bigger than
138 BS_NR_OF_STRINGS * BS_NR_OF_MODULES_PER_STRING * BS_NR_OF_CELL_BLOCKS_PER_MODULE -1 */
139 FAS_ASSERT(oneNumIdxOfVoltage < BS_NR_OF_STRINGS * BS_NR_OF_MODULES_PER_STRING * BS_NR_OF_CELL_BLOCKS_PER_MODULE);
140
141 /* The number of cell voltages starts from 1 while the index of the cell voltage starts from 0 */
142 uint16_t num_cellVoltages = oneNumIdxOfVoltage + 1;
143
144 /* Initialize the number of cell blocks(in the last string and the last module),
145 modules(in the last string), strings */
146 uint16_t num_cb = 1;
147 uint16_t num_m = 1;
148 uint16_t num_s = 1;
149
150 /* Initialize a temporal variable for calculating the minimum number of modules*/
151 uint16_t temp_m = 1;
152
153 /* Get the number of cell blocks */
154 num_cb = (uint16_t)(DECAN_ModifiedModuloFunction(num_cellVoltages, BS_NR_OF_CELL_BLOCKS_PER_MODULE));
155
156 /* In the case that more than one module exists */
157 if (num_cellVoltages > BS_NR_OF_CELL_BLOCKS_PER_MODULE) {
158 /* Get the number of module while ignoring the exist of string*/
159 if ((num_cellVoltages % BS_NR_OF_CELL_BLOCKS_PER_MODULE) == 0) {
160 temp_m = (uint16_t)(num_cellVoltages / BS_NR_OF_CELL_BLOCKS_PER_MODULE);
161 } else {
162 temp_m = (uint16_t)(num_cellVoltages / BS_NR_OF_CELL_BLOCKS_PER_MODULE) + 1;
163 }
164 num_m = temp_m;
165
166 /* If more than one string exists */
167 if (temp_m > BS_NR_OF_MODULES_PER_STRING) {
168 num_m = (uint16_t)(DECAN_ModifiedModuloFunction(temp_m, BS_NR_OF_MODULES_PER_STRING));
169 /* Get the number of strings */
170 if ((temp_m % BS_NR_OF_MODULES_PER_STRING) == 0) {
171 num_s = (uint16_t)(temp_m / BS_NR_OF_MODULES_PER_STRING);
172 } else {
173 num_s = (uint16_t)(temp_m / BS_NR_OF_MODULES_PER_STRING) + 1;
174 }
175 }
176 }
177
178 /* Get the indexes *s, *m, *cb of cell voltage */
179 *cb = num_cb - 1;
180 *m = num_m - 1;
181 *s = num_s - 1;
182}

234static STD_RETURN_TYPE_e DECAN_ReceiveCanCellVoltages(void) {
235 STD_RETURN_TYPE_e isSuccessful = STD_NOT_OK;
236 CAN_CAN2AFE_CELL_VOLTAGES_QUEUE_s decan_canCellVoltagesFromQueue = {0};
237
238 /* Get decan_canCellVoltagesFromQueue from queue */
239 OS_STD_RETURN_e receivedFromQueue = OS_ReceiveFromQueue(
240 ftsk_canToAfeCellVoltagesQueue, (void *)&decan_canCellVoltagesFromQueue, DECAN_CAN2AFE_QUEUE_TIMEOUT_MS);
241 if (receivedFromQueue == OS_SUCCESS) {
242 /* Initialize the indexes for cellVoltage_mV */
243 uint16_t s = 0u;
244 uint16_t m = 0u;
245 uint16_t cb = 0u;
246 uint16_t oneNumIdxOfVoltage = decan_canCellVoltagesFromQueue.muxValue *
247 CAN_NUM_OF_VOLTAGES_IN_CAN_CELL_VOLTAGES_MSG;
248 uint16_t upperBoundOneNumIdxOfVoltage =
249 BS_NR_OF_STRINGS * BS_NR_OF_MODULES_PER_STRING * BS_NR_OF_CELL_BLOCKS_PER_MODULE - 1;
250
251 /* Loop through all voltages in the received can message */
252 for (uint8_t i = 0u; i < CAN_NUM_OF_VOLTAGES_IN_CAN_CELL_VOLTAGES_MSG; i++) {
253 /* Check if the one number index of voltage surpass its upper bound */
254 if (oneNumIdxOfVoltage > upperBoundOneNumIdxOfVoltage) {
255 break;
256 }
257 /* Convert the one number index of temperature into s, m, ts */
258 DECAN_ConvertIndexForVoltage(&s, &m, &cb, oneNumIdxOfVoltage);
259
260 /* Write the received data */
261 decan_cellVoltage.cellVoltage_mV[s][m][cb] = (int16_t)decan_canCellVoltagesFromQueue.cellVoltage[i];
262 if (decan_canCellVoltagesFromQueue.invalidFlag[i] == DECAN_DATA_IS_VALID) {
263 decan_cellVoltage.invalidCellVoltage[s][m][cb] = false;
264 } else {
265 decan_cellVoltage.invalidCellVoltage[s][m][cb] = true;
266 }
267
268 /* Update one number index of the cell voltage */
269 oneNumIdxOfVoltage++;
270 }
271
272 /* Write this part of cell voltages and invalid flags to database,
273 read the total cell voltages and invalid flags, count the number of
274 invalid cell voltages and then write them back to database */
275 uint16_t nrValidCellVoltagesPerString = 0u;
276 int32_t stringVoltage_mV = 0;
277 int32_t moduleVoltage_mV = 0;
278 if ((DATA_WRITE_DATA(&decan_cellVoltage) == STD_OK) && (DATA_READ_DATA(&decan_cellVoltageFromRead) == STD_OK)) {
279 for (uint16_t idxString = 0u; idxString < BS_NR_OF_STRINGS; idxString++) {
280 nrValidCellVoltagesPerString = 0u;
281 stringVoltage_mV = 0;
282 for (uint16_t idxModule = 0u; idxModule < BS_NR_OF_MODULES_PER_STRING; idxModule++) {
283 moduleVoltage_mV = 0;
284 for (uint16_t idxCellBlocks = 0u; idxCellBlocks < BS_NR_OF_CELL_BLOCKS_PER_MODULE;
285 idxCellBlocks++) {
286 if (decan_cellVoltageFromRead.invalidCellVoltage[idxString][idxModule][idxCellBlocks] ==
287 false) {
288 nrValidCellVoltagesPerString++;
289 moduleVoltage_mV +=
290 (int32_t)decan_cellVoltageFromRead.cellVoltage_mV[idxString][idxModule][idxCellBlocks];
291 }
292 }
293 decan_cellVoltageFromRead.moduleVoltage_mV[idxString][idxModule] = moduleVoltage_mV;
294 stringVoltage_mV += moduleVoltage_mV;
295 }
296 decan_cellVoltageFromRead.stringVoltage_mV[idxString] = stringVoltage_mV;
297 decan_cellVoltageFromRead.nrValidCellVoltages[idxString] = nrValidCellVoltagesPerString;
298 }
299 isSuccessful = (void)DATA_WRITE_DATA(&decan_cellVoltageFromRead);
300 }
301 }
302
303 return isSuccessful;
304}

306static STD_RETURN_TYPE_e DECAN_ReceiveCanCellTemperatures(void) {
307 STD_RETURN_TYPE_e isSuccessful = STD_NOT_OK;
308
309 /* Get can cell temperatures from queue */
310 CAN_CAN2AFE_CELL_TEMPERATURES_QUEUE_s decan_canCellTemperaturesFromQueue = {0};
311 OS_STD_RETURN_e receivedFromQueue = OS_ReceiveFromQueue(
312 ftsk_canToAfeCellTemperaturesQueue,
313 (void *)&decan_canCellTemperaturesFromQueue,
314 DECAN_CAN2AFE_QUEUE_TIMEOUT_MS);
315 if (receivedFromQueue == OS_SUCCESS) {
316 /* Initialize the indexes for cellTemperature_ddegC */
317 uint16_t s = 0u;
318 uint16_t m = 0u;
319 uint16_t ts = 0u;
320 uint16_t oneNumIdxOfTemperature = decan_canCellTemperaturesFromQueue.muxValue *
321 CAN_NUM_OF_TEMPERATURES_IN_CAN_CELL_TEMPERATURES_MSG;
322 uint16_t upperBoundOneNumIdxOfTemperature =
323 BS_NR_OF_STRINGS * BS_NR_OF_MODULES_PER_STRING * BS_NR_OF_TEMP_SENSORS_PER_MODULE - 1;
324
325 /* Loop through all temperatures in the received can message */
326 for (uint16_t i = 0u; i < CAN_NUM_OF_TEMPERATURES_IN_CAN_CELL_TEMPERATURES_MSG; i++) {
327 /* Check if the one number index of temperature surpass its upper bound */
328 if (oneNumIdxOfTemperature > upperBoundOneNumIdxOfTemperature) {
329 break;
330 }
331
332 /* Convert the one number index of temperature into s, m, ts */
333 DECAN_ConvertIndexForTemperature(&s, &m, &ts, oneNumIdxOfTemperature);
334
335 /* Write the received data */
336 decan_cellTemperature.cellTemperature_ddegC[s][m][ts] =
337 decan_canCellTemperaturesFromQueue.cellTemperature[i] * 10;
338 if (decan_canCellTemperaturesFromQueue.invalidFlag[i] == DECAN_DATA_IS_VALID) {
339 decan_cellTemperature.invalidCellTemperature[s][m][ts] = false;
340 } else {
341 decan_cellTemperature.invalidCellTemperature[s][m][ts] = true;
342 }
343
344 /* Update one number index */
345 oneNumIdxOfTemperature++;
346 }
347
348 /* Write this part of cell temperatures and invalid flags to database,
349 read the total cell temperatures and invalid flags, count the number of
350 invalid cell temperatures and then write them back to database */
351 if ((DATA_WRITE_DATA(&decan_cellTemperature) == STD_OK) &&
352 (DATA_READ_DATA(&decan_cellTemperatureFromRead) == STD_OK)) {
353 for (uint16_t idxString = 0u; idxString < BS_NR_OF_STRINGS; idxString++) {
354 uint16_t nrValidCellTemperaturesPerString = 0u;
355 for (uint16_t idxModule = 0u; idxModule < BS_NR_OF_MODULES_PER_STRING; idxModule++) {
356 for (uint16_t idxTemperatureSensor = 0u; idxTemperatureSensor < BS_NR_OF_TEMP_SENSORS_PER_MODULE;
357 idxTemperatureSensor++) {
358 if (decan_cellTemperatureFromRead
359 .invalidCellTemperature[idxString][idxModule][idxTemperatureSensor] == false) {
360 nrValidCellTemperaturesPerString++;
361 }
362 }
363 }
364 decan_cellTemperatureFromRead.nrValidTemperatures[idxString] = nrValidCellTemperaturesPerString;
365 }
366 isSuccessful = (void)DATA_WRITE_DATA(&decan_cellTemperatureFromRead);
367 }
368 }
369
370 return isSuccessful;
371}

373/*========== Extern Function Implementations ================================*/
374extern STD_RETURN_TYPE_e DECAN_Initialize(void) {
375 /* Initialize entry of cell voltage and cell temperature in database */
376 for (uint16_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
377 for (uint16_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
378 for (uint16_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
379 decan_cellVoltage.cellVoltage_mV[s][m][cb] = 0;
380 decan_cellVoltage.invalidCellVoltage[s][m][cb] = true;
381 }
382 }
383 }
384 for (uint16_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
385 for (uint16_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
386 for (uint16_t ts = 0u; ts < BS_NR_OF_TEMP_SENSORS_PER_MODULE; ts++) {
387 decan_cellTemperature.cellTemperature_ddegC[s][m][ts] = 0;
388 decan_cellTemperature.invalidCellTemperature[s][m][ts] = true;
389 }
390 }
391 }
392 /* Give other tasks time to execute */
393 uint32_t current_time = OS_GetTickCount();
394 OS_DelayTaskUntil(&current_time, 10u);
395 return STD_OK;
396}

398extern STD_RETURN_TYPE_e DECAN_TriggerAfe(void) {
399 STD_RETURN_TYPE_e isSuccessful = STD_NOT_OK;
400 STD_RETURN_TYPE_e isSuccessful_receiveCanCellVoltages = STD_NOT_OK;
401 STD_RETURN_TYPE_e isSuccessful_receiveCanCellTemperatures = STD_NOT_OK;
402
403 isSuccessful_receiveCanCellVoltages = DECAN_ReceiveCanCellVoltages();
404 isSuccessful_receiveCanCellTemperatures = DECAN_ReceiveCanCellTemperatures();
405 if ((isSuccessful_receiveCanCellVoltages == STD_OK) && (isSuccessful_receiveCanCellTemperatures == STD_OK)) {
406 isSuccessful = STD_OK;
407 }
408
409 uint32_t currentTime = OS_GetTickCount();
410
411 /* Block task without possibility to wake up */
412 /* +1: to wait at least waitTime*/
413 OS_DelayTaskUntil(&currentTime, 10u);
414 return isSuccessful;
415}
