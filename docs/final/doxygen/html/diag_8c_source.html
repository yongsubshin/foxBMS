117/**
118 * @brief DIAG_Reset resets all needed structures
119 * @details This function gets called during initialization of the diagnose
120 * module. It clears memory and counters used by diag later on.
121 */
122
123static void DIAG_Reset(void) {
124 diag_locked = 1;
125
126 /* Reset counter */
127 for (uint32_t i = 0u; i < sizeof(diag.entry_cnt); i++) {
128 diag.entry_cnt[i] = 0;
129 }
130 diag.errcnttotal = 0;
131
132 /* Reset occurrence counter */
133 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
134 for (uint32_t i = 0u; i < DIAG_ID_MAX; i++) {
135 diag.occurrenceCounter[s][i] = 0;
136 }
137 }
138
139 diag_locked = 0;
140}

142/**
143 * @brief DIAG_SetFatalErrorById checks whether the error has been send before and if not
144 * send it per CAN. DO NOT use DIAG_ID_MAX!
145 * @details Checks if a certain event ID has already been marked as error and send via CAN.
146 * If not we send it and note is in the diag_activeFatalErrors array.
147 */
148
149static void DIAG_SetFatalErrorById(DIAG_ID_e xEventID) {
150 FAS_ASSERT(xEventID < DIAG_ID_MAX);
151 if (diag_activeFatalErrors[xEventID] == 0u) {
152 (void)CANTX_SendFatalErrorId(xEventID);
153 diag_activeFatalErrors[xEventID] = 1u;
154 if (diag_activeFatalErrorCount == 0u) {
155 (void)TIMER_Start(diag_fatalErrorResendTimer, 0u);
156 }
157 diag_activeFatalErrorCount++;
158 }
159}
160
161/**
162 * @brief DIAG_ClearFatalErrorById checks whether the error has been cleared before
163 * and if not sends a clear message per CAN
164 * DO NOT use DIAG_ID_MAX!
165 * @details Checks if a certain event ID has already been marked as error and send via CAN.
166 * If it is we send it a cleared error message via CAN and mark it
167 * in the diag_activeFatalErrors array.
168 */
169
170static void DIAG_ClearFatalErrorById(DIAG_ID_e xEventID) {
171 FAS_ASSERT(xEventID < DIAG_ID_MAX);
172 if (diag_activeFatalErrors[xEventID] != 0u) {
173 (void)CANTX_SendFatalErrorId(DIAG_ID_MAX);
174 diag_activeFatalErrors[xEventID] = 0u;
175 diag_activeFatalErrorCount--;
176 if (diag_activeFatalErrorCount == 0u) {
177 (void)TIMER_Stop(diag_fatalErrorResendTimer, 0u);
178 }
179 }
180}

182/**
183 * @brief Callback to check all active errors and resend them via CAN
184 * @details Checks which errors are marked as active and resends them via CAN.
185 * For better performance we first check the active fatal error count
186 * to se if iterating is even necessary.
187 */
188static void DIAG_ResendFatalErrors(TimerHandle_t xTimer) {
189 /* AXIVION Routine Generic-MissingParameterAssert: xTimer: parameter accept whole range */
190 (void)xTimer;
191 if (diag_activeFatalErrorCount > 0u) {
192 for (uint32_t i = 0u; i < DIAG_ID_MAX; i++) {
193 if (diag_activeFatalErrors[i] == 1u) {
194 (void)CANTX_SendFatalErrorId(i);
195 }
196 }
197 }
198}

201STD_RETURN_TYPE_e DIAG_Initialize(DIAG_DEV_s *diag_dev_pointer) {
202 FAS_ASSERT(diag_dev_pointer != NULL_PTR);
203
204 STD_RETURN_TYPE_e retval = STD_OK;
205 uint8_t id_nr = (uint8_t)DIAG_ID_MAX;
206 /* take assumptions on the value of DIAG_ID_MAX */
207 FAS_ASSERT((uint16_t)DIAG_ID_MAX < UINT8_MAX);
208 uint32_t tmperr_Check[((uint16_t)DIAG_ID_MAX + 31u) / 32u] = {0};
209
210 diag_devptr = diag_dev_pointer;
211
212 diag.state = DIAG_STATE_UNINITIALIZED;
213 uint16_t checkfail = 0u;
214
215 /* Fill lookup table id2ch */
216 for (uint8_t c = 0; c < diag_dev_pointer->nrOfConfiguredDiagnosisEntries; c++) {
217 id_nr = diag_dev_pointer->pConfigurationOfDiagnosisEntries[c].id;
218 if (id_nr < (uint16_t)DIAG_ID_MAX) {
219 diag.id2ch[id_nr] = c; /* e.g. diag.id2ch[DIAG_ID_90] = configured channel index */
220 } else {
221 /* Configuration error -> set retval to #STD_NOT_OK */
222 checkfail |= 0x20u;
223 retval = STD_NOT_OK;
224 }
225 }
226
227 for (uint8_t i = 0; i < (uint8_t)(((uint16_t)DIAG_ID_MAX + 31u) / 32u); i++) {
228 tmperr_Check[i] = 0u;
229 }
230
231 /* Fill enable array err_enableflag */
232 for (uint8_t i = 0; i < diag_dev_pointer->nrOfConfiguredDiagnosisEntries; i++) {
233 if (diag_dev_pointer->pConfigurationOfDiagnosisEntries[i].enable_evaluate == DIAG_EVALUATION_DISABLED) {
234 /* Disable diagnosis entry */
235 tmperr_Check[diag_dev_pointer->pConfigurationOfDiagnosisEntries[i].id / 32] |=
236 (1 << (diag_dev_pointer->pConfigurationOfDiagnosisEntries[i].id % 32));
237 }
238 }
239
240 /* take over configured error enable masks*/
241 for (uint8_t c = 0; c < (uint8_t)(((uint16_t)DIAG_ID_MAX + 31u) / 32u); c++) {
242 diag.err_enableflag[c] = ~tmperr_Check[c];
243 }
244
245 /* Reset counter in case init function is called multiple times */
246 diag_dev_pointer->numberOfFatalErrors = 0u;
247 /* Fill pointer array with links to all diagnosis entries with a fatal error */
248 for (uint16_t entry = 0u; entry < (uint16_t)DIAG_ID_MAX; entry++) {
249 if (diag_diagnosisIdConfiguration[entry].severity == DIAG_FATAL_ERROR) {
250 diag_dev_pointer->pFatalErrorLinkTable[diag_dev_pointer->numberOfFatalErrors] =
251 &diag_diagnosisIdConfiguration[entry];
252 diag_dev_pointer->numberOfFatalErrors++;
253 }
254 }
255
256 /** Iterate over #diag_diagnosisIdConfiguration and check that a meaningful
257 * state transition time is configured if a severity of #DIAG_FATAL_ERROR
258 * is configured. */
259 for (uint16_t diagnosisEntry = 0u; diagnosisEntry < diag_dev_pointer->nrOfConfiguredDiagnosisEntries;
260 diagnosisEntry++) {
261 bool fatalErrorDetected = (bool)(diag_diagnosisIdConfiguration[diagnosisEntry].severity == DIAG_FATAL_ERROR);
262 bool discardDelay = (bool)(diag_diagnosisIdConfiguration[diagnosisEntry].delay_ms == DIAG_DELAY_DISCARD);
263 if (fatalErrorDetected && discardDelay) {
264 /* Configuration error. Fatal error configured but delay is discared.*/
265 FAS_ASSERT(FAS_TRAP);
266 }
267 }
268
269 /** Initialize and start timer to periodically resend the fatal errors */
270 diag_fatalErrorResendTimer = TIMER_Create(
271 "fatal_error_resend",
272 diag_fatalErrorResendPeriod,
273 true,
274 (void *)&diag_fatalErrorResendTimerID,
275 DIAG_ResendFatalErrors,
276 &diag_timerBuffer);
277
278 /* Check if configuration errors occurred during initialization */
279 if (checkfail > 0u) {
280 DIAG_Reset();
281 }
282
283 diag.state = DIAG_STATE_INITIALIZED;
284 return retval;
285}

287STD_RETURN_TYPE_e DIAG_GetDiagnosisEntryState(DIAG_ID_e diagnosisEntry) {
288 STD_RETURN_TYPE_e retval = STD_OK;
289 const uint16_t errorThreshold =
290 diag_devptr->pConfigurationOfDiagnosisEntries[diag.id2ch[(uint16_t)diagnosisEntry]].threshold;
291
292 /* Error if active if threshold counter is larger than configured error threshold */
293 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
294 uint16_t thresholdCounter = diag.occurrenceCounter[s][(uint16_t)diagnosisEntry];
295 if (thresholdCounter > errorThreshold) {
296 /* error-threshold exceeded -> error is active */
297 retval = STD_NOT_OK;
298 }
299 }
300 return retval;
301}

306static uint8_t DIAG_EntryWrite(uint8_t eventID, DIAG_EVENT_e event, uint32_t data) {
307 uint8_t ret_val = 0;
308 if (diag_locked > 0u) {
309 return ret_val; /* only locked when clearing the diagnosis memory */
310 }
311
312 if (diag.entry_event[eventID] == event) {
313 /* same event of same error type already recorded before -> ignore until event toggles */
314 return ret_val;
315 }
316 if ((diag.entry_event[eventID] == DIAG_EVENT_OK) && (event == DIAG_EVENT_RESET)) {
317 /* do record DIAG_EVENT_RESET-event only if last event was an error (re-initialization) */
318 /* meaning: DIAG_EVENT_RESET-event at first time call or after DIAG_EVENT_OK-event will not be recorded */
319 return ret_val;
320 }
321
322 if (++diag.entry_cnt[eventID] > DIAG_MAX_ENTRIES_OF_ERROR) {
323 /* this type of error has been recorded too many times -> ignore to avoid filling buffer with same failure codes */
324 diag.entry_cnt[eventID] = DIAG_MAX_ENTRIES_OF_ERROR;
325 return ret_val;
326 }
327
328 /* now record failure code */
329 ret_val = 0xFF;
330
331 /* counts of (new) diagnosis entry records which is still not been read by
332 * external Tool which will reset this value to 0 after having read all
333 * new entries which means <acknowledged by user> */
334 ++diag.errcntreported;
335 ++diag.errcnttotal; /* total counts of diagnosis entry records */
336 diag.entry_event[eventID] = event;
337
338 return ret_val;
339}

341DIAG_RETURNTYPE_e DIAG_Handler(DIAG_ID_e diagId, DIAG_EVENT_e event, DIAG_IMPACT_LEVEL_e impact, uint32_t data) {
342 DIAG_RETURNTYPE_e ret_val = DIAG_HANDLER_RETURN_UNKNOWN;
343 uint32_t *u32ptr_errCodemsk = NULL_PTR;
344 uint32_t *u32ptr_warnCodemsk = NULL_PTR;
345 uint16_t *u16ptr_threshcounter = NULL_PTR;
346 uint16_t cfg_threshold = 0;
347 uint16_t err_enable_idx = 0;
348 uint32_t err_enable_bitmask = 0;
349
350 DIAG_EVALUATE_e evaluate_enabled;
351
352 if (diag_devptr == NULL_PTR) {
353 FAS_ASSERT(FAS_TRAP);
354 }
355
356 if (diag.state == DIAG_STATE_UNINITIALIZED) {
357 return DIAG_HANDLER_RETURN_NOT_READY;
358 }
359
360 if (diagId >= DIAG_ID_MAX) {
361 return DIAG_HANDLER_RETURN_WRONG_ID;
362 }
363
364 if (!((impact == DIAG_SYSTEM) || (impact == DIAG_STRING))) {
365 return DIAG_HANDLER_INVALID_ERR_IMPACT;
366 }
367
368 if ((impact == DIAG_STRING) && (data >= BS_NR_OF_STRINGS)) {
369 return DIAG_HANDLER_INVALID_DATA;
370 }
371
372 /* Determine a stringID, for impact level #DIAG_SYSTEM this is
373 always 0. This stringID is used to access the #DIAG_DIAGNOSIS_STATE_s::occurrenceCounter
374 2D-array.
375 */
376 uint8_t stringID = 0u;
377 if (impact == DIAG_STRING) {
378 stringID = data;
379 }
380
381 err_enable_idx = (uint16_t)diagId / 32u; /* array index of diag.err_enableflag[..] */
382 err_enable_bitmask = 1u << ((uint16_t)diagId % 32u); /* bit number (mask) of diag.err_enableflag[idx] */
383
384 u32ptr_errCodemsk = &diag.errflag[err_enable_idx];
385 u32ptr_warnCodemsk = &diag.warnflag[err_enable_idx];
386 u16ptr_threshcounter = &diag.occurrenceCounter[stringID][diagId];
387 cfg_threshold = diag_devptr->pConfigurationOfDiagnosisEntries[diag.id2ch[diagId]].threshold;
388 evaluate_enabled = diag_devptr->pConfigurationOfDiagnosisEntries[diag.id2ch[diagId]].enable_evaluate;
389
390 if (event == DIAG_EVENT_OK) {
391 if ((diag.err_enableflag[err_enable_idx] & err_enable_bitmask) > 0u) {
392 /* if (((*u16ptr_threshcounter) == 0) && (*u32ptr_errCodemsk == 0)) */
393 if (((*u16ptr_threshcounter) == 0)) {
394 /* everything ok, nothing to be handled */
395 } else if ((*u16ptr_threshcounter) > 1) {
396 (*u16ptr_threshcounter)--; /* Error did not occur, decrement Error-Counter */
397 } else if ((*u16ptr_threshcounter) == 1) {
398 /* else if ((*u16ptr_threshcounter) <= 1) */
399 /* Error did not occur, now decrement to zero and clear Error- or Warning-Flag and make recording if enabled */
400 *u32ptr_errCodemsk &= ~err_enable_bitmask; /* ERROR: clear corresponding bit in errflag[idx] */
401 *u32ptr_warnCodemsk &= ~err_enable_bitmask; /* WARNING: clear corresponding bit in warnflag[idx] */
402 (*u16ptr_threshcounter) = 0;
403 /* Make entry in error-memory (error disappeared) */
404 DIAG_EntryWrite(diagId, event, data);
405 /* Check if error would have been fatal and send clear CAN message*/
406 if (diag_devptr->pConfigurationOfDiagnosisEntries[diag.id2ch[(uint16_t)diagId]].severity ==
407 DIAG_FATAL_ERROR) {
408 DIAG_ClearFatalErrorById(diagId);
409 }
410
411 if (evaluate_enabled == DIAG_EVALUATION_ENABLED) {
412 /* Call callback function and reset error */
413 diag_diagnosisIdConfiguration[diag.id2ch[diagId]].fpCallback(
414 diagId, DIAG_EVENT_RESET, &diag_kDatabaseShim, data);
415 }
416 }
417 }
418 ret_val = DIAG_HANDLER_RETURN_OK; /* Function does not return an error-message! */
419 } else if (event == DIAG_EVENT_NOT_OK) {
420 if ((diag.err_enableflag[err_enable_idx] & err_enable_bitmask) > 0u) {
421 if ((*u16ptr_threshcounter) < cfg_threshold) {
422 (*u16ptr_threshcounter)++; /* error-threshold not exceeded yet, increment Error-Counter */
423 ret_val = DIAG_HANDLER_RETURN_OK; /* Function does not return an error-message! */
424 } else if ((*u16ptr_threshcounter) == cfg_threshold) {
425 /* Error occurred AND error-threshold exceeded */
426 (*u16ptr_threshcounter)++;
427 *u32ptr_errCodemsk |= err_enable_bitmask; /* ERROR: set corresponding bit in errflag[idx] */
428 *u32ptr_warnCodemsk &= ~err_enable_bitmask; /* WARNING: clear corresponding bit in warnflag[idx] */
429
430 /* Make entry in error-memory (error occurred) */
431 DIAG_EntryWrite(diagId, event, data);
432
433 /* Check if error is fatal*/
434 if (diag_devptr->pConfigurationOfDiagnosisEntries[diag.id2ch[(uint16_t)diagId]].severity ==
435 DIAG_FATAL_ERROR) {
436 DIAG_SetFatalErrorById(diagId);
437 }
438
439 if (evaluate_enabled == DIAG_EVALUATION_ENABLED) {
440 /* Call callback function and set error */
441 diag_diagnosisIdConfiguration[diag.id2ch[diagId]].fpCallback(
442 diagId, DIAG_EVENT_NOT_OK, &diag_kDatabaseShim, data);
443 }
444 /* Function returns an error-message! */
445 ret_val = DIAG_HANDLER_RETURN_ERR_OCCURRED;
446 } else if (((*u16ptr_threshcounter) > cfg_threshold)) {
447 /* error-threshold already exceeded, nothing to be handled */
448 ret_val = DIAG_HANDLER_RETURN_ERR_OCCURRED;
449 }
450 } else {
451 /* Error occurred BUT NOT enabled by mask */
452 *u32ptr_errCodemsk &= ~err_enable_bitmask; /* ERROR: clear corresponding bit in errflag[idx] */
453 *u32ptr_warnCodemsk |= err_enable_bitmask; /* WARNING: set corresponding bit in warnflag[idx] */
454 ret_val = DIAG_HANDLER_RETURN_WARNING_OCCURRED; /* Function returns an error-message! */
455 }
456 } else if (event == DIAG_EVENT_RESET) {
457 if ((diag.err_enableflag[err_enable_idx] & err_enable_bitmask) > 0u) {
458 /* clear counter, Error-, Warning-Flag and make recording if enabled */
459 *u32ptr_errCodemsk &= ~err_enable_bitmask; /* ERROR: clear corresponding bit in errflag[idx] */
460 *u32ptr_warnCodemsk &= ~err_enable_bitmask; /* WARNING: clear corresponding bit in warnflag[idx] */
461 (*u16ptr_threshcounter) = 0;
462 /* Make entry in error-memory (error disappeared) if error was recorded before */
463 DIAG_EntryWrite(diagId, event, data);
464
465 if (evaluate_enabled == DIAG_EVALUATION_ENABLED) {
466 /* Call callback function and reset error */
467 diag_diagnosisIdConfiguration[diag.id2ch[diagId]].fpCallback(
468 diagId, DIAG_EVENT_RESET, &diag_kDatabaseShim, data);
469 }
470 }
471 ret_val = DIAG_HANDLER_RETURN_OK; /* Function does not return an error-message! */
472 }
473
474 return ret_val;
475}

494bool DIAG_IsAnyFatalErrorSet(void) {
495 bool fatalErrorActive = false;
496 for (uint16_t entry = 0u; entry < diag_device.numberOfFatalErrors; entry++) {
497 const STD_RETURN_TYPE_e diagnosisState =
498 DIAG_GetDiagnosisEntryState(diag_device.pFatalErrorLinkTable[entry]->id);
499 if (STD_NOT_OK == diagnosisState) {
500 fatalErrorActive = true;
501 }
502 }
503
504 return fatalErrorActive;
505}
