73extern void DIAG_ErrorOvertemperatureCharge(
74 DIAG_ID_e diagId,
75 DIAG_EVENT_e event,
76 const DIAG_DATABASE_SHIM_s *const kpkDiagShim,
77 uint32_t stringNumber) {
78 FAS_ASSERT(
79 (diagId == DIAG_ID_TEMP_OVERTEMPERATURE_CHARGE_MSL) || (diagId == DIAG_ID_TEMP_OVERTEMPERATURE_CHARGE_RSL) ||
80 (diagId == DIAG_ID_TEMP_OVERTEMPERATURE_CHARGE_MOL));
81 FAS_ASSERT((event == DIAG_EVENT_OK) || (event == DIAG_EVENT_NOT_OK) || (event == DIAG_EVENT_RESET));
82 FAS_ASSERT(kpkDiagShim != NULL_PTR);
83 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
84
85 if (diagId == DIAG_ID_TEMP_OVERTEMPERATURE_CHARGE_MSL) {
86 if (event == DIAG_EVENT_RESET) {
87 kpkDiagShim->pTableMsl->overtemperatureCharge[stringNumber] = 0;
88 }
89 if (event == DIAG_EVENT_NOT_OK) {
90 kpkDiagShim->pTableMsl->overtemperatureCharge[stringNumber] = 1;
91 }
92 } else if (diagId == DIAG_ID_TEMP_OVERTEMPERATURE_CHARGE_RSL) {
93 if (event == DIAG_EVENT_RESET) {
94 kpkDiagShim->pTableRsl->overtemperatureCharge[stringNumber] = 0;
95 }
96 if (event == DIAG_EVENT_NOT_OK) {
97 kpkDiagShim->pTableRsl->overtemperatureCharge[stringNumber] = 1;
98 }
99 } else if (diagId == DIAG_ID_TEMP_OVERTEMPERATURE_CHARGE_MOL) {
100 if (event == DIAG_EVENT_RESET) {
101 kpkDiagShim->pTableMol->overtemperatureCharge[stringNumber] = 0;
102 }
103 if (event == DIAG_EVENT_NOT_OK) {
104 kpkDiagShim->pTableMol->overtemperatureCharge[stringNumber] = 1;
105 }
106 } else {
107 /* MISRA C:2012 Rule 15.7 - All if-else-if chains must have terminal else clause */
108 FAS_ASSERT(FAS_TRAP);
109 }
110}
