72extern void DIAG_ErrorOvervoltage(
73 DIAG_ID_e diagId,
74 DIAG_EVENT_e event,
75 const DIAG_DATABASE_SHIM_s *const kpkDiagShim,
76 uint32_t stringNumber) {
77 FAS_ASSERT(
78 (diagId == DIAG_ID_CELL_VOLTAGE_OVERVOLTAGE_MSL) || (diagId == DIAG_ID_CELL_VOLTAGE_OVERVOLTAGE_RSL) ||
79 (diagId == DIAG_ID_CELL_VOLTAGE_OVERVOLTAGE_MOL));
80 FAS_ASSERT((event == DIAG_EVENT_OK) || (event == DIAG_EVENT_NOT_OK) || (event == DIAG_EVENT_RESET));
81 FAS_ASSERT(kpkDiagShim != NULL_PTR);
82 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
83
84 if (diagId == DIAG_ID_CELL_VOLTAGE_OVERVOLTAGE_MSL) {
85 if (event == DIAG_EVENT_RESET) {
86 kpkDiagShim->pTableMsl->overVoltage[stringNumber] = 0u;
87 }
88 if (event == DIAG_EVENT_NOT_OK) {
89 kpkDiagShim->pTableMsl->overVoltage[stringNumber] = 1u;
90 }
91 } else if (diagId == DIAG_ID_CELL_VOLTAGE_OVERVOLTAGE_RSL) {
92 if (event == DIAG_EVENT_RESET) {
93 kpkDiagShim->pTableRsl->overVoltage[stringNumber] = 0u;
94 }
95 if (event == DIAG_EVENT_NOT_OK) {
96 kpkDiagShim->pTableRsl->overVoltage[stringNumber] = 1u;
97 }
98 } else if (diagId == DIAG_ID_CELL_VOLTAGE_OVERVOLTAGE_MOL) {
99 if (event == DIAG_EVENT_RESET) {
100 kpkDiagShim->pTableMol->overVoltage[stringNumber] = 0u;
101 }
102 if (event == DIAG_EVENT_NOT_OK) {
103 kpkDiagShim->pTableMol->overVoltage[stringNumber] = 1u;
104 }
105 } else { /* LCOV_EXCL_LINE */
106 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
107 }
108}

148extern void DIAG_ErrorHighVoltageMeasurement(
149 DIAG_ID_e diagId,
150 DIAG_EVENT_e event,
151 const DIAG_DATABASE_SHIM_s *const kpkDiagShim,
152 uint32_t stringNumber) {
153 FAS_ASSERT(
154 (diagId == DIAG_ID_CURRENT_SENSOR_V1_MEASUREMENT_TIMEOUT) ||
155 (diagId == DIAG_ID_CURRENT_SENSOR_V2_MEASUREMENT_TIMEOUT) ||
156 (diagId == DIAG_ID_CURRENT_SENSOR_V3_MEASUREMENT_TIMEOUT));
157 FAS_ASSERT((event == DIAG_EVENT_OK) || (event == DIAG_EVENT_NOT_OK) || (event == DIAG_EVENT_RESET));
158 FAS_ASSERT(kpkDiagShim != NULL_PTR);
159 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
160
161 switch (diagId) {
162 case DIAG_ID_CURRENT_SENSOR_V1_MEASUREMENT_TIMEOUT:
163 if (event == DIAG_EVENT_RESET) {
164 kpkDiagShim->pTableError->currentSensorVoltage1TimeoutError[stringNumber] = false;
165 } else if (event == DIAG_EVENT_NOT_OK) {
166 kpkDiagShim->pTableError->currentSensorVoltage1TimeoutError[stringNumber] = true;
167 } else {
168 /* no relevant event, do nothing */
169 }
170 break;
171 case DIAG_ID_CURRENT_SENSOR_V2_MEASUREMENT_TIMEOUT:
172 if (event == DIAG_EVENT_RESET) {
173 kpkDiagShim->pTableError->currentSensorVoltage2TimeoutError[stringNumber] = false;
174 } else if (event == DIAG_EVENT_NOT_OK) {
175 kpkDiagShim->pTableError->currentSensorVoltage2TimeoutError[stringNumber] = true;
176 } else {
177 /* no relevant event, do nothing */
178 }
179 break;
180 case DIAG_ID_CURRENT_SENSOR_V3_MEASUREMENT_TIMEOUT:
181 if (event == DIAG_EVENT_RESET) {
182 kpkDiagShim->pTableError->currentSensorVoltage3TimeoutError[stringNumber] = false;
183 } else if (event == DIAG_EVENT_NOT_OK) {
184 kpkDiagShim->pTableError->currentSensorVoltage3TimeoutError[stringNumber] = true;
185 } else {
186 /* no relevant event, do nothing */
187 }
188 break;
189
190 default: /* LCOV_EXCL_LINE */
191 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
192 break; /* LCOV_EXCL_LINE */
193 }
194}
