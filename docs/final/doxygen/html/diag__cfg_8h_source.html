82/** ---------------- DEFINES FOR ERROR STATE TRANSITION DELAY----------------
83 * These defines configure the delay before the transition to the error state
84 * occurs if a fault is detected. During this time the BMS can alert a
85 * superordinate control unit that the contactors will be opened soon. The
86 * superior control unit can take action and e.g., reduce the current until
87 * the transition to error state takes place.
88 *
89 * The delay is not taken into account if severity level #DIAG_FATAL_ERROR of
90 * type #DIAG_SEVERITY_LEVEL_e is configured in config array #diag_diagnosisIdConfiguration.
91 * For any other severity, #DIAG_DELAY_DISCARD can be used as a dummy value.
92 */
93#define DIAG_DELAY_DISCARD (UINT32_MAX)
94/** no delay after error is detected, open contactors instantaneous */
95#define DIAG_NO_DELAY (0u)
96

99#define DIAG_DELAY_200ms (200u)
100#define DIAG_DELAY_1000ms (1000u)
101#define DIAG_DELAY_2000ms (1000u)
