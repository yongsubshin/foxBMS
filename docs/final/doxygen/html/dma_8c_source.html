159 /* Configuration for SPI */
160
161 for (uint8_t i = 0u; i < DMA_NUMBER_SPI_INTERFACES; i++) {
162 /* assign dma request to Tx channel */
163 dmaReqAssign((dmaChannel_t)dma_spiDmaChannels[i].txChannel, (dmaRequest_t)dma_spiDmaRequests[i].txRequest);
164 /* assign dma request to Rx channel */
165 dmaReqAssign((dmaChannel_t)dma_spiDmaChannels[i].rxChannel, (dmaRequest_t)dma_spiDmaRequests[i].rxRequest);
166
167 /* Enable Interrupt after reception of data
168 Group A - Interrupts (FTC, LFS, HBC, and BTC) are routed to the ARM CPU
169 User software should configure only Group A interrupts */
170 /**
171 * Use Tx interrupt to transfer last word with CSHOLD = 0
172 * DO NOT ACTIVATE FOR SLAVE SPI NODES (here SPI4 used as slave)
173 * */
174 if (i != SPI_GetSpiIndex(spiREG4)) {
175 dmaEnableInterrupt(
176 (dmaChannel_t)(dmaChannel_t)dma_spiDmaChannels[i].txChannel,
177 (dmaInterrupt_t)BTC,
178 (dmaIntGroup_t)DMA_INTA);
179 }
180 /* Use Rx to determine when SPI over DMA transaction is finished */
181 dmaEnableInterrupt(
182 (dmaChannel_t)(dmaChannel_t)dma_spiDmaChannels[i].rxChannel, (dmaInterrupt_t)BTC, (dmaIntGroup_t)DMA_INTA);
183
184 dma_controlPacketSpiTx.DADD = (uint32_t)(&(dma_spiInterfaces[i]->DAT1)) + DMA_BIG_ENDIAN_ADDRESS_16BIT;
185 dma_controlPacketSpiRx.SADD = (uint32_t)(&(dma_spiInterfaces[i]->BUF)) + DMA_BIG_ENDIAN_ADDRESS_16BIT;
186
187 /* Set dma control packet for Tx */
188 dmaSetCtrlPacket((dmaChannel_t)dma_spiDmaChannels[i].txChannel, dma_controlPacketSpiTx);
189
190 /* Set dma control packet for Rx */
191 dmaSetCtrlPacket((dmaChannel_t)dma_spiDmaChannels[i].rxChannel, dma_controlPacketSpiRx);
192
193 /* Set the dma channels to trigger on h/w request */
194 dmaSetChEnable((dmaChannel_t)dma_spiDmaChannels[i].txChannel, (dmaTriggerType_t)DMA_HW);
195 dmaSetChEnable((dmaChannel_t)dma_spiDmaChannels[i].rxChannel, (dmaTriggerType_t)DMA_HW);

254void dmaGroupANotification(dmaInterrupt_t inttype, uint32 channel) {
255 /* AXIVION Routine Generic-MissingParameterAssert: inttype: unchecked in interrupt */
256 /* AXIVION Routine Generic-MissingParameterAssert: channel: unchecked in interrupt */
257 if (inttype == (dmaInterrupt_t)BTC) {
258 uint16_t timeoutIterations = 0u;
259 uint8_t spiIndex = 0u;
260 BaseType_t xHigherPriorityTaskWoken = pdFALSE;
261 bool success = true;
262 switch (channel) {
263 /* DMA for SPI Tx */
264 case DMA_CHANNEL_SPI1_TX:
265 case DMA_CHANNEL_SPI2_TX:
266 case DMA_CHANNEL_SPI3_TX:
267 case DMA_CHANNEL_SPI4_TX:
268 case DMA_CHANNEL_SPI5_TX:
269 /* Search for SPI index with DMA TX channel */
270 for (uint8_t i = 0u; i < DMA_NUMBER_SPI_INTERFACES; i++) {
271 if (channel == (uint32_t)dma_spiDmaChannels[i].txChannel) {
272 spiIndex = i;
273 break;
274 }
275 }
276 /**
277 * TX SPI DMA interrupt: last but one word transmitted,
278 * last word is transmitted manually (in order to write CSHOLD field)
279 */
280 timeoutIterations = SPI_TX_EMPTY_TIMEOUT_ITERATIONS;
281 /* Wait until TX buffer is free for the last word */
282 while (((dma_spiInterfaces[spiIndex]->FLG &
283 (uint32)((uint32_t)1u << SPI_TX_BUFFER_EMPTY_FLAG_POSITION)) == 0u) &&
284 (timeoutIterations > 0u)) {
285 timeoutIterations--;
286 }
287 SPI_DmaSendLastByte(spiIndex);
288 break;
289
290 /* DMA for SPI Rx */
291 case DMA_CHANNEL_SPI1_RX:
292 case DMA_CHANNEL_SPI2_RX:
293 case DMA_CHANNEL_SPI3_RX:
294 case DMA_CHANNEL_SPI4_RX:
295 case DMA_CHANNEL_SPI5_RX:
296 /* Search for SPI index with DMA RX channel */
297 for (uint8_t i = 0u; i < DMA_NUMBER_SPI_INTERFACES; i++) {
298 if (channel == (uint32_t)dma_spiDmaChannels[i].rxChannel) {
299 spiIndex = i;
300 break;
301 }
302 }
303 /* RX SPI DMA interrupt: last word received, means SPI transmission is finished */
304 if (spiIndex == SPI_GetSpiIndex(spiREG4)) { /** SPI configured as slave */
305 /* RX DMA interrupt, transmission finished, disable DMA */
306 dma_spiInterfaces[spiIndex]->INT0 &= ~DMA_REQUEST_ENABLE_BIT;
307 /* Disable SPI to prevent unwanted reception */
308 dma_spiInterfaces[spiIndex]->GCR1 &= ~DMA_SPI_ENABLE_BIT;
309 /* Set slave SPI Chip Select pins as GIO to deactivate slave SPI Chip Select pins */
310 dma_spiInterfaces[spiIndex]->PC0 &= SPI_PC0_CLEAR_HW_CS_MASK;
311
312 /* Specific call for AFEs */
313 AFE_DmaCallback(spiIndex);
314 } else { /* SPI configured as master */
315 /* RX DMA interrupt, transmission finished, disable DMA */
316 dma_spiInterfaces[spiIndex]->INT0 &= ~DMA_REQUEST_ENABLE_BIT;
317
318 /* Specific call for AFEs */
319 if (spiIndex == SPI_GetSpiIndex(spiREG1)) {
320 AFE_DmaCallback(spiIndex);
321 }
322 spi_busyFlags[spiIndex] = SPI_IDLE;
323 }
324 break;
325
326 /* DMA for I2C Tx */
327 case DMA_CHANNEL_I2C1_TX:
328 i2cREG1->DMACR &= ~((uint32_t)I2C_TX_DMA_ENABLE);
329 (void)xTaskNotifyIndexedFromISR(
330 I2C_TASK_HANDLE,
331 I2C_NOTIFICATION_TX_INDEX,
332 I2C_TX_NOTIFIED_VALUE,
333 eSetValueWithOverwrite,
334 &xHigherPriorityTaskWoken);
335 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
336 break;
337 case DMA_CHANNEL_I2C2_TX:
338 i2cREG2->DMACR &= ~((uint32_t)I2C_TX_DMA_ENABLE);
339 (void)xTaskNotifyIndexedFromISR(
340 I2C_TASK_HANDLE,
341 I2C_NOTIFICATION_TX_INDEX,
342 I2C_TX_NOTIFIED_VALUE,
343 eSetValueWithOverwrite,
344 &xHigherPriorityTaskWoken);
345 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
346 break;
347
348 /* DMA for I2C Rx */
349 case DMA_CHANNEL_I2C1_RX:
350 i2cREG1->DMACR &= ~((uint32_t)I2C_RX_DMA_ENABLE);
351 /* Received all but the last byte, now wait until receive buffer is filled with the last byte to receive
352 */
353 success = I2C_WaitReceive(i2cREG1, I2C_TIMEOUT_us);
354 if (success == false) {
355 /* Set Stop condition */
356 i2cREG1->MDR |= (uint32_t)I2C_REPEATMODE;
357 i2cSetStop(i2cREG1);
358 (void)xTaskNotifyIndexedFromISR(
359 I2C_TASK_HANDLE,
360 I2C_NOTIFICATION_RX_INDEX,
361 I2C_RX_NOTCOME_VALUE,
362 eSetValueWithOverwrite,
363 &xHigherPriorityTaskWoken);
364 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
365 } else {
366 i2c_rxLastByteInterface1 = I2C_ReadLastRxByte(i2cREG1);
367 (void)xTaskNotifyIndexedFromISR(
368 I2C_TASK_HANDLE,
369 I2C_NOTIFICATION_RX_INDEX,
370 I2C_RX_NOTIFIED_VALUE,
371 eSetValueWithOverwrite,
372 &xHigherPriorityTaskWoken);
373 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
374 }
375 break;
376 case DMA_CHANNEL_I2C2_RX:
377 i2cREG2->DMACR &= ~((uint32_t)I2C_RX_DMA_ENABLE);
378 /* Received all but the last byte, now wait until receive buffer is filled with the last byte to receive
379 */
380 success = I2C_WaitReceive(i2cREG2, I2C_TIMEOUT_us);
381 if (success == false) {
382 /* Set Stop condition */
383 i2cREG2->MDR |= (uint32_t)I2C_REPEATMODE;
384 i2cSetStop(i2cREG2);
385 (void)xTaskNotifyIndexedFromISR(
386 I2C_TASK_HANDLE,
387 I2C_NOTIFICATION_RX_INDEX,
388 I2C_RX_NOTCOME_VALUE,
389 eSetValueWithOverwrite,
390 &xHigherPriorityTaskWoken);
391 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
392 } else {
393 i2c_rxLastByteInterface2 = I2C_ReadLastRxByte(i2cREG2);
394 (void)xTaskNotifyIndexedFromISR(
395 I2C_TASK_HANDLE,
396 I2C_NOTIFICATION_RX_INDEX,
397 I2C_RX_NOTIFIED_VALUE,
398 eSetValueWithOverwrite,
399 &xHigherPriorityTaskWoken);
400 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
401 }
402 break;
403 default:
404 break;
405 }
406 }
407 /* Interrupt before last transfer is started */
408 if (inttype == (dmaInterrupt_t)LFS) {
409 if (channel == DMA_CHANNEL_I2C1_RX) {
410 /* Before last byte is received, set stop to generate NACK*/
411 i2cSetStop(i2cREG1);
412 }
413 if (channel == DMA_CHANNEL_I2C2_RX) {
414 /* Before last byte is received, set stop to generate NACK*/
415 i2cSetStop(i2cREG2);
416 }
417 }
418}
