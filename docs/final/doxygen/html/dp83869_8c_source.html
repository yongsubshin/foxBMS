135/**
136 * @brief Reads register from the Extended Address Space.
137 * @details The DP83869HM SMI function supports read or write access to the
138 * extended register set using registers REGCR (0x000Dh) and ADDAR
139 * (0x000Eh) and the MDIO Manageable Device (MMD) indirect method
140 * defined in IEEE 802.3ah Draft for clause 22 for accessing the
141 * clause 45 extended register set.
142 * The standard register set, MDIO registers 0 to 31, is accessed
143 * using the normal direct-MDIO access or the indirect method, except
144 * for register REGCR (0x000Dh) and ADDAR (0x000Eh) which is accessed
145 * only using the normal MDIO transaction. The SMI function ignores
146 * indirect accesses to these registers.
147 * @param mdioBaseAddress Base Address of the MDIO Module Registers.
148 * @param phyAddress PHY Address.
149 * @param regNum Register Number to be read.
150 * @param pData Pointer where the read value shall be written.
151 * @return true if reading was successful, false otherwise
152 **/

206static void PHY_SelectSpeed(uint32_t mdioBaseAddress, uint32_t phyAddress, PHY_SPEED_SEL_e speed) {
207 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
208 FAS_ASSERT(phyAddress < 32u);
209 /* AXIVION Routine Generic-MissingParameterAssert: speed: parameter accepts whole range */
210
211 uint16_t regVal = 0x0000u;
212 uint16_t *regPtr = &regVal;
213
214 MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, regPtr);
215 regVal &= ~(1 << 13);
216 regVal |= (speed & 2) << 12; /* Set SPEED_SEL_LSB */
217 regVal &= ~(1 << 6);
218 regVal |= (speed & 1) << 6; /* Set SPEED_SEL_MSB */
219 MDIOPhyRegWrite(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, regVal);
220}

223 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
224 FAS_ASSERT(phyAddress < 32u);
225
226 uint16_t regVal = 0u;
227 uint16_t *regPtr = &regVal;
228
229 MDIOPhyRegWrite(mdioBaseAddress, phyAddress, PHY_GEN_CTRL, PHY_SWRESTART);
230
231 (void)MDIOPhyRegRead(mdioBaseAddress, phyAddress, PHY_GEN_CTRL, regPtr);
232 /* : This bit is self-clearing and returns 1 until the reset process is complete. */

239extern STD_RETURN_TYPE_e PHY_Initialize(uint32_t mdioBaseAddressess) {
240 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
241
242 STD_RETURN_TYPE_e result = STD_OK;
243 uint32_t phyID = 0u;
244 uint32_t phyIdReadCount = 0xFu;
245 uint32_t phyLinkRetries = 0xFu;
246
247 /* Reset all registers before start */
248 PHY_ResetHardware();
249
250 /* Get PHY ID and check MDIO connection */
251 while ((phyID == 0u) && (phyIdReadCount > 0u)) {
252 phyID = PHY_GetId(mdioBaseAddressess, PHY_ADDRESS);
253 phyIdReadCount--;
254 /* Don't block other tasks to much */
255 OS_DelayTask(10u);
256 }
257
258 phyID = phyID >> 4; /* Don't compare the Revision Number */
259 if (0u == phyID) {
260 result = STD_NOT_OK;
261 } else if (PHY_PHY_ID >> 4 != phyID) {
262 result = STD_NOT_OK; /* Wrong PHY */
263 }
264
265 /* Check PHY alive status */
266 if (result == STD_OK) {
267 if (((MDIOPhyAliveStatusGet(mdioBaseAddressess) >> PHY_ADDRESS) & (uint32_t)0x01u) == (uint32_t)0u) {
268 result = STD_NOT_OK;
269 } else {
270 phy_tablePhy.aliveStatus = true;
271 }
272 }
273
274 /* Set PHY to MII-mode */
275 if ((result == STD_OK) && (PHY_SetMiiMode(mdioBaseAddressess, PHY_ADDRESS) == false)) {
276 result = STD_NOT_OK;
277 }
278
279 /* Check for network link */
280 if ((result == STD_OK) &&
281 (PHY_GetLinkStatus(mdioBaseAddressess, PHY_ADDRESS, (uint32_t)phyLinkRetries) != STD_OK)) {
282 result = STD_NOT_OK;
283 }
284
285 (void)DATA_WRITE_DATA(&phy_tablePhy);
286 return result;
287}

289extern STD_RETURN_TYPE_e PHY_GetLinkStatus(uint32_t mdioBaseAddress, uint32_t phyAddress, volatile uint32_t retries) {
290 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
291 FAS_ASSERT(phyAddress < 32u);
292 /* AXIVION Routine Generic-MissingParameterAssert: retries: parameter accepts whole range */
293
294 volatile uint16_t linkStatus = 0u;
295 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
296 bool readSuccess = false;
297
298 while (retries > 0u) {
299 /* First try to read the BMSR of the PHY*/
300 readSuccess = MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMSR, &linkStatus);
301 if (readSuccess == true) {
302 if ((linkStatus & PHY_LINK_STATUS) != 0u) {
303 /* Link status okay */
304 retVal = STD_OK;
305 phy_tablePhy.linkStatus = true;
306 break;
307 } else {
308 retries--;
309 /* Don't block other tasks to much */
310 OS_DelayTask(100u);
311 }
312 } else {
313 retries--;
314 /* Don't block other tasks to much */
315 OS_DelayTask(100u);
316 }
317 }
318 if (retVal == STD_NOT_OK) {
319 phy_tablePhy.linkStatus = false;
320 }
321 (void)DATA_WRITE_DATA(&phy_tablePhy);
322 return retVal;
323}

325extern uint32_t PHY_GetId(uint32_t mdioBaseAddress, uint32_t phyAddress) {
326 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
327 FAS_ASSERT(phyAddress < 32u);
328
329 uint32_t id = 0u;
330 uint16_t data = 0u;
331
332 /* read the ID1 register */
333 (void)MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_PHYIDR1, &data);
334
335 /* update the ID1 value */
336 id = (uint32_t)data;
337 id = (uint32_t)((uint32_t)id << PHY_ID_SHIFT);
338
339 /* read the ID2 register */
340 (void)MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_PHYIDR2, &data);
341
342 /* update the ID2 value */
343 id |= data;
344
345 /* return the ID in ID1:ID2 format */
346 return id;
347}

349extern STD_RETURN_TYPE_e PHY_AutoNegotiate(uint32_t mdioBaseAddress, uint32_t phyAddress, uint16_t advVal) {
350 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
351 FAS_ASSERT(phyAddress < 32u);
352 /* AXIVION Routine Generic-MissingParameterAssert: advVal: parameter accepts whole range */
353
354 volatile uint16_t data = 0u;
355 volatile uint16_t anar = 0u;
356 STD_RETURN_TYPE_e retVal = STD_OK;
357 uint32_t phyNegTries = 100u;
358
359 if (MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, &data) != true) {
360 retVal = STD_NOT_OK;
361 }
362
363 if (retVal == STD_OK) {
364 /* Enable Auto Negotiation */
365 data |= PHY_AUTONEG_ENABLE;
366 MDIOPhyRegWrite(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, data);
367 if (MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, &data) != true) {
368 retVal = STD_NOT_OK;
369 }
370 }
371
372 if (retVal == STD_OK) {
373 /* Write Auto Negotiation capabilities */
374 (void)MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_ANAR, &anar);
375 anar &= (uint16_t)(~0xFF10u);
376 MDIOPhyRegWrite(mdioBaseAddress, phyAddress, (uint32_t)PHY_ANAR, (anar | advVal));
377 /* Start Auto Negotiation */
378 data |= PHY_AUTONEG_RESTART;
379 MDIOPhyRegWrite(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, data);
380 data = 0u;
381 /* Get the auto negotiation status*/
382 if (MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMSR, &data) != true) {
383 retVal = STD_NOT_OK;
384 }
385 }
386
387 if (retVal == STD_OK) {
388 /* Wait till auto negotiation is complete */
389 while ((((uint16_t)(PHY_AUTONEG_INCOMPLETE)) == (data & (uint16_t)(PHY_AUTONEG_STATUS))) &&
390 (phyNegTries > 0u)) {
391 (void)MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMSR, &data);
392 phyNegTries--;
393 OS_DelayTask(PHY_AUTO_NEG_DELAY_TIME);
394 }
395
396 /* Check if the PHY is able to perform auto negotiation */
397 if ((data & PHY_AUTONEG_ABLE) == 0u) {
398 retVal = STD_NOT_OK;
399 }
400 }
401
402 return retVal;
403}

413extern bool PHY_SetMiiMode(uint32_t mdioBaseAddress, uint32_t phyAddress) {
414 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
415 FAS_ASSERT(phyAddress < 32u);
416
417 bool xReturn = false;
418 uint16_t regVal = 0x0000u;
419 uint16_t *regPtr = &regVal;
420
421 /* Reset PHY*/
422 PHY_ResetSoftware(mdioBaseAddress, phyAddress);
423
424 PHY_ReadExtendedAddressSpaceRegister(mdioBaseAddress, phyAddress, (uint32_t)PHY_OP_MODE, regPtr);
425
426 /* Setting MII mode. */
427 regVal = PHY_MII_MODE_SET;
428 PHY_WriteExtendedAddressSpaceRegister(mdioBaseAddress, phyAddress, (uint32_t)PHY_OP_MODE, regVal);
429
430 PHY_RestartSoftware(mdioBaseAddress, phyAddress);
431 regVal = 0x0000u;
432 PHY_ReadExtendedAddressSpaceRegister(mdioBaseAddress, phyAddress, (uint32_t)PHY_OP_MODE, regPtr);
433
434 if (PHY_MII_MODE_SET == regVal) {
435 xReturn = true;
436 }
437 return xReturn;
438}

455extern bool PHY_EnableLoopback(uint32_t mdioBaseAddress, uint32_t phyAddress) {
456 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
457 FAS_ASSERT(phyAddress < 32u);
458
459 uint16_t regVal = 0x0000u;
460 uint16_t *regPtr = &regVal;
461 boolean retVal = true;
462 uint8_t speed;
463
464 /* Disabling Auto Negotiate. */
465 (void)MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, regPtr);
466 regVal &= (uint16_t)(~((uint16_t)PHY_AUTONEG_ENABLE));
467 MDIOPhyRegWrite(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, regVal);
468
469 /* Force 10 Mbps mode */
470 PHY_SelectSpeed(mdioBaseAddress, phyAddress, mbps10);
471
472 /* Enabling Loopback. */
473 regVal |= PHY_LPBK_ENABLE;
474
475 MDIOPhyRegWrite(mdioBaseAddress, phyAddress, (uint32_t)PHY_BMCR, regVal);
476
477 /* In 100Base-TX mode after MII loopback is enabled through register 0x00, it is necessary
478 to write 0x0004 to register 0x16 for proper operation of MII Loopback*/
479 regVal = 0u;
480 MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_STATUS, regPtr);
481 speed = regVal >> 13u;
482 if (speed == mbps100) {
483 regVal = 0u;
484 MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BIST, regPtr);
485
486 /* Enable Digital Loopback*/
487 regVal |= 0x4u;
488 MDIOPhyRegWrite(mdioBaseAddress, phyAddress, (uint32_t)PHY_BIST, regVal);
489
490 regVal = 0u;
491 if (MDIOPhyRegRead(mdioBaseAddress, phyAddress, (uint32_t)PHY_BIST, regPtr) != true) {
492 retVal = FALSE;
493 }
494 }
495
496 /* When configuring loopback modes, the Loopback Configuration Register (LOOPCR), address 0x00FE, should be
497 set to 0xE720. */
498 PHY_WriteExtendedAddressSpaceRegister(mdioBaseAddress, phyAddress, 0x00FE, 0xE720);
499
500 /* Force 10 Mbps mode */
501 PHY_SelectSpeed(mdioBaseAddress, phyAddress, mbps10);
502
503 return retVal;
504}

529extern void PHY_GetOperationMode(uint32_t mdioBaseAddress, uint32_t phyAddress, PHY_OPERATION_MODE_s *pOperationMode) {
530 /* AXIVION Routine Generic-MissingParameterAssert: mdioBaseAddress: parameter accepts whole range */
531 FAS_ASSERT(phyAddress < 32u);
532 FAS_ASSERT(pOperationMode != NULL_PTR);
533
534 uint16_t regVal = 0x0000u;
535 uint16_t *regPtr = &regVal;
536
537 PHY_ReadExtendedAddressSpaceRegister(mdioBaseAddress, phyAddress, PHY_STRAP_STS, regPtr);
538
539 pOperationMode->rgmii2miiEnable = regVal & 0x1;
540 pOperationMode->autoNegotiationEnable = ~(regVal & 0x2);
541 pOperationMode->phyAddressess = (regVal >> 4) & 0x0F;
542 pOperationMode->operationMode = (regVal >> 9) & 0x07;
543 pOperationMode->mirrorEnable = (regVal >> 12);
544 pOperationMode->linkLossPassEnable = (uint8_t)~(regVal >> 13);
545}
