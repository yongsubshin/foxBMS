95/* Structure to hold the values of the PHY Opmode Register. */
96typedef struct {
97
98 bool rgmii2miiEnable; /*!< RGMII to MII Enable Strap: 0x0 = RGMII mode, 0x1 = MII Mode */
99 bool autoNegotiationEnable; /*!< Auto Negotiation Enable Strap: 0x0 = Enable, 0x1 = Disable*/
100 uint8_t phyAddressess; /*!< PHY Address Strap */
101 uint8_t operationMode; /*!< OPMODE Strap 0x0 = RGMII To Copper
102 0x1 = RGMII to 1000Base-X
103 0x2 = RGMII to 100Base-FX
104 0x3 = RGMII-SGMII Bridge
105 0x4 = 1000Base-T to 1000Base-X
106 0x5 = 100Base-T to 100Base-FX
107 0x6 = SGMII to Copper
108 0x7 = JTAG for Boundary Scan */
109 bool mirrorEnable; /*!< Mirror Mode Enable StraP. Refer to strap configuration section as this strap
110 also decides MAC interface in Bridge Mode applications: 0x0 = Disable, 0x1 = Enable*/
111 bool linkLossPassEnable; /*!< Link Loss Pass Through Enable Strap: 0x0 = Enable, 0x1 = Disable */
112
113} PHY_OPERATION_MODE_s;
