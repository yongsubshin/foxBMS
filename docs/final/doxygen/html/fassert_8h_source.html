42/**
43 * @file fassert.h
44 * @author foxBMS Team
45 * @date 2020-03-20 (date of creation)
46 * @updated 2025-08-07 (date of last update)
47 * @version v1.10.0
48 * @ingroup MAIN
49 * @prefix FAS
50 *
51 * @brief Assert macro implementation
52 * @details The default implementation accommodates three behaviors,
53 * based on FAS_ASSERT_LEVEL symbol:
54 *
55 * - When the FAS_ASSERT_LEVEL symbol is defined as
56 * FAS_ASSERT_LEVEL_INF_LOOP_AND_DISABLE_INTERRUPTS, the FAS_ASSERT
57 * macro is implemented as an infinite loop in which all interrupts
58 * are disabled. This will definitely trigger a watchdog reset.
59 *
60 * - When FAS_ASSERT_LEVEL symbol is defined as
61 * FAS_ASSERT_LEVEL_INF_LOOP_FOR_DEBUG, the validation performed by
62 * the FAS_ASSERT macro is enabled, and a failed validation triggers
63 * an infinite loop. This configuration is recommended for
64 * development environments, as it prevents further execution.
65 * Depending on the configuration this might lead to a watchdog
66 * reset. This mode is intended for investigation of problems by a
67 * developer.
68 *
69 * - When FAS_ASSERT_LEVEL symbol is defined as
70 * FAS_ASSERT_LEVEL_NO_OPERATION, the FAS_ASSERT macro is defined as
71 * empty and does nothing. It might be necessary to activate this
72 * mode in resource-constrained situations. Generally it is not
73 * recommended to use this option as it will not notice the
74 * undefined system-states that the assert should catch.
75 *
76 * Generally assertions should be used for asserting states and values
77 * that should occur when the program is executed correctly. Do *not*
78 * assert anything that *might* be false. As an example it is
79 * perfectly good practice to assert that the input to a function is
80 * not a null pointer if the function is not designed to accept null
81 * pointers, but it is not good to check if a communication error has
82 * occurred somewhere. Assertions should never trip in a bug-free
83 * program.
84 *
85 * Usage is FAS_ASSERT(condition that should be true).
86 */

154/**
155 * @def FAS_ASSERT_LEVEL_INF_LOOP_AND_DISABLE_INTERRUPTS
156 * @brief This assert level traps the complete program by going into an
157 * infinite loop and disabling all interrupts.
158 *
159 * @def FAS_ASSERT_LEVEL_INF_LOOP_FOR_DEBUG
160 * @brief This assert level traps the current task by going into an

179#if FAS_ASSERT_LEVEL == FAS_ASSERT_LEVEL_INF_LOOP_AND_DISABLE_INTERRUPTS
180/** Assert macro will trigger a watchdog reset */
181static inline void FAS_InfiniteLoop(void) {
182 /* disable IRQ interrupts */
183 FAS_DisableInterrupts();
184 /* AXIVION Next Codeline Style MisraC2012-2.2 Generic-NoEmptyLoops: an
185 infinite loop is intended to stop further code execution */
186 while (true) { /* Stay here until watchdog reset happens */
187 }
188}
189#elif FAS_ASSERT_LEVEL == FAS_ASSERT_LEVEL_INF_LOOP_FOR_DEBUG
190/** Assert macro will stay in infinite loop */
191static inline void FAS_InfiniteLoop(void) {
192 while (true) {
193 /* Stay here to ease debugging */
194 }
195}
196#elif FAS_ASSERT_LEVEL == FAS_ASSERT_LEVEL_NO_OPERATION
197static inline void FAS_InfiniteLoop(void) {
198}
199#else
200#error "Invalid value for FAS_ASSERT_LEVEL"
201#endif

213/**
214 * @brief Record the assert location
215 * @details Retrieves the program counter (with __curpc()) and line-number at
216 * the current location and passes it to #FAS_StoreAssertLocation().
217 *
218 * It is important that this is a macro in order to insert it directly
219 * at he assert location in the code
220 */
221/* AXIVION Next Construct Style MisraC2012-11.5: The program counter needs to be casted to platform register width */
222#define FAS_ASSERT_RECORD() \
223 do { \
224 uint32_t *pc = (uint32_t *)__curpc(); \
225 FAS_StoreAssertLocation(pc, __LINE__); \
226 } while (false)
227
228/**
229 * @def FAS_ASSERT(x)
230 * @brief Assertion macro that asserts that x is true
231 * @details This macro asserts the taken argument x. If the assertion fails
232 * it calls #FAS_ASSERT_RECORD() and then #FAS_InfiniteLoop().
233 *
234 * In unit tests this is replace by an exception that is thrown in
235 * order to be able to test for a failed assertion.
236 *
237 * Use this macro if you want to assert. If the assertion fails
238 * the macro will take action based on the configuration of this module.
239 * See #FAS_ASSERT_LEVEL for reference.
240 *
241 * If the macro passes, it is just ignored. If you want to definitely
242 * fail, you can use the value #FAS_TRAP as an argument which is
243 * designed to fail in all cases.
244 */
245#ifdef UNITY_UNIT_TEST
246#include "CException.h"
247#define FAS_ASSERT(x) \
248 if (!(x)) \
249 Throw(0)
250#else
251#define FAS_ASSERT(x) \
252 do { \
253 if (!(x)) { \
254 FAS_ASSERT_RECORD(); \
255 FAS_InfiniteLoop(); \
256 } \
257 } while (false)
258#endif
