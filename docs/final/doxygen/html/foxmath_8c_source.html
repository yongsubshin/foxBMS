93extern float_t MATH_LinearInterpolation(
94 const float_t x1,
95 const float_t y1,
96 const float_t x2,
97 const float_t y2,
98 const float_t x_interpolate) {
99 float_t slope = 0.0f;
100
101 if (fabsf(x1 - x2) >= FLT_EPSILON) {
102 /* Calculate slope */
103 slope = (y2 - y1) / (x2 - x1);
104 }
105 /* In the case that the if clause is not entered (x values are identical)
106 * -> no interpolation possible: return y1 value
107 * -> slope takes initialization value of 0.0f
108 */
109
110 /* Interpolate starting from x1/y1 */
111 float_t y_interpolate = y1 + (slope * (x_interpolate - x1));
112
113 return y_interpolate;
114}

120extern uint32_t MATH_SwapBytesUint32_t(const uint32_t val) {
121 const uint32_t alternating2PatternStartFF = 0xFF00FF00u;
122 const uint32_t alternating2PatternStart00 = 0x00FF00FFu;
123 const uint32_t intermediate = ((val << UTIL_SHIFT_ONE_BYTE) & alternating2PatternStartFF) |
124 ((val >> UTIL_SHIFT_ONE_BYTE) & alternating2PatternStart00);
125 return (intermediate << UTIL_SHIFT_TWO_BYTES) | (intermediate >> UTIL_SHIFT_TWO_BYTES);
126}
127
128extern uint64_t MATH_SwapBytesUint64_t(const uint64_t val) {
129 const uint64_t alternating2PatternStartFF = 0xFF00FF00FF00FF00uLL;
130 const uint64_t alternating2PatternStart00 = 0x00FF00FF00FF00FFuLL;
131 const uint64_t alternating4PatternStartFFFF = 0xFFFF0000FFFF0000uLL;
132 const uint64_t alternating4PatternStart0000 = 0x0000FFFF0000FFFFuLL;
133
134 uint64_t intermediate = ((val << UTIL_SHIFT_ONE_BYTE) & alternating2PatternStartFF) |
135 ((val >> UTIL_SHIFT_ONE_BYTE) & alternating2PatternStart00);
136 intermediate = ((intermediate << UTIL_SHIFT_TWO_BYTES) & alternating4PatternStartFFFF) |
137 ((intermediate >> UTIL_SHIFT_TWO_BYTES) & alternating4PatternStart0000);
138 return (intermediate << UTIL_SHIFT_FOUR_BYTES) | (intermediate >> UTIL_SHIFT_FOUR_BYTES);
139}
