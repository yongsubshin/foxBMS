99extern void FRAM_Initialize(void) {
100 uint32_t address = 0u;
101
102 /* Reset error flag at startup */
103 (void)DIAG_Handler(DIAG_ID_FRAM_READ_CRC_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
104 /* find address of all variables in FRAM by parsing length of data*/
105 for (uint8_t i = 0u; i < (uint8_t)FRAM_BLOCK_MAX; i++) {
106 (fram_databaseHeader[i]).address = address;
107 address += (fram_databaseHeader[i]).datalength + FRAM_CRC_HEADER_SIZE;
108 }
109
110 /* ASSERT that size of variables does not exceed FRAM size */
111 FAS_ASSERT(!(address > FRAM_MAX_ADDRESS));
112
113 /* Read FRAM version struct - Set CRC error flag if this fails */
114 if (FRAM_ACCESS_OK != FRAM_ReadData(FRAM_BLOCK_ID_VERSION)) {
115 (void)DIAG_Handler(DIAG_ID_FRAM_READ_CRC_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
116 }
117}

119extern STD_RETURN_TYPE_e FRAM_ReinitializeAllEntries(void) {
120 STD_RETURN_TYPE_e retVal = STD_OK;
121
122 /* Reset FRAM version struct information */
123 fram_version.project = FRAM_PROJECT_ID_FOXBMS_BASELINE;
124 fram_version.major = ver_versionInformation.major;
125 fram_version.minor = ver_versionInformation.minor;
126 fram_version.patch = ver_versionInformation.patch;
127
128 for (uint8_t i = 0u; i < (uint8_t)FRAM_BLOCK_MAX; i++) {
129 if (FRAM_WriteData((FRAM_BLOCK_ID_e)i) != FRAM_ACCESS_OK) {
130 retVal = STD_NOT_OK;
131 }
132 }
133 return retVal;
134}

136extern FRAM_RETURN_TYPE_e FRAM_WriteData(FRAM_BLOCK_ID_e blockId) {
137 FAS_ASSERT(blockId < FRAM_BLOCK_MAX);
138
139 uint16_t read = 0u;
140 uint64_t crc = 0u;
141 FRAM_RETURN_TYPE_e retVal = FRAM_ACCESS_OK;
142
143 /* FRAM must use SW Chip Select configuration*/
144 FAS_ASSERT(spi_framInterface.csType == SPI_CHIP_SELECT_SOFTWARE);
145
146 uint32_t address = (fram_databaseHeader[blockId]).address;
147 uint32_t size = (fram_databaseHeader[blockId]).datalength;
148 uint8_t *pWrite = (uint8_t *)(fram_databaseHeader[blockId].blockptr);
149
150 STD_RETURN_TYPE_e crcRetVal = CRC_CalculateCrc(&crc, pWrite, size);
151
152 if (crcRetVal == STD_OK) {
153 STD_RETURN_TYPE_e spiRetVal = SPI_Lock(SPI_GetSpiIndex(spi_framInterface.pNode));
154
155 if (spiRetVal == STD_OK) {
156 /* send write enable command */
157 IO_PinReset(spi_framInterface.pGioPort, spi_framInterface.csPin);
158 uint16_t write = FRAM_WRITE_ENABLE_COMMAND;
159 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
160 IO_PinSet(spi_framInterface.pGioPort, spi_framInterface.csPin);
161 MCU_Delay_us(FRAM_DELAY_AFTER_WRITE_ENABLE_US);
162
163 /* send data to write */
164 /* set chip select low to start transmission */
165 IO_PinReset(spi_framInterface.pGioPort, spi_framInterface.csPin);
166
167 /* send write command */
168 write = FRAM_WRITE_COMMAND;
169 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
170
171 /* send in this order: upper, middle and lower part of address */
172 write = (uint16_t)((address & FRAM_ADDRESS_UPPER_PART) >> UTIL_SHIFT_TWO_BYTES);
173 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
174 write = (uint16_t)((address & FRAM_ADDRESS_MIDDLE_PART) >> UTIL_SHIFT_ONE_BYTE);
175 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
176 write = (uint16_t)((address & FRAM_ADDRESS_LOWER_PART) >> UTIL_SHIFT_ZERO_BYTES);
177 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
178
179 /* send CRC */
180 pWrite = (uint8_t *)(&crc);
181 for (uint8_t i = 0u; i < FRAM_CRC_HEADER_SIZE; i++) {
182 write = (uint16_t)(*pWrite);
183 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
184 pWrite++;
185 }
186
187 pWrite = (uint8_t *)(fram_databaseHeader[blockId].blockptr);
188
189 /* send data */
190 while (size > 0u) {
191 write = (uint16_t)(*pWrite);
192 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
193 pWrite++;
194 size--;
195 }
196
197 /* set chip select high to start transmission */
198 IO_PinSet(spi_framInterface.pGioPort, spi_framInterface.csPin);
199
200 SPI_Unlock(SPI_GetSpiIndex(spi_framInterface.pNode));
201 } else {
202 retVal = FRAM_ACCESS_SPI_BUSY;
203 }
204 } else {
205 retVal = FRAM_ACCESS_CRC_BUSY;
206 }
207 return retVal;
208}

210extern FRAM_RETURN_TYPE_e FRAM_ReadData(FRAM_BLOCK_ID_e blockId) {
211 FAS_ASSERT(blockId < FRAM_BLOCK_MAX);
212
213 uint16_t read = 0u;
214 FRAM_RETURN_TYPE_e retVal = FRAM_ACCESS_OK;
215
216 /* FRAM must use SW Chip Select configuration*/
217 FAS_ASSERT(spi_framInterface.csType == SPI_CHIP_SELECT_SOFTWARE);
218
219 STD_RETURN_TYPE_e spiRetVal = SPI_Lock(SPI_GetSpiIndex(spi_framInterface.pNode));
220
221 if (spiRetVal == STD_OK) {
222 uint32_t address = (fram_databaseHeader[blockId]).address;
223 uint32_t size = (fram_databaseHeader[blockId]).datalength;
224
225 /* get data to be read */
226 /* set chip select low to start transmission */
227 IO_PinReset(spi_framInterface.pGioPort, spi_framInterface.csPin);
228
229 /* send write command */
230 uint16_t write = FRAM_READ_COMMAND;
231 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
232
233 /* send upper part of address */
234 write = (address & 0x3F0000u) >> 16u;
235 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
236
237 /* send middle part of address */
238 write = (address & 0x00FF00u) >> 8u;
239 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
240
241 /* send lower part of address */
242 write = address & 0x0000FFu;
243 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
244
245 /* read CRC */
246 uint64_t crcRead = 0u;
247 uint8_t *pRead = (uint8_t *)(&crcRead);
248 for (uint8_t i = 0u; i < FRAM_CRC_HEADER_SIZE; i++) {
249 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
250 *pRead = (uint8_t)read;
251 pRead++;
252 }
253
254 pRead = (uint8_t *)(fram_databaseHeader[blockId].blockptr);
255
256 /* read data */
257 write = 0;
258 while (size > 0u) {
259 SPI_FramTransmitReceiveData(&spi_framInterface, &write, &read, 1u);
260 *pRead = read & (uint8_t)UINT8_MAX;
261 pRead++;
262 size--;
263 }
264
265 /* set chip select high to start transmission */
266 IO_PinSet(spi_framInterface.pGioPort, spi_framInterface.csPin);
267
268 SPI_Unlock(SPI_GetSpiIndex(spi_framInterface.pNode));
269
270 pRead = (uint8_t *)(fram_databaseHeader[blockId].blockptr);
271 size = (fram_databaseHeader[blockId]).datalength;
272 uint64_t crcCalculated = 0u;
273 STD_RETURN_TYPE_e crcRetVal = CRC_CalculateCrc(&crcCalculated, pRead, size);
274
275 if (crcRetVal == STD_OK) {
276 if (crcRead != crcCalculated) {
277 (void)DIAG_Handler(DIAG_ID_FRAM_READ_CRC_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, blockId);
278 retVal = FRAM_ACCESS_CRC_ERROR;
279 }
280 } else {
281 retVal = FRAM_ACCESS_CRC_BUSY;
282 }
283 } else {
284 retVal = FRAM_ACCESS_SPI_BUSY;
285 }
286
287 return retVal;
288}
