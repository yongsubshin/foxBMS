73/**
74 * @brief IDs for projects that use a memory layout
75 * @details This enum can be use to differentiate between projects. While an older
76 * version of the same project should be upgradeable, the entry of another
77 * project should be just discarded if a conflict occurs.
78 *
79 * In order for the ID to stay the same it is important that the defined
80 * macros are not changed and that each and every macro has a different
81 * value.
82 */
83typedef uint16_t FRAM_PROJECT_ID;
84
85/** this is the standard main development branch */
86#define FRAM_PROJECT_ID_FOXBMS_BASELINE ((FRAM_PROJECT_ID)0u)

88/** fram block identification numbers */
89typedef enum {
90 FRAM_ACCESS_OK, /* Transaction with FRAM was successful */
91 FRAM_ACCESS_SPI_BUSY, /* SPI busy, transaction with FRAM could not take place */
92 FRAM_ACCESS_CRC_BUSY, /* CRC hardware busy, transaction with FRAM could not take place */
93 FRAM_ACCESS_CRC_ERROR, /* Read CRC does not match with CRC computed on read data */
94} FRAM_RETURN_TYPE_e;

117/**
118 * @brief Stores the version of the memory layout of the FRAM
119 * @details This struct stores with which memory-layout version the FRAM has
120 * been written. This allows the BMS to recognize an incompatible
121 * memory layout.
122 */
123typedef struct {
124 FRAM_PROJECT_ID project; /*!< an identifier for the project, it is not
125 intended to migrate between different projects */
126 uint8_t major; /*!< major release version */
127 uint8_t minor; /*!< minor release version */
128 uint8_t patch; /*!< patch release version */
129} FRAM_VERSION_s;

137/**
138 * state of charge (SOC). Since SOC is voltage dependent, three different
139 * values are used, min, max and average. SOC defined as a float_t number between
140 * 0.0f and 100.0f (0% and 100%)
141 */
142typedef struct {
143 float_t minimumSoc_perc[BS_NR_OF_STRINGS]; /*!< minimum SOC */
144 float_t maximumSoc_perc[BS_NR_OF_STRINGS]; /*!< maximum SOC */
145 float_t averageSoc_perc[BS_NR_OF_STRINGS]; /*!< average SOC */
146 float_t chargeThroughput_As[BS_NR_OF_STRINGS]; /*!< Charge througput */
147 float_t dischargeThroughput_As[BS_NR_OF_STRINGS]; /*!< Discharge througput */
148} FRAM_SOC_s;
149
150/**
151 * state of energy (SOE). Since SOE is voltage dependent, three different
152 * values are used, min, max and average. SOE defined as a float_t number between
153 * 0.0f and 100.0f (0% and 100%)
154 */
155typedef struct {
156 float_t minimumSoe_perc[BS_NR_OF_STRINGS]; /*!< minimum SOE */
157 float_t maximumSoe_perc[BS_NR_OF_STRINGS]; /*!< maximum SOE */
158 float_t averageSoe_perc[BS_NR_OF_STRINGS]; /*!< average SOE */
159 float_t chargeEnergyThroughput_Wh[BS_NR_OF_STRINGS]; /*!< inflow of energy */
160 float_t dischargeEnergyThroughput_Wh[BS_NR_OF_STRINGS]; /*!< outflow of energy */
161} FRAM_SOE_s;
162
163/** flag to indicate if a deep-discharge in a string has been detected */
