109#ifndef UNITY_UNIT_TEST
110static resetSource_t STU_GetResetSourceWithoutFlagReset(void) {
111 register resetSource_t rst_source;
112
113 if ((SYS_EXCEPTION & (uint32)POWERON_RESET) != 0U) {
114 /* power-on reset condition */
115 rst_source = POWERON_RESET;
116 } else if ((SYS_EXCEPTION & (uint32)EXT_RESET) != 0U) {
117 /*** Check for other causes of EXT_RESET that would take precedence **/
118 if ((SYS_EXCEPTION & (uint32)OSC_FAILURE_RESET) != 0U) {
119 /* Reset caused due to oscillator failure. Add user code here to handle oscillator failure */
120 rst_source = OSC_FAILURE_RESET;
121 } else if ((SYS_EXCEPTION & (uint32)WATCHDOG_RESET) != 0U) {
122 /* Reset caused due watchdog violation */
123 rst_source = WATCHDOG_RESET;
124 } else if ((SYS_EXCEPTION & (uint32)WATCHDOG2_RESET) != 0U) {
125 /* Reset caused due watchdog violation */
126 rst_source = WATCHDOG2_RESET;
127 } else if ((SYS_EXCEPTION & (uint32)SW_RESET) != 0U) {
128 /* Reset caused due to software reset. */
129 rst_source = SW_RESET;
130 } else {
131 /* Reset caused due to External reset. */
132 rst_source = EXT_RESET;
133 }
134 } else if ((SYS_EXCEPTION & (uint32)DEBUG_RESET) != 0U) {
135 /* Reset caused due Debug reset request */
136 rst_source = DEBUG_RESET;
137 } else if ((SYS_EXCEPTION & (uint32)CPU0_RESET) != 0U) {
138 /* Reset caused due to CPU0 reset. CPU reset can be caused by CPU self-test completion, or by toggling the "CPU
139 * RESET" bit of the CPU Reset Control Register. */
140 rst_source = CPU0_RESET;
141 } else {
142 /* No_reset occurred. */
143 rst_source = NO_RESET;
144 }
145 return rst_source;
146}
147#endif

153/* SourceId : STARTUP_SourceId_001 */
154/* DesignId : STARTUP_DesignId_001 */
155/* Requirements : HL_CONQ_STARTUP_SR1 */
156void _c_int00(void) {
157 register resetSource_t rstSrc;
158
159 /* Initialize Core Registers to avoid CCM Error */
160 _coreInitRegisters_();
161
162 /* Initialize Stack Pointers */
163 _coreInitStackPointer_();
164
165 /* Reset handler: the following instructions read from the system exception status register
166 * to identify the cause of the CPU reset.
167 */
168 /* Changed in comparison to TI _c_int00 implementation. Readout flags but do NOT clear them! */
169 rstSrc = STU_GetResetSourceWithoutFlagReset();
170
171 switch (rstSrc) {
172 case POWERON_RESET:
173 /* Initialize L2RAM to avoid ECC errors right after power on */
174 _memInit_();
175 /*SAFETYMCUSW 62 S MR:15.2, 15.5 <APPROVED> "Need to continue to handle POWERON Reset" */
176 /* FALLTHRU */
177 case DEBUG_RESET:
178 case EXT_RESET:
179 /* Initialize L2RAM to avoid ECC errors right after power on */
180 if (rstSrc != POWERON_RESET) {
181 _memInit_();
182 }
183
184 /* Enable CPU Event Export */
185 /* This allows the CPU to signal any single-bit or double-bit errors detected
186 * by its ECC logic for accesses to program flash or data RAM.
187 */
188 _coreEnableEventBusExport_();
189
190 /* Check if there were ESM group3 errors during power-up.
191 * These could occur during eFuse auto-load or during reads from flash OTP
192 * during power-up. Device operation is not reliable and not recommended
193 * in this case. */
194 if ((esmREG->SR1[2]) != 0U) {
195 esmGroup3Notification(esmREG, esmREG->SR1[2]);
196 }
197
198 /* Initialize System - Clock, Flash settings with Efuse self check */
199 systemInit();
200
201 /* Enable IRQ offset via Vic controller */
202 _coreEnableIrqVicOffset_();
203
204 /* Initialize VIM table */
205 vimInit();
206
207 /* Check if there were ESM group3 errors during power-up.
208 * These could occur during eFuse auto-load or during reads from flash OTP
209 * during power-up. Device operation is not reliable and not recommended
210 * in this case. */
211 esmInit();
212 break;
213
214 case OSC_FAILURE_RESET:
215 break;
216
217 case WATCHDOG_RESET:
218 case WATCHDOG2_RESET:
219 break;
220
221 case CPU0_RESET:
222 /* Enable CPU Event Export */
223 /* This allows the CPU to signal any single-bit or double-bit errors detected
224 * by its ECC logic for accesses to program flash or data RAM.
225 */
226 _coreEnableEventBusExport_();
227 break;
228
229 case SW_RESET:
230 break;
231
232 default:
233 break;
234 }
