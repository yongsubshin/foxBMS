160extern void FTSK_InitializeUserCodeEngine(void) {
161 /* Warning: Do not change the content of this function */
162 /* See function definition doxygen comment for details */
163 STD_RETURN_TYPE_e retval = DATA_Initialize();
164
165 if (retval == STD_NOT_OK) {
166 FAS_ASSERT(FAS_TRAP);
167 }
168
169 /* Suspend AFE task if unused, otherwise it will preempt all lower priority tasks */
170#if (FOXBMS_AFE_DRIVER_TYPE_FSM == 1)
171 OS_SuspendTask(ftsk_taskHandleAfe);
172#endif
173
174 /* Init FRAM */
175 FRAM_Initialize();
176
177 retval = SYSM_Initialize();
178
179 if (retval == STD_NOT_OK) {
180 FAS_ASSERT(FAS_TRAP);
181 }
182
183 /* Warning: Do not change the content of this function */
184 /* See function definition doxygen comment for details */
185}

196extern void FTSK_InitializeUserCodePreCyclicTasks(void) {
197 /* Init Sys */
198 SYS_RETURN_TYPE_e sys_retVal = SYS_SetStateRequest(SYS_STATE_INITIALIZATION_REQUEST);
199
200 /* Init port expander */
201 PEX_Initialize();
202
203 /* Set 3rd PE pin 0 to activate temperature/humidity sensor */
204 PEX_SetPinDirectionOutput(PEX_PORT_EXPANDER3, PEX_PORT_0_PIN_0);
205 PEX_SetPin(PEX_PORT_EXPANDER3, PEX_PORT_0_PIN_0);
206
207 CONT_Initialize();
208 SPS_Initialize();
209 (void)MEAS_Initialize(); /* cast to void as the return value is unused */
210
211 /* Initialize redundancy module */
212 (void)MRC_Initialize();
213
214 MDIOInit(PHY_MDIO_BASE, MDIO_FREQ_INPUT, MDIO_FREQ_OUTPUT);
215
216 (void)PHY_Initialize(PHY_MDIO_BASE);
217
218 /* This function operates under the assumption that it is called when
219 * the operating system is not yet running.
220 * In this state the return value of #SYS_SetStateRequest should
221 * always be #SYS_OK. Therefore we trap otherwise.
222 */
223 FAS_ASSERT(sys_retVal == SYS_OK);
224
225 /* System started correctly -> Start toggling of debug LED */
226 LED_SetToggleTime(LED_NORMAL_OPERATION_ON_OFF_TIME_ms);
227}

229extern void FTSK_RunUserCodeCyclic1ms(void) {
230 /* Increment of operating system timer */
231 /* This must not be changed, add user code only below */
232 OS_IncrementTimer();
233 DIAG_UpdateFlags();
234 /* user code */
235#if (FOXBMS_AFE_DRIVER_TYPE_FSM == 1)
236 MEAS_Control();
237#endif
238 CAN_ReadRxBuffer();
239}

241extern void FTSK_RunUserCodeCyclic10ms(void) {
242 static uint8_t ftsk_cyclic10msCounter = 0;
243 /* user code */
244 SYSM_UpdateFramData();
245 SYS_Trigger(&sys_state);
246 ILCK_Trigger();
247 ADC_Control();
248 SPS_Ctrl();
249 CAN_MainFunction();
250 SOF_Calculation();
251 ALGO_MonitorExecutionTime();
252 SBC_Trigger(&sbc_stateMcuSupervisor);
253
254 if (ftsk_cyclic10msCounter == TASK_10MS_COUNTER_FOR_50MS) {
255 MRC_ValidateAfeMeasurement();
256 MRC_ValidatePackMeasurement();
257 ftsk_cyclic10msCounter = 0;
258 }
259 /* Call BMS_Trigger function at the end of the 10ms task to allow previously
260 * called modules in this task to update respectively evaluate their new.
261 * This minimizes the delay between data evaluation and the reaction from
262 * the BMS module */
263 BMS_Trigger();
264 ftsk_cyclic10msCounter++;
265}

267extern void FTSK_RunUserCodeCyclic100ms(void) {
268 /* user code */
269 static uint8_t ftsk_cyclic100msCounter = 0;
270
271 /** Perform SOC and SOE calculations only every 1s. Not suited if analog
272 * integration of current sensor is NOT used. Manual integration of current
273 * requires a higher frequency.
274 */
275 if (ftsk_cyclic100msCounter == TASK_100MS_COUNTER_FOR_1S) {
276 SE_RunStateEstimations();
277 ftsk_cyclic100msCounter = 0;
278 }
279
280 BAL_Trigger();
281 IMD_Trigger();
282 LED_Trigger();
283 MINFO_CheckSupplyVoltageClamp30c();
284
285 ftsk_cyclic100msCounter++;
286}
