108/**
109 * Definition of single measurement command
110 * data sheet February 2019 - Version 6, table 9, page 10
111 *
112 * Clock stretching can be used to let sensor tell that the measurement
113 * is not finished by pulling the clock low
114 * Not recommended: if system is stopped during clock stretching,
115 * sensor could be blocked in this mode and render I2C bus unusable.
116 * !!!---WARNING---!!!: if clock stretching is used, the timeout in the
117 * receive loop of the function #I2C_Read() must be adapted,
118 * the current value is too small compared to the time the slave will
119 * make the master wait in the clock stretching mode.
120 *
121 * Depending on the clock stretching setting, the code for the instructions
122 * changes:
123 * -> With stretching
124 * - 1 MSB
125 * - 1 LSB for high repeatability
126 * - 1 LSB for medium repeatability
127 * - 1 LSB for low repeatability
128 * -> Without stretching
129 * - 1 MSB
130 * - 1 LSB for high repeatability
131 * - 1 LSB for medium repeatability
132 * - 1 LSB for low repeatability
133 * This results in 6 possible configurations
134 * Durability changes the measurement duration.
135 * @{
136 */

142#define HTSEN_LOW_REPEATABILITY (0x16u)
143#else

145#define HTSEN_HIGH_REPEATABILITY (0x06u)
146#define HTSEN_MEDIUM_REPEATABILITY (0x0Du)
147#define HTSEN_LOW_REPEATABILITY (0x10u)

192static uint8_t HTSEN_CalculateCrc8(const uint8_t *data, uint32_t length) {
193 FAS_ASSERT(data != NULL_PTR);
194 FAS_ASSERT(length <= UINT32_MAX);
195
196 uint16_t crc = HTSEN_CRC_SEED;
197
198 for (uint8_t i = 0u; i < length; i++) {
199 crc ^= data[i];
200 for (uint8_t j = 0u; j < HTSEN_CRC_BYTE_SHIFT; j++) {
201 if ((crc & HTSEN_CRC_MSB_MASK) != 0u) {
202 crc = ((crc << 1u) & HTSEN_CRC_8BIT_MASK) ^ HTSEN_CRC_POLYNOMIAL;
203 } else {
204 crc = (crc & HTSEN_CRC_8BIT_MASK) << 1u;
205 }
206 }
207 }
208 return (uint8_t)(crc & HTSEN_CRC_8BIT_MASK);
209}

211static int16_t HTSEN_ConvertRawTemperature(uint16_t data) {
212 /* AXIVION Routine Generic-MissingParameterAssert: data: parameter accepts whole range */
213 float_t temperature_ddeg = HTSEN_TEMP_DEG_TO_DDEG *
214 (HTSEN_TEMP_OFFSET + (HTSEN_TEMP_SCALING * (((float_t)data) / HTSEN_FULL_SCALE)));
215 return (int16_t)temperature_ddeg;
216}
217
218static uint8_t HTSEN_ConvertRawHumidity(uint16_t data) {
219 float_t humidity_perc = (HTSEN_HUMIDITY_SCALING * (((float_t)data) / HTSEN_FULL_SCALE));
220 return (uint8_t)humidity_perc;
221}

225extern void HTSEN_Trigger(void) {
226 static HTSEN_STATE_e htsenState = HTSEN_START_MEAS;
227 STD_RETURN_TYPE_e htsenReturnValue = STD_OK;
228 uint32_t current_time = OS_GetTickCount();
229
230 switch (htsenState) {
231 case HTSEN_START_MEAS:
232 /* Trigger a measurement */
233 htsenReturnValue = I2C_WriteDma(HTSEN_I2C_INTERFACE, HTSEN_I2C_ADDRESS, 2u, htsen_i2cWriteBuffer);
234 OS_DelayTaskUntil(&current_time, 2u);
235 if (htsenReturnValue == STD_OK) {
236 htsenState = HTSEN_READ_RESULTS;
237 }
238 htsen_readTries = HTSEN_READ_TRIES;
239 break;
240 case HTSEN_READ_RESULTS:
241 /* Try to read values */
242 htsenReturnValue = I2C_ReadDma(
243 HTSEN_I2C_INTERFACE, HTSEN_I2C_ADDRESS, HTSEN_TOTAL_DATA_LENGTH_IN_BYTES, htsen_i2cReadBuffer);
244 OS_DelayTaskUntil(&current_time, 2u);
245 if (htsenReturnValue == STD_OK) {
246 /* If sensor acknowledges on I2C bus, results are available */
247 /* Check if CRC valid */
248 /* Only take temperature value if CRC valid */
249 if (htsen_i2cReadBuffer[HTSEN_TEMPERATURE_BYTE_CRC] ==
250 HTSEN_CalculateCrc8(
251 &htsen_i2cReadBuffer[HTSEN_TEMPERATURE_MSB], HTSEN_MEASUREMENT_LENGTH_IN_BYTES)) {
252 htsen_data.temperature_ddegC = HTSEN_ConvertRawTemperature(
253 (((uint16_t)htsen_i2cReadBuffer[HTSEN_TEMPERATURE_MSB]) << HTSEN_BYTE_SHIFT) |
254 (uint16_t)htsen_i2cReadBuffer[HTSEN_TEMPERATURE_LSB]);
255 }
256 /* Only take humidity value if CRC valid */
257 if (htsen_i2cReadBuffer[HTSEN_HUMIDITY_BYTE_CRC] ==
258 HTSEN_CalculateCrc8(&htsen_i2cReadBuffer[HTSEN_HUMIDITY_MSB], HTSEN_MEASUREMENT_LENGTH_IN_BYTES)) {
259 htsen_data.humidity_perc = HTSEN_ConvertRawHumidity(
260 (((uint16_t)htsen_i2cReadBuffer[HTSEN_HUMIDITY_MSB]) << HTSEN_BYTE_SHIFT) |
261 (uint16_t)htsen_i2cReadBuffer[HTSEN_HUMIDITY_LSB]);
262 }
263 (void)DATA_WRITE_DATA(&htsen_data);
264 htsenState = HTSEN_START_MEAS;
265 } else {
266 /* If sensor does not acknowledge on I2C bus, results are not available ye */
267 if (htsen_readTries > 0u) {
268 htsen_readTries--;
269 } else {
270 htsenState = HTSEN_START_MEAS;
271 }
272 }
273 break;
274 default:
275 /* invalid state */
276 FAS_ASSERT(FAS_TRAP);
277 break;
278 }
279}
