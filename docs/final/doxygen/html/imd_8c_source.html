195static IMD_RETURN_TYPE_e IMD_SetStateRequest(IMD_STATE_s *pImdState, IMD_STATE_REQUEST_e stateRequest) {
196 FAS_ASSERT(pImdState != NULL_PTR);
197 /* AXIVION Routine Generic-MissingParameterAssert: stateRequest: parameter accepts whole range */
198 IMD_RETURN_TYPE_e retVal = IMD_REQUEST_OK;
199
200 OS_EnterTaskCritical();
201 retVal = IMD_CheckStateRequest(pImdState, stateRequest);
202
203 if (retVal == IMD_REQUEST_OK) {
204 pImdState->stateRequest = stateRequest;
205 }
206 OS_ExitTaskCritical();
207
208 return retVal;
209}

211static IMD_RETURN_TYPE_e IMD_CheckStateRequest(IMD_STATE_s *pImdState, IMD_STATE_REQUEST_e stateRequest) {
212 FAS_ASSERT(pImdState != NULL_PTR);
213 /* AXIVION Routine Generic-MissingParameterAssert: stateRequest: parameter accepts whole range */
214 IMD_RETURN_TYPE_e retval = IMD_REQUEST_PENDING;
215 if (pImdState->stateRequest == IMD_STATE_NO_REQUEST) {
216 /* init only allowed from the uninitialized state */
217 if (stateRequest == IMD_STATE_INITIALIZE_REQUEST) {
218 if (pImdState->currentState == IMD_FSM_STATE_UNINITIALIZED) {
219 retval = IMD_REQUEST_OK;
220 } else {
221 retval = IMD_ALREADY_INITIALIZED;
222 }
223 } else if (stateRequest == IMD_STATE_SWITCH_ON_REQUEST) {
224 if (pImdState->currentState == IMD_FSM_STATE_SHUTDOWN) {
225 retval = IMD_REQUEST_OK;
226 } else if (pImdState->currentState == IMD_FSM_STATE_IMD_ENABLE) {
227 retval = IMD_REQUEST_OK;
228 } else {
229 retval = IMD_ILLEGAL_REQUEST;
230 }
231 } else {
232 retval = IMD_ILLEGAL_REQUEST;
233 }
234 } else {
235 /* Request pending */
236 }
237 return retval;
238}

249static IMD_CHECK_MULTIPLE_CALLS_e IMD_CheckMultipleCalls(IMD_STATE_s *pImdState) {
250 FAS_ASSERT(pImdState != NULL_PTR);
251 IMD_CHECK_MULTIPLE_CALLS_e multipleCalls = IMD_MULTIPLE_CALLS_NO;
252 OS_EnterTaskCritical();
253 if (pImdState->triggerEntry == 0u) {
254 pImdState->triggerEntry++;
255 } else {
256 multipleCalls = IMD_MULTIPLE_CALLS_YES; /* multiple call of function IMD_Trigger for instance pImdState */
257 }
258 OS_ExitTaskCritical();
259 return multipleCalls;
260}

262static void IMD_SetState(
263 IMD_STATE_s *pImdState,
264 IMD_FSM_STATES_e nextState,
265 IMD_FSM_SUBSTATES_e nextSubstate,
266 uint16_t idleTime) {
267 FAS_ASSERT(pImdState != NULL_PTR);
268 /* AXIVION Routine Generic-MissingParameterAssert: idleTime: parameter accepts whole range */
269 /* AXIVION Routine Generic-MissingParameterAssert: nextState: parameter accepts whole range */
270 /* AXIVION Routine Generic-MissingParameterAssert: nextSubstate: parameter accepts whole range */
271 bool earlyExit = false;
272
273 pImdState->timer = idleTime;
274
275 if ((pImdState->currentState == nextState) && (pImdState->currentSubstate == nextSubstate)) {
276 /* Next state and next substate equal to current state and substate: nothing to do */
277 pImdState->nextState = IMD_FSM_STATE_DUMMY; /* no state transition required -> reset */
278 pImdState->nextSubstate = IMD_FSM_SUBSTATE_DUMMY; /* no substate transition required -> reset */
279 earlyExit = true;
280 }
281
282 if (earlyExit == false) {
283 if (pImdState->currentState != nextState) {
284 /* Next state is different: switch to it and set substate to entry value */
285 pImdState->previousState = pImdState->currentState;
286 pImdState->currentState = nextState;
287 pImdState->previousSubstate = pImdState->currentSubstate;
288 pImdState->currentSubstate = IMD_FSM_SUBSTATE_ENTRY; /* Use entry state after a top level state change */
289 pImdState->nextState = IMD_FSM_STATE_DUMMY; /* no state transition required -> reset */
290 pImdState->nextSubstate = IMD_FSM_SUBSTATE_DUMMY; /* no substate transition required -> reset */
291 } else if (pImdState->currentSubstate != nextSubstate) {
292 /* Only the next substate is different, switch to it */
293 IMD_SetSubstate(pImdState, nextSubstate, idleTime);
294 } else {
295 ;
296 }
297 }
298}

300static void IMD_SetSubstate(IMD_STATE_s *pImdState, IMD_FSM_SUBSTATES_e nextSubstate, uint16_t idleTime) {
301 FAS_ASSERT(pImdState != NULL_PTR);
302 /* AXIVION Routine Generic-MissingParameterAssert: nextSubstate: parameter accepts whole range */
303 /* AXIVION Routine Generic-MissingParameterAssert: idleTime: parameter accepts whole range */
304 pImdState->timer = idleTime;
305 pImdState->previousSubstate = pImdState->currentSubstate;
306 pImdState->currentSubstate = nextSubstate;
307 pImdState->nextSubstate = IMD_FSM_SUBSTATE_DUMMY; /* substate has been set, now reset value for nextSubstate */
308}

310static STD_RETURN_TYPE_e IMD_RunStateMachine(IMD_STATE_s *pImdState) {
311 FAS_ASSERT(pImdState != NULL_PTR);
312 STD_RETURN_TYPE_e ranStateMachine = STD_OK;
313 IMD_FSM_STATES_e nextState = IMD_FSM_STATE_DUMMY;
314 IMD_STATE_REQUEST_e stateRequest = IMD_STATE_NO_REQUEST;
315 switch (pImdState->currentState) {
316 /********************************************** STATE: HAS NEVER RUN */
317 case IMD_FSM_STATE_HAS_NEVER_RUN:
318 /* Nothing to do, just transfer */
319 IMD_SetState(pImdState, IMD_FSM_STATE_UNINITIALIZED, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
320 break;
321
322 /********************************************** STATE: UNINITIALIZED */
323 case IMD_FSM_STATE_UNINITIALIZED:
324 /* Transition to initialization state, one the state request has been received */
325 stateRequest = IMD_TransferStateRequest(pImdState);
326 if (stateRequest == IMD_STATE_INITIALIZE_REQUEST) {
327 IMD_SetState(pImdState, IMD_FSM_STATE_INITIALIZATION, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
328 } else {
329 /* Do nothing. Stay in state. */
330 }
331 break;
332
333 /********************************************* STATE: INITIALIZATION */
334 case IMD_FSM_STATE_INITIALIZATION:
335 nextState = IMD_ProcessInitializationState();
336 if (nextState == IMD_FSM_STATE_INITIALIZATION) {
337 /* staying in state, processed by state function */
338 } else if (nextState == IMD_FSM_STATE_ERROR) {
339 IMD_SetState(pImdState, IMD_FSM_STATE_ERROR, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
340 } else if (nextState == IMD_FSM_STATE_IMD_ENABLE) {
341 pImdState->information.isStateMachineInitialized = true;
342 IMD_SetState(pImdState, IMD_FSM_STATE_IMD_ENABLE, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
343 } else {
344 FAS_ASSERT(FAS_TRAP); /* Something went wrong */
345 }
346 break;
347
348 case IMD_FSM_STATE_IMD_ENABLE:
349 /* Transition to running state, once the state request has been received */
350 stateRequest = IMD_TransferStateRequest(pImdState);
351 if (stateRequest == IMD_STATE_SWITCH_ON_REQUEST) {
352 pImdState->information.switchImdDeviceOn = true;
353 }
354 if (pImdState->information.switchImdDeviceOn == true) {
355 nextState = IMD_ProcessEnableState();
356 if (nextState == IMD_FSM_STATE_RUNNING) {
357 /* Set flag that IMD is running. Do not update the database entry here. The database entry will be
358 * updated after first successful measurement in IMD_FSM_STATE_RUNNING state */
359 pImdState->pTableImd->isImdRunning = true;
360 pImdState->information.switchImdDeviceOn = false;
361 IMD_SetState(pImdState, IMD_FSM_STATE_RUNNING, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
362 } else if (nextState == IMD_FSM_STATE_ERROR) {
363 IMD_SetState(pImdState, IMD_FSM_STATE_ERROR, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
364 } else {
365 /* Do nothing as the process to activate the IMD device is ongoing */
366 }
367 } else {
368 /* Do nothing. Stay in state. */
369 }
370 break;
371
372 /**************************************************** STATE: RUNNING */
373 case IMD_FSM_STATE_RUNNING:
374 stateRequest = IMD_TransferStateRequest(pImdState);
375 if (stateRequest == IMD_STATE_SHUTDOWN_REQUEST) {
376 IMD_SetState(pImdState, IMD_FSM_STATE_SHUTDOWN, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
377 } else {
378 nextState = IMD_ProcessRunningState(pImdState->pTableImd);
379 /* Evaluate measurement results */
380 IMD_EvaluateInsulationMeasurement(pImdState->pTableImd);
381 if (nextState == IMD_FSM_STATE_RUNNING) {
382 /* staying in state, processed by state function */
383 } else {
384 IMD_SetState(pImdState, IMD_FSM_STATE_ERROR, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
385 }
386 }
387 break;
388
389 case IMD_FSM_STATE_SHUTDOWN:
390 nextState = IMD_ProcessShutdownState();
391 if (nextState == IMD_FSM_STATE_IMD_ENABLE) {
392 /* Set flag, that IMD is deactivated. Update database entry */
393 pImdState->pTableImd->isImdRunning = false;
394 (void)DATA_WRITE_DATA(&imd_tableInsulationMonitoring);
395 IMD_SetState(pImdState, IMD_FSM_STATE_IMD_ENABLE, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
396 } else if (nextState == IMD_FSM_STATE_ERROR) {
397 IMD_SetState(pImdState, IMD_FSM_STATE_ERROR, IMD_FSM_SUBSTATE_ENTRY, IMD_FSM_SHORT_TIME);
398 } else {
399 /* staying in state, processed by state function */
400 }
401 break;
402
403 /****************************************************** STATE: ERROR */
404 case IMD_FSM_STATE_ERROR:
405 /* Issue: 621 */
406 FAS_ASSERT(FAS_TRAP);
407 break;
408
409 /**************************************************** STATE: DEFAULT */
410 default:
411 /* all cases must be processed, trap if unknown state arrives */
412 FAS_ASSERT(FAS_TRAP);
413 break;
414 }
415 /* Increment general purpose counter */
416 pImdState->counter++;
417 return ranStateMachine;
418}

420static STD_RETURN_TYPE_e IMD_EvaluateInsulationMeasurement(
421 DATA_BLOCK_INSULATION_MONITORING_s *pTableInsulationMonitoring) {
422 FAS_ASSERT(pTableInsulationMonitoring != NULL_PTR);
423 /* Assume resistance value as good if no valid measurement values are detected */
424 bool lowResistanceDetected = false;
425
426 /* Check if measured resistance value is valid */
427 if (pTableInsulationMonitoring->isInsulationMeasurementValid == true) {
428 (void)DIAG_Handler(DIAG_ID_INSULATION_MEASUREMENT_VALID, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
429 if (pTableInsulationMonitoring->insulationResistance_kOhm < IMD_ERROR_THRESHOLD_INSULATION_RESISTANCE_kOhm) {
430 lowResistanceDetected = true;
431 }
432 } else {
433 (void)DIAG_Handler(DIAG_ID_INSULATION_MEASUREMENT_VALID, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
434 }
435
436 /* Check if flags of IMD are valid, e.g. status pin */
437 if (pTableInsulationMonitoring->areDeviceFlagsValid == true) {
438 if (pTableInsulationMonitoring->dfIsCriticalResistanceDetected == true) {
439 lowResistanceDetected = true;
440 }
441 if (pTableInsulationMonitoring->dfIsChassisFaultDetected == true) {
442 (void)DIAG_Handler(DIAG_ID_INSULATION_GROUND_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
443 } else {
444 (void)DIAG_Handler(DIAG_ID_INSULATION_GROUND_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
445 }
446 }
447
448 /* Set diagnosis entry depending on measured insulation resistance and critical threshold flag */
449 if (lowResistanceDetected == true) {
450 (void)DIAG_Handler(DIAG_ID_LOW_INSULATION_RESISTANCE_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
451 } else {
452 (void)DIAG_Handler(DIAG_ID_LOW_INSULATION_RESISTANCE_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
453 }
454
455 /* Set warning threshold flag depending on flag */
456 if (pTableInsulationMonitoring->dfIsWarnableResistanceDetected == true) {
457 (void)DIAG_Handler(DIAG_ID_LOW_INSULATION_RESISTANCE_WARNING, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
458 } else {
459 (void)DIAG_Handler(DIAG_ID_LOW_INSULATION_RESISTANCE_WARNING, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
460 }
461
462 /* Write database entry */
463 (void)DATA_WRITE_DATA(pTableInsulationMonitoring);
464
465 /* Issue: 621 */
466
467 return STD_OK;
468}

487extern STD_RETURN_TYPE_e IMD_Trigger(void) {
488 bool earlyExit = false;
489 STD_RETURN_TYPE_e returnValue = STD_OK;
490
491 /* Check multiple calls of function */
492 if (IMD_MULTIPLE_CALLS_YES == IMD_CheckMultipleCalls(&imd_state)) {
493 returnValue = STD_NOT_OK;
494 earlyExit = true;
495 }
496
497 if (earlyExit == false) {
498 if (imd_state.timer > 0u) {
499 if ((--imd_state.timer) > 0u) {
500 imd_state.triggerEntry--;
501 returnValue = STD_OK;
502 earlyExit = true;
503 }
504 }
505 }
506
507 if (earlyExit == false) {
508 IMD_RunStateMachine(&imd_state);
509 imd_state.triggerEntry--;
510 }
511 return returnValue;
512}
