153/*========== Static Function Implementations ================================*/
154static ILCK_RETURN_TYPE_e ILCK_CheckStateRequest(ILCK_STATE_REQUEST_e statereq) {
155 ILCK_RETURN_TYPE_e stateRequestCheck = ILCK_ILLEGAL_REQUEST;
156 if (!((statereq == ILCK_STATE_INITIALIZATION_REQUEST) || (statereq == ILCK_STATE_NO_REQUEST))) {
157 stateRequestCheck = ILCK_ILLEGAL_REQUEST;
158 } else if (ilck_state.statereq == ILCK_STATE_NO_REQUEST) {
159 /* init only allowed from the uninitialized state */
160 if (statereq == ILCK_STATE_INITIALIZATION_REQUEST) {
161 if (ilck_state.state == ILCK_STATEMACHINE_UNINITIALIZED) {
162 stateRequestCheck = ILCK_OK;
163 } else {
164 stateRequestCheck = ILCK_ALREADY_INITIALIZED;
165 }
166 }
167 } else {
168 stateRequestCheck = ILCK_REQUEST_PENDING;
169 }
170 return stateRequestCheck;
171}

173static ILCK_STATE_REQUEST_e ILCK_TransferStateRequest(void) {
174 ILCK_STATE_REQUEST_e retval = ILCK_STATE_NO_REQUEST;
175
176 OS_EnterTaskCritical();
177 retval = ilck_state.statereq;
178 ilck_state.statereq = ILCK_STATE_NO_REQUEST;
179 OS_ExitTaskCritical();
180
181 return retval;
182}

184static uint8_t ILCK_CheckReEntrance(void) {
185 uint8_t retval = 0;
186
187 OS_EnterTaskCritical();
188 if (!ilck_state.triggerentry) {
189 ilck_state.triggerentry++;
190 } else {
191 retval = 0xFF; /* multiple calls of function */
192 }
193 OS_ExitTaskCritical();
194
195 return retval;
196}

198static void ILCK_InitializePins(void) {
199 /* Configure diagnostic supply enable pin as output */
200 IO_SetPinDirectionToOutput(&ILCK_IO_REG_DIR, ILCK_INTERLOCK_CONTROL_PIN_IL_HS_ENABLE);

202 IO_PinReset(&ILCK_IO_REG_PORT->DOUT, ILCK_INTERLOCK_CONTROL_PIN_IL_HS_ENABLE);
203 /* Configure interlock feedback pin as input */
204 IO_SetPinDirectionToInput(&ILCK_IO_REG_DIR, ILCK_INTERLOCK_FEEDBACK_PIN_IL_STATE);

207static ILCK_ELECTRICAL_STATE_TYPE_e ILCK_GetInterlockFeedback(void) {
208 ILCK_ELECTRICAL_STATE_TYPE_e measuredInterlockState = ILCK_SWITCH_UNDEF;
209
210 OS_EnterTaskCritical();
211 const STD_PIN_STATE_e pinState = IO_PinGet(&ILCK_IO_REG_PORT->DIN, ILCK_INTERLOCK_FEEDBACK_PIN_IL_STATE);
212 OS_ExitTaskCritical();
213
214 /** Local variable containing voltages measured on TMS570 ADC1 inputs */
215 DATA_BLOCK_ADC_VOLTAGE_s ilck_tableAdcVoltages = {.header.uniqueId = DATA_BLOCK_ID_ADC_VOLTAGE};
216 (void)DATA_READ_DATA(&ilck_tableAdcVoltages);
217
218 /** Pin low: interlock closed, pin high: interlock open */
219 if (pinState == STD_PIN_HIGH) {
220 measuredInterlockState = ILCK_SWITCH_OFF;
221 } else if (pinState == STD_PIN_LOW) {
222 measuredInterlockState = ILCK_SWITCH_ON;
223 } else {
224 /* MISRA C:2012 Rule 15.7 - All if-else-if chains must have terminal else clause */
225 /* Pin state undefined - keep default ILCK_SWITCH_UNDEF value */
226 }
227
228 ilck_tableFeedback.interlockVoltageFeedback_IL_HS_VS_mV =
229 ilck_tableAdcVoltages.adc1ConvertedVoltages_mV[ILCK_ADC_INPUT_HIGH_SIDE_VOLTAGE_SENSE] *
230 ILCK_VOLTAGE_DIVIDER_FACTOR;
231 ilck_tableFeedback.interlockVoltageFeedback_IL_LS_VS_mV =
232 ilck_tableAdcVoltages.adc1ConvertedVoltages_mV[ILCK_ADC_INPUT_LOW_SIDE_VOLTAGE_SENSE] *
233 ILCK_VOLTAGE_DIVIDER_FACTOR;
234 ilck_tableFeedback.interlockCurrentFeedback_IL_HS_CS_mA =
235 ilck_tableAdcVoltages.adc1ConvertedVoltages_mV[ILCK_ADC_INPUT_HIGH_SIDE_CURRENT_SENSE] *
236 ILCK_FACTOR_IL_HS_CS_1_ohm;
237 ilck_tableFeedback.interlockCurrentFeedback_IL_LS_CS_mA =
238 ilck_tableAdcVoltages.adc1ConvertedVoltages_mV[ILCK_ADC_INPUT_LOW_SIDE_CURRENT_SENSE] *
239 ILCK_FACTOR_IL_LS_CS_1_ohm;
240
241 ilck_tableFeedback.interlockFeedback_IL_STATE = measuredInterlockState;
242
243 (void)DATA_WRITE_DATA(&ilck_tableFeedback);
244
245#if (BS_IGNORE_INTERLOCK_FEEDBACK == true)
246 measuredInterlockState = ILCK_SWITCH_ON;
247#endif
248
249 return measuredInterlockState;
250}

257ILCK_RETURN_TYPE_e ILCK_SetStateRequest(ILCK_STATE_REQUEST_e statereq) {
258 ILCK_RETURN_TYPE_e retVal = ILCK_ILLEGAL_REQUEST;
259
260 OS_EnterTaskCritical();
261 retVal = ILCK_CheckStateRequest(statereq);
262
263 if (retVal != ILCK_ILLEGAL_REQUEST) {
264 ilck_state.statereq = statereq;
265 }
266 OS_ExitTaskCritical();
267
268 return retVal;
269}

271void ILCK_Trigger(void) {
272 ILCK_STATE_REQUEST_e statereq = ILCK_STATE_NO_REQUEST;
273 ILCK_ELECTRICAL_STATE_TYPE_e interlockState = ILCK_SWITCH_UNDEF;
274
275 /* Check re-entrance of function */
276 if (ILCK_CheckReEntrance() > 0u) {
277 return;
278 }
279
280 if (ilck_state.timer > 0u) {
281 if ((--ilck_state.timer) > 0u) {
282 ilck_state.triggerentry--;
283 return; /* handle state machine only if timer has elapsed */
284 }
285 }
286
287 switch (ilck_state.state) {
288 /****************************UNINITIALIZED***********************************/
289 case ILCK_STATEMACHINE_UNINITIALIZED:
290 /* waiting for Initialization Request */
291 statereq = ILCK_TransferStateRequest();
292 if (statereq == ILCK_STATE_INITIALIZATION_REQUEST) {
293 ILCK_SAVE_LAST_STATES();
294 ILCK_InitializePins();
295 ilck_state.timer = ILCK_STATEMACH_SHORTTIME;
296 ilck_state.state = ILCK_STATEMACHINE_INITIALIZED;
297 ilck_state.substate = ILCK_ENTRY;
298 } else if (statereq == ILCK_STATE_NO_REQUEST) {
299 /* no actual request pending */
300 } else {
301 ilck_state.ErrRequestCounter++; /* illegal request pending */
302 }
303 break;
304
305 /****************************INITIALIZED*************************************/
306 case ILCK_STATEMACHINE_INITIALIZED:
307 ILCK_SAVE_LAST_STATES();
308 ilck_state.timer = ILCK_STATEMACH_SHORTTIME;
309 interlockState = ILCK_GetInterlockFeedback();
310 if (interlockState == ILCK_SWITCH_ON) {
311 (void)DIAG_Handler(DIAG_ID_INTERLOCK_FEEDBACK, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
312 } else {
313 (void)DIAG_Handler(DIAG_ID_INTERLOCK_FEEDBACK, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
314 }
315 break;
316
317 default:
318 /* this is an undefined state that should never be reached */
319 FAS_ASSERT(FAS_TRAP);
320 break;
321 } /* end switch (ilck_state.state) */
322
323 ilck_state.triggerentry--;
324}
