104extern STD_PIN_STATE_e IO_PinGet(const volatile uint32_t *pRegisterAddress, uint32_t pin) {
105 FAS_ASSERT(pRegisterAddress != NULL_PTR);
106 FAS_ASSERT(pin <= MCU_LARGEST_PIN_NUMBER);
107
108 STD_PIN_STATE_e retval = STD_PIN_UNDEFINED;
109 uint8_t pinState = (uint8_t)((*pRegisterAddress & ((uint32_t)1u << (pin))) >> pin);
110
111 /* pinState 0 equals a low level */
112 if (pinState == 0u) {
113 retval = STD_PIN_LOW;
114 } else {
115 retval = STD_PIN_HIGH;
116 }
117 return retval;
118}
