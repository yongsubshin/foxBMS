295/**
296 * @brief in the database, initializes the fields related to the LTC drivers.
297 *
298 * This function loops through all the LTC-related data fields in the database
299 * and sets them to 0. It should be called in the initialization or re-initialization
300 * routine of the LTC driver.
301 *
302 * @param ltc_state: state of the ltc state machine
303 *
304 */
305static void LTC_InitializeDatabase(LTC_STATE_s *ltc_state) {
306 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
307 ltc_state->ltcData.cellVoltage->state = 0u;
308 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
309 for (uint8_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
310 ltc_state->ltcData.cellVoltage->cellVoltage_mV[s][m][cb] = 0;
311 }
312 }
313 for (uint16_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_STRING; cb++) {
314 ltc_state->ltcData.openWireDetection->openWirePup[s][cb] = 0;
315 ltc_state->ltcData.openWireDetection->openWirePdown[s][cb] = 0;
316 ltc_state->ltcData.openWireDetection->openWireDelta[s][cb] = 0;
317 }
318
319 ltc_state->ltcData.cellTemperature->state = 0;
320 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
321 for (uint8_t ts = 0u; ts < BS_NR_OF_TEMP_SENSORS_PER_MODULE; ts++) {
322 ltc_state->ltcData.cellTemperature->cellTemperature_ddegC[s][m][ts] = 0;
323 }
324 }
325
326 ltc_state->ltcData.allGpioVoltages->state = 0u;
327 for (uint16_t gpio = 0u; gpio < (BS_NR_OF_MODULES_PER_STRING * SLV_NR_OF_GPIOS_PER_MODULE); gpio++) {
328 ltc_state->ltcData.allGpioVoltages->gpioVoltages_mV[s][gpio] = 0;
329 }
330
331 for (uint16_t ow = 0u; ow < (BS_NR_OF_MODULES_PER_STRING * (BS_NR_OF_CELL_BLOCKS_PER_MODULE + 1)); ow++) {
332 ltc_state->ltcData.openWire->openWire[s][ow] = 0u;
333 }
334 ltc_state->ltcData.openWire->state = 0u;
335 }
336 (void)DATA_WRITE_DATA(ltc_state->ltcData.cellVoltage, ltc_state->ltcData.cellTemperature, ltc_state->ltcData.openWire);
337}

382 * @param substate_nok substate to transition into if retVal is #STD_NOT_OK

405/*========== Extern Function Implementations ================================*/
406extern void LTC_SaveVoltages(LTC_STATE_s *ltc_state, uint8_t stringNumber) {
407 /* Pointer validity check */
408 FAS_ASSERT(ltc_state != NULL_PTR);
409
410 /* Iterate over all cell to:
411 *
412 * 1. Check open-wires and set respective cell measurements to invalid
413 * 2. Perform minimum/maximum measurement value plausibility check
414 * 3. Calculate string values
415 */
416 STD_RETURN_TYPE_e cellVoltageMeasurementValid = STD_OK;
417 int32_t stringVoltage_mV = 0;
418 uint16_t numberValidMeasurements = 0;
419 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
420 for (uint8_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
421 /* ------- 1. Check open-wires -----------------
422 * Is cell N input not open wire &&
423 * Is cell N+1 input not open wire &&
424 * Is cell voltage valid because of previous PEC error
425 * If so, everything okay, else set cell voltage measurement to invalid.
426 */
427 if ((ltc_state->ltcData.openWire
428 ->openWire[stringNumber][(m * (BS_NR_OF_CELL_BLOCKS_PER_MODULE + 1u)) + cb] == 0u) &&
429 (ltc_state->ltcData.openWire
430 ->openWire[stringNumber][(m * (BS_NR_OF_CELL_BLOCKS_PER_MODULE + 1u)) + cb + 1u] == 0u) &&
431 (ltc_state->ltcData.cellVoltage->invalidCellVoltage[stringNumber][m][cb] == false)) {
432 /* Cell voltage is valid -> perform minimum/maximum plausibility check */
433
434 /* ------- 2. Perform minimum/maximum measurement range check ---------- */
435 if (STD_OK == AFE_PlausibilityCheckVoltageMeasurementRange(
436 ltc_state->ltcData.cellVoltage->cellVoltage_mV[stringNumber][m][cb],
437 ltc_plausibleCellVoltages6806)) {
438 /* Cell voltage is valid -> calculate string voltage */
439 /* -------- 3. Calculate string values ------------- */
440 stringVoltage_mV += ltc_state->ltcData.cellVoltage->cellVoltage_mV[stringNumber][m][cb];
441 numberValidMeasurements++;
442 } else {
443 /* Invalidate cell voltage measurement */
444 ltc_state->ltcData.cellVoltage->invalidCellVoltage[stringNumber][m][cb] = true;
445 cellVoltageMeasurementValid = STD_NOT_OK;
446 }
447 } else {
448 /* Set cell voltage measurement value invalid, if not already invalid because of PEC Error */
449 ltc_state->ltcData.cellVoltage->invalidCellVoltage[stringNumber][m][cb] = true;
450 cellVoltageMeasurementValid = STD_NOT_OK;
451 }
452 }
453 }
454 DIAG_CheckEvent(cellVoltageMeasurementValid, DIAG_ID_AFE_CELL_VOLTAGE_MEAS_ERROR, DIAG_STRING, stringNumber);
455 ltc_state->ltcData.cellVoltage->stringVoltage_mV[stringNumber] = stringVoltage_mV;

624void LTC_Trigger(LTC_STATE_s *ltc_state) {
625 STD_RETURN_TYPE_e retVal = STD_OK;
626 LTC_REQUEST_s statereq = {.request = LTC_STATE_NO_REQUEST, .string = 0x0u};
627 uint8_t tmpbusID = 0;
628 LTC_ADCMODE_e tmpadcMode = LTC_ADCMODE_UNDEFINED;
629 LTC_ADCMEAS_CHAN_e tmpadcMeasCh = LTC_ADCMEAS_UNDEFINED;
630 STD_RETURN_TYPE_e continueFunction = STD_OK;
631
632 FAS_ASSERT(ltc_state != NULL_PTR);
633
634 /* Check re-entrance of function */
635 if (LTC_CheckReEntrance(ltc_state) > 0u) {
636 continueFunction = STD_NOT_OK;
637 }
638
639 if (ltc_state->check_spi_flag == STD_NOT_OK) {
640 if (ltc_state->timer > 0u) {
641 if ((--ltc_state->timer) > 0u) {
642 ltc_state->triggerentry--;
643 continueFunction = STD_NOT_OK; /* handle state machine only if timer has elapsed */
644 }
645 }
646 } else {
647 if (AFE_IsTransmitOngoing(ltc_state) == true) {
648 if (ltc_state->timer > 0u) {
649 if ((--ltc_state->timer) > 0u) {
650 ltc_state->triggerentry--;
651 continueFunction = STD_NOT_OK; /* handle state machine only if timer has elapsed */
652 }
653 }
654 }
655 }
656
657 if (continueFunction == STD_OK) {
658 switch (ltc_state->state) {
659 /****************************UNINITIALIZED***********************************/
660 case LTC_STATEMACH_UNINITIALIZED:
661 /* waiting for Initialization Request */
662 statereq = LTC_TransferStateRequest(ltc_state, &tmpbusID, &tmpadcMode, &tmpadcMeasCh);
663 if (statereq.request == LTC_STATE_INIT_REQUEST) {
664 LTC_SaveLastStates(ltc_state);
665 LTC_InitializeDatabase(ltc_state);
666 LTC_ResetErrorTable(ltc_state);
667 LTC_StateTransition(
668 ltc_state, LTC_STATEMACH_INITIALIZATION, LTC_INIT_STRING, LTC_STATEMACH_SHORTTIME);
669 ltc_state->adcMode = tmpadcMode;
670 ltc_state->adcMeasCh = tmpadcMeasCh;
671 } else if (statereq.request == LTC_STATE_NO_REQUEST) {
672 /* no actual request pending */
673 } else {
674 ltc_state->ErrRequestCounter++; /* illegal request pending */
675 }
676 break;
677
678 /****************************INITIALIZATION**********************************/
679 case LTC_STATEMACH_INITIALIZATION:
680
681 LTC_SetTransferTimes(ltc_state);
682 if (ltc_state->substate == LTC_INIT_STRING) {
683 LTC_SaveLastStates(ltc_state);
684 ltc_state->spiSeqPtr = ltc_state->ltcData.pSpiInterface;
685 ltc_state->spiNumberInterfaces = BS_NR_OF_STRINGS;
686 ltc_state->spiSeqEndPtr = ltc_state->ltcData.pSpiInterface + BS_NR_OF_STRINGS;
687 LTC_StateTransition(
688 ltc_state, LTC_STATEMACH_INITIALIZATION, LTC_ENTRY_INITIALIZATION, LTC_STATEMACH_SHORTTIME);
689 } else if (ltc_state->substate == LTC_ENTRY_INITIALIZATION) {
690 LTC_SaveLastStates(ltc_state);
691 retVal =
692 LTC_TRANSMIT_WAKE_UP(ltc_state->spiSeqPtr); /* Send dummy byte to wake up the daisy chain */
693 LTC_CondBasedStateTransition(
694 ltc_state,
695 retVal,
696 DIAG_ID_AFE_SPI,
697 LTC_STATEMACH_INITIALIZATION,
698 LTC_RE_ENTRY_INITIALIZATION,
699 LTC_STATEMACH_DAISY_CHAIN_FIRST_INITIALIZATION_TIME,
700 LTC_STATEMACH_INITIALIZATION,
701 LTC_ENTRY_INITIALIZATION,
702 LTC_STATEMACH_SHORTTIME);
703 } else if (ltc_state->substate == LTC_RE_ENTRY_INITIALIZATION) {
704 LTC_SaveLastStates(ltc_state);
705 retVal = LTC_TRANSMIT_WAKE_UP(
706 ltc_state->spiSeqPtr); /* Send dummy byte again to wake up the daisy chain */
707 LTC_CondBasedStateTransition(
708 ltc_state,
709 retVal,
710 DIAG_ID_AFE_SPI,
711 LTC_STATEMACH_INITIALIZATION,
712 LTC_START_INIT_INITIALIZATION,
713 LTC_STATEMACH_DAISY_CHAIN_SECOND_INITIALIZATION_TIME,
714 LTC_STATEMACH_INITIALIZATION,
715 LTC_RE_ENTRY_INITIALIZATION,
716 LTC_STATEMACH_SHORTTIME);
717 } else if (ltc_state->substate == LTC_START_INIT_INITIALIZATION) {
718 LTC_SaveLastStates(ltc_state);
719 ltc_state->check_spi_flag = STD_OK;
720 AFE_SetTransmitOngoing(ltc_state);
721 retVal = LTC_Init(
722 ltc_state->spiSeqPtr,
723 ltc_state->ltcData.txBuffer,
724 ltc_state->ltcData.rxBuffer,
725 ltc_state->ltcData.frameLength); /* Initialize main LTC loop */
726 ltc_state->lastSubstate = ltc_state->substate;
727 DIAG_CheckEvent(retVal, DIAG_ID_AFE_SPI, DIAG_STRING, ltc_state->currentString);
728 LTC_StateTransition(
729 ltc_state,
730 LTC_STATEMACH_INITIALIZATION,
731 LTC_CHECK_INITIALIZATION,
732 ltc_state->commandDataTransferTime + LTC_TRANSMISSION_TIMEOUT);
733 } else if (ltc_state->substate == LTC_CHECK_INITIALIZATION) {
734 /* Read values written in config register, currently unused */
735 LTC_SaveLastStates(ltc_state);
736 AFE_SetTransmitOngoing(ltc_state);
737 retVal = LTC_ReadRegister(
738 ltc_cmdRDCFG,
739 ltc_state->spiSeqPtr,
740 ltc_state->ltcData.txBuffer,
741 ltc_state->ltcData.rxBuffer,
742 ltc_state->ltcData.frameLength); /* Read config register */
743 LTC_StateTransition(
744 ltc_state,
745 LTC_STATEMACH_INITIALIZATION,
746 LTC_EXIT_INITIALIZATION,
747 ltc_state->commandDataTransferTime + LTC_TRANSMISSION_TIMEOUT);
748 } else if (ltc_state->substate == LTC_EXIT_INITIALIZATION) {
749 LTC_SaveLastStates(ltc_state);
750 ++ltc_state->spiSeqPtr;
751 ++ltc_state->currentString;
752 if (ltc_state->spiSeqPtr >= ltc_state->spiSeqEndPtr) {
753 LTC_StateTransition(
754 ltc_state, LTC_STATEMACH_INITIALIZED, LTC_ENTRY_INITIALIZATION, LTC_STATEMACH_SHORTTIME);
755 } else {
756 LTC_StateTransition(
757 ltc_state, LTC_STATEMACH_INITIALIZATION, LTC_ENTRY_INITIALIZATION, LTC_STATEMACH_SHORTTIME);
758 }
759 }
760 break;

776 ltc_state->spiSeqEndPtr = ltc_state->ltcData.pSpiInterface + BS_NR_OF_STRINGS;

791 LTC_STATEMACH_SHORTTIME);

793 break;

807 retVal,

810 LTC_READ_VOLTAGE_REGISTER_A_RDCVA_READVOLTAGE,

1158 ltc_state,

1246 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {

1251 ->openWireDelta[ltc_state->requestedString]
1252 [c + (m * BS_NR_OF_CELL_BLOCKS_PER_MODULE)] < LTC_ADOW_THRESHOLD) {
1253 ltc_state->ltcData.openWire->openWire[ltc_state->requestedString]
1254 [c + (m * BS_NR_OF_CELL_BLOCKS_PER_MODULE)] = 1;
1255 }
1256 }
1257 }

1263 ltc_state->ltcData.openWire->nrOpenWires[ltc_state->requestedString]++;
1264 }
1265 }
1266
1267 /* Write database entry */
1268 (void)DATA_WRITE_DATA(ltc_state->ltcData.openWire);
1269 /* Start new measurement cycle */
1270 LTC_StateTransition(ltc_state, LTC_STATEMACH_STARTMEAS, LTC_ENTRY, LTC_STATEMACH_SHORTTIME);
1271 }
1272 break;
1273
1274 /****************************DEFAULT**************************/
1275 default:
1276 /* invalid state */
1277 FAS_ASSERT(FAS_TRAP);
1278 break;
1279 }
1280
1281 ltc_state->triggerentry--; /* reentrance counter */
1282 } /* continueFunction */

1285/**
1286 * @brief saves the voltage values read from the LTC daisy-chain.
1287 *
1288 * After a voltage measurement was initiated to measure the voltages of the cells,
1289 * the result is read via SPI from the daisy-chain.
1290 * There are 6 register to read _(A,B,C,D,E,F,G,H,I) to get all cell voltages.
1291 * Only one register can be read at a time.
1292 * This function is called to store the result from the transmission in a buffer.
1293 *
1294 * @param ltc_state state of the ltc state machine
1295 * @param pRxBuff receive buffer
1296 * @param registerSet voltage register that was read (voltage register A,B,C,D,E,F,G,H or I)
1297 * @param stringNumber string addressed
1298 *
1299 */
1300static void LTC_SaveRXToVoltageBuffer_Fuelcell(
1301 LTC_STATE_s *ltc_state,
1302 uint16_t *pRxBuff,
1303 uint8_t registerSet,
1304 uint8_t stringNumber) {
1305 uint16_t cellOffset = 0;
1306 uint16_t val_ui = 0;
1307 int16_t voltage_mV = 0;
1308 uint16_t buffer_LSB = 0;
1309 uint16_t buffer_MSB = 0;
1310
1311 cellOffset = registerSet * LTC_NUMBER_OF_CELL_VOLTAGES_PER_REGISTER;
1312
1313 /* Reinitialize index counter at begin of cycle */
1314 if (cellOffset == 0u) {
1315 (ltc_state->ltcData.usedCellIndex[stringNumber]) = 0u;
1316 }
1317
1318 /* Retrieve data without command and CRC*/
1319 for (uint8_t m = 0u; m < LTC_N_LTC; m++) {
1320 uint8_t increment = 0u;
1321
1322 /* Parse all four voltages (4 * 12bits) contained in one register */
1323 for (uint8_t c = 0u; c < 4u; c++) {
1324 switch (c) {
1325 case 0u:
1326 buffer_MSB = pRxBuff[4u + (m * 8u)];
1327 buffer_LSB = (pRxBuff[4u + 1u + (m * 8u)]) >> 4u;
1328 val_ui = (uint16_t)(buffer_LSB | (buffer_MSB << 4u));
1329 break;
1330 case 1u:
1331 buffer_MSB = pRxBuff[4u + 1u + (m * 8u)] & 0x0Fu;
1332 buffer_LSB = (pRxBuff[4u + 2u + (m * 8u)]);
1333 val_ui = (uint16_t)(buffer_LSB | (buffer_MSB << 8u));
1334 break;
1335 case 2u:
1336 buffer_MSB = pRxBuff[4u + 3u + (m * 8u)];
1337 buffer_LSB = (pRxBuff[4u + 4u + (m * 8u)]) >> 4u;
1338 val_ui = (uint16_t)(buffer_LSB | (buffer_MSB << 4u));
1339 break;
1340 case 3u:
1341 buffer_MSB = pRxBuff[4u + 4u + (m * 8u)] & 0x0Fu;
1342 buffer_LSB = (pRxBuff[4u + 5u + (m * 8u)]);
1343 val_ui = (uint16_t)(buffer_LSB | (buffer_MSB << 8u));
1344 break;
1345 default:
1346 break;
1347 }
1348
1349 /* Check signed bit if measured value is negative or not */
1350 if ((val_ui & (1u << (12u - 1u))) == 0u) {
1351 voltage_mV = (int16_t)(((val_ui & 0x7FFu)) * LTC_FUEL_CELL_LSB_RESOLUTION_mV); /* Unit mV */
1352 } else {
1353 voltage_mV = (int16_t)(((((~val_ui) + 1) & 0x7FF)) * (-LTC_FUEL_CELL_LSB_RESOLUTION_mV)); /* Unit mV */
1354 }
1355
1356 if (ltc_state->ltcData.errorTable->PEC_valid[stringNumber][m] == true) {
1357 ltc_state->ltcData.cellVoltage
1358 ->cellVoltage_mV[stringNumber][m][ltc_state->ltcData.usedCellIndex[stringNumber]] = voltage_mV;
1359 for (uint8_t i = 0u; i < LTC_NUMBER_OF_CELL_VOLTAGES_PER_REGISTER; i++) {
1360 ltc_state->ltcData.cellVoltage
1361 ->invalidCellVoltage[stringNumber][(m / LTC_NUMBER_OF_LTC_PER_MODULE)][cellOffset + i] = false;
1362 }
1363 } else {
1364 /* PEC_valid == false: Invalidate only flags of this voltage register */
1365 for (uint8_t i = 0u; i < LTC_NUMBER_OF_CELL_VOLTAGES_PER_REGISTER; i++) {
1366 ltc_state->ltcData.cellVoltage
1367 ->invalidCellVoltage[stringNumber][(m / LTC_NUMBER_OF_LTC_PER_MODULE)][cellOffset + i] = true;
1368 }
1369 }
1370
1371 (ltc_state->ltcData.usedCellIndex[stringNumber])++;
1372 increment++;
1373
1374 if ((ltc_state->ltcData.usedCellIndex[stringNumber]) > BS_NR_OF_CELL_BLOCKS_PER_MODULE) {
1375 break;
1376 }
1377 }
1378
1379 /* Restore start value for next module in the daisy-chain. Only
1380 * decrement used cell index if current module is not the last
1381 * module in the daisy-chain. */
1382 if ((m + 1u) < LTC_N_LTC) {
1383 (ltc_state->ltcData.usedCellIndex[stringNumber]) -= increment;
1384 }
1385 }
1386}

1388/**
1389 * @brief initialize the daisy-chain.
1390 *
1391 * To initialize the LTC6804 daisy-chain, a dummy byte (0x00) is sent.
1392 *
1393 * @param pSpiInterface pointer to SPI configuration
1394 * @param pTxBuff transmit buffer
1395 * @param pRxBuff receive buffer
1396 * @param frameLength number of words to transmit
1397 *
1398 * @return retVal #STD_OK if dummy byte was sent correctly by SPI, #STD_NOT_OK otherwise
1399 *
1400 */
1401static STD_RETURN_TYPE_e LTC_Init(
1402 SPI_INTERFACE_CONFIG_s *pSpiInterface,
1403 uint16_t *pTxBuff,
1404 uint16_t *pRxBuff,
1405 uint32_t frameLength) {
1406 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
1407
1408 uint8_t PEC_Check[LTC_DATA_SIZE_IN_BYTES];
1409 uint16_t PEC_result = 0;
1410
1411 /* now construct the message to be sent: it contains the wanted data, PLUS the needed PECs */
1412 pTxBuff[0] = ltc_cmdWRCFG[0];
1413 pTxBuff[1] = ltc_cmdWRCFG[1];
1414 pTxBuff[2] = ltc_cmdWRCFG[2];
1415 pTxBuff[3] = ltc_cmdWRCFG[3];
1416
1417 /* set REFON bit to 1 */
1418 /* data for the configuration */
1419 for (uint16_t i = 0u; i < LTC_N_LTC; i++) {
1420 /* 3F = disable all pull-downs, 40: REFON = 1 */

1519/**
1520 * @brief checks if the data received from the daisy-chain is not corrupt.
1521 *
1522 * This function computes the PEC (CRC) from the data received by the daisy-chain.
1523 * It compares it with the PEC sent by the LTCs.
1524 * If there are errors, the array the error table is updated to locate the LTCs in daisy-chain
1525 * that transmitted corrupt data.
1526 *
1527 * @param ltc_state state of the ltc state machine
1528 * @param DataBufferSPI_RX_with_PEC data obtained from the SPI transmission
1529 * @param stringNumber string addressed
1530 *
1531 * @return retVal STD_OK if PEC check is OK, STD_NOT_OK otherwise
1532 *
1533 */
1534static STD_RETURN_TYPE_e LTC_CheckPec(
1535 LTC_STATE_s *ltc_state,
1536 uint16_t *DataBufferSPI_RX_with_PEC,
1537 uint8_t stringNumber) {
1538 uint16_t i = 0;
1539 STD_RETURN_TYPE_e retVal = STD_OK;
1540 uint8_t PEC_TX[2];
1541 uint16_t PEC_result = 0;
1542 uint8_t PEC_Check[LTC_DATA_SIZE_IN_BYTES] = {0x00, 0x00, 0x00, 0x00, 0x00, 0x00};
1543
1544 /* check all PECs and put data without command and PEC in DataBufferSPI_RX (easier to use) */
1545 for (i = 0; i < LTC_N_LTC; i++) {
1546 PEC_Check[0] = DataBufferSPI_RX_with_PEC[4u + (i * 8u)];
1547 PEC_Check[1] = DataBufferSPI_RX_with_PEC[5u + (i * 8u)];
1548 PEC_Check[2] = DataBufferSPI_RX_with_PEC[6u + (i * 8u)];
1549 PEC_Check[3] = DataBufferSPI_RX_with_PEC[7u + (i * 8u)];
1550 PEC_Check[4] = DataBufferSPI_RX_with_PEC[8u + (i * 8u)];
1551 PEC_Check[5] = DataBufferSPI_RX_with_PEC[9u + (i * 8u)];
1552
1553 PEC_result = LTC_CalculatePec15(LTC_DATA_SIZE_IN_BYTES, PEC_Check);
1554 PEC_TX[0] = (uint8_t)((PEC_result >> 8u) & 0xFFu);
1555 PEC_TX[1] = (uint8_t)(PEC_result & 0xFFu);
1556
1557 /* if calculated PEC not equal to received PEC */
1558 if ((PEC_TX[0] != DataBufferSPI_RX_with_PEC[10u + (i * 8u)]) ||
1559 (PEC_TX[1] != DataBufferSPI_RX_with_PEC[11u + (i * 8u)])) {
1560 /* update error table of the corresponding LTC only if PEC check is activated */
1561 if (LTC_DISCARD_PEC == false) {
1562 ltc_state->ltcData.errorTable->PEC_valid[stringNumber][i] = false;
1563 retVal = STD_NOT_OK;
1564 } else {
1565 ltc_state->ltcData.errorTable->PEC_valid[stringNumber][i] = true;
1566 }
1567 } else {
1568 /* update error table of the corresponding LTC */
1569 ltc_state->ltcData.errorTable->PEC_valid[stringNumber][i] = true;
1570 }
1571 }
1572 return retVal;

1631/**
1632 * @brief gets the frequency of the SPI clock.
1633 *
1634 * This function reads the configuration from the SPI handle directly.
1635 *
1636 * @param pSpiInterface pointer to SPI configuration
1637 *
1638 * @return frequency of the SPI clock
1639 */
1640static uint32_t LTC_GetSpiClock(SPI_INTERFACE_CONFIG_s *pSpiInterface) {
1641 uint32_t SPI_Clock = 0;
1642 uint32_t prescaler = 0;
1643
1644 /* if (LTC_SPI_INSTANCE == SPI2 || LTC_SPI_INSTANCE == SPI3) { */
1645 /* SPI2 and SPI3 are connected to APB1 (PCLK1) */
1646 /* The prescaler setup bits LTC_SPI_PRESCALER corresponds to the bits 5:3 in the SPI_CR1 register */
1647 /* Reference manual p.909 */
1648 /* The shift by 3 puts the bits 5:3 to the first position */
1649 /* Division are made by powers of 2 which corresponds to shifting to the right */
1650 /* Then 0 corresponds to divide by 2, 1 corresponds to divide by 4... so 1 has to be added to the value of the
1651 * configuration bits */
1652
1653 /* SPI_Clock = HAL_RCC_GetPCLK1Freq()>>((LTC_SPI_PRESCALER>>3)+1);
1654 } */
1655
1656 /* if (LTC_SPI_INSTANCE == SPI1 || LTC_SPI_INSTANCE == SPI4 || LTC_SPI_INSTANCE == SPI5 || LTC_SPI_INSTANCE == SPI6)
1657 * {
1658 */
1659 /* SPI1, SPI4, SPI5 and SPI6 are connected to APB2 (PCLK2) */
1660 /* The prescaler setup bits LTC_SPI_PRESCALER corresponds to the bits 5:3 in the SPI_CR1 register */
1661 /* Reference manual p.909 */
1662 /* The shift by 3 puts the bits 5:3 to the first position */
1663 /* Division are made by powers of 2 which corresponds to shifting to the right */
1664 /* Then 0 corresponds to divide by 2, 1 corresponds to divide by 4... so 1 has to be added to the value of the
1665 * configuration bits */
1666
1667 /* SPI_Clock = HAL_RCC_GetPCLK2Freq()>>((LTC_SPI_PRESCALER>>3)+1);
1668 } */
1669
1670 /* Get SPI prescaler */
1671 prescaler = ((pSpiInterface->pNode->FMT0) >> 8u) & 0xFFu;
1672 SPI_Clock = (uint32_t)(AVCLK1_FREQ * 1000000u) / (prescaler + 1u);
1673
1674 return SPI_Clock;
1675}
