93#ifndef UNITY_UNIT_TEST
94int main(void)
95#else
96int unit_test_main(void)
97#endif
98{
99 MINFO_SetResetSource(getResetSource()); /* Get reset source and clear respective flags */
100 muxInit();
101 gioInit();
102 SPI_Initialize();
103 adcInit();
104 hetInit();
105 etpwmInit();
106 crcInit();
107 LED_SetDebugLed();
108 I2C_Initialize();
109 DMA_Initialize();
110 PWM_Initialize();
111 DIAG_Initialize(&diag_device);
112 MATH_StartupSelfTest();
113 const STD_RETURN_TYPE_e checkTimeHasPassedSelfTestReturnValue = OS_CheckTimeHasPassedSelfTest();
114 FAS_ASSERT(checkTimeHasPassedSelfTestReturnValue == STD_OK);
115
116 OS_InitializeOperatingSystem();
117
118 /* Enable IRQ interrupt after creating the AFE task to prevent the DMA interrupt,
119 because the function called on DMA interrupts require an valid AFE task handle,
120 which is NULL before creating the AFE task. */
121 _enable_IRQ_interrupt_();
122
123 if (OS_INIT_PRE_OS != os_boot) {
124 /* Could not create Queues, Mutexes, Events and Tasks do not boot further from this point on */
125 FAS_ASSERT(FAS_TRAP);
126 }
127
128 if (STD_OK != CHK_ValidateChecksum()) {
129 if (DIAG_HANDLER_RETURN_OK != DIAG_Handler(DIAG_ID_FLASHCHECKSUM, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u)) {
130 /* Could not validate checksum do not boot further from this point on */
131 FAS_ASSERT(FAS_TRAP);
132 }
133 }
134
135 os_schedulerStartTime = OS_GetTickCount();
136
137 OS_StartScheduler();
138 /* we must never get here; there is no way to determine the exit state of this program,
139 * but for the sake of correctness we exit with an error code */
140 return 1;
141}
