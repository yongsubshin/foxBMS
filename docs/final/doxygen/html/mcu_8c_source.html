99void MCU_Delay_us(uint32_t delay_us) {
100 /* AXIVION Routine Generic-MissingParameterAssert: delay_us: parameter accepts whole range */
101 /**
102 * First the time of one FRC0 increment in microseconds is computed with
103 * increment_time = (1/mcu_frcClock_Hz)*1e6
104 * The 1e6 factor is to get the period in microseconds.
105 *
106 * The number of increments needed to reach 1 microsecond is then
107 * computed with
108 * nr_increments_for_1us = (1us/increment_time)
109 *
110 * To avoid floating point computation problems, the number of
111 * increments needed to reach 1 microsecond is computed at once with:
112 */
113 const uint32_t rti_nrOfCounts_us = (uint32_t)(((float_t)mcu_frcClock_Hz) / 1e6f);
114
115 /* Get current value of FRC0 counter */
116 const uint32_t startValue = MCU_RTI_CNT0_FRC0_REG;
117
118 uint32_t timeOut = 0;
119 while (timeOut < MCU_US_WAIT_TIMEOUT) {
120 /**
121 * Poll FRC0 value until the value corresponding to the desired
122 * waiting timing has been reached.
123 * The timeout is to ensure the code does not get trapped here.
124 */
125 const uint32_t checkValue = MCU_RTI_CNT0_FRC0_REG;
126 if ((checkValue - startValue) >= (delay_us * rti_nrOfCounts_us)) {
127 break;
128 }
129 timeOut++;
130 }
131}

137extern uint32_t MCU_ConvertFrcDifferenceToTimespan_us(uint32_t count) {
138 /* AXIVION Routine Generic-MissingParameterAssert: count: parameter accepts whole range */
139 const uint32_t rti_nrOfCounts_us = (uint32_t)(((float_t)mcu_frcClock_Hz) / 1e6f);
140 return count / rti_nrOfCounts_us;
141}

143extern bool MCU_IsTimeElapsed(uint32_t startCounter, uint32_t timeout_us) {
144 bool retVal = false;
145 uint32_t counter = MCU_GetFreeRunningCount();
146 uint32_t timeDifference_us = MCU_ConvertFrcDifferenceToTimespan_us(counter - startCounter);
147 if (timeDifference_us < timeout_us) {
148 retVal = false;
149 } else {
150 retVal = true;
151 }
152 return retVal;
153}
