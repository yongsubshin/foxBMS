181/*========== Extern Function Implementations ================================*/
182extern void ALGO_MovingAverage(void) {
183 static uint8_t curCounter = 0u;
184 static uint8_t powCounter = 0u;
185 static DATA_BLOCK_CURRENT_SENSOR_s curPow_tab = {.header.uniqueId = DATA_BLOCK_ID_CURRENT_SENSOR};
186 static DATA_BLOCK_MOVING_AVERAGE_s movingAverage_tab = {.header.uniqueId = DATA_BLOCK_ID_MOVING_AVERAGE};
187 static uint8_t curInit = 0u; /* bit0: 1s, bit1: 5s, bit2: 10s, bit3: 30s, bit4: 60s, bit5: cfg */
188 static uint8_t powInit = 0u; /* bit0: 1s, bit1: 5s, bit2: 10s, bit3: 30s, bit4: 60s, bit5: cfg */
189 static uint8_t newValues = 0u;
190 float_t divider = 0.0f;
191 bool validValues = true;
192
193 (void)DATA_READ_DATA(&curPow_tab);
194 (void)DATA_READ_DATA(&movingAverage_tab);
195
196 /* Check if new current value */
197 if (curCounter != curPow_tab.newCurrent) {
198 curCounter = curPow_tab.newCurrent;
199
200 /* Check if valid values */
201 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
202 if (curPow_tab.invalidCurrentMeasurement[s] != 0u) {
203 validValues = false;
204 }
205 }
206
207 if (validValues == true) {
208 /* new Values -> Save later in database */
209 newValues = 1u;
210
211 int32_t packCurrent = 0;
212 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
213 packCurrent += curPow_tab.current_mA[s];
214 }
215
216 /* Add value to array and calculate new moving average values */
217 *pMovingAverageCurrentNew = packCurrent;
218
219 /* Calculate new moving average - first add new value */
220 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_1s;
221 movingAverage_tab.movingAverageCurrent1sInterval_mA += (*pMovingAverageCurrentNew) / divider;
222 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_5s;
223 movingAverage_tab.movingAverageCurrent5sInterval_mA += (*pMovingAverageCurrentNew) / divider;
224 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_10s;
225 movingAverage_tab.movingAverageCurrent10sInterval_mA += (*pMovingAverageCurrentNew) / divider;
226 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_30s;
227 movingAverage_tab.movingAverageCurrent30sInterval_mA += (*pMovingAverageCurrentNew) / divider;
228 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_60s;
229 movingAverage_tab.movingAverageCurrent60sInterval_mA += (*pMovingAverageCurrentNew) / divider;
230 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_CFG;
231 movingAverage_tab.movingAverageCurrentConfigurableInterval_mA += (*pMovingAverageCurrentNew) / divider;
232
233 /* Then, increment pointer and subtract oldest value when respective window is filled with data */
234 pMovingAverageCurrentNew++;
235 if ((curInit & 0x01u) == 0x01u) {
236 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_1s;
237 movingAverage_tab.movingAverageCurrent1sInterval_mA -= ((*pMovingAverageCurrent_1s) / divider);
238 pMovingAverageCurrent_1s++;
239 } else {
240 if (pMovingAverageCurrentNew == &curValues[ALGO_NUMBER_AVERAGE_VALUES_CUR_1s]) {
241 curInit |= 0x01u;
242 }
243 }
244 if ((curInit & 0x02u) == 0x02u) {
245 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_5s;
246 movingAverage_tab.movingAverageCurrent5sInterval_mA -= (*pMovingAverageCurrent_5s) / divider;
247 pMovingAverageCurrent_5s++;
248 } else {
249 if (pMovingAverageCurrentNew == &curValues[ALGO_NUMBER_AVERAGE_VALUES_CUR_5s]) {
250 curInit |= 0x02u;
251 }
252 }
253 if ((curInit & 0x04u) == 0x04u) {
254 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_10s;
255 movingAverage_tab.movingAverageCurrent10sInterval_mA -= (*pMovingAverageCurrent_10s) / divider;
256 pMovingAverageCurrent_10s++;
257 } else {
258 if (pMovingAverageCurrentNew == &curValues[ALGO_NUMBER_AVERAGE_VALUES_CUR_10s]) {
259 curInit |= 0x04u;
260 }
261 }
262 if ((curInit & 0x08u) == 0x08u) {
263 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_30s;
264 movingAverage_tab.movingAverageCurrent30sInterval_mA -= (*pMovingAverageCurrent_30s) / divider;
265 pMovingAverageCurrent_30s++;
266 } else {
267 if (pMovingAverageCurrentNew == &curValues[ALGO_NUMBER_AVERAGE_VALUES_CUR_30s]) {
268 curInit |= 0x08u;
269 }
270 }
271 if ((curInit & 0x10u) == 0x10u) {
272 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_60s;
273 movingAverage_tab.movingAverageCurrent60sInterval_mA -= (*pMovingAverageCurrent_60s) / divider;
274 pMovingAverageCurrent_60s++;
275 } else {
276 if (pMovingAverageCurrentNew == &curValues[ALGO_NUMBER_AVERAGE_VALUES_CUR_60s]) {
277 curInit |= 0x10u;
278 }
279 }
280 if ((curInit & 0x20u) == 0x20u) {
281 divider = ALGO_NUMBER_AVERAGE_VALUES_CUR_CFG;
282 movingAverage_tab.movingAverageCurrentConfigurableInterval_mA -= (*pMovingAverageCurrent_cfg) / divider;
283 pMovingAverageCurrent_cfg++;
284 } else {
285 if (pMovingAverageCurrentNew == &curValues[ALGO_NUMBER_AVERAGE_VALUES_CUR_CFG]) {
286 curInit |= 0x20u;
287 }
288 }
289
290 /* Check pointer for buffer overflow */
291 if (pMovingAverageCurrentNew > &curValues[movingAverageCurrentLength - 1u]) {
292 pMovingAverageCurrentNew = &curValues[0u];
293 }
294 if (pMovingAverageCurrent_1s > &curValues[movingAverageCurrentLength - 1u]) {
295 pMovingAverageCurrent_1s = &curValues[0u];
296 }
297 if (pMovingAverageCurrent_5s > &curValues[movingAverageCurrentLength - 1u]) {
298 pMovingAverageCurrent_5s = &curValues[0u];
299 }
300 if (pMovingAverageCurrent_10s > &curValues[movingAverageCurrentLength - 1u]) {
301 pMovingAverageCurrent_10s = &curValues[0u];
302 }
303 if (pMovingAverageCurrent_30s > &curValues[movingAverageCurrentLength - 1u]) {
304 pMovingAverageCurrent_30s = &curValues[0u];
305 }
306 if (pMovingAverageCurrent_60s > &curValues[movingAverageCurrentLength - 1u]) {
307 pMovingAverageCurrent_60s = &curValues[0u];
308 }
309 if (pMovingAverageCurrent_cfg > &curValues[movingAverageCurrentLength - 1u]) {
310 pMovingAverageCurrent_cfg = &curValues[0u];
311 }
312 }
313 }
314
315 validValues = true;
316
317 /* Check if new power value */
318 if (powCounter != curPow_tab.newPower) {
319 powCounter = curPow_tab.newPower;
320
321 /* Check if valid values */
322 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
323 if (curPow_tab.invalidPowerMeasurement[s] != 0u) {
324 validValues = false;
325 }
326 }
327 if (validValues == true) {
328 newValues = 1u;
329
330 int32_t packPower = 0;
331 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
332 packPower += curPow_tab.power_W[s];
333 }
334
335 /* Add value to array and calculate new moving mean values */
336 *pMovingAveragePowerNew = packPower;
337
338 /* Calculate new moving means - first add new value */
339 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_1s;
340 movingAverage_tab.movingAveragePower1sInterval_mA += (*pMovingAveragePowerNew) / divider;
341 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_5s;
342 movingAverage_tab.movingAveragePower5sInterval_mA += (*pMovingAveragePowerNew) / divider;
343 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_10s;
344 movingAverage_tab.movingAveragePower10sInterval_mA += (*pMovingAveragePowerNew) / divider;
345 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_30s;
346 movingAverage_tab.movingAveragePower30sInterval_mA += (*pMovingAveragePowerNew) / divider;
347 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_60s;
348 movingAverage_tab.movingAveragePower60sInterval_mA += (*pMovingAveragePowerNew) / divider;
349 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_CFG;
350 movingAverage_tab.movingAveragePowerConfigurableInterval_mA += (*pMovingAveragePowerNew) / divider;
351
352 /* Then, increment pointer and subtract oldest value when respective window is filled with data */
353 pMovingAveragePowerNew++;
354 if ((powInit & 0x01u) == 0x01u) {
355 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_1s;
356 movingAverage_tab.movingAveragePower1sInterval_mA -= ((*pMovingAveragePower_1s) / divider);
357 pMovingAveragePower_1s++;
358 } else {
359 if (pMovingAveragePowerNew == &powValues[ALGO_NUMBER_AVERAGE_VALUES_POW_1s]) {
360 powInit |= 0x01u;
361 }
362 }
363 if ((powInit & 0x02u) == 0x02u) {
364 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_5s;
365 movingAverage_tab.movingAveragePower5sInterval_mA -= ((*pMovingAveragePower_5s) / divider);
366 pMovingAveragePower_5s++;
367 } else {
368 if (pMovingAveragePowerNew == &powValues[ALGO_NUMBER_AVERAGE_VALUES_POW_5s]) {
369 powInit |= 0x02u;
370 }
371 }
372 if ((powInit & 0x04u) == 0x04u) {
373 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_10s;
374 movingAverage_tab.movingAveragePower10sInterval_mA -= ((*pMovingAveragePower_10s) / divider);
375 pMovingAveragePower_10s++;
376 } else {
377 if (pMovingAveragePowerNew == &powValues[ALGO_NUMBER_AVERAGE_VALUES_POW_10s]) {
378 powInit |= 0x04u;
379 }
380 }
381 if ((powInit & 0x08u) == 0x08u) {
382 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_30s;
383 movingAverage_tab.movingAveragePower30sInterval_mA -= ((*pMovingAveragePower_30s) / divider);
384 pMovingAveragePower_30s++;
385 } else {
386 if (pMovingAveragePowerNew == &powValues[ALGO_NUMBER_AVERAGE_VALUES_POW_30s]) {
387 powInit |= 0x08u;
388 }
389 }
390 if ((powInit & 0x10u) == 0x10u) {
391 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_60s;
392 movingAverage_tab.movingAveragePower60sInterval_mA -= ((*pMovingAveragePower_60s) / divider);
393 pMovingAveragePower_60s++;
394 } else {
395 if (pMovingAveragePowerNew == &powValues[ALGO_NUMBER_AVERAGE_VALUES_POW_60s]) {
396 powInit |= 0x10u;
397 }
398 }
399 if ((powInit & 0x20u) == 0x20u) {
400 divider = ALGO_NUMBER_AVERAGE_VALUES_POW_CFG;
401 movingAverage_tab.movingAveragePowerConfigurableInterval_mA -= ((*pMovingAveragePower_cfg) / divider);
402 pMovingAveragePower_cfg++;
403 } else {
404 if (pMovingAveragePowerNew == &powValues[ALGO_NUMBER_AVERAGE_VALUES_POW_CFG]) {
405 powInit |= 0x20u;
406 }
407 }
408
409 /* Check pointer for buffer overflow */
410 if (pMovingAveragePowerNew > &powValues[movingAveragePowerLength - 1u]) {
411 pMovingAveragePowerNew = &powValues[0u];
412 }
413 if (pMovingAveragePower_1s > &powValues[movingAveragePowerLength - 1u]) {
414 pMovingAveragePower_1s = &powValues[0u];
415 }
416 if (pMovingAveragePower_5s > &powValues[movingAveragePowerLength - 1u]) {
417 pMovingAveragePower_5s = &powValues[0u];
418 }
419 if (pMovingAveragePower_10s > &powValues[movingAveragePowerLength - 1u]) {
420 pMovingAveragePower_10s = &powValues[0u];
421 }
422 if (pMovingAveragePower_30s > &powValues[movingAveragePowerLength - 1u]) {
423 pMovingAveragePower_30s = &powValues[0u];
424 }
425 if (pMovingAveragePower_60s > &powValues[movingAveragePowerLength - 1u]) {
426 pMovingAveragePower_60s = &powValues[0u];
427 }
428 if (pMovingAveragePower_cfg > &powValues[movingAveragePowerLength - 1u]) {
429 pMovingAveragePower_cfg = &powValues[0u];
430 }
431 }
432 }
433
434 if (newValues == 1u) {
435 newValues = 0;
436
437 (void)DATA_WRITE_DATA(&movingAverage_tab);
438 }
439}
