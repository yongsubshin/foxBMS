85/**
86 * Defines for calculating the ADC voltage on the ends of the operating range.
87 * The ADC voltage is calculated with the following formula:
88 *
89 * V_adc = ( ( V_supply * R_ntc ) / ( R + R_ntc ) )
90 *
91 * Depending on the position of the NTC in the voltage resistor (R_1/R_2),
92 * different R_ntc values are used for the calculation.
93 * @{
94 */
95#if defined(TS_MURATA_NCXXXXH103_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && \
96 (TS_MURATA_NCXXXXH103_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
97#define TS_MURATA_NCXXXXH103_ADC_VOLTAGE_V_MAX_V \
98 ((float_t)((TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * \
99 ts_ncxxxxh103Lut[ts_ncxxxxh103LutSize - 1u].resistance_Ohm) / \
100 (ts_ncxxxxh103Lut[ts_ncxxxxh103LutSize - 1u].resistance_Ohm + \
101 TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))
102#define TS_MURATA_NCXXXXH103_ADC_VOLTAGE_V_MIN_V \
103 ((float_t)((TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * ts_ncxxxxh103Lut[0u].resistance_Ohm) / \
104 (ts_ncxxxxh103Lut[0u].resistance_Ohm + TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))
105#else /* TS_MURATA_NCXXXXH103_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 is false */
106#define TS_MURATA_NCXXXXH103_ADC_VOLTAGE_V_MIN_V \
107 ((float_t)((TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * \
108 ts_ncxxxxh103Lut[ts_ncxxxxh103LutSize - 1u].resistance_Ohm) / \
109 (ts_ncxxxxh103Lut[ts_ncxxxxh103LutSize - 1u].resistance_Ohm + \
110 TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))
111#define TS_MURATA_NCXXXXH103_ADC_VOLTAGE_V_MAX_V \

123extern int16_t TS_Mur00GetTemperatureFromLut(uint16_t adcVoltage_mV) {
124 int16_t temperature_ddegC = INT16_MIN;
125 float_t adcVoltage_V = (float_t)adcVoltage_mV / TS_SCALING_FACTOR_1V_IN_MV_FLOAT; /* Convert mV to V */
126
127 /* Check for valid ADC measurements to prevent undefined behavior */
128 if (adcVoltage_V > TS_MURATA_NCXXXXH103_ADC_VOLTAGE_V_MAX_V) {
129 /* Invalid measured ADC voltage -> sensor out of operating range or disconnected/shorted */
130 temperature_ddegC = INT16_MIN;
131 } else if (adcVoltage_V < TS_MURATA_NCXXXXH103_ADC_VOLTAGE_V_MIN_V) {
132 /* Invalid measured ADC voltage -> sensor out of operating range or shorted/disconnected */
133 temperature_ddegC = INT16_MAX;
134 } else {
135 /* Calculate NTC resistance based on measured ADC voltage */
136#if defined(TS_MURATA_NCXXXXH103_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && \
137 (TS_MURATA_NCXXXXH103_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
138 /* R_1 = R_2 * ( ( V_supply / V_adc ) - 1 ) */
139 const float_t resistance_Ohm = TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
140 ((TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V / adcVoltage_V) - 1);
141#else /* TS_MURATA_NCXXXXH103_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 is false */
142 /* formula: R_2 = R_1 * ( V_2 / ( V_supply - V_adc ) ) */
143 const float_t resistance_Ohm =
144 TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
145 (adcVoltage_V / (TS_MURATA_NCXXXXH103_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V - adcVoltage_V));
146#endif /* TS_MURATA_NCXXXXH103_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 */
147
148 /* Variables for interpolating LUT value */
149 uint16_t between_high = 0;
150 uint16_t between_low = 0;
151 for (uint16_t i = 0u; i < ts_ncxxxxh103LutSize; i++) {
152 if (resistance_Ohm < ts_ncxxxxh103Lut[i].resistance_Ohm) {
153 between_low = i + 1u;
154 between_high = i;
155 }
156 }
157
158 /* Interpolate between LUT values, but do not extrapolate LUT! */
159 if (!(((between_high == 0u) && (between_low == 0u)) || /* measured resistance > maximum LUT resistance */
160 (between_low >= ts_ncxxxxh103LutSize))) { /* measured resistance < minimum LUT resistance */
161 temperature_ddegC = (int16_t)MATH_LinearInterpolation(
162 ts_ncxxxxh103Lut[between_low].resistance_Ohm,
163 (float_t)ts_ncxxxxh103Lut[between_low].temperature_ddegC,
164 ts_ncxxxxh103Lut[between_high].resistance_Ohm,
165 (float_t)ts_ncxxxxh103Lut[between_high].temperature_ddegC,
166 resistance_Ohm);
167 }
168 }
169
170 /* Return temperature based on measured NTC resistance */
171 return temperature_ddegC;
172}
