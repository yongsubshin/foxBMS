288static STD_RETURN_TYPE_e MXM_41BRegisterWrite(
289 MXM_41B_INSTANCE_s *pInstance,
290 MXM_41B_REG_ADD_t command,
291 const uint8_t *const kpkPayload,
292 uint8_t lengthPayload) {
293 /* sanity check: state-pointer may not be null */
294 FAS_ASSERT(pInstance != NULL_PTR);
295 FAS_ASSERT(lengthPayload < (MXM_SPI_TX_BUFFER_LENGTH - 1u));
296 /* AXIVION Routine Generic-MissingParameterAssert: command: parameter accepts whole range */
297 /* AXIVION Routine Generic-MissingParameterAssert: kpkPayload: pointer may be NULL */
298
299 STD_RETURN_TYPE_e retval = STD_NOT_OK;
300 /* check if command is a write command (write addresses in MAX17841B are even) */
301 if ((command % 2u) == 0u) {
302 /* construct tx buffer */
303 pInstance->spiTxBuffer[0] = command;
304 /* message has payload --> copy into buffer */
305 if ((kpkPayload != NULL_PTR) && (lengthPayload != 0u)) {
306 for (uint8_t i = 0u; i < lengthPayload; i++) {
307 pInstance->spiTxBuffer[i + 1u] = kpkPayload[i];
308 }
309 /* null rest of tx buffer */
310 for (uint8_t i = lengthPayload + 1u; i < MXM_SPI_TX_BUFFER_LENGTH; i++) {
311 pInstance->spiTxBuffer[i] = 0u;
312 }
313 /* send command with payload */
314 retval = MXM_SendData(pInstance->spiTxBuffer, (uint16_t)lengthPayload + 1u);
315 } else if ((kpkPayload == NULL_PTR) && (lengthPayload == 0u)) {
316 /* send command without payload */
317 retval = MXM_SendData(pInstance->spiTxBuffer, 1);
318 } else {
319 /* invalid configuration */
320 }
321 }
322 return retval;
323}

325static STD_RETURN_TYPE_e MXM_41BRegisterRead(
326 MXM_41B_INSTANCE_s *pInstance,
327 MXM_41B_REG_ADD_t command,
328 uint16_t *pRxBuffer,
329 uint8_t length) {
330 /* sanity check: state-pointer may not be null */
331 FAS_ASSERT(pInstance != NULL_PTR);
332 /* RX Buffer may not be NULL pointer for this function */
333 FAS_ASSERT(pRxBuffer != NULL_PTR);
334 FAS_ASSERT(length <= MXM_SPI_RX_BUFFER_LENGTH);
335 /* AXIVION Routine Generic-MissingParameterAssert: command: parameter accepts whole range */
336
337 STD_RETURN_TYPE_e retval = STD_NOT_OK;
338 /* check if command is a read command (read addresses in MAX17841B are odd) */
339 if ((command % 2u) != 0u) {
340 /* construct tx buffer */
341 pInstance->spiTxBuffer[0] = command;
342 /* null rest of tx buffer */
343 for (uint16_t i = 1u; i < MXM_SPI_TX_BUFFER_LENGTH; i++) {
344 pInstance->spiTxBuffer[i] = 0u;
345 }
346 /* send command with payload */
347 retval = MXM_ReceiveData(pInstance->spiTxBuffer, pRxBuffer, ((uint16_t)length + 1u));
348 }
349 return retval;

353 /* sanity check: state-pointer may not be null */

372static STD_RETURN_TYPE_e MXM_41BBufferWrite(
373 MXM_41B_INSTANCE_s *pInstance,
374 const uint16_t *const kpkMessage,
375 uint8_t messageLength,
376 uint8_t extendMessage) {
377 /* sanity check: state-pointer may not be null */
378 FAS_ASSERT(pInstance != NULL_PTR);
379 /* check if message-pointer is valid */
380 FAS_ASSERT(kpkMessage != NULL_PTR);
381 FAS_ASSERT(messageLength >= 1u);
382 FAS_ASSERT(messageLength <= 6u);
383 /* AXIVION Routine Generic-MissingParameterAssert: extendMessage: parameter accepts whole range */
384
385 /* write address and length to buffer */
386 pInstance->spiTxBuffer[0] = (uint16_t)MXM_BUF_WR_LD_Q_0;
387 pInstance->spiTxBuffer[1] = (uint16_t)messageLength + extendMessage;
388 /* iterate of complete TX buffer and
389 * write into proper fields, null rest
390 */
391 for (uint8_t i = 0; i < (MXM_SPI_TX_BUFFER_LENGTH - 2u); i++) {
392 if (i < messageLength) {
393 pInstance->spiTxBuffer[i + 2u] = kpkMessage[i];
394 } else {
395 pInstance->spiTxBuffer[i + 2u] = 0x00u;
396 }
397 }
398
399 /* send data */
400 return MXM_SendData(pInstance->spiTxBuffer, ((uint16_t)messageLength + 2u));
401}

430static void MXM_41BStateHandlerInit(MXM_41B_INSTANCE_s *pInstance) {
431 FAS_ASSERT(pInstance != NULL_PTR);
432
433 if (pInstance->substate == MXM_41B_ENTRY_SUBSTATE) {
434 /* entry of state --> set to first substate */
435 pInstance->substate = MXM_41B_INIT_RESET_BRIDGE_IC;
436 }
437
438 if (pInstance->substate == MXM_41B_INIT_RESET_BRIDGE_IC) {
439 MXM_ShutDownBridgeIc();
440 pInstance->shutdownTimeStamp = OS_GetTickCount();
441 pInstance->substate = MXM_41B_INIT_START_BRIDGE_IC;
442 } else if (pInstance->substate == MXM_41B_INIT_START_BRIDGE_IC) {
443 const bool resetTimeHasPassed =
444 OS_CheckTimeHasPassed(pInstance->shutdownTimeStamp, MXM_41B_BRIDGE_RESET_TIME_MS);
445 if (resetTimeHasPassed) {
446 MXM_EnableBridgeIc();
447 pInstance->substate = MXM_41B_INIT_WRITE_DEFAULT_VALUES;
448 }
449 } else if (pInstance->substate == MXM_41B_INIT_WRITE_DEFAULT_VALUES) {
450 /* set default register values according to data sheet */
451 MXM_41BInitializeRegisterCopies(pInstance);
452 const STD_RETURN_TYPE_e retval = MXM_41BConfigRegisterWrite(pInstance);
453 if (retval == STD_OK) {
454 pInstance->substate = MXM_41B_INIT_READ_CONFIG_REGISTERS;
455 }
456 } else if (pInstance->substate == MXM_41B_INIT_READ_CONFIG_REGISTERS) {
457 const STD_RETURN_TYPE_e retval = MXM_41BRegisterRead(
458 pInstance, MXM_REG_RX_INTERRUPT_ENABLE_R, pInstance->spiRxBuffer, MXM_41B_CONFIG_REGISTER_LENGTH);
459
460 if (retval == STD_OK) {
461 pInstance->substate = MXM_41B_INIT_CHECK_INITIALIZATION;
462 }
463 } else if (pInstance->substate == MXM_41B_INIT_CHECK_INITIALIZATION) {
464 STD_RETURN_TYPE_e retval = STD_OK;
465
466 if (MXM_GetSPIStateReady() == STD_OK) {
467 /**
468 * @brief Default values for the configuration and interrupt registers
469 * @details This constant array contains the default values to which the
470 * configuration is compared when resetting it.
471 * The array is 7 entries long in order to fit onto the following
472 * registers which are placed in succession in the memory of
473 * MAX17841B:
474 * - RX_Interrupt_Enable
475 * - TX_Interrupt_Enable
476 * - RX_Interrupt_Flags
477 * - TX_Interrupt_Flags
478 * - Configuration_1
479 * - Configuration_2
480 * - Configuration_3
481 */
482 const uint8_t mxm_41B_reg_default_values[MXM_41B_CONFIG_REGISTER_LENGTH] = {
483 MXM_41B_RX_INT_ENABLE_DEFAULT_VALUE,
484 MXM_41B_TX_INT_ENABLE_DEFAULT_VALUE,
485 MXM_41B_RX_INT_FLAG_DEFAULT_VALUE,
486 MXM_41B_TX_INT_FLAG_DEFAULT_VALUE,
487 MXM_41B_CONFIG_1_DEFAULT_VALUE,
488 MXM_41B_CONFIG_2_DEFAULT_VALUE,
489 MXM_41B_CONFIG_3_DEFAULT_VALUE};
490
491 for (uint8_t i = 0; i < MXM_41B_CONFIG_REGISTER_LENGTH; i++) {
492 if (pInstance->spiRxBuffer[i + 1u] != mxm_41B_reg_default_values[i]) {
493 retval = STD_NOT_OK;
494 }
495 }
496 } else {
497 retval = STD_NOT_OK;
498 }
499
500 if (retval == STD_NOT_OK) {
501 MXM_41BTransitionToIdleError(pInstance);
502 } else {
503 MXM_41BTransitionToIdleSuccess(pInstance);
504 }
505 } else {
506 MXM_41BTransitionToIdleError(pInstance);
507 }
508}

510static void MXM_41BStateHandlerGetVersion(MXM_41B_INSTANCE_s *pInstance) {
511 FAS_ASSERT(pInstance != NULL_PTR);
512 if (pInstance->substate == MXM_41B_ENTRY_SUBSTATE) {
513 pInstance->substate = MXM_41B_VERSION_REQUEST_REGISTER;
514 }
515
516 if (pInstance->substate == MXM_41B_VERSION_REQUEST_REGISTER) {
517 /* read two byte in order to read also the adjacent version register */
518 const STD_RETURN_TYPE_e retval = MXM_41BRegisterRead(pInstance, MXM_REG_MODEL_R, pInstance->spiRxBuffer, 2);
519
520 if (retval == STD_OK) {
521 pInstance->substate = MXM_41B_VERSION_VERIFY;
522 }
523 } else if (pInstance->substate == MXM_41B_VERSION_VERIFY) {
524 if (MXM_GetSPIStateReady() == STD_OK) {
525 /* get model from model register and high nibble of mask revision (should be 0x8410) */
526 pInstance->hardwareModel =
527 (uint16_t)((pInstance->spiRxBuffer[1] & MXM_41B_BIT_MASK_ONE_BYTE) << MXM_41B_BIT_SHIFT_HALF_BYTE);
528 pInstance->hardwareModel |=
529 (uint16_t)((pInstance->spiRxBuffer[2] & MXM_41B_BIT_MASK_HIGH_NIBBLE) >> MXM_41B_BIT_SHIFT_HALF_BYTE);
530 /* extract mask revision from low nibble */
531 pInstance->hardwareMaskRevision = (uint8_t)(pInstance->spiRxBuffer[2] & MXM_41B_BIT_MASK_LOW_NIBBLE);
532
533 MXM_41BTransitionToIdleSuccess(pInstance);
534 }
535 } else {
536 /* something is very broken */
537 MXM_41BTransitionToIdleError(pInstance);
538 }
539}

585static void MXM_41BStateHandlerUartTransaction(MXM_41B_INSTANCE_s *pInstance) {
586 FAS_ASSERT(pInstance != NULL_PTR);
587 if (pInstance->substate == MXM_41B_ENTRY_SUBSTATE) {
588 /* entry of state --> set to first substate */
589 pInstance->substate = MXM_41B_UART_READ_RX_SPACE;
590 }
591
592 if (pInstance->substate == MXM_41B_UART_READ_RX_SPACE) {
593 const STD_RETURN_TYPE_e retval = MXM_41BRegisterRead(pInstance, MXM_REG_RX_SPACE_R, pInstance->spiRxBuffer, 1u);
594 if (retval == STD_NOT_OK) {
595 MXM_41BTransitionToIdleError(pInstance);
596 } else {
597 pInstance->substate = MXM_41B_UART_READ_RX_SPACE_PARSE;
598 }
599 } else if (pInstance->substate == MXM_41B_UART_READ_RX_SPACE_PARSE) {
600 pInstance->regRxSpace = (uint8_t)(pInstance->spiRxBuffer[1] & MXM_41B_BIT_MASK_ONE_BYTE);
601 pInstance->substate = MXM_41B_UART_WRITE_LOAD_QUEUE;
602
603 } else if (pInstance->substate == MXM_41B_UART_WRITE_LOAD_QUEUE) {
604 /* load queue with message */
605 const STD_RETURN_TYPE_e retval =
606 MXM_41BBufferWrite(pInstance, pInstance->pPayload, pInstance->payloadLength, pInstance->extendMessageBytes);
607
608 if (retval == STD_NOT_OK) {
609 MXM_41BTransitionToIdleError(pInstance);
610 } else {
611 pInstance->substate = MXM_41B_UART_READ_LOAD_QUEUE;
612 }
613 } else if (pInstance->substate == MXM_41B_UART_READ_LOAD_QUEUE) {
614 /* check assumption that incremented payload length fits into uint8_t */
615 FAS_ASSERT(pInstance->payloadLength < (uint8_t)UINT8_MAX);
616 const uint8_t payloadLength = pInstance->payloadLength + 1u;
617 /* send read load queue */
618 const STD_RETURN_TYPE_e retval =
619 MXM_41BRegisterRead(pInstance, MXM_BUF_RD_LD_Q_0, pInstance->spiRxBuffer, payloadLength);
620
621 if (retval == STD_NOT_OK) {
622 MXM_41BTransitionToIdleError(pInstance);
623 } else {
624 pInstance->substate = MXM_41B_UART_VERIFY_LOAD_QUEUE_AND_TRANSMIT;
625 }
626 } else if (pInstance->substate == MXM_41B_UART_VERIFY_LOAD_QUEUE_AND_TRANSMIT) {
627 /* verify load queue */
628 STD_RETURN_TYPE_e retval = STD_OK;
629 /* check message length */
630 if (pInstance->spiRxBuffer[1] != (pInstance->payloadLength + (uint16_t)pInstance->extendMessageBytes)) {
631 retval = STD_NOT_OK;
632 }
633 for (uint8_t i = 0; i < pInstance->payloadLength; i++) {
634 FAS_ASSERT(pInstance->pPayload != NULL_PTR);
635 if (pInstance->spiRxBuffer[i + 2u] != pInstance->pPayload[i]) {
636 /* message corrupted during SPI transfer */
637 retval = STD_NOT_OK;
638 }
639 }
640 if (retval == STD_NOT_OK) {
641 MXM_41BTransitionToIdleError(pInstance);
642 } else {
643 /* transmit queue */
644 retval = MXM_41BRegisterWrite(pInstance, MXM_BUF_WR_NXT_LD_Q_0, NULL_PTR, 0);
645
646 if (retval == STD_NOT_OK) {
647 MXM_41BTransitionToIdleError(pInstance);
648 } else {
649 pInstance->substate = MXM_41B_UART_WAIT_FOR_RX_STATUS_CHANGE_WRITE;
650 }
651 }
652 } else if (pInstance->substate == MXM_41B_UART_WAIT_FOR_RX_STATUS_CHANGE_WRITE) {
653 /* poll RX status change */
654 const STD_RETURN_TYPE_e retval = MXM_41BRegisterRead(pInstance, MXM_REG_RX_STATUS_R, pInstance->spiRxBuffer, 1);
655
656 if (retval == STD_NOT_OK) {
657 MXM_41BTransitionToIdleError(pInstance);
658 } else {
659 pInstance->substate = MXM_41B_UART_WAIT_FOR_RX_STATUS_CHANGE_READ_AND_READ_BACK_RCV_BUF;
660 }
661 } else if (pInstance->substate == MXM_41B_UART_WAIT_FOR_RX_STATUS_CHANGE_READ_AND_READ_BACK_RCV_BUF) {
662 /* update RX status register copy with received buffer */
663 pInstance->regRxStatus = (uint8_t)(pInstance->spiRxBuffer[1] & MXM_41B_BIT_MASK_ONE_BYTE);
664 /* check if RX_OVERFLOW_Status is 1 */
665 MXM_41B_REG_BIT_VALUE rx_overflow_status_value = MXM_41B_REG_FALSE;
666 const STD_RETURN_TYPE_e resultWrongRegisterOverflow =
667 MXM_41BReadRegisterFunction(pInstance, MXM_41B_REG_FUNCTION_RX_OVERFLOW_STATUS, &rx_overflow_status_value);
668 FAS_ASSERT(resultWrongRegisterOverflow == STD_OK);
669 /* check if RX_STOP_Status is 1 */
670 MXM_41B_REG_BIT_VALUE rx_stop_status_value = MXM_41B_REG_FALSE;
671 const STD_RETURN_TYPE_e resultWrongRegisterStop =
672 MXM_41BReadRegisterFunction(pInstance, MXM_41B_REG_FUNCTION_RX_STOP_STATUS, &rx_stop_status_value);
673 FAS_ASSERT(resultWrongRegisterStop == STD_OK);
674 if (rx_overflow_status_value == MXM_41B_REG_TRUE) {
675 /* overflow, we have to discard the rx buffer */
676 const STD_RETURN_TYPE_e retval = MXM_41BRegisterWrite(pInstance, MXM_BUF_CLR_RX_BUF, NULL_PTR, 0);
677 if (retval == STD_NOT_OK) {
678 MXM_41BTransitionToIdleError(pInstance);
679 } else {
680 MXM_41BTransitionToIdleError(pInstance);
681 }
682 } else if (rx_stop_status_value == MXM_41B_REG_TRUE) {
683 /* received full UART frame --> continue */
684 /* check assumption that payload length fits into uint8_t */
685 FAS_ASSERT((pInstance->payloadLength + (uint16_t)1u + pInstance->extendMessageBytes) <= (uint8_t)UINT8_MAX);
686 const uint8_t payloadLength = pInstance->payloadLength + 1u + pInstance->extendMessageBytes;
687 /* read back receive buffer */
688 const STD_RETURN_TYPE_e retval =
689 MXM_41BRegisterRead(pInstance, MXM_BUF_RD_NXT_MSG, pInstance->spiRxBuffer, payloadLength);
690
691 if (retval == STD_NOT_OK) {
692 MXM_41BTransitionToIdleError(pInstance);
693 } else {
694 pInstance->substate = MXM_41B_UART_READ_BACK_RECEIVE_BUFFER_SAVE;
695 pInstance->waitCounter = 0u;
696 }
697 } else {
698 /* no UART frame received yet --> check again */
699 pInstance->substate = MXM_41B_UART_WAIT_FOR_RX_STATUS_CHANGE_WRITE;
700 /* increment wait counter (only to 1 above MXM_41B_WAIT_COUNTER_THRESHOLD,
701 then other parts of the code will reset). */
702 if (pInstance->waitCounter <= MXM_41B_WAIT_COUNTER_THRESHOLD) {
703 pInstance->waitCounter++;
704 }
705 }
706 } else if (pInstance->substate == MXM_41B_UART_READ_BACK_RECEIVE_BUFFER_SAVE) {
707 if ((pInstance->spiRxBuffer != NULL_PTR) && (pInstance->pRxBuffer != NULL_PTR)) {
708 for (uint16_t i = 0; i < ((uint16_t)pInstance->payloadLength + pInstance->extendMessageBytes); i++) {
709 if (i < pInstance->rxBufferLength) {
710 pInstance->pRxBuffer[i] = pInstance->spiRxBuffer[i + 1u];
711 }
712 }
713 }
714 MXM_41BTransitionToIdleSuccess(pInstance);
715 } else {
716 /* we should not be here */
717 FAS_ASSERT(FAS_TRAP);
718 }
719}
