189/**
190 * @brief Write a register function
191 *
192 * Functions of the ASCI are mapped onto parts of the register.
193 * This function is used to write the proper values into the register copy
194 * of the ASCI that is inside the driver.
195 * In order to use the function, provide the name of the ASCI function that you
196 * want to write (#MXM_41B_REG_FUNCTION_e) and the value that
197 * should be written (#MXM_41B_REG_BIT_VALUE).
198 * Currently, only a relevant subset of the ASCI features is implemented.
199 *
200 * After updating the register copy, it has to be written to the ASCI memory.
201 * For the configuration registers this is done with the function
202 * MXM_41BConfigRegisterWrite(). This function is for example called
203 * in state #MXM_STATEMACH_41B_WRITE_CONF_AND_INT_REGISTER.
204 *
205 * @param[in,out] pInstance pointer to the state of the
206 * MAX17841B-state-machine
207 * @param[in] registerFunction name of the function that shall be
208 * written
209 * @param[in] value value of the bits that shall be written
210 * @return #STD_NOT_OK for unknown or unimplemented register
211 * functions, otherwise #STD_OK
212 */
213extern STD_RETURN_TYPE_e MXM_41BWriteRegisterFunction(
214 MXM_41B_INSTANCE_s *pInstance,
215 MXM_41B_REG_FUNCTION_e registerFunction,
216 MXM_41B_REG_BIT_VALUE value);
217
218/**
219 * @brief Read the value of a register function
220 *
221 * Corresponding read-function to MXM_41BWriteRegisterFunction().
222 * This function reads a value from the register-copy of MAX17841B and writes
223 * it into a supplied pointer.
224 * The register function is specified by the
225 * corresponding #MXM_41B_REG_FUNCTION_e value.
226 *
227 * This function implements only the needed subset of register functions.
228 *
229 * @param[in,out] kpkInstance pointer to the state of the
230 * MAX17841B-state-machine
231 * @param[in] registerFunction name of the function that shall be
232 * written
233 * @param[out] pValue pointer to a variable in which the bits
234 * shall be written

236 * otherwise #STD_OK

238extern STD_RETURN_TYPE_e MXM_41BReadRegisterFunction(

251/**
252 * @brief Set state transition for MAX17841B-state-machine.
253 *
254 * This function is used to transition into another state of the
255 * state-machine which is handled in MXM_41BStateMachine().
256 * The user sets a state that should be transitioned into.
257 * For most states a payload and a payload-length can be supplied.
258 * The payload-length has to be consistent with the length in payload.
259 * Typically, this supplied payload describes the byte-sequence that
260 * should be written into the load-queue.
261 *
262 * For the case of a e.g. READALL-command the message-length has to
263 * be stretched by the number of connected monitoring ICs times two.
264 * This is achieved by passing on extendMessageBytes.
265 *
266 * In case that the state generates data that has to be handled in the
267 * higher layers, the pointer pRxBuffer will be used. Please make sure that
268 * the length of pRxBuffer is also consistent.
269 *
270 * In order to determine success or failure of the requested state,
271 * the higher layer supplies a pointer into which the current status
272 * of the requested state will be written.
273 *
274 * @param[in,out] pInstance pointer to the state of the
275 * MAX17841B-state-machine
276 * @param[in] state state into which the state-machine
277 * shall transition
278 * @param[in] pPayload pointer to an array with data that will
279 * be processed in the requested state
280 * @param[in] payloadLength length of the payload-array
281 * @param[in] extendMessageBytes number of bytes that shall be appended
282 * by the ASCI
283 * @param[out] pRxBuffer pointer to an array that will be filled
284 * by the requested state with data that
285 * has to be processed in higher layers
286 * @param[in] rxBufferLength length of the RX buffer array
287 * @param[out] processed pointer of the status of the requested
288 * state
289 * @return #STD_NOT_OK for inconsistent input or forbidden state
290 * transitions, otherwise #STD_OK
291 */
292extern STD_RETURN_TYPE_e MXM_41BSetStateRequest(
293 MXM_41B_INSTANCE_s *pInstance,
294 MXM_STATEMACH_41B_e state,
295 uint16_t *pPayload,
296 uint8_t payloadLength,
297 uint8_t extendMessageBytes,
298 uint16_t *pRxBuffer,
299 uint16_t rxBufferLength,
300 MXM_41B_STATE_REQUEST_STATUS_e *processed);
