123/**
124 * @brief Parse voltage values from a READALL receive buffer into an
125 * array.
126 * @details This function expects a preprocessed RX buffer with an answer
127 * to a READALL command. It will parse the data bytes from this
128 * message into an array structured like mxm_local_cellVoltages.
129 * The offset of the measured cell has to be defined. For example
130 * cell 1 has an offset of 0 and cell 4 has an offset of 3. The
131 * offset for the READALL command is always defined in reference
132 * to module 0.
133 *
134 * If is_temperature_measurement is set to true, the function
135 * expects an auxiliary measurement (e.g. temperatures).
136 * @param[in] kpkVoltRxBuffer array-pointer to the RX buffer
137 * @param[in] voltRxBufferLength length of the RX buffer
138 * @param[in] measurementOffset offset of the data in the cell voltage
139 * array (target)
140 * @param[in] conversionType type of conversion that has been used
141 * for the measured data
142 * @param[out] pVoltagesTarget array-pointer to the cell voltages
143 * array
144 * @param[in] measurementType whether the measurement is temperature
145 * or cell
146 * @param[in] fullScaleReference_mV reference voltage of full scale
147 */
148static void MXM_ParseVoltageLineReadall(
149 const uint8_t *const kpkVoltRxBuffer,
150 uint16_t voltRxBufferLength,
151 uint8_t measurementOffset,
152 MXM_CONVERSION_TYPE_e conversionType,
153 uint16_t *pVoltagesTarget,
154 MXM_MEASURE_TYPE_e measurementType,
155 uint32_t fullScaleReference_mV);

157/**
158 * @brief Parse a RX buffer containing voltage values.

205static void MXM_ParseVoltageLineReadall(
206 const uint8_t *const kpkVoltRxBuffer,
207 uint16_t voltRxBufferLength,
208 uint8_t measurementOffset,
209 MXM_CONVERSION_TYPE_e conversionType,
210 uint16_t *pVoltagesTarget,
211 MXM_MEASURE_TYPE_e measurementType,
212 uint32_t fullScaleReference_mV) {
213 FAS_ASSERT(kpkVoltRxBuffer != NULL_PTR);
214 FAS_ASSERT(pVoltagesTarget != NULL_PTR);
215 /* AXIVION Routine Generic-MissingParameterAssert: voltRxBufferLength: parameter accepts whole range */
216 /* AXIVION Routine Generic-MissingParameterAssert: measurementOffset: parameter accepts whole range */
217 /* AXIVION Routine Generic-MissingParameterAssert: conversionType: parameter accepts whole range */
218 /* AXIVION Routine Generic-MissingParameterAssert: measurementType: parameter accepts whole range */
219 /* AXIVION Routine Generic-MissingParameterAssert: fullScaleReference_mV: parameter accepts whole range */
220
221 /* assert that assumptions behind computation of numberOfConnectedDevices
222 are correct */
223 FAS_ASSERT(voltRxBufferLength >= (2u + 2u));
224 FAS_ASSERT(((voltRxBufferLength - 2u - 2u) / 2u) <= (uint16_t)UINT8_MAX);
225 /* buffer-length - length of start - length of end divided by two (LSB and MSB) */
226 const uint8_t numberOfConnectedDevices = (uint8_t)(((voltRxBufferLength - 2u - 2u) / 2u) & (uint8_t)UINT8_MAX);
227 /* TODO impact of alive counter on rxBufferLength
228 * otherwise offset at the end of message is currently 2 (DATACHECKBYTE and CRC) */
229 for (uint8_t i = 2u; i < (voltRxBufferLength - 2u); i = i + 2u) {
230 const uint8_t calculatedModuleNumber = numberOfConnectedDevices - ((i / 2u) - 1u) - 1u;
231 uint16_t calculatedModulePosition = 0u;
232 switch (measurementType) {
233 case MXM_MEASURE_TEMP:
234 calculatedModulePosition = (uint16_t)calculatedModuleNumber * MXM_MAXIMUM_NR_OF_AUX_PER_MODULE;
235 break;
236 case MXM_MEASURE_CELL_VOLTAGE:
237 calculatedModulePosition = (uint16_t)calculatedModuleNumber * MXM_MAXIMUM_NR_OF_CELLS_PER_MODULE;
238 break;
239 case MXM_MEASURE_BLOCK_VOLTAGE:
240 calculatedModulePosition = calculatedModuleNumber;
241 break;
242 default:
243 FAS_ASSERT(FAS_TRAP);
244 break; /* LCOV_EXCL_LINE */
245 }
246 FAS_ASSERT((calculatedModulePosition + measurementOffset) <= (uint16_t)UINT16_MAX);
247 uint16_t calculatedArrayPosition = calculatedModulePosition + measurementOffset;
248
249 /* check calculated array position */
250 switch (measurementType) {
251 case MXM_MEASURE_TEMP:
252 FAS_ASSERT(calculatedArrayPosition < (MXM_MAXIMUM_NR_OF_MODULES * MXM_MAXIMUM_NR_OF_AUX_PER_MODULE));
253 break;
254 case MXM_MEASURE_CELL_VOLTAGE:
255 FAS_ASSERT(calculatedArrayPosition < (MXM_MAXIMUM_NR_OF_MODULES * MXM_MAXIMUM_NR_OF_CELLS_PER_MODULE));
256 break;
257 case MXM_MEASURE_BLOCK_VOLTAGE:
258 FAS_ASSERT(calculatedArrayPosition < (MXM_MAXIMUM_NR_OF_MODULES));
259 break;
260 default:
261 FAS_ASSERT(FAS_TRAP);
262 break; /* LCOV_EXCL_LINE */
263 }
264
265 FAS_ASSERT(i <= voltRxBufferLength);
266 FAS_ASSERT((i + (uint16_t)1u) <= voltRxBufferLength);
267 MXM_Convert(
268 kpkVoltRxBuffer[i],
269 kpkVoltRxBuffer[i + 1u],
270 &pVoltagesTarget[calculatedArrayPosition],
271 conversionType,
272 fullScaleReference_mV);
273 }

991extern STD_RETURN_TYPE_e MXM_ProcessOpenWire(const MXM_MONITORING_INSTANCE_s *const kpkInstance) {
992 FAS_ASSERT(kpkInstance != NULL_PTR);
993 FAS_ASSERT(kpkInstance->pOpenWire_table != NULL_PTR);
994
995 const uint8_t numberOfSatellites = MXM_5XGetNumberOfSatellites(kpkInstance->pInstance5X);
996 FAS_STATIC_ASSERT(
997 (((uint16_t)BATTERY_MANAGEMENT_TX_LENGTH_READALL + (2u * MXM_MAXIMUM_NR_OF_MODULES)) <= (uint8_t)UINT8_MAX),
998 "please check assumptions: code cannot handle number of modules");
999 const uint8_t messageLength = BATTERY_MANAGEMENT_TX_LENGTH_READALL + (2u * numberOfSatellites);
1000 /* step over every byte-tuple in the RX-buffer */
1001 for (uint8_t i = 2u; i < (messageLength - 2u); i = i + 2u) {
1002 /* calculate current module number (first in RX-buffer is last in line) */
1003 const uint8_t calculatedModuleNumberInDaisyChain = numberOfSatellites - ((i / 2u) - 1u) - 1u;
1004 uint8_t stringNumber = 0u;
1005 uint16_t moduleNumber = 0u;
1006 MXM_ConvertModuleToString(calculatedModuleNumberInDaisyChain, &stringNumber, &moduleNumber);
1007 const uint16_t calculatedModulePosition = moduleNumber * MXM_MAXIMUM_NR_OF_CELLS_PER_MODULE;
1008
1009 /* step over every cell in the module and update the openWire struct accordingly */
1010 for (uint8_t c = 0u; c < MXM_MAXIMUM_NR_OF_CELLS_PER_MODULE; c++) {
1011 if (c < MXM_CELLS_IN_LSB) {
1012 /* cell numbers under 8 can be found in the LSB */
1013 const uint8_t mask = 1u << c;
1014 if ((uint8_t)(mask & kpkInstance->rxBuffer[i]) > 0u) {
1015 kpkInstance->pOpenWire_table->openWire[stringNumber][calculatedModulePosition + c] = 1;
1016 } else {
1017 kpkInstance->pOpenWire_table->openWire[stringNumber][calculatedModulePosition + c] = 0;
1018 }
1019 } else {
1020 /* cell numbers over or equal 8 can be found in the MSB */
1021 const uint8_t mask = 1u << (c - MXM_CELLS_IN_LSB);
1022 if ((uint8_t)(mask & kpkInstance->rxBuffer[i + 1u]) > 0u) {
