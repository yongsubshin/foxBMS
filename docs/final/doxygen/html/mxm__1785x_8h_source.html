42/**
43 * @file mxm_1785x.h
44 * @author foxBMS Team
45 * @date 2019-01-15 (date of creation)
46 * @updated 2025-08-07 (date of last update)
47 * @version v1.10.0
48 * @ingroup DRIVERS
49 * @prefix MXM
50 *
51 * @brief Headers for the driver for the MAX17841B ASCI and MAX1785x
52 * analog front-end
53 * @details Battery monitoring driver for MAX1785x battery monitoring ICs.
54 *
55 * This module supplies a driver for the Battery Monitoring ICs of the
56 * MAX1785x-family by Maxim Integrated.
57 *
58 * Entry point for the module is the function #MXM_Tick() in
59 * mxm_afe.c.
60 * It handles the measurement flow and the coordination of the
61 * underlying state-machines.
62 * Below this layer two state-machines are implemented.
63 *
64 * The state-machine in mxm_battery_management.c is executed with the
65 * #MXM_5XStateMachine()-function.
66 * This state-machine exposes commands of the Maxim Battery Management
67 * Protocol to the upper layers.
68 * Below, it translates these commands into state-changes for the
69 * underlying state-machine.
70 * This state-machine is implemented in mxm_17841b.c and executed with
71 * #MXM_41BStateMachine().
72 * It handles the register- and buffer-transactions required for the
73 * MAX17841B communication interface (Maxim calls this analog
74 * front-end ASCI).
75 */

170/**
171 * @brief Handle the state machine-transactions for a READALL
172 * @details Call this function and pass on the state-variable, the
173 * register to be read and the next state. The function will
174 * handle the communication with the lower state-machine and
175 * will transition into the next state, if the command has
176 * been sent successfully. Moreover it will return true when
177 * transitioning. The return value has to be checked and used
178 * to execute additional code if necessary.
179 * @param[in,out] pInstance pointer to instance of the mxm
180 * monitoring state-machine
181 * @param[in] registerName register that should be read
182 * @param[in] nextState state that should be entered upon successful
183 * completion
184 * @return true when the state has been handled, false otherwise, use
185 * this to execute additional code when the message has been
186 * read.
187 */

225/**
226 * @brief Encapsulation for reading voltages from a register
227 * @details This function encapsulates the request of state-changes and
228 * following conversion for the reading of an arbitrary
229 * measurement voltage of the daisy-chain. Its parameters are
230 * a variable for tracking the state of the underlying
231 * state-machines and the register address that has to be
232 * queried. It returns whether the action has been successful
233 * or not.
234 * In order to obtain all cell voltages this function has to
235 * be called for every relevant register address.
236 * @param[in,out] pState pointer to the state-machine struct
237 * @param[in] regAddress register address that shall be queried
238 * @return current state of the action:
239 * - MXM_MON_STATE_PASS upon completion
240 * - MXM_MON_STATE_PENDING as long as the action is
241 * ongoing
242 * - MXM_MON_STATE_FAIL if the function failed and could
243 * not recover on its own
244 */

248/**
249 * @brief Copies measured voltage data into the database.
250 * @details This function copies the acquired voltage data from
251 * #MXM_MONITORING_INSTANCE_s::localVoltages into the database-struct
252 * #mxm_tableCellVoltages and copies this struct into the database.
253 * This action is required due to different data layouts. This driver
254 * always stores its cell-voltages in an array with 14*32 = 448 cells
255 * in order to reduce the amount of different configurations and
256 * variants.
257 *
258 * This function maps these values into the database-struct which
259 * scales with the number of connected cells and monitoring ICs.
260 * @param[in] kpkInstance pointer to the #MXM_MONITORING_INSTANCE_s struct
261 * @return #STD_OK if the action was successful or #STD_NOT_OK otherwise
262 */
263extern STD_RETURN_TYPE_e MXM_ParseVoltagesIntoDB(const MXM_MONITORING_INSTANCE_s *const kpkInstance);
