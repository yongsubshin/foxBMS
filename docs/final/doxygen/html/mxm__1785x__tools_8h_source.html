286/**
287 * @brief Convert a measurement value to a voltage value.
288 * @details This function converts measurement values from the monitoring
289 * IC into a voltage value. It assumes that measurement values are
290 * spread over the LSB and MSB according to the data sheet. This
291 * means that the two lowest bits are 0 and the 14-bit measurement
292 * is spread over the 14 highest bits.
293 *
294 * The conversion type describes the kind of measurement that has
295 * been executed.
296 * @param[in] lsb LSB of the value
297 * @param[in] msb MSB of the value
298 * @param[out] pTarget pointer in which the converted value
299 * shall be written
300 * @param[in] convType type of conversion (bipolar or unipolar)
301 * @param[in] fullScaleReference_mV reference voltage of full scale
302 * (for voltage measurements 5000)
303 */
304extern void MXM_Convert(
305 uint8_t lsb,
306 uint8_t msb,
307 uint16_t *pTarget,
308 MXM_CONVERSION_TYPE_e convType,
309 uint32_t fullScaleReference_mV);
310
311/**
312 * @brief Test the #MXM_Convert()-function.
313 * @details This test-function is intended as a self-check. It tests whether
314 * the #MXM_Convert()-function computes the values as expected.
315 * @return #STD_OK on successful execution, otherwise #STD_NOT_OK
316 */
317extern STD_RETURN_TYPE_e GEN_MUST_CHECK_RETURN MXM_ConvertTest(void);
318
319/**
320 * @brief Extract a value from a single register.
321 * @details This function extracts a single value from a single register.
322 * The register has to be supplied as MSB and LSB. The bitmask has to
323 * be of type #MXM_REG_BM. The extracted value will be written into
324 * value.
325 *
326 * The function can only extract values from single registers
327 * (MSB and LSB)and does not handle values that span multiple
328 * registers.
329 * @param[in] lsb least significant byte of the register
330 * @param[in] msb most significant byte of the register
331 * @param[in] bitmask bitmask for the value
332 * @param[out] pValue pointer to a variable in which the value
333 * should be written
334 */
335extern void MXM_ExtractValueFromRegister(uint8_t lsb, uint8_t msb, MXM_REG_BM bitmask, uint16_t *pValue);

375/**
376 * @brief Get the string and module number from a global module number

380 * strings).
381 *
382 * This function assumes that a string contains #BS_NR_OF_MODULES_PER_STRING modules

385 * @param[in] moduleNumber number of the module in the daisy-chain (starts with 0)
386 * @param[out] pStringNumber pointer where the string number will be written
387 * @param[out] pModuleNumberInString module number in the string
388 */
389extern void MXM_ConvertModuleToString(
390 const uint16_t moduleNumber,
391 uint8_t *pStringNumber,
