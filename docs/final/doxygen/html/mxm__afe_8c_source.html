131/**
132 * @brief Tick function
133 * @details This function is the central entry-point for this driver. It calls
134 * the state-machines via the functions #MXM_StateMachine(),
135 * #MXM_5XStateMachine() and #MXM_41BStateMachine().
136 *
137 * The state-machines are called in such a way that the main blocking
138 * point is the SPI interface. (That means the order is optimized so
139 * that every cycle a SPI command should be available for sending or
140 * receiving as this is the limiting resource.)
141 */

153static void MXM_Tick(void) {
154 MXM_CheckIfErrorCounterCanBeReset(&mxm_state);
155 MXM_StateMachine(&mxm_state);
156 MXM_5XStateMachine(mxm_state.pInstance41B, mxm_state.pInstance5X);
157 MXM_41BStateMachine(mxm_state.pInstance41B);
158 /* execute battery management state machine once (low cost and prepares for next cycle) */
159 MXM_5XStateMachine(mxm_state.pInstance41B, mxm_state.pInstance5X);
160
161 /* check if init is stuck */
162 if ((mxm_state.state != MXM_STATEMACHINE_STATES_UNINITIALIZED) &&
163 (mxm_state.state != MXM_STATEMACHINE_STATES_IDLE) && (mxm_state.state != MXM_STATEMACHINE_STATES_OPERATION) &&
164 (mxm_state.resetNecessary == false)) {
165 /* check if we spend too long in initialization and try to resolve with a reset */
166 if (OS_CheckTimeHasPassed(mxm_state.timestampInit, MXM_MAXIMUM_INIT_TIME_ms) == true) {
167 MXM_ErrorHandlerReset(&mxm_state, true);
168 }
169 }
170}

172static void MXM_SetStateStructDefaultValues(void) {
173 MXM_41BInitializeStateStruct(&mxm_41bState);
174 MXM_5X_InitializeStateStruct(&mxm_5xState);
175 MXM_InitializeStateStruct(&mxm_balancingState, &mxm_state);
176}

179extern STD_RETURN_TYPE_e AFE_TriggerIc(void) {
180 if (mxm_state.resetNecessary == true) {
181 /* a reset has been requested */
182
183 OS_EnterTaskCritical();
184 /* save allowStartup and operationRequested */
185 const bool allowStartup = mxm_state.allowStartup;
186 const bool operationRequested = mxm_state.operationRequested;
187 const bool firstMeasurementDone = mxm_state.firstMeasurementDone;
188 OS_ExitTaskCritical();
189
190 (void)AFE_Initialize();
191
192 /* restore relevant flags */
193 mxm_state.allowStartup = allowStartup;
194 /* if a first measurement has been done, operation has been requested before */
195 mxm_state.operationRequested = (operationRequested || firstMeasurementDone);
196 } else {
197 MXM_Tick();
198 }
199
200 return STD_OK;
201}

203extern STD_RETURN_TYPE_e AFE_Initialize(void) {
204 MXM_SetStateStructDefaultValues();
205 MXM_InitializeMonitoringPins();
206 /* call pre init self check so that we can do these costly tests before the main cycle begins (results are stored)
207 */
208 (void)MXM_PreInitSelfCheck(&mxm_state);
209 return STD_OK;
