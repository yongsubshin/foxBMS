259static STD_RETURN_TYPE_e MXM_5XIsUserAccessibleRegister(uint8_t regAddress, MXM_MODEL_ID_e model) {
260 FAS_ASSERT(model <= MXM_MODEL_ID_invalid);
261 /* AXIVION Routine Generic-MissingParameterAssert: regAddress: parameter accepts whole range */
262
263 STD_RETURN_TYPE_e retval = STD_NOT_OK;
264
265 switch (model) {
266 case MXM_MODEL_ID_MAX17852:
267 retval = MXM_52IsUserAccessibleRegister(regAddress);
268 break;
269 case MXM_MODEL_ID_MAX17853:
270 retval = MXM_53IsUserAccessibleRegister(regAddress);
271 break;
272 case MXM_MODEL_ID_MAX17854:
273 case MXM_MODEL_ID_NONE:
274 case MXM_MODEL_ID_invalid:
275 /* not implemented or invalid model id */
276 break;
277 default:
278 /* invalid state, should not happen */
279 FAS_ASSERT(FAS_TRAP);
280 break;
281 }
282 return retval;
283}

285static STD_RETURN_TYPE_e MXM_52IsUserAccessibleRegister(uint8_t regAddress) {
286 /* AXIVION Routine Generic-MissingParameterAssert: regAddress: parameter accepts whole range */
287
288 STD_RETURN_TYPE_e retval = STD_NOT_OK;
289 /* check if regAddress is outside user-accessible area */
290 /* AXIVION Disable Style Generic-NoMagicNumbers: memory limits of ICs are specific and unchangeable, therefore hard
291 * coded */
292 bool registerAddressIsInvalid = (regAddress == 0x5Du);
293 registerAddressIsInvalid = registerAddressIsInvalid || (regAddress == 0x5Eu);
294 registerAddressIsInvalid = registerAddressIsInvalid || (regAddress > 0x98u);
295 /* AXIVION Enable Style Generic-NoMagicNumbers: */
296
297 if (registerAddressIsInvalid == false) {
298 /* valid MAX17852 register address */
299 retval = STD_OK;
300 }
301 return retval;
302}

331static STD_RETURN_TYPE_e MXM_5XConstructCommandBufferWriteall(MXM_5X_INSTANCE_s *pInstance) {
332 FAS_ASSERT(pInstance != NULL_PTR);
333 STD_RETURN_TYPE_e retval = STD_NOT_OK;
334
335 const MXM_5X_COMMAND_PAYLOAD_s *const pPayload = &pInstance->commandPayload;
336 FAS_ASSERT(pPayload != NULL_PTR);
337
338 if (MXM_5XIsUserAccessibleRegister((uint8_t)pPayload->regAddress, pPayload->model) == STD_OK) {
339 /* clear command buffer */
340 MXM_5XClearCommandBuffer(pInstance);
341
342 /* construct command buffer */
343 pInstance->commandBuffer[0] = BATTERY_MANAGEMENT_WRITEALL;
344 pInstance->commandBuffer[1] = (uint8_t)pPayload->regAddress;
345 pInstance->commandBuffer[2] = pPayload->lsb;
346 pInstance->commandBuffer[3] = pPayload->msb;
347 /* PEC byte */
348 pInstance->commandBuffer[4] = MXM_CRC8(pInstance->commandBuffer, 4);
349 /* TODO alive-counter? */
350 pInstance->commandBufferCurrentLength = 5;
351 retval = STD_OK;
352 }
353
354 return retval;
355}

357static STD_RETURN_TYPE_e MXM_5XConstructCommandBufferWriteDevice(MXM_5X_INSTANCE_s *pInstance) {
358 FAS_ASSERT(pInstance != NULL_PTR);
359 STD_RETURN_TYPE_e retval = STD_NOT_OK;
360
361 const MXM_5X_COMMAND_PAYLOAD_s *const pPayload = &pInstance->commandPayload;
362 FAS_ASSERT(pPayload != NULL_PTR);
363
364 if (MXM_5XIsUserAccessibleRegister((uint8_t)pPayload->regAddress, pPayload->model) == STD_OK) {
365 /* clear command buffer */
366 MXM_5XClearCommandBuffer(pInstance);
367
368 /* construct command buffer */
369
370 /* commandBuffer[0] = Device address in a daisy chain + 0b100
371 * DA = deviceAddress
372 * Bit | 7 | 6 | 5 | 4 | 3 | 2 | 1 | 0
373 * Content | DA[4] | DA[3] | DA[2] | DA[1] | DA[0] | 1 | 0 | 0
374 */
375 pInstance->commandBuffer[0] =
376 ((((uint16_t)pPayload->deviceAddress << 3u) & MXM_5X_BIT_MASK_WRITE_DEVICE_ADDRESS) |
377 (uint16_t)BATTERY_MANAGEMENT_WRITEDEVICE);
378 pInstance->commandBuffer[1] = (uint8_t)pPayload->regAddress;
379 pInstance->commandBuffer[2] = pPayload->lsb;
380 pInstance->commandBuffer[3] = pPayload->msb;
381 /* PEC byte */
382 pInstance->commandBuffer[4] = MXM_CRC8(pInstance->commandBuffer, 4);
383 /* TODO alive-counter? */
384 pInstance->commandBufferCurrentLength = 5;
385 retval = STD_OK;
386 }
387
388 return retval;
389}

461static void MXM_5XStateHandler41BFmeaCheck(MXM_5X_INSTANCE_s *pInstance5x, MXM_41B_INSTANCE_s *pInstance41b) {
462 FAS_ASSERT(pInstance5x != NULL_PTR);
463 FAS_ASSERT(pInstance41b != NULL_PTR);
464 if (pInstance5x->substate == MXM_5X_ENTRY_SUBSTATE) {
465 /* entry of state --> set to first substate */
466 MXM_5XTransitionToSubstate(pInstance5x, MXM_5X_41B_FMEA_REQUEST);
467 }
468
469 if (pInstance5x->substate == MXM_5X_41B_FMEA_REQUEST) {
470 const STD_RETURN_TYPE_e stateRequestReturn = MXM_41BSetStateRequest(
471 pInstance41b, MXM_STATEMACH_41B_CHECK_FMEA, NULL_PTR, 0, 0, NULL_PTR, 0, &pInstance5x->status41b);
472 FAS_ASSERT(stateRequestReturn == STD_OK);
473 MXM_5XTransitionToSubstate(pInstance5x, MXM_5X_41B_FMEA_CHECK);
474 } else if (pInstance5x->substate == MXM_5X_41B_FMEA_CHECK) {
475 if (pInstance5x->status41b == MXM_41B_STATE_UNPROCESSED) {
476 /* wait for processing */
477 } else if (pInstance5x->status41b == MXM_41B_STATE_ERROR) {
478 /* failure in FMEA; signal error */
479 MXM_5XSignalError(pInstance5x);
480 } else if (pInstance5x->status41b == MXM_41B_STATE_PROCESSED) {
481 MXM_5XSignalSuccess(pInstance5x);
482 } else {
483 FAS_ASSERT(FAS_TRAP);
484 }
485 } else {
486 FAS_ASSERT(FAS_TRAP);
487 }
488}

490static void MXM_5XStateHandlerInit(MXM_5X_INSTANCE_s *pInstance5x, MXM_41B_INSTANCE_s *pInstance41b) {
491 FAS_ASSERT(pInstance5x != NULL_PTR);
492 FAS_ASSERT(pInstance41b != NULL_PTR);
493 if (pInstance5x->substate == MXM_5X_ENTRY_SUBSTATE) {
494 /* entry of state --> set to first substate */
495 MXM_5XTransitionToSubstate(pInstance5x, MXM_5X_INIT_41B_INIT);
496 }
497
498 if (pInstance5x->substate == MXM_5X_INIT_41B_INIT) {
499 if (pInstance5x->status41b == MXM_41B_STATE_UNSENT) {
500 const STD_RETURN_TYPE_e stateRequestReturn = MXM_41BSetStateRequest(
501 pInstance41b, MXM_STATEMACH_41B_INIT, NULL_PTR, 0, 0, NULL_PTR, 0, &pInstance5x->status41b);
502 FAS_ASSERT(stateRequestReturn == STD_OK);
503 } else if (pInstance5x->status41b == MXM_41B_STATE_UNPROCESSED) {
504 /* wait for processing */
505 } else if (pInstance5x->status41b == MXM_41B_STATE_ERROR) {
506 MXM_5XHandle41BErrorState(pInstance5x);
507 } else if (pInstance5x->status41b == MXM_41B_STATE_PROCESSED) {
508 MXM_5XTransitionToSubstate(pInstance5x, MXM_5X_INIT_41B_GET_VERSION);
509 pInstance5x->resetWaitTimestamp = OS_GetTickCount();
510 } else {
511 FAS_ASSERT(FAS_TRAP);
512 }
513 } else if (pInstance5x->substate == MXM_5X_INIT_41B_GET_VERSION) {
514 if (pInstance5x->status41b == MXM_41B_STATE_UNSENT) {
515 const STD_RETURN_TYPE_e stateRequestReturn = MXM_41BSetStateRequest(
516 pInstance41b, MXM_STATEMACH_41B_GET_VERSION, NULL_PTR, 0, 0, NULL_PTR, 0, &pInstance5x->status41b);
517 FAS_ASSERT(stateRequestReturn == STD_OK);
518 } else if (pInstance5x->status41b == MXM_41B_STATE_UNPROCESSED) {
519 /* wait for processing */
520 } else if (pInstance5x->status41b == MXM_41B_STATE_ERROR) {
521 MXM_5XHandle41BErrorState(pInstance5x);
522 } else if (pInstance5x->status41b == MXM_41B_STATE_PROCESSED) {
523 MXM_5XTransitionToSubstate(pInstance5x, MXM_5X_INIT_WAIT_FOR_RESET);
524 } else {
525 FAS_ASSERT(FAS_TRAP);
526 }
527 } else if (pInstance5x->substate == MXM_5X_INIT_WAIT_FOR_RESET) {
528 /* wait so that shutdown low of the satellites discharges and they switch off */
529 const bool shutdownTimeoutHasPassed =
530 OS_CheckTimeHasPassed(pInstance5x->resetWaitTimestamp, MXM_5X_SLAVE_SHUTDOWN_TIMEOUT_MS);
531 if (shutdownTimeoutHasPassed) {
