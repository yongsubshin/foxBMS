150/**
151 * @brief States of the Battery Management Protocol state-machine
152 */
153typedef enum {
154 MXM_STATEMACH_5X_UNINITIALIZED, /*!< Uninitialized state that the state-machine starts in */
155 MXM_STATEMACH_5X_INIT, /*!< Initialization state */
156 MXM_STATEMACH_5X_41B_FMEA_CHECK, /*!< Requests a FMEA-check from the 41B-state-machine */
157 MXM_STATEMACH_5X_IDLE, /*!< Idle state, transitions are possible in this state */
158 MXM_STATEMACH_5X_WRITEALL, /*!< WRITEALL state, sends a WRITEALL to the daisy-chain */
159 MXM_STATEMACH_5X_WRITE_DEVICE, /*!< WRITEDEVICE state, sends a WRITEDEVICE to a specific address */
160 MXM_STATEMACH_5X_READALL, /*!< READALL state, sends a READALL to the daisy-chain */
161 MXM_STATEMACH_5X_MAXSTATE, /*!< Highest state */
162} MXM_STATEMACHINE_5X_e;
163
164/**
165 * @brief Sub-states of the Battery Management Protocol state-machine
166 */
167typedef enum {
168 MXM_5X_INIT_41B_INIT, /*!< initialization of the MAX17841 bridge IC */
169 MXM_5X_INIT_41B_GET_VERSION, /*!< retrieves the version of the bridge IC */
170 /** waits for the slave devices to be reset (due to disabled preambles) */
171 MXM_5X_INIT_WAIT_FOR_RESET,
172 /** substate for the initialization, enable RX interrupt flags */
173 MXM_5X_INIT_ENABLE_RX_INTERRUPT_FLAGS,
174 /** substate for the wake-up routine, clear the receive buffer in the
175 * bridge IC */
176 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_CLEAR_RECEIVE_BUFFER,
177 /** substate for the wake-up routine, enable the sending of preambles in
178 * order to wake up the daisy-chain devices */
179 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_EN_PREAMBLES,
180 MXM_5X_INIT_ENABLE_KEEP_ALIVE, /*!< enables keep alive mode (which will take over after en preambles is off) */
181 /** substate for the wake-up routine, wait for the daisy-chain devices to
182 * return an answer, resulting in status RX busy of the bridge IC */
183 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_WAIT_FOR_RX_STATUS_BUSY,
184 /** substate for the wake-up routine, disable the sending of preambles */
185 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_DIS_PREAMBLES,
186 /** substate for the wake-up routine, wait for the RX transmission to
187 * finish */
188 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_WAIT_FOR_RX_STATUS_EMPTY,
189 /** substate for the wake-up routine, clear the transmit buffer */
190 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_CLEAR_TRANSMIT_BUFFER,
191 /** substate for the wake-up routine, clear the receive buffer */
192 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_CLEAR_RECEIVE_BUFFER_2,
193 /** substate for the wake-up routine, sending helloAll to the daisy-chain */
194 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_HELLO_ALL,
195 /** substate for the wake-up routine, verifying the result of helloAll */
196 MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_HELLO_ALL_VERIFY_MSG_AND_COUNT,
197 /** substate for requesting an FMEA check on the bridge IC */
198 MXM_5X_41B_FMEA_REQUEST,
199 /** substate for the verification of the FMEA check done in
200 * #MXM_5X_41B_FMEA_REQUEST */
201 MXM_5X_41B_FMEA_CHECK,
202 /** substate for the initiation of a uart transaction for the WRITEALL
203 * and WRITEDEVICE command */
204 MXM_5X_WRITE_UART_TRANSACTION,
205 /** substate for checking the received CRC in a WRITEALL/WRITEDEVICE transaction */
206 MXM_5X_WRITE_CHECK_CRC,
207 /** substate for the initiation of a uart transaction for the READALL
208 * command */
209 MXM_5X_READALL_UART_TRANSACTION,
210 /** substate for checking the received CRC in a READALL transaction */
211 MXM_5X_READALL_CHECK_CRC,
212 /** substate for retrieving the data check byte from a READALL transaction */
213 MXM_5X_READALL_GET_DC,
214 /** default value indicating no selected substate; has to be last substate in enum */
215 MXM_5X_ENTRY_SUBSTATE,
216} MXM_5X_SUBSTATES_e;

218/**
219 * @brief Request status of Battery Management Protocol states.
220 * @details This enum describes the states that a state in the
221 * #MXM_5XStateMachine() can have.
222 *
223 * Alternating bit patterns are used for better distinction
224 * and to make sure that there is no accidental overlap.
225 */
226typedef enum {
227 MXM_5X_STATE_UNSENT = 0x56, /*!< The request has not been sent to the state-machine yet. */
228 MXM_5X_STATE_UNPROCESSED =
229 0x67, /*!< The request has been received by the state-machine, but not been processed yet. */
230 MXM_5X_STATE_PROCESSED = 0x9A, /*!< The request has been process successfully */
231 MXM_5X_STATE_ERROR = 0xAB, /*!< An error has occurred during processing of the request. */
232} MXM_5X_STATE_REQUEST_STATUS_e;

261 * #BS_NR_OF_MODULES_PER_STRING.
262 */
263 uint8_t numberOfSatellites;
264 /**
265 * @brief Number of monitoring ICs matches the expected number.
266 * @details This status variable indicates whether the number of found
267 * monitoring ICs in #MXM_5X_INSTANCE_s::numberOfSatellites
268 * matches with the configured number of monitoring ICs in
269 * #BS_NR_OF_MODULES_PER_STRING. This variable is updated during execution of
270 * the substate
271 * #MXM_5X_INIT_WAKE_UP_SATELLITE_DEVICES_HELLO_ALL_VERIFY_MSG_AND_COUNT
272 * in the state #MXM_STATEMACH_5X_INIT in #MXM_5XStateMachine().
273 *
274 * In case the numbers are the same, the variable will read #STD_OK,
275 * otherwise it will be set to #STD_NOT_OK.
276 */

283 uint32_t resetWaitTimestamp; /*!< timestamp of the moment when waiting for the devices to be reset has started */
284 /**

287 */
288 uint8_t commandBufferCurrentLength;
289 /**
290 * @brief Command Buffer

297 * #MXM_5X_INSTANCE_s::commandBufferCurrentLength.
298 */
299 uint16_t commandBuffer[COMMAND_BUFFER_LENGTH];
300 uint16_t rxBuffer[MXM_5X_RX_BUFFER_LEN]; /*!< array containing the buffer for received data */
301} MXM_5X_INSTANCE_s;
