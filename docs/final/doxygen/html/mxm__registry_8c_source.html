81extern void MXM_MonRegistryInit(MXM_MONITORING_INSTANCE_s *pState) {
82 FAS_ASSERT(pState != NULL_PTR);
83 for (uint8_t i = 0; i < MXM_MAXIMUM_NR_OF_MODULES; i++) {
84 MXM_REGISTRY_ENTRY_s *entry = &pState->registry[i];
85 entry->connected = false;
86 entry->deviceAddress = 0u;
87 entry->deviceID = 0u;
88 entry->model = MXM_MODEL_ID_NONE;
89 entry->siliconVersion = MXM_SILICON_VERSION_0;
90 entry->registerStatus1 = MXM_MON_REGISTRY_STATUS_FMEA_DEFAULT;
91 entry->registerStatus2 = MXM_MON_REGISTRY_STATUS_FMEA_DEFAULT;
92 entry->registerStatus3 = MXM_MON_REGISTRY_STATUS_FMEA_DEFAULT;
93 entry->registerFmea1 = MXM_MON_REGISTRY_STATUS_FMEA_DEFAULT;
94 entry->registerFmea2 = MXM_MON_REGISTRY_STATUS_FMEA_DEFAULT;
95 }
96}

98extern STD_RETURN_TYPE_e MXM_MonRegistryConnectDevices(MXM_MONITORING_INSTANCE_s *pState, uint8_t numberOfDevices) {
99 FAS_ASSERT(pState != NULL_PTR);
100 /* AXIVION Routine Generic-MissingParameterAssert: numberOfDevices: parameter accepts whole range */
101
102 STD_RETURN_TYPE_e retval = STD_OK;
103 if (numberOfDevices > MXM_MAXIMUM_NR_OF_MODULES) {
104 retval = STD_NOT_OK;
105 } else {
106 for (uint8_t i = 0; i < numberOfDevices; i++) {
107 pState->registry[i].connected = true;
108 pState->registry[i].deviceAddress = HELLO_ALL_START_SEED + i;
109 pState->highest5xDevice = HELLO_ALL_START_SEED + i;
110 }
111 }
112
113 return retval;
114}

123extern void MXM_MonRegistryParseIdIntoDevices(
124 MXM_MONITORING_INSTANCE_s *pState,
125 uint8_t rxBufferLength,
126 MXM_REG_NAME_e type) {
127 FAS_ASSERT(pState != NULL_PTR);
128 FAS_ASSERT(rxBufferLength <= MXM_RX_BUFFER_LENGTH);
129 /* only ID1 or ID2 are valid */
130 FAS_ASSERT((type == MXM_REG_ID1) || (type == MXM_REG_ID2));
131
132 /* find highest connected device */
133 const uint8_t highestConnectedDevice =
134 MATH_MinimumOfTwoUint8_t(MXM_MonRegistryGetHighestConnected5XDevice(pState), MXM_MAXIMUM_NR_OF_MODULES);
135
136 const uint8_t startBytes = 2u;
137
138 for (uint8_t i = 0; i <= highestConnectedDevice; i++) {
139 const uint8_t bufferPosition = startBytes + (i * 2u);
140 MXM_REGISTRY_ENTRY_s *currentDevice = &pState->registry[highestConnectedDevice - i];
141 FAS_ASSERT((bufferPosition + 1u) <= rxBufferLength);
142 uint16_t id = 0;
143 MXM_ExtractValueFromRegister(
144 pState->rxBuffer[bufferPosition], pState->rxBuffer[bufferPosition + 1u], MXM_BM_WHOLE_REG, &id);
145 if (type == MXM_REG_ID1) {
146 currentDevice->deviceID = id;
147 } else {
148 /* intended condition: type is MXM_REG_ID2 */
149 currentDevice->deviceID = ((uint32_t)id << 16u) | currentDevice->deviceID;
150 }
151 }
152}

154extern void MXM_MonRegistryParseVersionIntoDevices(MXM_MONITORING_INSTANCE_s *pState, uint8_t rxBufferLength) {
155 FAS_ASSERT(pState != NULL_PTR);
156 FAS_ASSERT(pState->rxBuffer != NULL_PTR);
157 FAS_ASSERT(rxBufferLength <= MXM_RX_BUFFER_LENGTH);
158
159 /* find highest connected device */
160 const uint8_t highestConnectedDevice =
161 MATH_MinimumOfTwoUint8_t(MXM_MonRegistryGetHighestConnected5XDevice(pState), MXM_MAXIMUM_NR_OF_MODULES);
162
163 for (uint8_t i = 0; i <= highestConnectedDevice; i++) {
164 const uint8_t startBytes = 2u;
165 const uint8_t bufferPosition = startBytes + (i * 2u);
166 MXM_REGISTRY_ENTRY_s *currentDevice = &pState->registry[highestConnectedDevice - i];
167 FAS_ASSERT((bufferPosition + 1u) <= rxBufferLength);
168 uint16_t model = 0;
169 MXM_ExtractValueFromRegister(
170 pState->rxBuffer[bufferPosition], pState->rxBuffer[bufferPosition + 1u], MXM_REG_VERSION_MOD, &model);
171 if (model >= (uint16_t)MXM_MODEL_ID_invalid) {
172 currentDevice->model = MXM_MODEL_ID_invalid;
173 } else if (
174 (model != (uint16_t)MXM_MODEL_ID_MAX17852) && (model != (uint16_t)MXM_MODEL_ID_MAX17853) &&
175 (model != (uint16_t)MXM_MODEL_ID_MAX17854)) {
176 currentDevice->model = MXM_MODEL_ID_invalid;
177 } else {
178 /* AXIVION Next Codeline Style MisraC2012-10.5: All invalid values have been cleared. */
179 currentDevice->model = (MXM_MODEL_ID_e)model;
180 }
181
182 uint16_t version = 0;
183 MXM_ExtractValueFromRegister(
184 pState->rxBuffer[bufferPosition], pState->rxBuffer[bufferPosition + 1u], MXM_REG_VERSION_VER, &version);
185 if (version >= (uint16_t)MXM_SILICON_VERSION_INVALID) {
186 currentDevice->siliconVersion = MXM_SILICON_VERSION_INVALID;
187 } else {
188 /* AXIVION Next Codeline Style MisraC2012-10.5: All invalid values have been cleared. */
189 currentDevice->siliconVersion = (MXM_SILICON_VERSION_e)version;
190 }
191 }

194extern void MXM_MonRegistryParseStatusFmeaIntoDevices(MXM_MONITORING_INSTANCE_s *pState, uint8_t rxBufferLength) {
195 FAS_ASSERT(pState != NULL_PTR);
196 FAS_ASSERT(pState->rxBuffer != NULL_PTR);
197 FAS_ASSERT(rxBufferLength <= MXM_RX_BUFFER_LENGTH);
198
199 /* find highest connected device */
200 const uint8_t highestConnectedDevice =
201 MATH_MinimumOfTwoUint8_t(MXM_MonRegistryGetHighestConnected5XDevice(pState), MXM_MAXIMUM_NR_OF_MODULES);
202
203 /* detect which register has been read */
204 const uint8_t currentRegister = pState->rxBuffer[1u];
205
206 for (uint8_t i = 0; i <= highestConnectedDevice; i++) {
207 const uint8_t startBytes = 2u;
208 const uint8_t bufferPosition = startBytes + (i * 2u);
209 MXM_REGISTRY_ENTRY_s *currentDevice = &pState->registry[highestConnectedDevice - i];
210 FAS_ASSERT((bufferPosition + 1u) <= rxBufferLength);
211 uint16_t registerValue = 0u;
212 MXM_ExtractValueFromRegister(
213 pState->rxBuffer[bufferPosition], pState->rxBuffer[bufferPosition + 1u], MXM_BM_WHOLE_REG, &registerValue);
214
215 switch (currentRegister) {
216 case (uint8_t)MXM_REG_STATUS1:
217 currentDevice->registerStatus1 = registerValue;
218 break;
219 case (uint8_t)MXM_REG_STATUS2:
220 currentDevice->registerStatus2 = registerValue;
221 break;
222 case (uint8_t)MXM_REG_STATUS3:
223 currentDevice->registerStatus3 = registerValue;
224 break;
225 case (uint8_t)MXM_REG_FMEA1:
226 currentDevice->registerFmea1 = registerValue;
227 break;
228 case (uint8_t)MXM_REG_FMEA2:
229 currentDevice->registerFmea2 = registerValue;
230 break;
231 case (uint8_t)MXM_REG_BALSTAT:
232 currentDevice->registerBalstat = registerValue;
233 break;
234 default:
235 /* unknown register, just discard */
236 break;
237 }
238 }

241extern bool MXM_CheckIfADeviceHasBeenReset(const MXM_MONITORING_INSTANCE_s *const kpkState) {
242 FAS_ASSERT(kpkState != NULL_PTR);
243
244 /* find highest connected device */
245 const uint8_t highestConnectedDevice =
246 MATH_MinimumOfTwoUint8_t(MXM_MonRegistryGetHighestConnected5XDevice(kpkState), MXM_MAXIMUM_NR_OF_MODULES);
247
248 bool deviceReset = false;
249
250 for (uint8_t i = 0; i <= highestConnectedDevice; i++) {
251 const uint16_t status1 = kpkState->registry[i].registerStatus1;
252 const uint8_t lsb = (uint8_t)(status1 & MXM_BM_LSB);
253 const uint8_t msb = (uint8_t)((status1 & MXM_BM_MSB) >> MXM_REGISTRY_SHIFT_ONE_BYTE);
254 uint16_t alertReset = 0u;
255 MXM_ExtractValueFromRegister(lsb, msb, MXM_REG_STATUS1_ALRTRST, &alertReset);
256 if (alertReset == 1u) {
257 deviceReset = true;
258 }
259 }
260
261 return deviceReset;
262}
