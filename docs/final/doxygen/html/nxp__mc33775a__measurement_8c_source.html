107static void N775_InitMeasurement(N77X_STATE_s *pState) {
108 FAS_ASSERT(pState != NULL_PTR);
109
110 /* Send capture command. This ends the last cycle and starts a new one */
111 N77x_CommunicationWrite(
112 N77X_BROADCAST_ADDRESS,
113 MC3377X_ALLM_APP_CTRL_OFFSET,
114 (1u << MC3377X_ALLM_APP_CTRL_PAUSEBAL_POS) | (0x3FFu << MC3377X_PRMM_APP_CTRL_CAPVC_POS) |
115 (MC3377X_PRMM_APP_CTRL_VCOLNUM_DISABLED_ENUM_VAL << MC3377X_PRMM_APP_CTRL_VCOLNUM_POS),
116 pState->pSpiTxSequence);
117 /* Wait for measurements to take place */
118 N77x_Wait(N77X_MEASUREMENT_CAPTURE_TIME_MS);
119 /* Send capture command. This ends the last cycle and starts a new one */
120 N77x_CommunicationWrite(
121 N77X_BROADCAST_ADDRESS,
122 MC3377X_ALLM_APP_CTRL_OFFSET,
123 (0u << MC3377X_ALLM_APP_CTRL_PAUSEBAL_POS) | (0x3FFu << MC3377X_PRMM_APP_CTRL_CAPVC_POS) |
124 (MC3377X_PRMM_APP_CTRL_VCOLNUM_DISABLED_ENUM_VAL << MC3377X_PRMM_APP_CTRL_VCOLNUM_POS),
125 pState->pSpiTxSequence);
126 /* Wait for measurements to be ready */
127 N77x_Wait(N77X_MEASUREMENT_READY_TIME_MS);
128}

130static void N775_RetrieveVoltages(N77X_STATE_s *pState, uint8_t m, uint16_t *primaryRawValues, bool *gpio03Error) {
131 int16_t primaryValues[20] = {0u};
132
133 for (uint8_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
134 /* Store cell voltages */
135 if (N77X_INVALID_REGISTER_VALUE != primaryRawValues[cb + 1u]) {
136 primaryValues[cb + 1u] = (int16_t)primaryRawValues[cb + 1u];
137 pState->n77xData.cellVoltage->cellVoltage_mV[pState->currentString][m][cb] =
138 (((float_t)primaryValues[cb + 1u]) * 154.0e-6f * UNIT_CONVERSION_FACTOR_1000_FLOAT);
139 /* String voltage measurement is calculated as sum of individual cell voltage measurement */
140 pState->n77xData.cellVoltage->stringVoltage_mV[pState->currentString] +=
141 pState->n77xData.cellVoltage->cellVoltage_mV[pState->currentString][m][cb];
142 pState->n77xData.cellVoltage->invalidCellVoltage[pState->currentString][m][cb] = false;
143 pState->n77xData.cellVoltage->nrValidCellVoltages[pState->currentString]++;
144 } else {
145 pState->n77xData.cellVoltage->cellVoltage_mV[pState->currentString][m][cb] =
146 AFE_DEFAULT_CELL_VOLTAGE_INVALID_VALUE;
147 pState->n77xData.cellVoltage->invalidCellVoltage[pState->currentString][m][cb] = true;
148 }
149 }
150 for (uint8_t g = 0u; g < 4u; g++) {
151 /* Store GPIO voltages */
152 if (N77X_INVALID_REGISTER_VALUE != primaryRawValues[g + 16u]) {
153 primaryValues[g + 16u] = (int16_t)primaryRawValues[g + 16u];
154 pState->n77xData.allGpioVoltage
155 ->gpioVoltages_mV[pState->currentString][g + (m * SLV_NR_OF_GPIOS_PER_MODULE)] =
156 (((float_t)primaryValues[g + 16u]) * 154.0e-6f * 1000.0f);
157 } else {
158 *gpio03Error = true;
159 }
160 }
161 /* Store module voltage */
162 if (N77X_INVALID_REGISTER_VALUE != primaryRawValues[15u]) {
163 primaryValues[15u] = (int16_t)primaryRawValues[15u];
164 pState->n77xData.cellVoltage->moduleVoltage_mV[pState->currentString][m] =
165 (((float_t)primaryValues[15u]) * 2.58e-3f * 1000.0f);
166 }
167}

169static void N775_RetrieveTemperatures(N77X_STATE_s *pState, uint8_t m, bool *gpio03Error, bool *gpio47Error) {
170 /* Set temperature values */
171 if (N77X_USE_MUX_FOR_TEMP == true) {
172 /* Mux case */
173 if (*gpio03Error == false) {
174 pState->n77xData.cellTemperature
175 ->cellTemperature_ddegC[pState->currentString][m][pState->currentMux[pState->currentString]] =
176 N77x_ConvertVoltagesToTemperatures(
177 pState->n77xData.allGpioVoltage
178 ->gpioVoltages_mV[pState->currentString]
179 [N77X_MULTIPLEXER_TEMP_GPIO_POSITION + (m * SLV_NR_OF_GPIOS_PER_MODULE)]);
180 pState->n77xData.cellTemperature
181 ->invalidCellTemperature[pState->currentString][m][pState->currentMux[pState->currentString]] = false;
182 } else {
183 pState->n77xData.cellTemperature
184 ->cellTemperature_ddegC[pState->currentString][m][pState->currentMux[pState->currentString]] = 0;
185 pState->n77xData.cellTemperature
186 ->invalidCellTemperature[pState->currentString][m][pState->currentMux[pState->currentString]] = true;
187 }
188 } else if (N77X_USE_MUX_FOR_TEMP == false) {
189 /* No mux case */
190 if ((*gpio03Error == false) && (*gpio47Error == false)) {
191 for (uint8_t ts = 0u; ts < BS_NR_OF_TEMP_SENSORS_PER_MODULE; ts++) {
192 pState->n77xData.cellTemperature->cellTemperature_ddegC[pState->currentString][m][ts] =
193 N77x_ConvertVoltagesToTemperatures(
194 pState->n77xData.allGpioVoltage
195 ->gpioVoltages_mV[pState->currentString][ts + (m * SLV_NR_OF_GPIOS_PER_MODULE)]);
196 pState->n77xData.cellTemperature->invalidCellTemperature[pState->currentString][m][ts] = false;
197 }
198 } else {
199 for (uint8_t ts = 0u; ts < BS_NR_OF_TEMP_SENSORS_PER_MODULE; ts++) {
200 pState->n77xData.cellTemperature->cellTemperature_ddegC[pState->currentString][m][ts] = 0;
201 pState->n77xData.cellTemperature->invalidCellTemperature[pState->currentString][m][ts] = true;
202 }
203 }
204 } else {
205 /* Invalid value for switch case */
206 FAS_ASSERT(FAS_TRAP);
207 }
208}

210static void N775_RetrieveMeasurement(N77X_STATE_s *pState) {
211 FAS_ASSERT(pState != NULL_PTR);
212
213 uint16_t primaryRawValues[20] = {0u};
214 uint16_t secondaryRawValues[4] = {0u};
215 uint16_t currentRawValue = 0u;
216 int16_t secondaryValues[4] = {0u};
217 int16_t currentValue = 0u;
218 N77X_COMMUNICATION_STATUS_e retValPrimary = N77X_COMMUNICATION_OK;
219 N77X_COMMUNICATION_STATUS_e retValSecondary = N77X_COMMUNICATION_OK;
220 N77X_COMMUNICATION_STATUS_e retValSupplyCurrent = N77X_COMMUNICATION_OK;
221 bool gpio03Error = false;
222 bool gpio47Error = false;
223
224 /* Reset number previous measurement results on string-level */
225 pState->n77xData.cellVoltage->stringVoltage_mV[pState->currentString] = 0u;
226 pState->n77xData.cellVoltage->nrValidCellVoltages[pState->currentString] = 0u;
227 pState->n77xData.cellTemperature->nrValidTemperatures[pState->currentString] = 0u;
228
229 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
230 uint8_t deviceAddress = m + 1u;
231 retValPrimary = N77x_CommunicationReadMultiple(
232 deviceAddress, 20, 4u, MC3377X_PRMM_APP_VC_CNT_OFFSET, primaryRawValues, pState);
233 retValSecondary = N77x_CommunicationReadMultiple(
234 deviceAddress, 4, 4u, MC3377X_SECM_APP_AIN4_OFFSET, secondaryRawValues, pState);
235 if (N77X_CHECK_SUPPLY_CURRENT == true) {
236 retValSupplyCurrent =
237 N77x_CommunicationRead(deviceAddress, MC3377X_SECM_PER_NPNISENSE_OFFSET, &currentRawValue, pState);
238 }
239
240 N77x_ErrorHandling(pState, retValPrimary, m);
241 if (retValPrimary == N77X_COMMUNICATION_OK) {
242 N775_RetrieveVoltages(pState, m, primaryRawValues, &gpio03Error);
243 } else {
244 /* Reset voltage values and the relevant invalid flag if the communication is not ok */
245 for (uint8_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
246 pState->n77xData.cellVoltage->cellVoltage_mV[pState->currentString][m][cb] = 0;
247 pState->n77xData.cellVoltage->invalidCellVoltage[pState->currentString][m][cb] = true;
248 }
249 }
250
251 N77x_ErrorHandling(pState, retValSecondary, m);
252 if (retValSecondary == N77X_COMMUNICATION_OK) {
253 for (uint8_t g = 4u; g < 8u; g++) {
254 if (N77X_INVALID_REGISTER_VALUE != secondaryRawValues[g - 4u]) {
255 secondaryValues[g - 4u] = (int16_t)secondaryRawValues[g - 4u];
256 pState->n77xData.allGpioVoltage
257 ->gpioVoltages_mV[pState->currentString][g + (m * SLV_NR_OF_GPIOS_PER_MODULE)] =
258 (((float_t)secondaryValues[g - 4u]) * 154.0e-6f * 1000.0f);
259 } else {
260 gpio47Error = true;
261 }
262 }
263 }
264
265 N77x_ErrorHandling(pState, retValPrimary, m);
266 if (retValPrimary == N77X_COMMUNICATION_OK) {
267 N775_RetrieveTemperatures(pState, m, &gpio03Error, &gpio47Error);
268 } else {
269 /* Reset temperature values the relevant invalid flag if the communication is not ok */
270 for (uint8_t ts = 0u; ts < BS_NR_OF_TEMP_SENSORS_PER_MODULE; ts++) {
271 pState->n77xData.cellTemperature->cellTemperature_ddegC[pState->currentString][m][ts] = 0;
272 pState->n77xData.cellTemperature->invalidCellTemperature[pState->currentString][m][ts] = true;
273 }
274 }
275
276 if (N77X_CHECK_SUPPLY_CURRENT == true) {
277 N77x_ErrorHandling(pState, retValSupplyCurrent, m);
278 if (retValSupplyCurrent == N77X_COMMUNICATION_OK) {
279 if (N77X_INVALID_REGISTER_VALUE != currentRawValue) {
280 currentValue = (int16_t)currentRawValue;
281 pState->n77xData.supplyCurrent->current[pState->currentString][m] =
282 (((float_t)currentValue) * 7.69e-6f * 1000.0f);
283 }
284 }
285 }
286 }
287
288 (void)DATA_WRITE_DATA(pState->n77xData.cellVoltage, pState->n77xData.cellTemperature, pState->n77xData.allGpioVoltage);
289}

299extern void N77x_StartMeasurement(N77X_STATE_s *pState) {
300 FAS_ASSERT(pState != NULL_PTR);
301
302 /* Enable cell voltage measurements */
303 N77x_CommunicationWrite(N77X_BROADCAST_ADDRESS, MC3377X_ALLM_VCVB_CFG_OFFSET, 0x3FFF, pState->pSpiTxSequence);
304 N77x_Wait(N77X_T_WAIT_CYC_SOC_MS);
305 /* Enable analog inputs 0-3 and module voltage measurement */
306 N77x_CommunicationWrite(N77X_BROADCAST_ADDRESS, MC3377X_PRMM_AIN_CFG_OFFSET, 0x1F, pState->pSpiTxSequence);
307 N77x_Wait(N77X_T_WAIT_CYC_SOC_MS);
308 /* Enable analog inputs 4-7 measurement */
309 N77x_CommunicationWrite(N77X_BROADCAST_ADDRESS, MC3377X_SECM_AIN_CFG_OFFSET, 0x0F, pState->pSpiTxSequence);
310 N77x_Wait(N77X_T_WAIT_CYC_SOC_MS);
311 /* Set pause of balancing before measurement start, enable the measurement units simultaneously */
312 N77x_CommunicationWrite(
313 N77X_BROADCAST_ADDRESS,
314 MC3377X_ALLM_CFG_OFFSET,
315 (N77X_BALPAUSELEN_10US << MC3377X_ALLM_CFG_BALPAUSELEN_POS) | (1 << MC3377X_ALLM_CFG_MEASEN_POS),
316 pState->pSpiTxSequence);
317 N77x_Wait(N77X_T_WAIT_CYC_SOC_MS);
318}
