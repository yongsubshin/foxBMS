59/** Number of words (16 bits) used for write messages */

62#define N77X_READ_HEADER_SPI_BUFFER_SIZE (3u)

64 * Number of words (16 bits) in read answers that contain data.
65 * Based on maximum payload, which is 4 register per answer frame.

68/**
69 * Maximum number of groups of registers that can be read,
70 * to limit buffer size.
71 * Groups can be of size 1 to 4 registers.
72 * One frame = header (e.g., register address) +
73 * payload (data, 1 to 4 registers)
74 */
75#define N77X_MAX_ANSWER_FRAMES (30u)

82/**
83 * Timeout to wait in microseconds for reception of a read answer.
84 * After that time the communication is considered to have failed.

131static void N77x_ConvertMessageToBuffer(uint16_t *pBuffer, uc_msg_t message) {
132 FAS_ASSERT(pBuffer != NULL_PTR);
133 pBuffer[0u] = message.head;
134 pBuffer[1u] = message.data.dhead;
135 pBuffer[2u] = message.data.data[0u];
136 pBuffer[3u] = message.crc;
137}

139static uint32_t N77x_WaitForTxCompletedNotification(void) {
140 uint32_t notifiedValueTx = N77X_NO_NOTIFIED_VALUE;
141 /**
142 * Suspend task and wait for SPI send DMA RX finished notification,
143 * clear notification value on entry and exit
144 */
145 OS_WaitForNotificationIndexed(N77X_NOTIFICATION_TX_INDEX, &notifiedValueTx, N77X_NOTIFICATION_TX_TIMEOUT_ms);
146 return notifiedValueTx;
147}

161extern void N77x_CommunicationWrite(
162 uint16_t deviceAddress,
163 uint16_t registerAddress,
164 uint16_t value,
165 SPI_INTERFACE_CONFIG_s *pSpiInterface) {
166 FAS_ASSERT(pSpiInterface != NULL_PTR);
167 uc_msg_t message = {0}; /* Message for read command */
168
169 /** The function gets the device address
170 * The N77x_CommunicationComposeMessage function adds the chain address.
171 * Chain address = 1 used so (deviceAddress | (1u << 6u)) is used
172 */
173 N77x_CommunicationComposeMessage(
174 BMS1_CMD_WRITE, 0, (deviceAddress | (1u << 6u)), registerAddress, 0, &value, &message);
175 N77x_ConvertMessageToBuffer(N77xToTplTxBuffer, message);
176
177 /*
178 * Used to clear a pending Tx notification made by the read function
179 * when writing to registers: when reading, commands are sent on SPI1.
180 * In the SPI1 ISR, a notification is made, but it is not used: the system
181 * waits for the SPI Rx notification from SPI4. As a consequence, when the
182 * write function is called after the read function, there is already a
183 * notification pending. The function waiting for the notification in the
184 * ISR of SPI1 exits immediately instead of waiting for the end of the
185 * transmission.
186 */
187 (void)OS_ClearNotificationIndexed(N77X_NOTIFICATION_TX_INDEX);
188
189 (void)SPI_TransmitReceiveDataDma(pSpiInterface, N77xToTplTxBuffer, N77xToTplRxBuffer, 4u);
190
191 uint32_t notificationTx = N77x_WaitForTxCompletedNotification();
192 if (notificationTx != N77X_TX_NOTIFIED_VALUE) {
193 /* Tx DMA interrupt has not come, release Tx SPI interface */
194 const uint8_t spiIndex = SPI_GetSpiIndex(spiREG1);
195 OS_EnterTaskCritical();
196 spi_busyFlags[spiIndex] = SPI_IDLE;
197 OS_ExitTaskCritical();
198 }
199}

213extern N77X_COMMUNICATION_STATUS_e N77x_CommunicationReadMultiple(
214 uint16_t deviceAddress,
215 uint16_t numberOfItems,
216 uint16_t responseLength,
217 uint16_t registerAddress,
218 uint16_t *pValues,
219 N77X_STATE_s *pState) {
220 FAS_ASSERT(pValues != NULL_PTR);
221 FAS_ASSERT(pState != NULL_PTR);
222 FAS_ASSERT(pState->pSpiTxSequence != NULL_PTR);
223 FAS_ASSERT(pState->pSpiRxSequence != NULL_PTR);
224
225 /* Number of registers to read */
226 uint16_t itemsReadRemaining = numberOfItems;
227 uint16_t *pReadValues = pValues;
228
229 /**
230 * numberOfItems = 0 --> corresponds to one frame
231 * (max_frames - 1): because first frame is for the mirroring of Tx
232 * So (numberOfItems - 1u) is used in the following
233 */
234
235 /** The function gets the device address
236 * The pack_msg function adds the chain address.
237 * Chain address = 1 used so (deviceAddress | (1u << 6u)) is used in the following
238 * */
239 /* responseLength = 0 means one data word per answer frame so responseIndex is used in the following */
240
241 uc_msg_t txMessage = {0u}; /* Message for read command */
242 uc_msg_t rxMessage = {0u}; /* Message for response */
243
244 N77X_COMMUNICATION_STATUS_e communicationStatus; /* Reception return code */
245 uint16_t read_parameter; /* Read parameter (padding_en, responseLength, num_regs) */
246
247 uint16_t rsp_cmd; /* Response parts */
248 uint16_t rsp_mst_addr;
249 uint16_t rsp_dev_addr;
250 uint16_t rsp_reg_addr;
251 uint16_t rsp_length;
252 uint16_t rsp_values[4];
253
254 uint16_t responseIndex = responseLength - 1u;
255
256 read_parameter = ((uint16_t)1u << 10u) + /* Use padding */
257 (responseIndex << 8u) + /* responseLength */
258 ((numberOfItems - 1u) << 0u); /* (numberOfItems-1u) register to read */
259
260 N77x_CommunicationComposeMessage(
261 BMS1_CMD_READ, 0, (deviceAddress | (1u << 6u)), registerAddress, 0, &read_parameter, &txMessage);
262 N77x_ConvertMessageToBuffer(N77xToTplTxBuffer, txMessage);
263
264 /**
265 * After transmission to daisy-chain, daisy-chain will normally answer
266 * Already prepare SPI slave for reception
267 */
268
269 /**
270 * Compute the number of answer frames needed by the MC3377xA.
271 * Padding is used: always same number of registers in each answer frame
272 * (1,2,3 or 4 registers).
273 */
274 uint16_t nrAnswerFrames = (((numberOfItems - 1u) + 1u) / (responseIndex + 1u));
275 /**
276 * Additional answer frame if number of registers to read and
277 * number of registers per answer frame are not multiples
278 * Example: 22 registers to read, 4 registers per answer frame.
279 * Answer: (5*4 register) + (2 registers + 2 words padded with 0)
280 */
281 if ((((numberOfItems - 1u) + 1u) % (responseIndex + 1u)) != 0u) {
282 nrAnswerFrames++;
283 }
284 /**
285 * N77X_WRITE_SPI_BUFFER_SIZE (4u): because of Tx mirroring, the command is received, too
286 * N77X_READ_HEADER_SPI_BUFFER_SIZE + responseLength: size of one answer frame */
287
288 uint16_t rxBufferLength = N77X_WRITE_SPI_BUFFER_SIZE +
289 ((N77X_READ_HEADER_SPI_BUFFER_SIZE + responseLength) * nrAnswerFrames);
290
291 FAS_ASSERT(
292 rxBufferLength <=
293 (N77X_WRITE_SPI_BUFFER_SIZE +
294 ((N77X_READ_HEADER_SPI_BUFFER_SIZE + N77X_READ_PAYLOAD_SPI_BUFFER_SIZE) * N77X_MAX_ANSWER_FRAMES)));
295 SPI_SlaveSetReceiveDataDma(pState->pSpiRxSequence, N77xFromTplTxBuffer, N77xFromTplRxBuffer, rxBufferLength);
296
297 /* send message */
298 (void)SPI_TransmitReceiveDataDma(pState->pSpiTxSequence, N77xToTplTxBuffer, N77xToTplRxBuffer, N77X_RX_FRAME_LENGTH);
299 bool N77x_rxCompleted = true;
300
301 uint32_t notificationRx = N77x_WaitForRxCompletedNotification();
302 if (notificationRx != N77X_RX_NOTIFIED_VALUE) {
303 N77x_rxCompleted = false;
304 /* Rx has not come, release Tx SPI interface */
305 const uint8_t spiIndex = SPI_GetSpiIndex(spiREG1);
306 OS_EnterTaskCritical();
307 spi_busyFlags[spiIndex] = SPI_IDLE;
308 OS_ExitTaskCritical();
309 }
310
311 if (N77x_rxCompleted == false) {
312 pState->pSpiRxSequence->pNode->INT0 &= ~DMA_REQUEST_ENABLE_BIT;
313 pState->pSpiRxSequence->pNode->GCR1 &= ~DMA_SPI_ENABLE_BIT;
314 communicationStatus = N77X_COMMUNICATION_ERROR_TIMEOUT;
315 } else {
316 for (uint16_t i = 0; i < nrAnswerFrames; i++) {
317 /* Currently only support for 64 bit communication */
318 rxMessage.message_length = N77X_TX_MIRROR_OFFSET + (responseIndex);
319 /* +4u: because of Tx mirroring, the command is received, too */
320
321 rxMessage.head =
322 N77xFromTplRxBuffer[N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + responseIndex)) + 0u];
323 rxMessage.data.dhead =
324 N77xFromTplRxBuffer[N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + responseIndex)) + 1u];
325 rxMessage.data.data[0u] =
326 N77xFromTplRxBuffer[N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 2u];
327 if (rxMessage.message_length <= 4u) {
328 rxMessage.crc = N77xFromTplRxBuffer[N77X_TX_MIRROR_OFFSET + (i * N77X_TX_MIRROR_OFFSET) + 3u];
329 } else {
330 if (rxMessage.message_length == 5u) {
331 rxMessage.data.data[1u] = N77xFromTplRxBuffer
332 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 3u];
333 rxMessage.crc = N77xFromTplRxBuffer
334 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 4u];
335 } else {
336 if (rxMessage.message_length == 6u) {
337 rxMessage.data.data[1u] = N77xFromTplRxBuffer
338 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 3u];
339 rxMessage.data.data[2u] = N77xFromTplRxBuffer
340 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 4u];
341 rxMessage.crc = N77xFromTplRxBuffer
342 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 5u];
343 } else {
344 rxMessage.data.data[1u] = N77xFromTplRxBuffer
345 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 3u];
346 rxMessage.data.data[2u] = N77xFromTplRxBuffer
347 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 4u];
348 rxMessage.data.data[3u] = N77xFromTplRxBuffer
349 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 5u];
350 rxMessage.crc = N77xFromTplRxBuffer
351 [N77X_TX_MIRROR_OFFSET + (i * (N77X_TX_MIRROR_OFFSET + (responseIndex))) + 6u];
352 }
353 }
354 }
355
356 communicationStatus = N77x_CommunicationDecomposeMessage(
357 &rxMessage,
358 &rsp_cmd,
359 &rsp_mst_addr,
360 &rsp_dev_addr,
361 &rsp_reg_addr,
362 &rsp_length,
363 rsp_values,
364 pState->currentString);
365
366 if (communicationStatus == N77X_COMMUNICATION_OK) {
367 /* SM.e.30 : Communication - Unique ID */
368 if (rsp_dev_addr == (deviceAddress | (1u << 6u))) {
369 if (rsp_reg_addr == (registerAddress + (i * ((responseIndex) + 1u)))) {
370 for (uint16_t j = 0u; j < (rsp_length + 1u); j++) {
371 /* To take padded answer frames into account
372 * Stop before reaching the words padded with 0s
373 */
374 if (itemsReadRemaining > 0u) {
375 *pReadValues = rsp_values[j];
376 pReadValues++;
377 itemsReadRemaining--;
378 } else {
379 break;
380 }
381 }
382 } else {
383 communicationStatus = N77X_COMMUNICATION_ERROR_NOT_MATCHING_REGISTER_ADDRESS;
384 }
385 } else {
386 communicationStatus = N77X_COMMUNICATION_ERROR_NOT_MATCHING_DEVICE_ADDRESS;
387 }
388 }
389 }
390 }
391
392 return communicationStatus;
393}

402/* Pack message */
403extern void N77x_CommunicationComposeMessage(
404 uint16_t cmd,
405 uint16_t masterAddress,
406 uint16_t deviceAddress,
407 uint16_t registerAddress,
408 uint16_t length,
409 uint16_t *pValue,
410 uc_msg_t *pMessage) {
411 FAS_ASSERT(pValue != NULL_PTR);
412 FAS_ASSERT(pMessage != NULL_PTR);
413 /* Create message */
414 set_cmd(pMessage, cmd);
415 set_madd(pMessage, masterAddress);
416 set_cadd(pMessage, deviceAddress >> 6u);
417 set_devadd(pMessage, deviceAddress & 0x3Fu);
418 set_msgcnt(pMessage, 0); /* not used by MCU, therefore fixed 0 */
419 set_datalen(pMessage, length);
420 set_regadd(pMessage, registerAddress);
421 for (uint16_t i = 0; i <= length; i++) {
422 set_data(pMessage, pValue[i], i);
423 }
424 set_message_length(pMessage, length + 4u);
425
426 /* Create CRC */
427 set_crc(pMessage, calc_crc(pMessage));
428}

430/* Unpack a message */
431extern N77X_COMMUNICATION_STATUS_e N77x_CommunicationDecomposeMessage(
432 uc_msg_t *pMessage,
433 uint16_t *pCommand,
434 uint16_t *pMasterAddress,
435 uint16_t *pDeviceAddress,
436 uint16_t *pRegisterAddress,
437 uint16_t *pLength,
438 uint16_t *pValue,
439 uint8_t string) {
440 FAS_ASSERT(pMessage != NULL_PTR);
441 FAS_ASSERT(pCommand != NULL_PTR);
442 FAS_ASSERT(pMasterAddress != NULL_PTR);
443 FAS_ASSERT(pDeviceAddress != NULL_PTR);
444 FAS_ASSERT(pRegisterAddress != NULL_PTR);
445 FAS_ASSERT(pLength != NULL_PTR);
446 FAS_ASSERT(pValue != NULL_PTR);
447 uint16_t messageLength; /* length of received message */
448 uint16_t messageCount; /* message count from message */
449 uint16_t chainAddress; /* chain address */
450 uint16_t deviceAddressInChain; /* device address in the chain */
451
452 bool errorCodeMatch = false;
453 N77X_COMMUNICATION_STATUS_e communicationStatus = N77X_COMMUNICATION_OK;
454
455 /* check if we have any content */
456 /* SM.e.28 : Communication - Timeout monitoring */
457 get_message_length(pMessage, &messageLength);
458 if (messageLength == 0u) {
459 errorCodeMatch = true;
460 communicationStatus = N77X_COMMUNICATION_ERROR_TIMEOUT;
461 }
462
463 if ((errorCodeMatch == false) && (messageLength < 4u)) {
464 errorCodeMatch = true;
465 communicationStatus = N77X_COMMUNICATION_ERROR_SHORT_MESSAGE;
466 }
467
468 /* Check CRC */
469 /* SM.e.27 : Communication - Information redundancy */
470 if ((errorCodeMatch == false) && (!check_crc(pMessage, calc_crc(pMessage)))) {
471 errorCodeMatch = true;
472 uint16_t receivedCrc;
473 get_crc(pMessage, &receivedCrc);
474 communicationStatus = N77X_COMMUNICATION_ERROR_WRONG_CRC;
475 }
476
477 /* Extract message parts */
478 get_cmd(pMessage, pCommand);
479 get_madd(pMessage, pMasterAddress);
480 get_cadd(pMessage, &chainAddress);
481 get_devadd(pMessage, &deviceAddressInChain);
482 *pDeviceAddress = (chainAddress << 6u) | deviceAddressInChain;
483 get_msgcnt(pMessage, &messageCount);
484 get_datalen(pMessage, pLength);
485 get_regadd(pMessage, pRegisterAddress);
486
487 /* Check message counter */
488 /* SM.e.29 : Communication - Message counter */
489 if ((errorCodeMatch == false) && (messageCount != referenceMessageCounter[string][*pDeviceAddress])) {
490 errorCodeMatch = true;
491 referenceMessageCounter[string][*pDeviceAddress] = (messageCount + 1u) & 0x0Fu;
492 communicationStatus = N77X_COMMUNICATION_ERROR_WRONG_MESSAGE_COUNT;
493 }
494 /* Increment message counter */
495 referenceMessageCounter[string][*pDeviceAddress] = (referenceMessageCounter[string][*pDeviceAddress] + 1u) & 0x0Fu;
496
497 /* Check error address */
498 if ((errorCodeMatch == false) && (*pRegisterAddress == N77X_ERROR_REGISTER_ADDRESS)) {
499 communicationStatus = N77X_COMMUNICATION_ERROR_NO_ACCESS;
500 }
501
502 /* return data */
503 if (errorCodeMatch == false) {
504 for (uint16_t i = 0u; i <= *pLength; i++) {
505 get_data(pMessage, &pValue[i], i);
506 }
507 if (*pCommand != BMS1_CMD_RESP) { /* Check for response */
508 communicationStatus = N77X_COMMUNICATION_ERROR_NO_RESPONSE;
509 }
510 }
511
512 /* TODO: check for N77X_COMMUNICATION_ERROR_NOT_MATCHING_DEVICE_ADDRESS */
513 /* TODO: check for N77X_COMMUNICATION_ERROR_NOT_MATCHING_REGISTER_ADDRESS */
514
515 return communicationStatus;
516}
