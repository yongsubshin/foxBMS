157static STD_RETURN_TYPE_e N77x_Enumerate(N77X_STATE_s *pState) {
158 FAS_ASSERT(pState != NULL_PTR);
159 uint16_t readValue = 0u;
160 uint16_t uid[3u] = {0};
161 N77X_COMMUNICATION_STATUS_e returnedValue = N77X_COMMUNICATION_OK;
162 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
163
164 /** Parse all slaves in the daisy-chain */
165 for (uint8_t i = 1; i <= BS_NR_OF_MODULES_PER_STRING; i++) {
166 /* First send slave to deep sleep to reset message counter */
167 N77x_CommunicationWrite(
168 i,
169 MC3377X_SYS_MODE_OFFSET,
170 (MC3377X_SYS_MODE_TARGETMODE_DEEPSLEEP_ENUM_VAL << MC3377X_SYS_MODE_TARGETMODE_POS),
171 pState->pSpiTxSequence);
172 N77x_Wait(N77X_T_SW_ACT_DEEP_SLEEP_MS);
173
174 /* Wake up slave */
175 returnedValue = N77x_CommunicationRead(i, MC3377X_SYS_COM_CFG_OFFSET, &readValue, pState);
176 /* If slave is not enumerated */
177 if (returnedValue != N77X_COMMUNICATION_OK) {
178 /* Wait until the slave has woken up */
179 N77x_Wait(N77X_T_WAKE_COM_MS);
180
181 returnedValue = N77x_CommunicationRead(i, MC3377X_SYS_COM_CFG_OFFSET, &readValue, pState);
182 /* If slave is not enumerated */
183 if (returnedValue != N77X_COMMUNICATION_OK) {
184 /* Enumerate slave */
185 N77x_CommunicationWrite(
186 (0u << 6u) + 0u,
187 MC3377X_SYS_COM_CFG_OFFSET,
188 i + (N77X_DEFAULT_CHAIN_ADDRESS << 6) +
189 (MC3377X_SYS_COM_CFG_BUSFW_ENABLED_ENUM_VAL << MC3377X_SYS_COM_CFG_BUSFW_POS),
190 pState->pSpiTxSequence);
191 }
192
193 /* Reset the message counter of the driver */
194 N77x_ResetMessageCounter((N77X_DEFAULT_CHAIN_ADDRESS << 6) + i, pState->currentString);
195 /* Check that the device has been enumerated */
196 returnedValue = N77x_CommunicationRead(i, MC3377X_SYS_VERSION_OFFSET, &readValue, pState);
197 if (returnedValue == N77X_COMMUNICATION_OK) {
198 retVal = STD_OK;
199 }
200 } else {
201 /* Slave already has an address */
202 retVal = STD_OK;
203 }
204
205 /* Set timeout, enable/disable timeout */
206 N77x_CommunicationWrite(
207 i,
208 MC3377X_SYS_COM_TO_CFG_OFFSET,
209 (N77X_TIMEOUT_SWITCH << MC3377X_SYS_COM_TO_CFG_COMTODISABLE_POS) |
210 (N77X_TIMEOUT_TO_SLEEP_10MS << MC3377X_SYS_COM_TO_CFG_COMTO_POS),
211 pState->pSpiTxSequence);
212
213 /* read uid of each device */
214 returnedValue = N77x_CommunicationReadMultiple(i, 3u, 3u, MC3377X_SYS_UID_LOW_OFFSET, uid, pState);
215 if (returnedValue == N77X_COMMUNICATION_OK) {
216 pState->serialId[pState->currentString][i - 1u] = 0u;
217 for (uint8_t j = 0u; j < 3u; j++) {
218 pState->serialId[pState->currentString][i - 1u] |= ((uint64_t)uid[j]) << (16u * j);
219 }
220 }
221 }
222
223 return retVal;
224}

226static void N77x_Initialize(N77X_STATE_s *pState) {
227 FAS_ASSERT(pState != NULL_PTR);
228
229 /* Reset mux sequence */
230 N77x_ResetMuxIndex(pState);
231
232 /* Initialize daisy-chain */
233 if (STD_OK != N77x_Enumerate(pState)) {
234 /* error handling */}
235 N77x_StartMeasurement(pState);
236 N77x_InitializeI2c(pState);
237 N77x_BalanceSetup(pState);
238}

245 OS_ExitTaskCritical();

248static void N77x_IncrementStringSequence(N77X_STATE_s *pState) {
249 FAS_ASSERT(pState != NULL_PTR);
250
251 pState->currentString++;
252 pState->pSpiTxSequence = pState->pSpiTxSequenceStart + pState->currentString;
253 pState->pSpiRxSequence = pState->pSpiRxSequenceStart + pState->currentString;
254}

256static void N77x_ResetStringSequence(N77X_STATE_s *pState) {
257 FAS_ASSERT(pState != NULL_PTR);
258
259 pState->currentString = 0u;
260 pState->pSpiTxSequence = pState->pSpiTxSequenceStart + pState->currentString;
261 pState->pSpiRxSequence = pState->pSpiRxSequenceStart + pState->currentString;
262}

276extern void N77x_Measure(N77X_STATE_s *pState) {
277 FAS_ASSERT(pState != NULL_PTR);
278
279 N77x_InitializeDatabase(pState);
280 /* Initialize SPI sequence pointers */
281 pState->pSpiTxSequenceStart = spi_nxp77xInterfaceTx;
282 pState->pSpiRxSequenceStart = spi_nxp77xInterfaceRx;
283
284 /* Initialize each string */
285 N77x_ResetStringSequence(pState);
286 while (pState->currentString < BS_NR_OF_STRINGS) {
287 /* Initialize mux sequence pointers */
288 pState->pMuxSequenceStart[pState->currentString] = n77x_muxSequence;
289 N77x_Initialize(pState);
290 N77x_IncrementStringSequence(pState);
291 }
292
293 while (1) {
294 N77x_ResetStringSequence(pState);
295
296 while (pState->currentString < BS_NR_OF_STRINGS) {
297 if (N77X_USE_MUX_FOR_TEMP == true) {
298 /* Set mux channel according to mux sequence */
299 N77x_SetMuxChannel(pState);
300 }
301 N77x_CaptureMeasurement(pState);
302 if (N77X_USE_MUX_FOR_TEMP == true) {
303 /* Update index in mux sequence */
304 N77x_IncrementMuxIndex(pState);
305 }
306 N77x_BalanceControl(pState);
307
308 N77x_TransmitI2c(pState);
309
310 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
311 if (N77x_IsFirstMeasurementCycleFinished(pState) == true) {
312 if (pState->n77xData.errorTable->noCommunicationTimeout[pState->currentString][m] == 0u) {
313 /* Daisy-chain does not answer: initialize it again */
314 pState->pMuxSequenceStart[pState->currentString] = n77x_muxSequence;
315 N77x_Initialize(pState);
316 /* Daisy-chain re-initialized: no need to check further slaves */
317 break;
318 }
319 }
320 }
321
322 N77x_IncrementStringSequence(pState);
323 }
324
325 /* Set flag when all strings have been measured */
326 if ((pState->currentString == BS_NR_OF_STRINGS) && (N77x_IsFirstMeasurementCycleFinished(pState) == 0u)) {
327 N77x_SetFirstMeasurementCycleFinished(pState);
328 }
329 }
330}
