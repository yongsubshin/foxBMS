71extern void N77x_ErrorHandling(N77X_STATE_s *pState, N77X_COMMUNICATION_STATUS_e returnedValue, uint8_t module) {
72 FAS_ASSERT(pState != NULL_PTR);
73
74 if (returnedValue == N77X_COMMUNICATION_OK) {
75 pState->n77xData.errorTable->communicationOk[pState->currentString][module] = true;
76 pState->n77xData.errorTable->noCommunicationTimeout[pState->currentString][module] = true;
77 pState->n77xData.errorTable->crcIsValid[pState->currentString][module] = true;
78 } else {
79 pState->n77xData.errorTable->communicationOk[pState->currentString][module] = false;
80 switch (returnedValue) {
81 case N77X_COMMUNICATION_ERROR_TIMEOUT:
82 pState->n77xData.errorTable->noCommunicationTimeout[pState->currentString][module] = false;
83 break;
84 case N77X_COMMUNICATION_ERROR_WRONG_CRC:
85 pState->n77xData.errorTable->crcIsValid[pState->currentString][module] = false;
86 break;
87 default:
88 pState->n77xData.errorTable->communicationOk[pState->currentString][module] = false;
89 break;
90 }
91 }
92}

95 uint32_t current_time = OS_GetTickCount();
96 /* Block task without possibility to wake up */
