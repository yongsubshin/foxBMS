528/*========== Extern Function Implementations ================================*/
529extern STD_RETURN_TYPE_e FS85_InitializeFsPhase(FS85_STATE_s *pInstance) {
530 FAS_ASSERT(pInstance != NULL_PTR);
531
532 STD_RETURN_TYPE_e retval = STD_OK;
533 uint16_t registerMask = 0u;
534 uint16_t expRegisterValue = 0u;
535 uint16_t registerValue = 0u;
536
537 FAS_ASSERT(pInstance != NULL_PTR);
538 /* Default init value: #STD_NOT_OK, will be set to #STD_OK later in init if
539 no short-circuit between FIN and RSTB detected. Do not apply external
540 clock on this pin until finState == #STD_OK */
541 FAS_ASSERT(pInstance->fin.finState == STD_NOT_OK);
542
543 /** First: Verify following conditions:
544 1. Verify LBIST (logical-built-in-self-test) and ABIST1 (analog-built-in-self-test1) are pass
545 2. Verify Debug mode is not activated
546 3. Verify there is no OTP CRC error
547 4. Verify PGOOD was released: PGOOD is connected to power-on-reset of the MCU
548
549 Second: Configure FS_I and FS_I_NOT registers
550
551 - Write the desired data in the FS_I_Register_A (DATA)
552 - Write the opposite in the FS_I_NOT_Register_A (DATA_NOT)
553 - Only the utility bits must be inverted in the DATA_NOT content. The RESERVED bits are not considered and can be
554 written at '0'. If the comparison result is correct, then the REG_CORRUPT is set to '0'. If the comparison result
555 is wrong, then the REG_CORRUPT bit is set to '1'. The REG_CORRUPT monitoring is active as soon as the INIT_FS is
556 closed by the first good watchdog refresh. INIT_FS must be closed by the first good watchdog refresh before 256ms
557 timeout.
558
559 1. Configure VCOREMON_OV_UV impact on RSTB and FS0B
560 2. Configure VDDIO_OV_UV impact on RSTB and FS0B
561 3. Configure VMONx_OV_UV impact on RSTB and FS0B
562 4. Configure ABIST2 assignment
563 5. Configure the WD window period, the WD window duty cycle, the WD counters limits,
564 and its impact on RSTB and FS0B. Ensure that the configuration does not violate the
565 FTTI requirement at system level.
566 6. Configure the Fault Error Counter limit and its impact on RSTB and FS0B at
567 intermediate value
568 7. Configure the RSTB pulse duration
569 8. Configure MCU FCCU error monitoring and its impact on RSTB and FS0B
570 9. Configure Ext. IC error monitoring and its impact on RSTB and FS0B
571 10.Configure FS0B short to high impact on RSTB
572
573 Third: Execute
574
575 1. Close INIT_FS by sending the first good WD refresh
576 2. Execute ABIST2 and verify it is pass
577 3. Clear all the flags by writing in FS_DIAG_SAFETY, FS_OVUVREG_STATUS
578 4. Clear the fault error counter to 0 with consecutive good WD refresh
579 5. Perform RSTB path check (repeat steps 1 to 4 after RSTB is released)
580 6. Release FS0B pin
581 7. Perform FS0B safety path check
582 8. Refresh the WD according to its configuration
583 9. Check FS_GRL_FLAGS register after each WD refresh
584
585 The FS85 is now ready. If everything is OK for the MCU, it can release its own
586 safety path and the ECU starts.
587 */
588
589 /* --------------------- First: Verify --------------------------------- */
590
591 /** 1.: Verify LBIST and ABIST1 */
592 if (STD_OK != FS85_ReadBackRegister(pInstance, true, FS8X_FS_DIAG_SAFETY_ADDR)) {
593 retval = STD_NOT_OK;
594 }
595 registerMask = (FS8X_FS_LBIST_OK_MASK | FS8X_FS_ABIST1_OK_MASK);
596 expRegisterValue = (FS8X_FS_LBIST_OK_PASS | FS8X_FS_ABIST1_OK_PASS);
597 if (STD_OK != FS85_CheckRegisterValues((pInstance->fsRegister.diag_safety & registerMask), expRegisterValue)) {
598 retval = STD_NOT_OK;
599 }
600
601 /** Read FS STATES register */
602 if (STD_OK != FS85_ReadBackRegister(pInstance, true, FS8X_FS_STATES_ADDR)) {
603 retval = STD_NOT_OK;
604 }
605
606 /** 2.: Check if debug mode is active */
607 registerMask = FS8X_FS_DBG_MODE_MASK;
608 expRegisterValue = FS8X_FS_DBG_MODE_NO_DEBUG;
609 if (STD_OK != FS85_CheckRegisterValues((pInstance->fsRegister.states & registerMask), expRegisterValue)) {
610 /* Debug mode active: this can only be the case if debugger connected */
611 MINFO_SetDebugProbeConnectionState(MINFO_DEBUG_PROBE_CONNECTED);
612 pInstance->mode = FS85_DEBUG_MODE;
613 } else {
614 /* Debug mode not active: no debugger connected */
615 MINFO_SetDebugProbeConnectionState(MINFO_DEBUG_PROBE_NOT_CONNECTED);
616 pInstance->mode = FS85_NORMAL_MODE;

914extern bool FS85_CheckIgnitionSignal(FS85_STATE_s *pInstance) {
915 FAS_ASSERT(pInstance != NULL_PTR);
916
917 /* Default state */
918 bool ignitionSignalDetected = true;
919 /* variables for storing current and previous wake1 signal levels */
920 static uint16_t wake1Level = 0u;
921 static uint16_t oldWake1Level = 0u;
922
923 /* Read wake1 signal level */
924 fs8x_rx_frame_t rxTemp = {.deviceStatus = 0u, .readData = 0u};
925 if (fs8xStatusOk ==
926 FS8x_ReadRegister(pInstance->pSpiInterface, &(pInstance->configValues), false, FS8X_M_FLAG2_ADDR, &rxTemp)) {
927 wake1Level = rxTemp.readData & FS8X_M_WK1RT_WAKE1_HIGH;
928 }
929
930 /* Check for falling edge in wake1 signal */
931 if ((oldWake1Level != wake1Level) && (wake1Level == 0u)) {
932 if (STD_OK != FS85_GoToStandby(pInstance)) {
933 /* error in switch to standby mode, reset ole level to try again */
934 wake1Level = oldWake1Level;
935 } else {
936 ignitionSignalDetected = false;
937 }
938 }
939 /* update old level for edge detection */
940 oldWake1Level = wake1Level;
941 return ignitionSignalDetected;
942}

944extern STD_RETURN_TYPE_e FS85_InitializeNumberOfRequiredWatchdogRefreshes(
945 FS85_STATE_s *pInstance,
946 uint8_t *pRequiredWatchdogRefreshes) {
947 FAS_ASSERT(pInstance != NULL_PTR);
948 FAS_ASSERT(pRequiredWatchdogRefreshes != NULL_PTR);
949
950 STD_RETURN_TYPE_e retval = STD_OK;
951
952 /** Clear the fault error counter to 0 with consecutive good WD refreshes.
953 * The watchdog refresh counter is used to decrement the fault error counter. Each time the watchdog is properly
954 * refreshed, the watchdog refresh counter is incremented by '1'. Each time the watchdog refresh counter reaches
955 * its maximum value ('6' by default) and if next WD refresh is also good, the fault error counter is decremented
956 * by '1'. Whatever the position the watchdog refresh counter is in, each time there is a wrong refresh watchdog,
957 * the watchdog refresh counter is reset to '0'. */
958
959 /** Read out FS_I_WD_CFG register to get watchdog refresh counter limit and value */
960 if (STD_OK != FS85_ReadBackRegister(pInstance, true, FS8X_FS_I_WD_CFG_ADDR)) {
961 retval = STD_NOT_OK;
962 }
963
964 /** Get refresh counter value */
965 uint8_t watchdogRefreshLimit = 0u;
966 uint16_t watchdogRefreshCounter = FS8x_BO_GET_REG_VALUE(
967 pInstance->fsRegister.iWatchdogConfiguration, FS8X_FS_I_WD_RFR_CNT_MASK, FS8X_FS_I_WD_RFR_CNT_SHIFT);
968
969 /** Get refresh counter limit register value */
970 if (FS8X_FS_I_WD_RFR_LIMIT_6 ==
971 FS8x_BO_GET_REG_VALUE(
972 pInstance->fsRegister.iWatchdogConfiguration, FS8X_FS_I_WD_RFR_LIMIT_MASK, FS8X_FS_I_WD_RFR_LIMIT_SHIFT)) {
973 watchdogRefreshLimit = 6u;
974 } else if (
975 FS8X_FS_I_WD_RFR_LIMIT_4 ==
976 FS8x_BO_GET_REG_VALUE(
977 pInstance->fsRegister.iWatchdogConfiguration, FS8X_FS_I_WD_RFR_LIMIT_MASK, FS8X_FS_I_WD_RFR_LIMIT_SHIFT)) {
978 watchdogRefreshLimit = 4u;
979 } else if (
980 FS8X_FS_I_WD_RFR_LIMIT_2 ==
981 FS8x_BO_GET_REG_VALUE(
982 pInstance->fsRegister.iWatchdogConfiguration, FS8X_FS_I_WD_RFR_LIMIT_MASK, FS8X_FS_I_WD_RFR_LIMIT_SHIFT)) {
983 watchdogRefreshLimit = 2u;
984 } else {
985 watchdogRefreshLimit = 1u;
986 }
987
988 /** Get fault error counter */
989 uint8_t faultCounter = 0u;
990 if (fs8xStatusOk !=
991 FS8x_GetFaultErrorCounterValue(pInstance->pSpiInterface, &(pInstance->configValues), &faultCounter)) {
992 retval = STD_NOT_OK;
993 }
994
995 /** Get number of required watchdog refreshes to clear fault error counter to 0 */
996 *pRequiredWatchdogRefreshes = faultCounter + (watchdogRefreshLimit - watchdogRefreshCounter);
997
998 return retval;
999}

1161static STD_RETURN_TYPE_e FS85_PerformPathCheckRstb(FS85_STATE_s *pInstance) {
1162 FAS_ASSERT(pInstance != NULL_PTR);
1163
1164 STD_RETURN_TYPE_e retval = STD_OK;
1165 resetSource_t rstReason = MINFO_GetResetSource();
1166 bool test_assertionRSTB = false; /* Do not test RSTB assertion on default */
1167
1168 /** Assertion of RSTB pin will trigger reset, check if reset reason was power-cycle.
1169 If so, check if short circuit between FIN and RSTB pin exists */
1170 if (rstReason == POWERON_RESET) {
1171 /** First check if FIN is used */
1172 if (pInstance->fin.finUsed == true) {
1173 /** Write to NVRAM to determine after reset and if short-circuit
1174 * between RSTB and FIN present what exactly caused the reset. */
1175 pInstance->nvram.pData->phase = (uint8_t)FS85_FIN_TEST;
1176 (void)FRAM_WriteData(pInstance->nvram.entry);
1177
1178 /** MCU SBC is connected to ECLK1 -> privilege mode is required to access register */
1179 FSYS_RaisePrivilege();
1180
1181 /** Last reset was caused by power-cycle */
1182 /** Set level of FIN pin low and check if this generates reset */
1183 IO_PinReset(pInstance->fin.pGIOport, pInstance->fin.pin);
1184
1185 /** Pulses longer than 2000ns trigger reset -> wait 10us to check if
1186 reset is triggered by short between RSTB and FIN */
1187 MCU_Delay_us(10u);
1188
1189 /** If we reach this line of code, no reset has taken place.
1190 Everything okay. Set level of FIN pin back to high */
1191 IO_PinSet(pInstance->fin.pGIOport, pInstance->fin.pin);
1192
1193 /** No further register access required -> leave privilege mode */
1194 FSYS_SwitchToUserMode();
1195
1196 /** FIN state okay, no short circuit. Update also in nvram struct */
1197 (void)DIAG_Handler(DIAG_ID_SBC_FIN_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0);
1198 pInstance->fin.finState = STD_OK;
1199 pInstance->nvram.pData->finState = STD_OK;
1200
1201 /** Continue with RSTB assertion test */
1202 test_assertionRSTB = false;
1203 } else {
1204 /** Power-cycle but no FIN pin used -> continue with RSTB check */
1205 test_assertionRSTB = false;
1206 }
1207 } else if (rstReason == EXT_RESET) {
1208 /** Last reset reason was external reset via nRST pin (EXT_RESET)
1209 Readout FRAM to determine in which state the SBC was prior to reset */
1210 (void)FRAM_ReadData(pInstance->nvram.entry);
1211 if ((FS85_INIT_PHASE_e)pInstance->nvram.pData->phase == FS85_FIN_TEST) {
1212 /** Short-circuit between FIN and RSTB: Do not apply CLK on FIN */
1213 /** Update nvram FIN state */
1214 pInstance->nvram.pData->finState = STD_NOT_OK;
1215 /** FIN state not okay, but still in SBC init phase after power-cycle
1216 * continue now with RSTB assertion */
1217 test_assertionRSTB = false;
1218 } else if ((FS85_INIT_PHASE_e)pInstance->nvram.pData->phase == FS85_RSTB_ASSERTION_TEST) {
1219 /** Reset was triggered by SPI RSTB assertion test -> continue with SBC init phase */
1220 test_assertionRSTB = false;
1221
1222 /** Diagnosis of the RSTB pin/event is available by reading the FS_SAFE_IOs register:
1223 * RSTB_EVENT bit reports an activation of RSTB pin. */
1224 (void)FS85_ReadBackRegister(pInstance, true, FS8X_FS_SAFE_IOS_ADDR);
1225
1226 /** Check RSTB_EVENT if RSTB has been activated */
1227 if (STD_OK ==
1228 FS85_CheckRegisterValues(
1229 (pInstance->fsRegister.safeIos & FS8X_FS_RSTB_EVENT_MASK), FS8X_FS_RSTB_EVENT_RESET_OCCURRED)) {
1230 /** RSTB pin should be sensed high and no RSTB short to high */
1231 uint16_t mask = (FS8X_FS_RSTB_DIAG_MASK | FS8X_FS_RSTB_SNS_MASK | FS8X_FS_RSTB_DRV_MASK);
1232 uint16_t expRegisterValue =
1233 (FS8X_FS_RSTB_DIAG_NO_FAILURE | FS8X_FS_RSTB_SNS_PAD_SENSED_HIGH |
1234 FS8X_FS_RSTB_DRV_COMMAND_SENSED_HIGH);
1235 if (STD_OK != FS85_CheckRegisterValues((pInstance->fsRegister.safeIos & mask), expRegisterValue)) {
1236 retval = STD_NOT_OK;
1237 }
1238 /** Reset RSTB_EVENT flag */
1239 (void)FS85_WriteBackRegisterFsInit(pInstance, FS8X_FS_SAFE_IOS_ADDR, FS8X_FS_RSTB_EVENT_RESET_OCCURRED);
1240 /** Update diag flag */
1241 (void)DIAG_Handler(DIAG_ID_SBC_RSTB_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0);
1242 } else {
1243 /** RSTB has not been activated but this should have been the case */
1244 retval = STD_NOT_OK;
1245 (void)DIAG_Handler(DIAG_ID_SBC_RSTB_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0);
1246 }
1247 } else {
1248 /** Reset was not caused by SBC initialization or power-cycle.
1249 * Continue with SBC init phase as RSTB assertion is only tested
1250 * after power-cycle occurred */
1251 test_assertionRSTB = false;
1252 }
1253 /** Copy FIN state info from nvram variable into local state variable. This restores lost data from rest or
1254 updates local FIN state if short-circuit between FIN and RSTB has been detected */
1255 pInstance->fin.finState = pInstance->nvram.pData->finState;
1256 if (pInstance->fin.finState == STD_OK) {
1257 (void)DIAG_Handler(DIAG_ID_SBC_FIN_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0);
1258 } else {
1259 (void)DIAG_Handler(DIAG_ID_SBC_FIN_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0);
1260 }
1261 } else {
1262 /** Reset was not caused by power-cycle or SBC. SBC has already been initialized successfully after detected
1263 * power-cycle. Everything okay. Read FIN state from NVRAM and continue with normal operation */
1264 (void)FRAM_ReadData(pInstance->nvram.entry);
1265 test_assertionRSTB = false;
1266 pInstance->fin.finState = pInstance->nvram.pData->finState;
1267 if (pInstance->fin.finState == STD_OK) {
1268 (void)DIAG_Handler(DIAG_ID_SBC_FIN_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0);
1269 } else {
1270 (void)DIAG_Handler(DIAG_ID_SBC_FIN_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0);
1271 }
1272 }
1273
1274 /** Verify the hardware connection between the MCU reset pin and the FS85 reset pin */
1275 if (test_assertionRSTB == true) {
1276 /** Write to NVRAM to determine after reset and if RSTB was asserted correctly */
1277 pInstance->nvram.pData->phase = (uint8_t)FS85_RSTB_ASSERTION_TEST;
1278 (void)FRAM_WriteData(pInstance->nvram.entry);
1279
1280 (void)FS85_ReadBackRegister(pInstance, true, FS8X_FS_STATES_ADDR);
1281
1282 uint16_t registerValue = FS8X_FS_RSTB_REQ_RSTB_ASSERTION;
1283 (void)FS85_WriteRegisterFsInit(pInstance, FS8X_FS_SAFE_IOS_ADDR, registerValue);
1284
1285 while (true) {
1286 ;
1287 }
1288
1289 /* Controller should never reach this point as reset via RSTB is requested */
1290 }
1291 return retval;
1292}
