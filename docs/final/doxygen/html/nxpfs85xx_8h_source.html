141/**
142 * @brief Configures SBC during INIT_FS phase
143 * @details Checks first if SBC currently is in INIT_FS phase and if
144 * not transfers SBC back into INIT_FS. Verifies basic checks,
145 * configures fail-safe registers and closes INIT_FS
146 * afterwards.
147 * @param[in,out] pInstance SBC instance that is initialized
148 * @return #STD_OK if all checks were successful and SBC configured
149 * correctly, otherwise #STD_NOT_OK
150 */

153/**

176/**
177 * @brief Performs SBC safety path checks

186/**

192 * otherwise #STD_NOT_OK

194extern STD_RETURN_TYPE_e FS85_TriggerWatchdog(FS85_STATE_s *pInstance);
