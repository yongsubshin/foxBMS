172void OS_DelayTaskUntil(uint32_t *pPreviousWakeTime, uint32_t milliseconds) {
173 FAS_ASSERT(pPreviousWakeTime != NULL_PTR);
174 FAS_ASSERT(milliseconds > 0u);
175 uint32_t ticks = (milliseconds / OS_TICK_RATE_MS);
176 if ((uint32_t)ticks < 1u) {
177 ticks = 1u; /* Minimum delay is 1 tick */
178 }
179 vTaskDelayUntil((TickType_t *)pPreviousWakeTime, (TickType_t)ticks);
180}

186extern OS_STD_RETURN_e OS_WaitForNotification(uint32_t *pNotifiedValue, uint32_t timeout) {
187 /* AXIVION Routine Generic-MissingParameterAssert: timeout: parameter accepts whole range */
188 FAS_ASSERT(pNotifiedValue != NULL_PTR);
189
190 OS_STD_RETURN_e notificationReceived = OS_FAIL;
191 /* FreeRTOS: This function must not be used in an interrupt service routine. */
192 /* ulBitsToClearOnEntry and ulBitsToClearOnExit set to 0xffffffff
193 to clear all the bits in the task's notification value */
194 BaseType_t xNotificationReceived = xTaskNotifyWait(UINT32_MAX, UINT32_MAX, pNotifiedValue, timeout);
195 /* FreeRTOS:xTaskNotifyWait returns pdTRUE if notification was received (otherwise pdFALSE). */
196 if (xNotificationReceived == pdTRUE) {
197 notificationReceived = OS_SUCCESS;
198 }
199 return notificationReceived;
200}

202extern OS_STD_RETURN_e OS_NotifyFromIsr(TaskHandle_t taskToNotify, uint32_t notifiedValue) {
203 /* AXIVION Routine Generic-MissingParameterAssert: notifiedValue: parameter accepts whole range */
204 FAS_ASSERT(taskToNotify != NULL_PTR);
205
206 OS_STD_RETURN_e notification = OS_FAIL;
207 BaseType_t xHigherPriorityTaskWoken = pdFALSE;
208 /* Return value dependent on the value of the eAction parameter, here set to eSetValueWithOverwrite */
209 BaseType_t xNotification =
210 xTaskNotifyFromISR(taskToNotify, notifiedValue, eSetValueWithOverwrite, &xHigherPriorityTaskWoken);
211 if (xNotification == pdTRUE) {
212 notification = OS_SUCCESS;
213 }
214 /* Make the scheduler yield when notification made, so that unblocked tasks is run immediately
215 (if priorities allow it, instead of waiting for the next OS tick) */
216 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
217 return notification;
218}
