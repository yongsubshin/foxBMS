176static STD_RETURN_TYPE_e PEX_ReadInputs(void) {
177 STD_RETURN_TYPE_e retVal = STD_OK;
178
179 for (uint8_t i = 0u; i < PEX_NR_OF_PORT_EXPANDERS; i++) {
180 /*
181 * Input Port 0 as address, next read register will be Input Port 1
182 * data sheet: Rev. 9 - 8 November 2017
183 * "After the first byte is read, additional bytes may be read but
184 * the data will now reflect the information in the other register in the pair.
185 * For example, if you read Input port 1, then the next byte read would be Input port 0."
186 */
187 pex_i2cDataWrite[0u] = PEX_INPUT_PORT0_REGISTER_ADDRESS;
188 pex_i2cDataRead[0u] = 0u;
189 pex_i2cDataRead[1u] = 0u;
190 STD_RETURN_TYPE_e i2cReadReturn =
191 I2C_WriteReadDma(PEX_I2C_INTERFACE, pex_addressList[i], 1u, pex_i2cDataWrite, 2u, pex_i2cDataRead);
192 if (i2cReadReturn == STD_NOT_OK) {
193 retVal = STD_NOT_OK;
194 } else {
195 pex_inputPort0Local[i] = pex_i2cDataRead[0u];
196 pex_inputPort1Local[i] = pex_i2cDataRead[1u];
197 }
198 }
199 return retVal;
200}

202static STD_RETURN_TYPE_e PEX_WriteOutputs(void) {
203 STD_RETURN_TYPE_e retVal = STD_OK;
204 for (uint8_t i = 0u; i < PEX_NR_OF_PORT_EXPANDERS; i++) {
205 /**
206 * Out-port Port 0 as address, next read register will be Output Port 1
207 * data sheet: Rev. 9 - 8 November 2017
208 * Figure 10: one register pair can be written in one transaction
209 */
210 pex_i2cDataWrite[0u] = PEX_OUTPUT_PORT0_REGISTER_ADDRESS;
211 pex_i2cDataWrite[1u] = pex_outputPort0Local[i];
212 pex_i2cDataWrite[2u] = pex_outputPort1Local[i];
213 STD_RETURN_TYPE_e i2cWriteReturn = I2C_WriteDma(PEX_I2C_INTERFACE, pex_addressList[i], 3u, pex_i2cDataWrite);
214 if (i2cWriteReturn == STD_NOT_OK) {
215 retVal = STD_NOT_OK;
216 }
217 }
218 return retVal;
219}

283/**
284 * In the data sheet, pins are grouped in two sets, numbered from 00 to 07 and from 10 to 17.
285 * Each set is handled by a separate register, so pin 00 is numbered 0, pin 01 is numbered 01, ..,
286 * pin 07 is numbered 07. But also pin 10 is numbered 0, pin 11 is numbered 01, .., pin 17 is numbered 07.
287 * Using defines from 0 to 15, it is possible to get with pin number which register has to be addressed:
288 * 0 to 7 means register for 00 to 07, 8 to 15 means register for 10 to 17.
289 * Applying (pin%8) for 8 to 15 gives 0 to 7, so the numbering in register is obtained immediately.
290 */
291extern void PEX_Initialize(void) {
292 for (uint8_t i = 0u; i < PEX_NR_OF_PORT_EXPANDERS; i++) {
293 /** Default state of output registers data sheet: Rev. 9 - 8 November 2017, Table 7 and 8 */
294 pex_outputPort0[i] = PEX_DEFAULT_VALUE_ALL_1;
295 pex_outputPort1[i] = PEX_DEFAULT_VALUE_ALL_1;
296 pex_outputPort0Local[i] = PEX_DEFAULT_VALUE_ALL_1;
297 pex_outputPort1Local[i] = PEX_DEFAULT_VALUE_ALL_1;
298 /**
299 * Default state of polarity inversion registers data sheet: Rev. 9 - 8 November 2017, Table 9 and 10
300 */
301 pex_configPolarityPort0[i] = PEX_DEFAULT_VALUE_ALL_0;
302 pex_configPolarityPort1[i] = PEX_DEFAULT_VALUE_ALL_0;
303 pex_configPolarityPort0Local[i] = PEX_DEFAULT_VALUE_ALL_0;
304 pex_configPolarityPort1Local[i] = PEX_DEFAULT_VALUE_ALL_0;
305 /** Default state of direction registers data sheet: Rev. 9 - 8 November 2017, Table 11 and 12 */
306 pex_configDirectionPort0[i] = PEX_DEFAULT_VALUE_ALL_1;
307 pex_configDirectionPort1[i] = PEX_DEFAULT_VALUE_ALL_1;
308 pex_configDirectionPort0Local[i] = PEX_DEFAULT_VALUE_ALL_1;
309 pex_configDirectionPort1Local[i] = PEX_DEFAULT_VALUE_ALL_1;
310 }
311}

313extern void PEX_Trigger(void) {
314 PEX_CopyToLocalVariable();
315 uint32_t current_time = OS_GetTickCount();
316
317 STD_RETURN_TYPE_e writeConfigDirectionReturn = PEX_WriteConfigDirection();
318 OS_DelayTaskUntil(&current_time, 2u);
319 STD_RETURN_TYPE_e writeConfigPolarityReturn = PEX_WriteConfigPolarity();
320 OS_DelayTaskUntil(&current_time, 2u);
321 STD_RETURN_TYPE_e readInputsReturn = PEX_ReadInputs();
322 OS_DelayTaskUntil(&current_time, 2u);
323 STD_RETURN_TYPE_e writeOutputsReturn = PEX_WriteOutputs();
324 OS_DelayTaskUntil(&current_time, 2u);
325
326 /* notify diag if one of these functions failed, but continue normally */
327 if ((writeConfigDirectionReturn == STD_NOT_OK) || (writeConfigPolarityReturn == STD_NOT_OK) ||
328 (readInputsReturn == STD_NOT_OK) || (writeOutputsReturn == STD_NOT_OK)) {
329 (void)DIAG_Handler(DIAG_ID_I2C_PEX_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
330 } else {
331 (void)DIAG_Handler(DIAG_ID_I2C_PEX_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
332 }
333
334 PEX_GetFromLocalVariable();
335}

337extern void PEX_SetPin(uint8_t portExpander, uint8_t pin) {
338 FAS_ASSERT(portExpander < PEX_NR_OF_PORT_EXPANDERS);
339 FAS_ASSERT(pin <= PEX_PORT_1_PIN_7);
340
341 if (pin <= PEX_PORT_0_PIN_7) {
342 OS_EnterTaskCritical();
343 pex_outputPort0[portExpander] |= PEX_PIN_HIGH << pin;
344 OS_ExitTaskCritical();
345 } else { /* pin is in range PEX_PORT_1_PIN_0 to PEX_PORT_1_PIN_7 */
346 OS_EnterTaskCritical();
347 pex_outputPort1[portExpander] |= PEX_PIN_HIGH << (pin % PEX_NR_OF_PORTS_PER_REGISTER);
348 OS_ExitTaskCritical();
349 }
350}

353 FAS_ASSERT(portExpander < PEX_NR_OF_PORT_EXPANDERS);

363 OS_ExitTaskCritical();
364 }

367extern uint8_t PEX_GetPin(uint8_t portExpander, uint8_t pin) {
368 FAS_ASSERT(portExpander < PEX_NR_OF_PORT_EXPANDERS);
369 FAS_ASSERT(pin <= PEX_PORT_1_PIN_7);
370
371 uint8_t pinState = UINT8_MAX;
372
373 if (pin <= PEX_PORT_0_PIN_7) {
374 OS_EnterTaskCritical();
375 pinState = (pex_inputPort0[portExpander] >> pin) & 0x1u;
376 OS_ExitTaskCritical();
377 } else { /* pin is in range PEX_PORT_1_PIN_0 to PEX_PORT_1_PIN_7 */
378 OS_EnterTaskCritical();
379 pinState = (pex_inputPort1[portExpander] >> (pin % PEX_NR_OF_PORTS_PER_REGISTER)) & 0x1u;
380 OS_ExitTaskCritical();
381 }
382
383 if (pinState == 0u) {
384 pinState = PEX_PIN_LOW;
385 } else if (pinState == 1u) {
386 pinState = PEX_PIN_HIGH;
387 } else {
388 /* invalid pin state */
389 FAS_ASSERT(FAS_TRAP);
390 }
391
392 return pinState;
393}
