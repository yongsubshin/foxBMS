77extern STD_RETURN_TYPE_e PL_CheckStringVoltage(int32_t voltageAfe_mV, int32_t voltageCurrentSensor_mV) {
78 STD_RETURN_TYPE_e result = STD_NOT_OK;
79
80 /* Get deviation between these two measurements */
81 int32_t diff_mV = voltageAfe_mV - voltageCurrentSensor_mV;
82
83 if (abs(diff_mV) < PL_STRING_VOLTAGE_TOLERANCE_mV) {
84 result = STD_OK;
85 }
86 return result;
87}

89extern STD_RETURN_TYPE_e PL_CheckCellVoltage(
90 int16_t baseCellVoltage,
91 int16_t redundancy0CellVoltage,
92 int16_t *pCellVoltage) {
93 /* AXIVION Routine Generic-MissingParameterAssert: baseCellVoltage: parameter accepts whole range */
94 /* AXIVION Routine Generic-MissingParameterAssert: redundancy0CellVoltage: parameter accepts whole range */
95 /* Pointer validity check */
96 FAS_ASSERT(pCellVoltage != NULL_PTR);
97 STD_RETURN_TYPE_e retval = STD_OK;
98
99 if (abs(baseCellVoltage - redundancy0CellVoltage) > PL_CELL_VOLTAGE_TOLERANCE_mV) {
100 retval = STD_NOT_OK;
101 }
102 /* Take the average value of base and redundant measurement value */
103 *pCellVoltage = (baseCellVoltage + redundancy0CellVoltage) / 2;
104 return retval;
105}

126extern STD_RETURN_TYPE_e PL_CheckVoltageSpread(
127 DATA_BLOCK_CELL_VOLTAGE_s *pCellVoltages,
128 DATA_BLOCK_MIN_MAX_s *pMinMaxAverageValues) {
129 /* Pointer validity check */
130 FAS_ASSERT(pCellVoltages != NULL_PTR);
131 FAS_ASSERT(pMinMaxAverageValues != NULL_PTR);
132
133 STD_RETURN_TYPE_e retval = STD_OK;
134
135 /* Iterate over all cells */
136 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
137 STD_RETURN_TYPE_e plausibilityIssueDetected = STD_OK;
138 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
139 for (uint8_t cb = 0u; cb < BS_NR_OF_CELL_BLOCKS_PER_MODULE; cb++) {
140 /* Only do check for valid voltages */
141 if (pCellVoltages->invalidCellVoltage[s][m][cb] == false) {
142 if (abs(pCellVoltages->cellVoltage_mV[s][m][cb] - pMinMaxAverageValues->averageCellVoltage_mV[s]) >
143 PL_CELL_VOLTAGE_SPREAD_TOLERANCE_mV) {
144 /* Voltage difference too large */
145 plausibilityIssueDetected = STD_NOT_OK;
146 retval = STD_NOT_OK;
147 /* Set this cell voltage invalid */
148 pCellVoltages->invalidCellVoltage[s][m][cb] = true;
149 }
150 }
151 }
152 }
153 DIAG_CheckEvent(plausibilityIssueDetected, DIAG_ID_PLAUSIBILITY_CELL_VOLTAGE_SPREAD, DIAG_STRING, s);
154 }
155 return retval;
156}

158extern STD_RETURN_TYPE_e PL_CheckTemperatureSpread(
159 DATA_BLOCK_CELL_TEMPERATURE_s *pCellTemperatures,
160 DATA_BLOCK_MIN_MAX_s *pMinMaxAverageValues) {
161 /* Pointer validity check */
162 FAS_ASSERT(pCellTemperatures != NULL_PTR);
163 FAS_ASSERT(pMinMaxAverageValues != NULL_PTR);
164
165 STD_RETURN_TYPE_e retval = STD_OK;
166
167 /* Iterate over all cells */
168 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
169 STD_RETURN_TYPE_e plausibilityIssueDetected = STD_OK;
170 for (uint8_t m = 0u; m < BS_NR_OF_MODULES_PER_STRING; m++) {
171 for (uint8_t ts = 0u; ts < BS_NR_OF_TEMP_SENSORS_PER_MODULE; ts++) {
172 /* Only do check for valid temperatures */
173 if (pCellTemperatures->invalidCellTemperature[s][m][ts] == false) {
174 if (abs(pCellTemperatures->cellTemperature_ddegC[s][m][ts] -
175 (int16_t)pMinMaxAverageValues->averageTemperature_ddegC[s]) >
176 PL_CELL_TEMPERATURE_SPREAD_TOLERANCE_dK) {
177 /* temperature difference too large */
178 plausibilityIssueDetected = STD_NOT_OK;
179 retval = STD_NOT_OK;
180 /* Set this cell temperature invalid */
181 pCellTemperatures->invalidCellTemperature[s][m][ts] = true;
182 } else {
183 pCellTemperatures->nrValidTemperatures[s]++;
184 }
185 }
186 }
187 }
188 DIAG_CheckEvent(plausibilityIssueDetected, DIAG_ID_PLAUSIBILITY_CELL_TEMPERATURE_SPREAD, DIAG_STRING, s);
189 }
190 return retval;
191}
