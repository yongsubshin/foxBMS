174static void RTC_WriteTime(RTC_TIME_DATA_s rtcTime) {
175 /* AXIVION Routine Generic-MissingParameterAssert: rtcTime: parameter accepts whole range */
176 STD_RETURN_TYPE_e retValI2c = STD_OK;
177 STD_RETURN_TYPE_e retVal = STD_OK;
178
179 /* Write control_1 register to stop RTC */
180 rtc_i2cWriteBuffer[0u] = RTC_REG_CONTROL_1_ADDR;
181 /* Set stop bit, clear 12/24hour mode bit */
182 rtc_i2cWriteBuffer[1u] = 0u;
183 rtc_i2cWriteBuffer[1u] |= RTC_CTRL1_STOP_BIT_MASK;
184 rtc_i2cWriteBuffer[1u] &= ~((uint8_t)RTC_CTRL1_12_24HOUR_MODE_BIT_MASK);
185 retValI2c = I2C_WriteDma(
186 RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES, rtc_i2cWriteBuffer);
187 if (retValI2c == STD_NOT_OK) {
188 retVal = STD_NOT_OK;
189 }
190 /* Write 0xA4 to software reset register */
191 rtc_i2cWriteBuffer[0u] = RTC_REG_SOFTWARE_RESET_ADDR;
192 rtc_i2cWriteBuffer[1u] = RTC_SWRESET_CLEARPRESCALER;
193 /* Address counters rolls to time data */
194 rtc_i2cWriteBuffer[RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES + RTC_100TH_OF_SECONDS_OFFSET] =
195 ((rtcTime.hundredthOfSeconds / RTC_TENS_PLACE_FACTOR) << RTC_TENS_PLACE_OFFSET) |
196 (rtcTime.hundredthOfSeconds % RTC_TENS_PLACE_FACTOR);
197 rtc_i2cWriteBuffer[RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES + RTC_SECONDS_OFFSET] =
198 ((rtcTime.seconds / RTC_TENS_PLACE_FACTOR) << RTC_TENS_PLACE_OFFSET) |
199 (rtcTime.seconds % RTC_TENS_PLACE_FACTOR);
200 rtc_i2cWriteBuffer[RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES + RTC_MINUTES_OFFSET] =
201 ((rtcTime.minutes / RTC_TENS_PLACE_FACTOR) << RTC_TENS_PLACE_OFFSET) |
202 (rtcTime.minutes % RTC_TENS_PLACE_FACTOR);
203 rtc_i2cWriteBuffer[RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES + RTC_HOURS_OFFSET] =
204 ((rtcTime.hours / RTC_TENS_PLACE_FACTOR) << RTC_TENS_PLACE_OFFSET) | (rtcTime.hours % RTC_TENS_PLACE_FACTOR);
205 rtc_i2cWriteBuffer[RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES + RTC_DAYS_OFFSET] =
206 ((rtcTime.day / RTC_TENS_PLACE_FACTOR) << RTC_TENS_PLACE_OFFSET) | (rtcTime.day % RTC_TENS_PLACE_FACTOR);
207 rtc_i2cWriteBuffer[RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES + RTC_WEEKDAYS_OFFSET] =
208 ((rtcTime.weekday / RTC_TENS_PLACE_FACTOR) << RTC_TENS_PLACE_OFFSET) |
209 (rtcTime.weekday % RTC_TENS_PLACE_FACTOR);
210 rtc_i2cWriteBuffer[RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES + RTC_MONTHS_OFFSET] =
211 ((rtcTime.month / RTC_TENS_PLACE_FACTOR) << RTC_TENS_PLACE_OFFSET) | (rtcTime.month % RTC_TENS_PLACE_FACTOR);
212 rtc_i2cWriteBuffer[RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES + RTC_YEARS_OFFSET] =
213 ((rtcTime.year / RTC_TENS_PLACE_FACTOR) << RTC_TENS_PLACE_OFFSET) | (rtcTime.year % RTC_TENS_PLACE_FACTOR);
214 /* Write time data */
215 retValI2c =
216 I2C_WriteDma(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, RTC_MAX_I2C_TRANSACTION_SIZE_IN_BYTES, rtc_i2cWriteBuffer);
217 if (retValI2c == STD_NOT_OK) {
218 retVal = STD_NOT_OK;
219 }
220 /* Wait external reference */
221 /* Now write control_1 register again to start RTC */
222 rtc_i2cWriteBuffer[0u] = RTC_REG_CONTROL_1_ADDR;
223 rtc_i2cWriteBuffer[1u] = 0u;
224 rtc_i2cWriteBuffer[1u] &= ~((uint8_t)RTC_CTRL1_STOP_BIT_MASK);
225 rtc_i2cWriteBuffer[1u] &= ~((uint8_t)RTC_CTRL1_12_24HOUR_MODE_BIT_MASK);
226 retValI2c = I2C_WriteDma(
227 RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES, rtc_i2cWriteBuffer);
228 if (retValI2c == STD_NOT_OK) {
229 retVal = STD_NOT_OK;
230 }
231
232 if (retVal == STD_NOT_OK) {
233 (void)DIAG_Handler(DIAG_ID_I2C_RTC_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
234 } else {
235 (void)DIAG_Handler(DIAG_ID_I2C_RTC_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
236 }
237}

250static void RTC_AdjustTime(void) {
251 RTC_TIME_DATA_s rtcTimeFromTimerRtcFormat = {0};
252 RTC_TIME_DATA_s rtcTimeFromIcRtcFormat = {0};
253 time_t rtcTimeFromTimerEpochFormat = 0;
254 time_t rtcTimeFromIcEpochFormat = 0;
255 struct tm rtcTimeFromTimerTmFormat = {0};
256 struct tm rtcTimeFromIcTmFormat = {0};
257
258 rtcTimeFromIcRtcFormat = RTC_ReadTime();
259 /* Convert time and date from RTC to tm struct */
260 rtcTimeFromIcTmFormat = RTC_rtcFormatToTmFormat(rtcTimeFromIcRtcFormat);
261 /* Convert tm struct to timer in seconds since epoch */
262 rtcTimeFromIcEpochFormat = mktime(&rtcTimeFromIcTmFormat);
263
264 rtcTimeFromTimerRtcFormat = RTC_GetSystemTimeRtcFormat();
265 /* Convert time and date from RTC to tm struct */
266 rtcTimeFromTimerTmFormat = RTC_rtcFormatToTmFormat(rtcTimeFromTimerRtcFormat);
267 /* Convert tm struct to timer in seconds since epoch */
268 rtcTimeFromTimerEpochFormat = mktime(&rtcTimeFromTimerTmFormat);
269
270 if (abs(rtcTimeFromIcEpochFormat - rtcTimeFromTimerEpochFormat) > RTC_MAX_DIFFERENCE_BETWEEN_TIMER_AND_IC_s) {
271 /* Difference between RTC timer and RTC IC higher than limit: adjust RTC timer */
272 RTC_InitializeSystemTimeWithRtc();
273 }
274}

276static void RTC_CheckBatteryLowVoltageAlert(void) {
277 STD_RETURN_TYPE_e retValI2c = STD_OK;
278 STD_RETURN_TYPE_e retVal = STD_OK;
279 uint8_t batteryLowFlag = 0u; /* Battery Low Flag */
280
281 /* Set address to read from */
282 rtc_i2cWriteBuffer[0u] = RTC_REG_CONTROL_3_ADDR;
283 retValI2c = I2C_Write(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, 1u, rtc_i2cWriteBuffer);
284 if (retValI2c == STD_NOT_OK) {
285 retVal = STD_NOT_OK;
286 }
287 /* Address set, read register data */
288 retValI2c = I2C_Read(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, 1u, rtc_i2cReadBuffer);
289 if (retValI2c == STD_NOT_OK) {
290 retVal = STD_NOT_OK;
291 }
292
293 /* OSF bit is stored at position 7 in seconds register */
294 batteryLowFlag = (rtc_i2cReadBuffer[0u] & RTC_CTRL3_BATTERY_LOW_FLAG_BIT_MASK) >>
295 RTC_CTRL3_BATTERY_LOW_FLAG_BIT_POSITION;
296
297 if (retVal == STD_NOT_OK) {
298 (void)DIAG_Handler(DIAG_ID_I2C_RTC_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
299 } else {
300 (void)DIAG_Handler(DIAG_ID_I2C_RTC_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
301 /* If I2C communication successful, check Battery Low Flag bit */
302 if (batteryLowFlag != 0u) {
303 (void)DIAG_Handler(DIAG_ID_RTC_BATTERY_LOW_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
304 } else {
305 (void)DIAG_Handler(DIAG_ID_RTC_BATTERY_LOW_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
306 }
307 }
308}

353static RTC_TIME_DATA_s RTC_ReadTime(void) {
354 RTC_TIME_DATA_s rtcTime = {0};
355 STD_RETURN_TYPE_e retValI2c = STD_OK;
356 STD_RETURN_TYPE_e retVal = STD_OK;
357
358 uint8_t osfBit = 0u;
359
360 /* Set address to read from */
361 rtc_i2cWriteBuffer[0u] = RTC_REG_TIME_START_ADDR;
362 retValI2c = I2C_Write(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, 1u, rtc_i2cWriteBuffer);
363 if (retValI2c == STD_NOT_OK) {
364 retVal = STD_NOT_OK;
365 }
366 /* Address set, read time data */
367 retValI2c = I2C_ReadDma(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, RTC_NUMBER_OF_TIME_DATA_BYTES, rtc_i2cReadBuffer);
368 if (retValI2c == STD_NOT_OK) {
369 retVal = STD_NOT_OK;
370 }
371
372 rtcTime.hundredthOfSeconds =
373 (((rtc_i2cReadBuffer[RTC_100TH_OF_SECONDS_OFFSET] & RTC_HUNDREDTH_OF_SECONDS_TENS_PLACE_MASK) >>
374 RTC_TENS_PLACE_OFFSET) *
375 RTC_TENS_PLACE_FACTOR) +
376 (rtc_i2cReadBuffer[RTC_100TH_OF_SECONDS_OFFSET] & RTC_UNITS_PLACE_MASK);
377 rtcTime.seconds =
378 (((rtc_i2cReadBuffer[RTC_SECONDS_OFFSET] & RTC_SECONDS_TENS_PLACE_MASK) >> RTC_TENS_PLACE_OFFSET) *
379 RTC_TENS_PLACE_FACTOR) +
380 (rtc_i2cReadBuffer[RTC_SECONDS_OFFSET] & RTC_UNITS_PLACE_MASK);
381 /* OSF bit is stored at position 7 in seconds register */
382 osfBit = (rtc_i2cReadBuffer[RTC_SECONDS_OFFSET] & RTC_SECONDS_OSF_BIT_MASK) >> RTC_SECONDS_OSF_BIT_POSITION;
383 rtcTime.minutes =
384 (((rtc_i2cReadBuffer[RTC_MINUTES_OFFSET] & RTC_MINUTES_TENS_PLACE_MASK) >> RTC_TENS_PLACE_OFFSET) *
385 RTC_TENS_PLACE_FACTOR) +
386 (rtc_i2cReadBuffer[RTC_MINUTES_OFFSET] & RTC_UNITS_PLACE_MASK);
387 rtcTime.hours = (((rtc_i2cReadBuffer[RTC_HOURS_OFFSET] & RTC_HOURS_TENS_PLACE_MASK) >> RTC_TENS_PLACE_OFFSET) *
388 RTC_TENS_PLACE_FACTOR) +
389 (rtc_i2cReadBuffer[RTC_HOURS_OFFSET] & RTC_UNITS_PLACE_MASK);
390 rtcTime.day = (((rtc_i2cReadBuffer[RTC_DAYS_OFFSET] & RTC_DAYS_TENS_PLACE_MASK) >> RTC_TENS_PLACE_OFFSET) *
391 RTC_TENS_PLACE_FACTOR) +
392 (rtc_i2cReadBuffer[RTC_DAYS_OFFSET] & RTC_UNITS_PLACE_MASK);
393 rtcTime.weekday = (rtc_i2cReadBuffer[RTC_WEEKDAYS_OFFSET] & RTC_WEEKDAYS_UNITS_PLACE_MASK);
394 rtcTime.month = (((rtc_i2cReadBuffer[RTC_MONTHS_OFFSET] & RTC_MONTHS_TENS_PLACE_MASK) >> RTC_TENS_PLACE_OFFSET) *
395 RTC_TENS_PLACE_FACTOR) +
396 (rtc_i2cReadBuffer[RTC_MONTHS_OFFSET] & RTC_UNITS_PLACE_MASK);
397 rtcTime.year = (((rtc_i2cReadBuffer[RTC_YEARS_OFFSET] & RTC_YEARS_TENS_PLACE_MASK) >> RTC_TENS_PLACE_OFFSET) *
398 RTC_TENS_PLACE_FACTOR) +
399 (rtc_i2cReadBuffer[RTC_YEARS_OFFSET] & RTC_UNITS_PLACE_MASK);
400
401 if (retVal == STD_NOT_OK) {
402 (void)DIAG_Handler(DIAG_ID_I2C_RTC_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
403 } else {
404 (void)DIAG_Handler(DIAG_ID_I2C_RTC_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
405 /* If I2C communication successful, check OSF bit */
406 if (osfBit != 0u) {
407 (void)DIAG_Handler(DIAG_ID_RTC_CLOCK_INTEGRITY_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
408 } else {
409 (void)DIAG_Handler(DIAG_ID_RTC_CLOCK_INTEGRITY_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
410 }
411 }
412
413 return rtcTime;
414}

418extern void RTC_Trigger(void) {
419 static uint32_t initialTimeBatteryLowCheck = 0u;
420 static uint32_t initialTimeAdjustCheck = 0u;
421 uint32_t currentTime = OS_GetTickCount();
422
423 STD_RETURN_TYPE_e initFinished = STD_NOT_OK;
424
425 initFinished = RTC_Initialize();
426
427 if (initFinished == STD_OK) {
428 /* Check if an RTC CAN message came */
429 RTC_SetOverCanMessage();
430 /* Check battery low */
431 if ((currentTime - initialTimeBatteryLowCheck) >=
432 (RTC_TIME_BETWEEN_BATTERY_LOW_FLAG_CHECK_min * RTC_FACTOR_MIN_TO_MS)) {
433 RTC_CheckBatteryLowVoltageAlert();
434 initialTimeBatteryLowCheck = OS_GetTickCount();
435 }
436 /* Adjust time if necessary */
437 if ((currentTime - initialTimeAdjustCheck) >=
438 (RTC_TIME_BETWEEN_RTC_TIMER_ADJUSTMENT_min * RTC_FACTOR_MIN_TO_MS)) {
439 RTC_AdjustTime();
440 initialTimeAdjustCheck = OS_GetTickCount();
441 }
442 }
443}

445extern STD_RETURN_TYPE_e RTC_Initialize(void) {
446 STD_RETURN_TYPE_e returnValue = STD_NOT_OK;
447 static RTC_INIT_STATES_e rtcInitState = RTC_SET_SYSTEM_TIMER;
448 STD_RETURN_TYPE_e retValI2c = STD_OK;
449 STD_RETURN_TYPE_e retVal = STD_OK;
450 uint8_t otprBit = 1u;
451 uint8_t otprTimeout = RTC_OTPR_BIT_WAIT_TIMEOUT_ms;
452 static uint32_t initTime = 0u;
453 uint32_t currentTime = OS_GetTickCount();
454
455 switch (rtcInitState) {
456 case RTC_SET_SYSTEM_TIMER:
457 RTC_InitializeSystemTimeWithRtc();
458 rtcInitState = RTC_SET_BATTERY_LOW_FLAG;
459 OS_DelayTaskUntil(&currentTime, 2u);
460 break;
461 case RTC_SET_BATTERY_LOW_FLAG:
462 /* Write POWER_MANAGE bits */
463 /* Set address to write to */
464 rtc_i2cWriteBuffer[0u] = RTC_REG_CONTROL_3_ADDR;
465 /* Set data to write, to activate direct switching mode and battery low detection */
466 rtc_i2cWriteBuffer[1u] = 0u;
467 rtc_i2cWriteBuffer[1u] |= ((uint8_t)RTC_CTRL3_POWER_MANAGE_DIRECT_SWITCH_LOW_DETECTION_ENABLE_MODE)
468 << RTC_CTRL3_BATTERY_POWER_MANAGE_BITS_POSITION;
469 retValI2c = I2C_WriteDma(
470 RTC_I2C_INTERFACE,
471 RTC_I2C_ADDRESS,
472 RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES,
473 rtc_i2cWriteBuffer);
474 if (retValI2c == STD_NOT_OK) {
475 retVal = STD_NOT_OK;
476 }
477 rtcInitState = RTC_CLEAR_OTPR;
478 OS_DelayTaskUntil(&currentTime, 2u);
479 break;
480
481 case RTC_CLEAR_OTPR:
482 /* OTP refresh begin */
483 /* Clear OTPR bit */
484 /* Set address to write to */
485 rtc_i2cWriteBuffer[0u] = RTC_REG_CLKOUT_CTL_ADDR;
486 /* Set data to write, to clear OTPR to 0 */
487 rtc_i2cWriteBuffer[1u] = 0u;
488 rtc_i2cWriteBuffer[1u] &= ~((uint8_t)RTC_CLKOUT_CTL_OTPR_BIT_MASK);
489 retValI2c = I2C_WriteDma(
490 RTC_I2C_INTERFACE,
491 RTC_I2C_ADDRESS,
492 RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES,
493 rtc_i2cWriteBuffer);
494 if (retValI2c == STD_NOT_OK) {
495 retVal = STD_NOT_OK;
496 }
497 initTime = OS_GetTickCount();
498 rtcInitState = RTC_WAIT_CLEAR_OTPR;
499 OS_DelayTaskUntil(&currentTime, 2u);
500 break;
501
502 case RTC_WAIT_CLEAR_OTPR:
503 /* Read OTPR bit until cleared to 0 */
504
505 /* Set address to read from */
506 rtc_i2cWriteBuffer[0u] = RTC_REG_CLKOUT_CTL_ADDR;
507 retValI2c = I2C_WriteDma(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, 1u, rtc_i2cWriteBuffer);
508 if (retValI2c == STD_NOT_OK) {
509 retVal = STD_NOT_OK;
510 }
511 /* Address set, read data */
512 retValI2c = I2C_Read(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, 1u, rtc_i2cReadBuffer);
513 if (retValI2c == STD_NOT_OK) {
514 retVal = STD_NOT_OK;
515 }
516 otprBit = (rtc_i2cReadBuffer[0u] & RTC_CLKOUT_CTL_OTPR_BIT_MASK) >> RTC_CLKOUT_CTL_OTPR_BIT_POSITION;
517 if (((currentTime - initTime) > RTC_OTPR_BIT_WAIT_TIMEOUT_ms) || (otprTimeout == 0u)) {
518 rtcInitState = RTC_SET_OTPR;
519 }
520 OS_DelayTaskUntil(&currentTime, 2u);
521 break;
522
523 case RTC_SET_OTPR:
524 /* Set OTPR bit */
525 /* Set address to write to */
526 rtc_i2cWriteBuffer[0u] = RTC_REG_CLKOUT_CTL_ADDR;
527 /* Set data to write, to set OTPR to 1 */
528 rtc_i2cWriteBuffer[1u] = 0u;
529 rtc_i2cWriteBuffer[1u] |= RTC_CLKOUT_CTL_OTPR_BIT_MASK;
530 retValI2c = I2C_WriteDma(
531 RTC_I2C_INTERFACE,
532 RTC_I2C_ADDRESS,
533 RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES,
534 rtc_i2cWriteBuffer);
535 if (retValI2c == STD_NOT_OK) {
536 retVal = STD_NOT_OK;
537 }
538 initTime = OS_GetTickCount();
539 rtcInitState = RTC_WAIT_SET_OTPR;
540 OS_DelayTaskUntil(&currentTime, 2u);
541 break;
542
543 case RTC_WAIT_SET_OTPR:
544 /* Read OTPR bit until set to 1 */
545 /* Set address to read from */
546 rtc_i2cWriteBuffer[0u] = RTC_REG_CLKOUT_CTL_ADDR;
547 retValI2c = I2C_Write(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, 1u, rtc_i2cWriteBuffer);
548 if (retValI2c == STD_NOT_OK) {
549 retVal = STD_NOT_OK;
550 }
551 /* Address set, read data */
552 retValI2c = I2C_Read(RTC_I2C_INTERFACE, RTC_I2C_ADDRESS, 1u, rtc_i2cReadBuffer);
553 if (retValI2c == STD_NOT_OK) {
554 retVal = STD_NOT_OK;
555 }
556 otprBit = (rtc_i2cReadBuffer[0u] & RTC_CLKOUT_CTL_OTPR_BIT_MASK) >> RTC_CLKOUT_CTL_OTPR_BIT_POSITION;
557 if (((currentTime - initTime) > RTC_OTPR_BIT_WAIT_TIMEOUT_ms) || (otprBit == 1u)) {
558 /* OTP refresh end */
559 rtcInitState = RTC_INIT_SET_FINISHED;
560 }
561 OS_DelayTaskUntil(&currentTime, 2u);
562 break;
563
564 case RTC_INIT_SET_FINISHED:
565 /* RTC initialization finished */
566 rtc_moduleIsInitialized = true;
567 returnValue = STD_OK;
568 break;
569 default:
570 break;
571 }
572
573 if (retVal == STD_NOT_OK) {
574 (void)DIAG_Handler(DIAG_ID_I2C_RTC_ERROR, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, 0u);
575 } else {
576 (void)DIAG_Handler(DIAG_ID_I2C_RTC_ERROR, DIAG_EVENT_OK, DIAG_SYSTEM, 0u);
577 }
578
579 return returnValue;
580}
