69/** Time data of the RTC */
70typedef struct {
71 time_t secondsSinceEpoch;
72 uint16_t milliseconds;
73} RTC_SYSTEM_TIMER_EPOCH_s;
74
75/** RTC I2C interface */
76#define RTC_I2C_INTERFACE (i2cREG1)
77/* I2C address of the RTC IC */
78#define RTC_I2C_ADDRESS (0x53u)
79
80/** Time data of the RTC */
81typedef struct {
82 uint8_t hundredthOfSeconds; /* [0-99] */
83 uint8_t seconds; /* [0-59] */
84 uint8_t minutes; /* [0-59] */
85 uint8_t hours; /* [0-23], 24 hour mode is used */
86 uint8_t weekday; /* [0-6], 0 is Sunday, 1 is Monday */
87 uint8_t day; /* [1-31] */
88 uint8_t month; /* [1-12] */
89 uint8_t year; /* [0-99] */
90 uint8_t requestFlag; /* [0-2], 0 no request, 1 is valid, 2 is invalid */
91} RTC_TIME_DATA_s;
92
93/* Defines for the I2C communication */
94/**@{*/
95#define RTC_MAX_I2C_TRANSACTION_SIZE_IN_BYTES (10u)
96#define RTC_WRITE_REGISTER_I2C_TRANSACTION_SIZE_IN_BYTES (2u)
97/**@}*/
98
99/* Defines for the RTC control_1 register */
100/**@{*/
101#define RTC_REG_CONTROL_1_ADDR (0x0u)
102#define RTC_CTRL1_STOP_BIT_MASK (0x20u)
103#define RTC_CTRL1_12_24HOUR_MODE_BIT_MASK (0x04u)
104/**@}*/
105
106/* Defines for the RTC control_3 register */
107/**@{*/
108#define RTC_REG_CONTROL_3_ADDR (0x02u)
109#define RTC_CTRL3_BATTERY_POWER_MANAGE_BITS_MASK (0xE0u)
110#define RTC_CTRL3_BATTERY_POWER_MANAGE_BITS_POSITION (5u)
111#define RTC_CTRL3_BATTERY_LOW_FLAG_BIT_MASK (0x04u)
112#define RTC_CTRL3_BATTERY_LOW_FLAG_BIT_POSITION (2u)
113/**@}*/
114
115/* Power management configuration value to switch to battery voltage
116 and monitor battery voltage */
117#define RTC_CTRL3_POWER_MANAGE_DIRECT_SWITCH_LOW_DETECTION_ENABLE_MODE (0x03u)
118
119/* Defines for the RTC CLKOUT_ctl register */
120/**@{*/
121#define RTC_REG_CLKOUT_CTL_ADDR (0x13u)
122#define RTC_CLKOUT_CTL_OTPR_BIT_MASK (0x20u)
123#define RTC_CLKOUT_CTL_OTPR_BIT_POSITION (5u)
124/**@}*/
125
126/* Timeout in 10ms to wait for OTPR bit to be cleared or set */
127#define RTC_OTPR_BIT_WAIT_TIMEOUT_ms (150u)
128
129/* Mask for the OSF bit */
130#define RTC_SECONDS_OSF_BIT_MASK (0x80u)
131/* Position of the OSF bit */
132#define RTC_SECONDS_OSF_BIT_POSITION (7u)
133
134/* Defines for the RTC software reset register */
135/**@{*/
136#define RTC_REG_SOFTWARE_RESET_ADDR (0x5u)
137#define RTC_SWRESET_CLEARPRESCALER (0xA4u)
138/**@}*/
139
140/* Defines for the RTC time data registers */
141/**@{*/
142#define RTC_REG_TIME_START_ADDR (0x06u)
143#define RTC_100TH_OF_SECONDS_OFFSET (0x0u)
144#define RTC_SECONDS_OFFSET (0x1u)
145#define RTC_MINUTES_OFFSET (0x2u)
146#define RTC_HOURS_OFFSET (0x3u)
147#define RTC_DAYS_OFFSET (0x4u)
148#define RTC_WEEKDAYS_OFFSET (0x5u)
149#define RTC_MONTHS_OFFSET (0x6u)
150#define RTC_YEARS_OFFSET (0x7u)
151/**@}*/
152
153/* Number of bytes needed to store all the RTC time data */
154#define RTC_NUMBER_OF_TIME_DATA_BYTES (8u)
155
156/* Defines to access time data for ten's place */
157/**@{*/
158#define RTC_TENS_PLACE_FACTOR (10u)
159#define RTC_TENS_PLACE_OFFSET (4u)
160#define RTC_HUNDREDTH_OF_SECONDS_TENS_PLACE_MASK (0xF0u)
161#define RTC_SECONDS_TENS_PLACE_MASK (0x70u)
162#define RTC_MINUTES_TENS_PLACE_MASK (0x70u)
163#define RTC_HOURS_TENS_PLACE_MASK (0x30u)
164#define RTC_DAYS_TENS_PLACE_MASK (0x30u)
165#define RTC_MONTHS_TENS_PLACE_MASK (0x10u)
166#define RTC_YEARS_TENS_PLACE_MASK (0xF0u)
167/**@}*/
168
169/* Defines to access time data for unit's place */
170/**@{*/
171#define RTC_UNITS_PLACE_MASK (0x0Fu)
172#define RTC_WEEKDAYS_UNITS_PLACE_MASK (0x07u)
173/**@}*/
174
175/* Offset for the RTC year, because it is encoded with two digits */
176#define RTC_START_YEAR (2000)
177
178/* Defines for the C library time */
179/**@{*/
180#define RTC_CTIME_YEAR_START (1900)
181#define RTC_CTIME_MONTH_START (1)
182/**@}*/
183
184/* Max allowed difference between timer form RTC timer and IC time */
185#define RTC_MAX_DIFFERENCE_BETWEEN_TIMER_AND_IC_s (1)
186/* Factor to convert minutes to milliseconds */
187#define RTC_FACTOR_MIN_TO_MS (60000u)
188/* Time interval between two comparisons between RTC timer and RTC IC */
189#define RTC_TIME_BETWEEN_RTC_TIMER_ADJUSTMENT_min (60u)
190/* Time interval between two check of the battery low flag */
191#define RTC_TIME_BETWEEN_BATTERY_LOW_FLAG_CHECK_min (10u)

221extern void RTC_IncrementSystemTime(void);

223/**
224 * @brief initialization of the RTC IC.
225 * @return STD_OK if initialization is finished, STD_NOT_OK otherwise

229/**
230 * @brief initialize system time with RTC time.
231 * @details This function is called once at the startup of the system.
232 * It reads the time from the RTC IC over I2C and sets the RTC
233 * system timer with this value. It must be called once the
234 * 1ms task is already running, otherwise the RTC system timer
235 * will not be incremented and there will be a discrepancy between
236 * the RTC IC time and the RTC system timer.
237 */
238extern void RTC_InitializeSystemTimeWithRtc(void);
239
240/**
241 * @brief set the RTC system timer.
242 * @details This function is used to set the RTC system timer with
243 * RTC time data. It is called at startup by
244 * RTC_InitializeSystemTimeWithRtc().
245 * @param timeRtcFormat time data to set the RTC system timer
246 */
247extern void RTC_SetSystemTimeRtcFormat(RTC_TIME_DATA_s *timeRtcFormat);
248
249/**
250 * @brief set the RTC request flag.
251 * @details This function is used to set the RTC request flag with
252 * the value 0 when no request was made, 1 if the
253 * last request was valid and 2 when invalid.
254 * @param rtcRequest flag to set the RTC request flag
255 */
256extern void RTC_SetRtcRequestFlag(uint8_t rtcRequest);
257
258/**
259 * @brief trigger function for the RTC driver.
260 * @details This function uses RTC_SetOverCanMessage() to check in the RTC
261 * queue for requests to set the RTC time.
262 * If a request was made, the time of the RTC IC is set over I2C and
263 * the RTC system timer is set.
264 *
265 */
266extern void RTC_Trigger(void);
