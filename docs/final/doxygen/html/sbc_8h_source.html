68/**
69 * This define MUST represent the cycle time of the task in which context the
70 * functions run, e.g., if the #SBC_Trigger() is running in the 10 ms task
71 * then the define must be set to 10.
72 *
73 * This sets the minimum time between two subsequent executed states/substates.
74 *
75 * Define is only used for compile-time assertion, it has no programmatic
76 * influence on the actual code.
77 */

113/** State requests for the SYS state machine */
114typedef enum {
115 SBC_STATE_INIT_REQUEST, /*!< request to begin SBC initialization */
116 SBC_STATE_ERROR_REQUEST, /*!< request to switch SBC into error state */
117 SBC_STATE_NO_REQUEST, /*!< enum to clarify that currently no new request needs to be processed */
118} SBC_STATE_REQUEST_e;
119
120/** State of watchdog state */
121typedef enum {
122 SBC_PERIODIC_WATCHDOG_ACTIVATED,
123 SBC_PERIODIC_WATCHDOG_DEACTIVATED,
124} SBC_PERIODIC_WATCHDOG_STATE_e;
125
126/** Possible return values when state requests are made to the SYS state machine */
127typedef enum {
128 SBC_OK, /*!< sys --> ok */
129 SBC_BUSY_OK, /*!< sys busy --> ok */
130 SBC_REQUEST_PENDING, /*!< requested to be executed */
131 SBC_ILLEGAL_REQUEST, /*!< Request can not be executed */
132 SBC_ALREADY_INITIALIZED, /*!< Initialization of SBC already finished */
133 SBC_ILLEGAL_TASK_TYPE, /*!< Illegal */
134} SBC_RETURN_TYPE_e;
135
136/** States of the SBC state machine */
137typedef enum {
138 /* Init-Sequence */
139 SBC_STATEMACHINE_UNINITIALIZED, /*!< state machine start value */
140 SBC_STATEMACHINE_INITIALIZATION, /*!< SBC startup initialization process */
141 SBC_STATEMACHINE_RUNNING, /*!< error detected communicating with SBC */
142 SBC_STATEMACHINE_ERROR, /*!< default initialization value */
143 SBC_STATEMACHINE_UNDEFINED,
144} SBC_STATEMACHINE_e;
145
146/** Substates of the SBC state machine */
147typedef enum {
148 SBC_ENTRY, /*!< Substate entry state */
149 SBC_INIT_RESET_FAULT_ERROR_COUNTER_PART1, /*!< Substate during initialization to reset fault-error counter */
150 SBC_INIT_RESET_FAULT_ERROR_COUNTER_PART2, /*!< Substate during initialization to reset fault-error counter */
151 SBC_INITIALIZE_SAFETY_PATH_CHECK, /*!< Substate to initialize SBC */
152 SBC_INITIALIZE_VOLTAGE_SUPERVISOR_PART3, /*!< Substate to initialize SBC */
153 SBC_INITIALIZE_VOLTAGE_SUPERVISOR_PART4, /*!< Substate to initialize SBC */
154} SBC_STATEMACHINE_SUB_e;
155
156/**
157 * This structure contains all the variables relevant for the SBC state machine.
158 * The user can get the current state of the SBC state machine with this variable
159 */
160typedef struct {
161 uint16_t timer; /*!< time in ms before the state machine processes the next state, in counts of 10ms */
162 uint16_t watchdogTrigger; /*!< time in ms before the state machine triggers watchdog, in counts of 10ms */
163 SBC_STATE_REQUEST_e stateRequest; /*!< current state request made to the state machine */
164 SBC_STATEMACHINE_e state; /*!< state of Driver State Machine */
165 SBC_STATEMACHINE_SUB_e substate; /*!< current substate of the state machine */
166 SBC_STATEMACHINE_e lastState; /*!< previous state of the state machine */
167 SBC_STATEMACHINE_SUB_e lastSubstate; /*!< previous substate of the state machine */
168 uint32_t illegalRequestsCounter; /*!< counts the number of illegal requests to the SBC state machine */
169 uint8_t retryCounter; /*!< counter to retry subsystem initialization if fails */
170 uint8_t requestWatchdogTrigger; /*!< required watchdog triggers during init to correctly initialize SBC */
171 uint8_t triggerEntry; /*!< counter for re-entrance protection (function running flag) */
172 SBC_PERIODIC_WATCHDOG_STATE_e watchdogState; /*!< state if periodic watchdog trigger is required or not */
173 FS85_STATE_s *pFs85xxInstance; /*!< pointer to FS85xx instance */
174 uint16_t watchdogPeriod_10ms; /*!< watchdog trigger frequency in 10ms */
175 bool useIgnitionForPowerDown; /*!< configures whether ignition signal is used for power down requests */
176} SBC_STATE_s;

182/**
183 * @brief sets the current state request of passed state variable
184 * @details This function is used to make a state request to the state machine,
185 * e.g., start initialization. It calls #SBC_CheckStateRequest()
186 * to check if the request is valid. The state request is rejected if
187 * is not valid. The result of the check is returned immediately, so
188 * that the requester can act in case it made a non-valid state
189 * request.
190 *
191 * @param[in,out] pInstance SBC instance where different state is requested
192 * @param[in] stateRequest requested state
193 * @return If the request was successfully set, it returns the #SBC_OK, else
194 * the current state of requests (type #SBC_STATE_REQUEST_e)
195 */
196extern SBC_RETURN_TYPE_e SBC_SetStateRequest(SBC_STATE_s *pInstance, SBC_STATE_REQUEST_e stateRequest);
197
198/**
199 * @brief gets the current state of passed state variable
200 * @details This function is used in the functioning of the SBC state machine.
201 *
202 * @param[in] pInstance SBC instance where the current state is requested
203 *
204 * @return current state, taken from #SBC_STATEMACHINE_e
205 */
206extern SBC_STATEMACHINE_e SBC_GetState(SBC_STATE_s *pInstance);

208/**
