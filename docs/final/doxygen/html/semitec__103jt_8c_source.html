97/**
98 * @brief Defines for calculating the ADC voltage on the ends of the operating range.
99 * @details The ADC voltage is calculated with the following formula:
100 *
101 * V_adc = ((V_supply * R_ntc) / (R + R_ntc))
102 *
103 * Depending on the position of the NTC in the voltage resistor (R1/R2),
104 * different R_ntc values are used for the calculation.
105 */
106/**@{*/
107#if defined(TS_SEMITEC_103JT_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && \
108 (TS_SEMITEC_103JT_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
109#define TS_SEMITEC_103JT_ADC_VOLTAGE_V_MAX_V \
110 ((float_t)((TS_SEMITEC_103JT_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * \
111 ts_103jtLut[ts_103jtLutSize - 1].resistance_Ohm) / \
112 (ts_103jtLut[ts_103jtLutSize - 1].resistance_Ohm + \
113 TS_SEMITEC_103JT_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))
114#define TS_SEMITEC_103JT_ADC_VOLTAGE_V_MIN_V \
115 ((float_t)((TS_SEMITEC_103JT_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * ts_103jtLut[0].resistance_Ohm) / \
116 (ts_103jtLut[0].resistance_Ohm + TS_SEMITEC_103JT_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))
117#else /* TS_SEMITEC_103JT_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == false */
118#define TS_SEMITEC_103JT_ADC_VOLTAGE_V_MIN_V \
119 ((float_t)((TS_SEMITEC_103JT_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * \
120 ts_103jtLut[ts_103jtLutSize - 1].resistance_Ohm) / \
121 (ts_103jtLut[ts_103jtLutSize - 1].resistance_Ohm + \
122 TS_SEMITEC_103JT_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))
123#define TS_SEMITEC_103JT_ADC_VOLTAGE_V_MAX_V \

135extern int16_t TS_Sem00GetTemperatureFromLut(uint16_t adcVoltage_mV) {
136 /* AXIVION Routine Generic-MissingParameterAssert: adcVoltage_mV: parameter accepts whole range */
137
138 int16_t temperature_ddegC = 0;
139 float_t resistance_Ohm = 0.0f;
140 float_t adcVoltage_V = (float_t)adcVoltage_mV / 1000.0f; /* Convert mV to V */
141
142 /* Check for valid ADC measurements to prevent undefined behavior */
143 if (adcVoltage_V > TS_SEMITEC_103JT_ADC_VOLTAGE_V_MAX_V) {
144 /* Invalid measured ADC voltage -> sensor out of operating range or disconnected/shorted */
145 temperature_ddegC = INT16_MIN;
146 } else if (adcVoltage_V < TS_SEMITEC_103JT_ADC_VOLTAGE_V_MIN_V) {
147 /* Invalid measured ADC voltage -> sensor out of operating range or shorted/disconnected */
148 temperature_ddegC = INT16_MAX;
149 } else {
150 /* Calculate NTC resistance based on measured ADC voltage */
151#if defined(TS_SEMITEC_103JT_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && \
152 (TS_SEMITEC_103JT_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
153 /* R_1 = R_2 * ( ( V_supply / V_adc ) - 1 ) */
154 resistance_Ohm = TS_SEMITEC_103JT_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
155 ((TS_SEMITEC_103JT_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V / adcVoltage_V) - 1);
156#else /* TS_SEMITEC_103JT_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == false */
157 /* R_2 = R_1 * ( V_2 / ( V_supply - V_adc ) ) */
158 resistance_Ohm = TS_SEMITEC_103JT_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
159 (adcVoltage_V / (TS_SEMITEC_103JT_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V - adcVoltage_V));
160#endif /* TS_SEMITEC_103JT_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 */
161
162 /* Variables for interpolating LUT value */
163 uint16_t between_high = 0;
164 uint16_t between_low = 0;
165 for (uint16_t i = 1; i < ts_103jtLutSize; i++) {
166 if (resistance_Ohm < ts_103jtLut[i].resistance_Ohm) {
167 between_low = i + 1u;
168 between_high = i;
169 }
170 }
171
172 /* Interpolate between LUT values, but do not extrapolate LUT! */
173 if (!(((between_high == 0u) && (between_low == 0u)) || /* measured resistance > maximum LUT resistance */
174 (between_low > ts_103jtLutSize))) { /* measured resistance < minimum LUT resistance */
175 temperature_ddegC = (int16_t)MATH_LinearInterpolation(
176 ts_103jtLut[between_low].resistance_Ohm,
177 ts_103jtLut[between_low].temperature_ddegC,
178 ts_103jtLut[between_high].resistance_Ohm,
179 ts_103jtLut[between_high].temperature_ddegC,
180 resistance_Ohm);
181 }
182 }
183
184 /* Return temperature based on measured NTC resistance */
185 return temperature_ddegC;
186}
