71/** This structure contains all the variables relevant for the SOX */
72typedef struct {
73 bool socInitialized; /*!< true if the initialization has passed, false otherwise */
74 bool sensorCcUsed[BS_NR_OF_STRINGS]; /*!< bool if coulomb counting functionality from current sensor is used */
75 float_t ccScalingAverage[BS_NR_OF_STRINGS]; /*!< current sensor offset scaling for average SOC */

161 const float_t charge_mAs = (float_t)charge_As * UNIT_CONVERSION_FACTOR_1000_FLOAT;

165static void SOC_RecalibrateViaLookupTable(DATA_BLOCK_SOC_s *pTableSoc) {
166 FAS_ASSERT(pTableSoc != NULL_PTR);
167 DATA_BLOCK_MIN_MAX_s tableMinMaxCellVoltages = {.header.uniqueId = DATA_BLOCK_ID_MIN_MAX};
168 (void)DATA_READ_DATA(&tableMinMaxCellVoltages);
169
170 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
171 SOC_SetValue(
172 pTableSoc,
173 SE_GetStateOfChargeFromVoltage(tableMinMaxCellVoltages.minimumCellVoltage_mV[s]),
174 SE_GetStateOfChargeFromVoltage(tableMinMaxCellVoltages.maximumCellVoltage_mV[s]),
175 SE_GetStateOfChargeFromVoltage(tableMinMaxCellVoltages.averageCellVoltage_mV[s]),
176 s);
177 }
178 (void)FRAM_WriteData(FRAM_BLOCK_ID_SOC);
179}

181static void SOC_SetValue(
182 DATA_BLOCK_SOC_s *pTableSoc,
183 float_t socMinimumValue_perc,
184 float_t socMaximumValue_perc,
185 float_t socAverageValue_perc,
186 uint8_t stringNumber) {
187 FAS_ASSERT(pTableSoc != NULL_PTR);
188 /* Set database values */
189 pTableSoc->averageSoc_perc[stringNumber] = socAverageValue_perc;
190 pTableSoc->minimumSoc_perc[stringNumber] = socMinimumValue_perc;
191 pTableSoc->maximumSoc_perc[stringNumber] = socMaximumValue_perc;
192
193 if (soc_state.sensorCcUsed[stringNumber] == true) {
194 /* Current sensor database entry is read before the call of SOC_SetValue */
195 float_t ccOffset_perc =
196 SOC_GetStringSocPercentageFromCharge((uint32_t)abs(soc_tableCurrentSensor.currentCounter_As[stringNumber]));
197
198#if BS_POSITIVE_DISCHARGE_CURRENT == false
199 ccOffset_perc *= (-1.0f);
200#endif /* BS_POSITIVE_DISCHARGE_CURRENT == false */
201
202 /* Recalibrate scaling values */
203 soc_state.ccScalingAverage[stringNumber] = pTableSoc->averageSoc_perc[stringNumber] + ccOffset_perc;
204 soc_state.ccScalingMinimum[stringNumber] = pTableSoc->minimumSoc_perc[stringNumber] + ccOffset_perc;
205 soc_state.ccScalingMaximum[stringNumber] = pTableSoc->maximumSoc_perc[stringNumber] + ccOffset_perc;
206 }
207
208 /* Limit SOC values to [0.0, 100.0] */
209 SOC_CheckDatabaseSocPercentageLimits(pTableSoc, stringNumber);
210
211 /* Update non-volatile memory values */
212 SOC_UpdateNvmValues(pTableSoc, stringNumber);
213
214 (void)FRAM_WriteData(FRAM_BLOCK_ID_SOC);
215}

217static void SOC_CheckDatabaseSocPercentageLimits(DATA_BLOCK_SOC_s *pTableSoc, uint8_t stringNumber) {
218 FAS_ASSERT(pTableSoc != NULL_PTR);
219 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
220
221 if (pTableSoc->averageSoc_perc[stringNumber] > SOC_MAXIMUM_SOC_perc) {
222 pTableSoc->averageSoc_perc[stringNumber] = SOC_MAXIMUM_SOC_perc;
223 }
224 if (pTableSoc->averageSoc_perc[stringNumber] < SOC_MINIMUM_SOC_perc) {
225 pTableSoc->averageSoc_perc[stringNumber] = SOC_MINIMUM_SOC_perc;
226 }
227 if (pTableSoc->minimumSoc_perc[stringNumber] > SOC_MAXIMUM_SOC_perc) {
228 pTableSoc->minimumSoc_perc[stringNumber] = SOC_MAXIMUM_SOC_perc;
229 }
230 if (pTableSoc->minimumSoc_perc[stringNumber] < SOC_MINIMUM_SOC_perc) {
231 pTableSoc->minimumSoc_perc[stringNumber] = SOC_MINIMUM_SOC_perc;
232 }
233 if (pTableSoc->maximumSoc_perc[stringNumber] > SOC_MAXIMUM_SOC_perc) {
234 pTableSoc->maximumSoc_perc[stringNumber] = SOC_MAXIMUM_SOC_perc;
235 }
236 if (pTableSoc->maximumSoc_perc[stringNumber] < SOC_MINIMUM_SOC_perc) {
237 pTableSoc->maximumSoc_perc[stringNumber] = SOC_MINIMUM_SOC_perc;
238 }

253void SE_InitializeStateOfCharge(DATA_BLOCK_SOC_s *pSocValues, bool ccPresent, uint8_t stringNumber) {
254 FAS_ASSERT(pSocValues != NULL_PTR);
255 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
256 (void)DATA_READ_DATA(&soc_tableCurrentSensor);
257
258 (void)FRAM_ReadData(FRAM_BLOCK_ID_SOC);
259
260 if (ccPresent == true) {
261 soc_state.sensorCcUsed[stringNumber] = true;
262 soc_state.previousCurrentCountingValue_As[stringNumber] =
263 soc_tableCurrentSensor.currentCounter_As[stringNumber];
264
265 float_t scalingOffset_perc =
266 SOC_GetStringSocPercentageFromCharge((uint32_t)abs(soc_tableCurrentSensor.currentCounter_As[stringNumber]));
267
268 if (soc_tableCurrentSensor.currentCounter_As[stringNumber] < 0) {
269 scalingOffset_perc *= (-1.0f);
270 }
271
272#if BS_POSITIVE_DISCHARGE_CURRENT == false
273 scalingOffset_perc *= (-1.0f);
274#endif /* BS_POSITIVE_DISCHARGE_CURRENT == false */
275
276 soc_state.ccScalingAverage[stringNumber] = fram_soc.averageSoc_perc[stringNumber] + scalingOffset_perc;
277 soc_state.ccScalingMinimum[stringNumber] = fram_soc.minimumSoc_perc[stringNumber] + scalingOffset_perc;
278 soc_state.ccScalingMaximum[stringNumber] = fram_soc.maximumSoc_perc[stringNumber] + scalingOffset_perc;
279 } else {
280 soc_state.previousTimestamp[stringNumber] = soc_tableCurrentSensor.timestampCurrent[stringNumber];
281 soc_state.sensorCcUsed[stringNumber] = false;
282 }
283
284 pSocValues->averageSoc_perc[stringNumber] = fram_soc.averageSoc_perc[stringNumber];
285 pSocValues->minimumSoc_perc[stringNumber] = fram_soc.minimumSoc_perc[stringNumber];
286 pSocValues->maximumSoc_perc[stringNumber] = fram_soc.maximumSoc_perc[stringNumber];
287 pSocValues->dischargeThroughput_As[stringNumber] = fram_soc.dischargeThroughput_As[stringNumber];
288 pSocValues->chargeThroughput_As[stringNumber] = fram_soc.chargeThroughput_As[stringNumber];
289
290 SOC_CheckDatabaseSocPercentageLimits(pSocValues, stringNumber);
291
292 /* Alternatively, SOC can be initialized with {V,SOC} lookup table if available */
293 /* with the function SOC_Init_Lookup_Table() */
294
295 soc_state.socInitialized = true;
296}

298/* INCLUDE MARKER FOR THE DOCUMENTATION; DO NOT MOVE cc-documentation-start-include */
299void SE_CalculateStateOfCharge(DATA_BLOCK_SOC_s *pSocValues) {
300 /* INCLUDE MARKER FOR THE DOCUMENTATION; DO NOT MOVE cc-documentation-stop-include */
301 FAS_ASSERT(pSocValues != NULL_PTR);
302 bool continueFunction = true;
303
304 if (soc_state.socInitialized == false) {
305 /* Exit if SOC not initialized yet */
306 continueFunction = false;
307 }
308
309 if (continueFunction == true) {
310 /* Read current sensor entry for coulomb/current counting or CC recalibration */
311 (void)DATA_READ_DATA(&soc_tableCurrentSensor);
312
313 if (BMS_GetBatterySystemState() == BMS_AT_REST) {
314 /* Recalibrate SOC via LUT */
315 SOC_RecalibrateViaLookupTable(pSocValues);
316 } else {
317 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
318 if (soc_state.sensorCcUsed[s] == false) {
319 /* check if current measurement has been updated */
320 if (soc_state.previousTimestamp[s] != soc_tableCurrentSensor.timestampCurrent[s]) {
321 float_t timeStep_s =
322 ((float_t)(soc_tableCurrentSensor.timestampCurrent[s] - soc_state.previousTimestamp[s])) /
323 1000.0f;
324
325 if (timeStep_s > 0.0f) {
326 /* Current in charge direction negative means SOC increasing --> BAT naming, not ROB */
327
328 float_t deltaSOC_perc = (((float_t)soc_tableCurrentSensor.current_mA[s] * timeStep_s) /
329 SOC_STRING_CAPACITY_mAs) *
330 100.0f / 1000.0f; /* ((mA) * 1s) / 1As) * 100% */
331
332 float_t charge_As =
333 fabs((float_t)soc_tableCurrentSensor.current_mA[s] * timeStep_s / 1000.0f);
334
335#if BS_POSITIVE_DISCHARGE_CURRENT == false
336 deltaSOC_perc *= (-1.0f);
337#endif /* BS_POSITIVE_DISCHARGE_CURRENT == false */
338
339 pSocValues->averageSoc_perc[s] = pSocValues->averageSoc_perc[s] - deltaSOC_perc;
340 pSocValues->minimumSoc_perc[s] = pSocValues->minimumSoc_perc[s] - deltaSOC_perc;
341 pSocValues->maximumSoc_perc[s] = pSocValues->maximumSoc_perc[s] - deltaSOC_perc;
342 if (BMS_GetCurrentFlowDirection(soc_tableCurrentSensor.current_mA[s]) == BMS_CHARGING) {
343 pSocValues->chargeThroughput_As[s] = pSocValues->chargeThroughput_As[s] + charge_As;
344 } else {
345 /* When BMS_DISCHARGING and BMS_AT_REST add charge to dischargeThroughput*/
346 pSocValues->dischargeThroughput_As[s] = pSocValues->dischargeThroughput_As[s] +
347 charge_As;
348 }
349 /* Limit SOC calculation to 0% respectively 100% */
350 SOC_CheckDatabaseSocPercentageLimits(pSocValues, s);
351
352 /* Update values in non-volatile memory */
353 SOC_UpdateNvmValues(pSocValues, s);
354 }
355 soc_state.previousTimestamp[s] = soc_tableCurrentSensor.timestampCurrent[s];
356 } /* end check if current measurement has been updated */
357 /* update the variable for the next check */
358 } else {
359 /* check if cc measurement has been updated */
360 if (soc_state.previousTimestamp[s] != soc_tableCurrentSensor.timestampCurrentCounting[s]) {
361 float_t deltaSoc_perc =
362 ((float_t)soc_tableCurrentSensor.currentCounter_As[s] / SOC_STRING_CAPACITY_As) * 100.0f;
363
364 float_t chargeDifference_As = fabs(
365 (float_t)soc_tableCurrentSensor.currentCounter_As[s] -
366 soc_state.previousCurrentCountingValue_As[s]);
367
368#if BS_POSITIVE_DISCHARGE_CURRENT == false
369 deltaSoc_perc *= (-1.0f);
370#endif /* BS_POSITIVE_DISCHARGE_CURRENT == false */
371
372 pSocValues->averageSoc_perc[s] = soc_state.ccScalingAverage[s] - deltaSoc_perc;
373 pSocValues->minimumSoc_perc[s] = soc_state.ccScalingMinimum[s] - deltaSoc_perc;
374 pSocValues->maximumSoc_perc[s] = soc_state.ccScalingMaximum[s] - deltaSoc_perc;
375 if (BMS_GetCurrentFlowDirection(soc_tableCurrentSensor.current_mA[s]) == BMS_CHARGING) {
376 pSocValues->chargeThroughput_As[s] = pSocValues->chargeThroughput_As[s] +
377 chargeDifference_As;
378 } else {
379 /* When BMS_DISCHARGING and BMS_AT_REST add charge to dischargeThroughput*/
380 pSocValues->dischargeThroughput_As[s] = pSocValues->dischargeThroughput_As[s] +
381 chargeDifference_As;
382 }
383
384 /* Limit SOC values to [0.0, 100.0] */
385 SOC_CheckDatabaseSocPercentageLimits(pSocValues, s);
386
387 /* Update values in non-volatile memory */
388 SOC_UpdateNvmValues(pSocValues, s);
389 soc_state.previousCurrentCountingValue_As[s] = soc_tableCurrentSensor.currentCounter_As[s];
390 soc_state.previousTimestamp[s] = soc_tableCurrentSensor.timestampCurrentCounting[s];
391 } /* end check if cc measurement has been updated */
392 }
393 }
394 /* Update database and FRAM value */
395 (void)FRAM_WriteData(FRAM_BLOCK_ID_SOC);
396 }
397 }

400extern float_t SE_GetStateOfChargeFromVoltage(int16_t voltage_mV) {
401 float_t soc_perc = 0.50f;
402
403 /* Variables for interpolating LUT value */
404 uint16_t between_high = 0;
405 uint16_t between_low = 0;
406
407 /* Cell voltages are inserted in LUT in descending order -> start with 1 as we do not want to extrapolate. */
408 for (uint16_t i = 1u; i < bc_stateOfChargeLookupTableLength; i++) {
409 if (voltage_mV < bc_stateOfChargeLookupTable[i].voltage_mV) {
410 between_low = i + 1u;
411 between_high = i;
412 }
413 }
414
415 /* Interpolate between LUT values, but do not extrapolate LUT! */
416 if (!(((between_high == 0u) && (between_low == 0u)) || /* cell voltage > maximum LUT voltage */
417 (between_low >= bc_stateOfChargeLookupTableLength))) { /* cell voltage < minimum LUT voltage */
418 soc_perc = MATH_LinearInterpolation(
419 (float_t)bc_stateOfChargeLookupTable[between_low].voltage_mV,
420 bc_stateOfChargeLookupTable[between_low].value,
421 (float_t)bc_stateOfChargeLookupTable[between_high].voltage_mV,
422 bc_stateOfChargeLookupTable[between_high].value,
423 (float_t)voltage_mV);
424 } else if ((between_low >= bc_stateOfChargeLookupTableLength)) {
425 /* LUT SOE values are in descending order: cell voltage < minimum LUT voltage */
426 soc_perc = SOC_MINIMUM_SOC_perc;
427 } else {
428 /* cell voltage > maximum LUT voltage */
429 soc_perc = 100.0f;
430 }
431 return soc_perc;
432}
