71/**
72 * This structure contains all the variables relevant for the SOE.
73 */
74typedef struct {
75 bool soeInitialized; /*!< true if the initialization has passed, false otherwise */
76 bool sensorEcUsed[BS_NR_OF_STRINGS]; /*!< true if energy counting functionality of current sensor is used */
77 float_t ecScalingAverage[BS_NR_OF_STRINGS]; /*!< current sensor offset scaling for average SOE */
78 float_t ecScalingMinimum[BS_NR_OF_STRINGS]; /*!< current sensor offset scaling for minimum SOE */
79 float_t ecScalingMaximum[BS_NR_OF_STRINGS]; /*!< current sensor offset scaling for maximum SOE */
80 float_t chargeEnergyThroughput_Wh[BS_NR_OF_STRINGS]; /*!< inflow of energy */
81 float_t dischargeEnergyThroughput_Wh[BS_NR_OF_STRINGS]; /*!< outflow of energy */
82 float_t previousEnergyCount_Wh[BS_NR_OF_STRINGS];
83 uint32_t previousTimestamp[BS_NR_OF_STRINGS]; /*!< last used timestamp of current or energy counting value for SOE
84 estimation */
85} SOE_STATE_s;

178static float_t SOE_GetStringSoePercentageFromEnergy(uint32_t energy_Wh) {
179 float_t stringSoe_perc = 0.0f;
180 const float_t stringEnergy_Wh = (float_t)energy_Wh;
181 if (stringEnergy_Wh >= SOE_STRING_ENERGY_Wh) {
182 stringSoe_perc = MAXIMUM_SOE_PERC;
183 } else {
184 stringSoe_perc = UNIT_CONVERSION_FACTOR_100_FLOAT * (stringEnergy_Wh / SOE_STRING_ENERGY_Wh);
185 }
186 return stringSoe_perc;
187}
188
189static uint32_t SOE_GetStringEnergyFromSoePercentage(float_t stringSoe_perc) {
190 float_t energy_Wh = 0.0f;
191 if (stringSoe_perc >= MAXIMUM_SOE_PERC) {
192 energy_Wh = SOE_STRING_ENERGY_Wh;
193 } else if (stringSoe_perc <= MINIMUM_SOE_PERC) {
194 energy_Wh = MINIMUM_SOE_PERC;
195 } else {
196 energy_Wh = SOE_STRING_ENERGY_Wh * (stringSoe_perc / UNIT_CONVERSION_FACTOR_100_FLOAT);
197 }
198 return (uint32_t)energy_Wh;
199}

218static float_t SOE_GetFromVoltage(int16_t voltage_mV) {
219 float_t soe_perc = 50.0f;
220 /* Variables for interpolating LUT value */
221 uint16_t between_high = 0;
222 uint16_t between_low = 0;
223
224 /* Cell voltages are inserted in LUT in descending order -> start with 1 as we do not want to extrapolate. */
225 for (uint16_t i = 1u; i < bc_stateOfEnergyLookupTableLength; i++) {
226 if (voltage_mV < bc_stateOfEnergyLookupTable[i].voltage_mV) {
227 between_low = i + 1u;
228 between_high = i;
229 }
230 }
231
232 /* Interpolate between LUT values, but do not extrapolate LUT! */
233 if (!(((between_high == 0u) && (between_low == 0u)) || /* cell voltage > maximum LUT voltage */
234 (between_low >= bc_stateOfEnergyLookupTableLength))) { /* cell voltage < minimum LUT voltage */
235 soe_perc = MATH_LinearInterpolation(
236 (float_t)bc_stateOfEnergyLookupTable[between_low].voltage_mV,
237 bc_stateOfEnergyLookupTable[between_low].value,
238 (float_t)bc_stateOfEnergyLookupTable[between_high].voltage_mV,
239 bc_stateOfEnergyLookupTable[between_high].value,
240 (float_t)voltage_mV);
241 } else if ((between_low >= bc_stateOfEnergyLookupTableLength)) {
242 /* LUT SOE values are in descending order: cell voltage < minimum LUT voltage */
243 soe_perc = MINIMUM_SOE_PERC;
244 } else {
245 /* cell voltage > maximum LUT voltage */
246 soe_perc = MAXIMUM_SOE_PERC;
247 }
248 return soe_perc;
249}

251static void SOE_SetValue(
252 DATA_BLOCK_SOE_s *pSoeValues,
253 float_t soeMinimumValue_perc,
254 float_t soeMaximumValue_perc,
255 float_t soeAverageValue_perc,
256 uint8_t stringNumber) {
257 FAS_ASSERT(pSoeValues != NULL_PTR);
258
259 /* Update FRAM value */
260 fram_soe.averageSoe_perc[stringNumber] = soeAverageValue_perc;
261 fram_soe.minimumSoe_perc[stringNumber] = soeMinimumValue_perc;
262 fram_soe.maximumSoe_perc[stringNumber] = soeMaximumValue_perc;
263
264 /* Update database values */
265 pSoeValues->averageSoe_perc[stringNumber] = soeAverageValue_perc;
266 pSoeValues->minimumSoe_perc[stringNumber] = soeMinimumValue_perc;
267 pSoeValues->maximumSoe_perc[stringNumber] = soeMaximumValue_perc;
268
269 pSoeValues->maximumSoe_Wh[stringNumber] = SOE_GetStringEnergyFromSoePercentage(soeMaximumValue_perc);
270 pSoeValues->averageSoe_Wh[stringNumber] = SOE_GetStringEnergyFromSoePercentage(soeAverageValue_perc);
271 pSoeValues->minimumSoe_Wh[stringNumber] = SOE_GetStringEnergyFromSoePercentage(soeMinimumValue_perc);
272
273 /* Calculate scaling values depending on EC counting value and current SOE */
274 if (soe_state.sensorEcUsed[stringNumber] == true) {
275 (void)DATA_READ_DATA(&soe_tableCurrentSensor);
276
277 float_t ecOffset =
278 SOE_GetStringSoePercentageFromEnergy((uint32_t)abs(soe_tableCurrentSensor.energyCounter_Wh[stringNumber]));
279
280 if (soe_tableCurrentSensor.energyCounter_Wh[stringNumber] < 0) {
281 ecOffset *= (-1.0f);
282 }
283
284#if BS_POSITIVE_DISCHARGE_CURRENT == false
285 ecOffset *= (-1.0f); /* negate calculated delta SOE in perc */
286
287#endif /* BS_POSITIVE_DISCHARGE_CURRENT == false */
288
289 soe_state.ecScalingAverage[stringNumber] = fram_soe.averageSoe_perc[stringNumber] + ecOffset;
290 soe_state.ecScalingMinimum[stringNumber] = fram_soe.minimumSoe_perc[stringNumber] + ecOffset;
291 soe_state.ecScalingMaximum[stringNumber] = fram_soe.maximumSoe_perc[stringNumber] + ecOffset;
292 }

296 FAS_ASSERT(pTableSoe != NULL_PTR);
297 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
298
299 if (pTableSoe->averageSoe_perc[stringNumber] > MAXIMUM_SOE_PERC) {

305 if (pTableSoe->minimumSoe_perc[stringNumber] > MAXIMUM_SOE_PERC) {

310 }
311 if (pTableSoe->maximumSoe_perc[stringNumber] > MAXIMUM_SOE_PERC) {

321extern void SE_InitializeStateOfEnergy(DATA_BLOCK_SOE_s *pSoeValues, bool ec_present, uint8_t stringNumber) {
322 FAS_ASSERT(pSoeValues != NULL_PTR);
323 FAS_ASSERT(stringNumber < BS_NR_OF_STRINGS);
324 (void)FRAM_ReadData(FRAM_BLOCK_ID_SOE);
325
326 pSoeValues->averageSoe_perc[stringNumber] = fram_soe.averageSoe_perc[stringNumber];
327 pSoeValues->minimumSoe_perc[stringNumber] = fram_soe.minimumSoe_perc[stringNumber];
328 pSoeValues->maximumSoe_perc[stringNumber] = fram_soe.maximumSoe_perc[stringNumber];
329 pSoeValues->chargeEnergyThroughput_Wh[stringNumber] = fram_soe.chargeEnergyThroughput_Wh[stringNumber];
330 pSoeValues->dischargeEnergyThroughput_Wh[stringNumber] = fram_soe.dischargeEnergyThroughput_Wh[stringNumber];
331
332 /* Limit SOE values [0.0f, 100.0f] */
333 SOE_CheckDatabaseSoePercentageLimits(pSoeValues, stringNumber);
334
335 /* Calculate string energy in Wh */
336 pSoeValues->maximumSoe_Wh[stringNumber] =
337 SOE_GetStringEnergyFromSoePercentage(pSoeValues->maximumSoe_perc[stringNumber]);
338 pSoeValues->minimumSoe_Wh[stringNumber] =
339 SOE_GetStringEnergyFromSoePercentage(pSoeValues->minimumSoe_perc[stringNumber]);
340 pSoeValues->averageSoe_Wh[stringNumber] =
341 SOE_GetStringEnergyFromSoePercentage(pSoeValues->averageSoe_perc[stringNumber]);
342
343 if (ec_present == true) {
344 (void)DATA_READ_DATA(&soe_tableCurrentSensor);
345 soe_state.sensorEcUsed[stringNumber] = true;
346
347 /* Set scaling values */
348 float_t ecOffset =
349 SOE_GetStringSoePercentageFromEnergy((uint32_t)abs(soe_tableCurrentSensor.energyCounter_Wh[stringNumber]));
350
351 if (soe_tableCurrentSensor.energyCounter_Wh[stringNumber] < 0) {
352 ecOffset *= (-1.0f);
353 }
354
355#if BS_POSITIVE_DISCHARGE_CURRENT == false
356 ecOffset *= (-1.0f); /* negate calculated delta SOE in perc */
357
358#endif /* BS_POSITIVE_DISCHARGE_CURRENT == false */
359
360 soe_state.ecScalingMinimum[stringNumber] = fram_soe.minimumSoe_perc[stringNumber] + ecOffset;
361 soe_state.ecScalingMaximum[stringNumber] = fram_soe.maximumSoe_perc[stringNumber] + ecOffset;
362 soe_state.ecScalingAverage[stringNumber] = fram_soe.averageSoe_perc[stringNumber] + ecOffset;
363 }
364 soe_state.soeInitialized = true;
365}

367void SE_CalculateStateOfEnergy(DATA_BLOCK_SOE_s *pSoeValues) {
368 FAS_ASSERT(pSoeValues != NULL_PTR);
369 bool continueFunction = true;
370 if (soe_state.soeInitialized == false) {
371 /* Exit if SOE not initialized yet */
372 continueFunction = false;
373 }
374
375 if (continueFunction == true) {
376 /* Use energy counting/integrate */
377 (void)DATA_READ_DATA(&soe_tableCurrentSensor);
378
379 if (BMS_GetBatterySystemState() == BMS_AT_REST) {
380 /* Recalibrate SOE via LUT */
381 SOE_RecalibrateViaLookupTable(pSoeValues);
382 } else {
383 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
384 if (soe_state.sensorEcUsed[s] == false) {
385 /* no energy counting activated -> manually integrate energy */
386 uint32_t timestamp = soe_tableCurrentSensor.timestampCurrent[s];
387 uint32_t previous_timestamp = soe_tableCurrentSensor.previousTimestampCurrent[s];
388
389 /* check if current measurement has been updated */
390 if (soe_state.previousTimestamp[s] != timestamp) {
391 float_t time_step_s = (((float_t)timestamp - (float_t)previous_timestamp)) / 1000.0f;
392 if (time_step_s > 0.0f) {
393 /* Current in charge direction negative means SOE increasing --> BAT naming, not ROB */
394 float_t deltaSOE_Wh =
395 ((((float_t)soe_tableCurrentSensor.current_mA[s] / 1000.0f) * /* convert to A */
396 ((float_t)soe_tableCurrentSensor.highVoltage_mV[s][0] / 1000.0f)) / /* convert to V */
397 time_step_s) / /* unit: s */
398 3600.0f; /* convert Ws -> Wh */
399
400#if BS_POSITIVE_DISCHARGE_CURRENT == false
401 /* negate calculated delta SOE_Wh */
402 deltaSOE_Wh *= (-1.0f);
403#endif /* BS_POSITIVE_DISCHARGE_CURRENT == false */
404
405 pSoeValues->averageSoe_Wh[s] -= (uint32_t)deltaSOE_Wh;
406 pSoeValues->minimumSoe_Wh[s] -= (uint32_t)deltaSOE_Wh;
407 pSoeValues->maximumSoe_Wh[s] -= (uint32_t)deltaSOE_Wh;
408
409 if (BMS_GetCurrentFlowDirection(soe_tableCurrentSensor.current_mA[s]) == BMS_CHARGING) {
410 pSoeValues->chargeEnergyThroughput_Wh[s] = pSoeValues->chargeEnergyThroughput_Wh[s] +
411 fabs(deltaSOE_Wh);
412 } else {
413 /* When BMS_DISCHARGING and BMS_AT_REST add charge to dischargeThroughput*/
414 pSoeValues->dischargeEnergyThroughput_Wh[s] =
415 pSoeValues->dischargeEnergyThroughput_Wh[s] + fabs(deltaSOE_Wh);
416 }
417
418 pSoeValues->averageSoe_perc[s] =
419 SOE_GetStringSoePercentageFromEnergy(pSoeValues->averageSoe_Wh[s]);
420 pSoeValues->minimumSoe_perc[s] =
421 SOE_GetStringSoePercentageFromEnergy(pSoeValues->minimumSoe_Wh[s]);
422 pSoeValues->maximumSoe_perc[s] =
423 SOE_GetStringSoePercentageFromEnergy(pSoeValues->maximumSoe_Wh[s]);
424
425 /* update timestamp SOE state variable for next iteration */
426 soe_state.previousTimestamp[s] = timestamp;
427 }
428 } /* end check if current measurement has been updated */
429 } else {
430 /* check if ec measurement has been updated */
431 if (soe_state.previousTimestamp[s] != soe_tableCurrentSensor.timestampEnergyCounting[s]) {
432 /* Calculate SOE value with current sensor EC value */
433 float_t deltaSoe_perc =
434 (((float_t)soe_tableCurrentSensor.energyCounter_Wh[s] / SOE_STRING_ENERGY_Wh) *
435 UNIT_CONVERSION_FACTOR_100_FLOAT);
436
437 float_t energyDifference_Wh = fabs(
438 (float_t)soe_tableCurrentSensor.energyCounter_Wh[s] - soe_state.previousEnergyCount_Wh[s]);
439
440#if BS_POSITIVE_DISCHARGE_CURRENT == false
441 /* negate calculated delta SOE_perc */
442 deltaSoe_perc *= (-1.0f);
443#endif
444 /* Apply EC scaling offset to get actual string energy */
445 pSoeValues->averageSoe_perc[s] = soe_state.ecScalingAverage[s] - deltaSoe_perc;
446 pSoeValues->minimumSoe_perc[s] = soe_state.ecScalingMinimum[s] - deltaSoe_perc;
447 pSoeValues->maximumSoe_perc[s] = soe_state.ecScalingMaximum[s] - deltaSoe_perc;
448 if (BMS_GetCurrentFlowDirection(soe_tableCurrentSensor.current_mA[s]) == BMS_CHARGING) {
449 pSoeValues->chargeEnergyThroughput_Wh[s] = pSoeValues->chargeEnergyThroughput_Wh[s] +
450 energyDifference_Wh;
451 } else {
452 /* When BMS_DISCHARGING and BMS_AT_REST add charge to dischargeThroughput*/
453 pSoeValues->dischargeEnergyThroughput_Wh[s] = pSoeValues->dischargeEnergyThroughput_Wh[s] +
454 energyDifference_Wh;
455 }
456 /* Limit SOE values to [0.0, 100.0] */
457 SOE_CheckDatabaseSoePercentageLimits(pSoeValues, s);
458
459 /* Calculate new Wh values */
460 pSoeValues->maximumSoe_Wh[s] =
461 SOE_GetStringEnergyFromSoePercentage(pSoeValues->maximumSoe_perc[s]);
462 pSoeValues->averageSoe_Wh[s] =
463 SOE_GetStringEnergyFromSoePercentage(pSoeValues->averageSoe_perc[s]);
464 pSoeValues->minimumSoe_Wh[s] =
465 SOE_GetStringEnergyFromSoePercentage(pSoeValues->minimumSoe_perc[s]);
466
467 /* Update timestamp for next iteration */
468 soe_state.previousEnergyCount_Wh[s] = soe_tableCurrentSensor.energyCounter_Wh[s];
469 soe_state.previousTimestamp[s] = soe_tableCurrentSensor.timestampEnergyCounting[s];
470 }
471 }
472
473 fram_soe.averageSoe_perc[s] = pSoeValues->averageSoe_perc[s];
474 fram_soe.minimumSoe_perc[s] = pSoeValues->minimumSoe_perc[s];
475 fram_soe.maximumSoe_perc[s] = pSoeValues->maximumSoe_perc[s];
476 fram_soe.chargeEnergyThroughput_Wh[s] = pSoeValues->chargeEnergyThroughput_Wh[s];
477 fram_soe.dischargeEnergyThroughput_Wh[s] = pSoeValues->dischargeEnergyThroughput_Wh[s];
478 }
479
480 /* Update database and FRAM value */
481 (void)FRAM_WriteData(FRAM_BLOCK_ID_SOE);
482 }
483 }
