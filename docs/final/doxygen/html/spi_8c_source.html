130/*========== Static Function Implementations ================================*/
131static uint8_t SPI_GetHardwareChipSelectPin(uint8_t chipSelectPin) {
132 FAS_ASSERT(chipSelectPin < SPI_MAX_NUMBER_HW_CS);
133
134 uint8_t mappedHardwareChipSelectPin = SPI_HARDWARE_CHIP_SELECT_DISABLE_ALL;
135 switch (chipSelectPin) {
136 case SPI_HARDWARE_CHIP_SELECT_PIN_0:
137 mappedHardwareChipSelectPin = SPI_HARDWARE_CHIP_SELECT_0_ACTIVE;
138 break;
139 case SPI_HARDWARE_CHIP_SELECT_PIN_1:
140 mappedHardwareChipSelectPin = SPI_HARDWARE_CHIP_SELECT_1_ACTIVE;
141 break;
142 case SPI_HARDWARE_CHIP_SELECT_PIN_2:
143 mappedHardwareChipSelectPin = SPI_HARDWARE_CHIP_SELECT_2_ACTIVE;
144 break;
145 case SPI_HARDWARE_CHIP_SELECT_PIN_3:
146 mappedHardwareChipSelectPin = SPI_HARDWARE_CHIP_SELECT_3_ACTIVE;
147 break;
148 case SPI_HARDWARE_CHIP_SELECT_PIN_4:
149 mappedHardwareChipSelectPin = SPI_HARDWARE_CHIP_SELECT_4_ACTIVE;
150 break;
151 case SPI_HARDWARE_CHIP_SELECT_PIN_5:
152 mappedHardwareChipSelectPin = SPI_HARDWARE_CHIP_SELECT_5_ACTIVE;
153 break;
154 default: /* something went really wrong */
155 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
156 break; /* LCOV_EXCL_LINE */
157 }
158 return mappedHardwareChipSelectPin;
159}
160static uint8_t SPI_GetChipSelectPin(SPI_CHIP_SELECT_TYPE_e chipSelectType, uint32_t chipSelectPin) {
161 FAS_ASSERT((chipSelectType == SPI_CHIP_SELECT_HARDWARE) || (chipSelectType == SPI_CHIP_SELECT_SOFTWARE));
162 FAS_ASSERT(chipSelectPin <= MCU_LARGEST_PIN_NUMBER);
163
164 uint8_t mappedChipSelectPin = (uint8_t)chipSelectPin;
165 /* for software chips select, there is nothing more to sanitize or map, therefore it is just needed to further
166 analyze the hardware chip select setting */
167 if (chipSelectType == SPI_CHIP_SELECT_HARDWARE) {
168 mappedChipSelectPin = SPI_GetHardwareChipSelectPin(mappedChipSelectPin);
169 }
170
171 return mappedChipSelectPin;
172}

275extern STD_RETURN_TYPE_e SPI_TransmitData(SPI_INTERFACE_CONFIG_s *pSpiInterface, uint16 *pTxBuff, uint32 frameLength) {
276 FAS_ASSERT(pSpiInterface != NULL_PTR);
277 FAS_ASSERT(pTxBuff != NULL_PTR);
278 FAS_ASSERT(frameLength > 0u);
279 STD_RETURN_TYPE_e retval = STD_NOT_OK;
280
281 /* Lock SPI hardware to prevent concurrent read/write commands */
282 if (STD_OK == SPI_Lock(SPI_GetSpiIndex(pSpiInterface->pNode))) {
283 pSpiInterface->pNode->GCR1 |= DMA_SPI_ENABLE_BIT;
284
285 /** SW Chip Select */
286 if (pSpiInterface->csType == SPI_CHIP_SELECT_SOFTWARE) {
287 /** Set SPI Chip Select pins as GIOs */
288 pSpiInterface->pNode->PC0 &= SPI_PC0_CLEAR_HW_CS_MASK;
289 /** Activate Chip Select */
290 IO_PinReset(pSpiInterface->pGioPort, pSpiInterface->csPin);
291 }
292 /** HW Chip Select */
293 if (pSpiInterface->csType == SPI_CHIP_SELECT_HARDWARE) {
294 /**
295 * Activate HW Chip Select according to bitmask register CSNR
296 * by setting pins as SPI functional pins
297 */
298 /** First deactivate all HW Chip Selects */
299 pSpiInterface->pNode->PC0 &= SPI_PC0_CLEAR_HW_CS_MASK;
300 for (uint8_t csNumber = 0u; csNumber < SPI_MAX_NUMBER_HW_CS; csNumber++) {
301 if (((pSpiInterface->pConfig->CSNR >> csNumber) & 0x1u) == 0u) {
302 /** Bitmask = 0 --> HW CS active
303 * --> write to PC0 to set pin as SPI pin (and not GIO)
304 */
305 pSpiInterface->pNode->PC0 |= (uint32_t)1u << csNumber;
306 }
307 }
308 }
309 uint32_t spiRetval = spiTransmitData(pSpiInterface->pNode, pSpiInterface->pConfig, frameLength, pTxBuff);
310 /** SW Chip Select */
311 if (pSpiInterface->csType == SPI_CHIP_SELECT_SOFTWARE) {
312 /** Deactivate Chip Select */
313 IO_PinSet(pSpiInterface->pGioPort, pSpiInterface->csPin);
314 }
315
316 /* Unlock SPI hardware */
317 SPI_Unlock(SPI_GetSpiIndex(pSpiInterface->pNode));
318
319 /* Transmission successful */
320 if ((spiRetval & SPI_FLAG_REGISTER_TRANSMISSION_ERRORS) == 0u) {
321 retval = STD_OK;
322 }
323 }
324 return retval;
325}

398extern STD_RETURN_TYPE_e SPI_TransmitReceiveDataDma(
399 SPI_INTERFACE_CONFIG_s *pSpiInterface,
400 uint16_t *pTxBuff,
401 uint16_t *pRxBuff,
402 uint32_t frameLength) {
403 FAS_ASSERT(frameLength > 2u);
404 FAS_ASSERT(pSpiInterface != NULL_PTR);
405 FAS_ASSERT(pTxBuff != NULL_PTR);
406 FAS_ASSERT(pRxBuff != NULL_PTR);
407 /** SPI over DMA currently only compatible with HW Chip Select */
408 FAS_ASSERT(pSpiInterface->csType == SPI_CHIP_SELECT_HARDWARE);
409 const uint8_t spiIndex = SPI_GetSpiIndex(pSpiInterface->pNode);
410 FAS_ASSERT(spiIndex < spi_nrBusyFlags);
411
412 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
413
414 OS_EnterTaskCritical();
415 /* Lock SPI hardware to prevent concurrent read/write commands */
416 if (spi_busyFlags[spiIndex] == SPI_IDLE) {
417 spi_busyFlags[spiIndex] = SPI_BUSY;
418
419 /* Check that not SPI transmission over DMA is taking place */
420 if ((pSpiInterface->pNode->INT0 & DMA_REQUEST_ENABLE_BIT) == 0x0) {
421 /**
422 * Activate HW Chip Select according to bitmask register CSNR
423 * by setting pins as SPI functional pins
424 */
425 /** First deactivate all HW Chip Selects */
426 pSpiInterface->pNode->PC0 &= SPI_PC0_CLEAR_HW_CS_MASK;
427 for (uint8_t csNumber = 0u; csNumber < SPI_MAX_NUMBER_HW_CS; csNumber++) {
428 if (((pSpiInterface->pConfig->CSNR >> csNumber) & 0x1u) == 0u) {
429 /** Bitmask = 0 --> HW CS active
430 * --> write to PC0 to set pin as SPI pin (and not GIO)
431 */
432 pSpiInterface->pNode->PC0 |= (uint32_t)1u << csNumber;
433 }
434 }
435
436 /* The upper 16 bits will be written in the SPI DAT1 register where they serve as configuration */
437 uint32 Chip_Select_Hold = 0u;
438 if (pSpiInterface->pConfig->CS_HOLD == TRUE) {
439 Chip_Select_Hold = SPI_CSHOLD_BIT;
440 } else {
441 Chip_Select_Hold = 0U;
442 }
443 uint32 WDelay = 0u;
444 if (pSpiInterface->pConfig->WDEL == TRUE) {
445 WDelay = SPI_WDEL_BIT;
446 } else {
447 WDelay = 0U;
448 }
449 SPIDATAFMT_t DataFormat = pSpiInterface->pConfig->DFSEL;
450 uint8 ChipSelect = pSpiInterface->pConfig->CSNR;
451
452 /* Go to privilege mode to write DMA config registers */
453 (void)FSYS_RaisePrivilege();
454
455 spi_txLastWord[spiIndex] = pTxBuff[frameLength - 1u];
456 spi_txLastWord[spiIndex] |= ((uint32)DataFormat << SPI_DATA_FORMAT_FIELD_POSITION) |
457 ((uint32)ChipSelect << SPI_HARDWARE_CHIP_SELECT_FIELD_POSITION) | (WDelay);
458
459 /* Set Tx buffer address */
460 dmaRAMREG->PCP[(dmaChannel_t)dma_spiDmaChannels[spiIndex].txChannel].ISADDR =
461 (uint32_t)(&pTxBuff[1u]); /* First word sent manually to write configuration in SPIDAT1 register */
462 /**
463 * Set number of Tx words to send
464 * Last word sent in ISR to set CSHOLD = 0
465 */
466 dmaRAMREG->PCP[(dmaChannel_t)dma_spiDmaChannels[spiIndex].txChannel].ITCOUNT =
467 ((frameLength - 2u) << 16U) | 1U; /* Last word sent manually to write CSHOLD in SPIDAT1 register */
468
469 /* Set Rx buffer address */
470 dmaRAMREG->PCP[(dmaChannel_t)dma_spiDmaChannels[spiIndex].rxChannel].IDADDR = (uint32_t)pRxBuff;
471 /* Set number of Rx words to receive */
472 dmaRAMREG->PCP[(dmaChannel_t)dma_spiDmaChannels[spiIndex].rxChannel].ITCOUNT = (frameLength << 16U) | 1U;
473
474 /* Re-enable channels; because auto-init is disabled */
475 /* Disable otherwise transmission is constantly ongoing */
476 dmaSetChEnable((dmaChannel_t)dma_spiDmaChannels[spiIndex].txChannel, (dmaTriggerType_t)DMA_HW);
477 dmaSetChEnable((dmaChannel_t)dma_spiDmaChannels[spiIndex].rxChannel, (dmaTriggerType_t)DMA_HW);
478
479 /* DMA config registers written, leave privilege mode */
480 FSYS_SwitchToUserMode();
481
482 /* DMA_REQ_Enable */
483 /* Starts DMA requests if SPIEN is also set to 1 */
484 pSpiInterface->pNode->GCR1 |= DMA_SPI_ENABLE_BIT;
485 uint32_t txBuffer = pTxBuff[0u];
486 txBuffer |= ((uint32)DataFormat << 24U) | ((uint32)ChipSelect << 16U) | (WDelay) | (Chip_Select_Hold);
487 /**
488 * Send first word without DMA because when writing config to DAT1
489 * the HW CS pin are asserted immediately, even if SPIEN bit in GCR1 is 0.
490 * The C2TDELAY is then taken into account before the transmission.
491 */
492 pSpiInterface->pNode->DAT1 = txBuffer;
493 uint32_t timeoutIterations = SPI_TX_EMPTY_TIMEOUT_ITERATIONS;
494 while (((pSpiInterface->pNode->FLG & (uint32)((uint32_t)1u << SPI_TX_BUFFER_EMPTY_FLAG_POSITION)) == 0u) &&
495 (timeoutIterations > 0u)) {
496 timeoutIterations--;
497 }
498 pSpiInterface->pNode->INT0 |= DMA_REQUEST_ENABLE_BIT;
499
500 retVal = STD_OK;
501 }
502 }
503 OS_ExitTaskCritical();
504
505 return retVal;
506}

508extern STD_RETURN_TYPE_e SPI_Lock(uint8_t spi) {
509 /* AXIVION Routine Generic-MissingParameterAssert: spi: parameter accepts whole range */
510
511 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
512
513 OS_EnterTaskCritical();
514 if ((spi < spi_nrBusyFlags) && (spi_busyFlags[spi] == SPI_IDLE)) {
515 spi_busyFlags[spi] = SPI_BUSY;
516 retVal = STD_OK;
517 } else {
518 retVal = STD_NOT_OK;
519 }
520 OS_ExitTaskCritical();
521
522 return retVal;
523}

532 OS_ExitTaskCritical();

535extern void SPI_SetFunctional(spiBASE_t *pNode, enum spiPinSelect bit, bool hardwareControlled) {
536 FAS_ASSERT(pNode != NULL_PTR);
537 FAS_ASSERT(bit <= (enum spiPinSelect)MCU_LARGEST_PIN_NUMBER);
538
539 /* retrieve current configuration */
540 spi_config_reg_t configRegisterBuffer = {0};
541 if (pNode == spiREG1) {
542 spi1GetConfigValue(&configRegisterBuffer, CurrentValue);
543 } else if (pNode == spiREG2) {
544 spi2GetConfigValue(&configRegisterBuffer, CurrentValue);
545 } else if (pNode == spiREG3) {
546 spi3GetConfigValue(&configRegisterBuffer, CurrentValue);
547 } else if (pNode == spiREG4) {
548 spi4GetConfigValue(&configRegisterBuffer, CurrentValue);
549 } else if (pNode == spiREG5) {
550 spi5GetConfigValue(&configRegisterBuffer, CurrentValue);
551 } else {
552 /* invalid SPI node */
553 FAS_ASSERT(FAS_TRAP);
554 }
555 uint32_t newPc0 = configRegisterBuffer.CONFIG_PC0;
556
557 if (hardwareControlled == false) {
558 /* bit has to be cleared */

569extern STD_RETURN_TYPE_e SPI_SlaveSetReceiveDataDma(
570 SPI_INTERFACE_CONFIG_s *pSpiInterface,
571 uint16_t *pTxBuff,
572 uint16_t *pRxBuff,
573 uint32_t frameLength) {
574 FAS_ASSERT(pSpiInterface != NULL_PTR);
575 FAS_ASSERT(pTxBuff != NULL_PTR);
576 FAS_ASSERT(pRxBuff != NULL_PTR);
577 FAS_ASSERT(frameLength > 0u);
578 /** SPI receive works only with HW Chip Select */
579 FAS_ASSERT(pSpiInterface->csType == SPI_CHIP_SELECT_HARDWARE);
580
581 STD_RETURN_TYPE_e retVal = STD_OK;
582
583 OS_EnterTaskCritical();
584 /* Go to privilege mode to write DMA config registers */
585 (void)FSYS_RaisePrivilege();
586
587 /* DMA_REQ Disable */
588 pSpiInterface->pNode->INT0 &= ~DMA_REQUEST_ENABLE_BIT;
589 pSpiInterface->pNode->GCR1 &= ~DMA_SPI_ENABLE_BIT;
590
591 /* Write FMT configuration in DAT1; as SPI is configured as slave, this does not provoke a transmission */
592 SPIDATAFMT_t DataFormat = pSpiInterface->pConfig->DFSEL;
593 pSpiInterface->pNode->DAT1 |= ((uint32)DataFormat << SPI_DATA_FORMAT_FIELD_POSITION);
594
595 /* Set Tx buffer address */
596 dmaRAMREG->PCP[(dmaChannel_t)dma_spiDmaChannels[SPI_GetSpiIndex(pSpiInterface->pNode)].txChannel].ISADDR =
597 (uint32_t)pTxBuff;
598 /* Set number of Tx bytes to send */
599 dmaRAMREG->PCP[(dmaChannel_t)dma_spiDmaChannels[SPI_GetSpiIndex(pSpiInterface->pNode)].txChannel].ITCOUNT =
600 (frameLength << 16U) | 1U;
601
602 /* Set Rx buffer address */
603 dmaRAMREG->PCP[(dmaChannel_t)dma_spiDmaChannels[SPI_GetSpiIndex(pSpiInterface->pNode)].rxChannel].IDADDR =
604 (uint32_t)pRxBuff;
605 /* Set number of Rx bytes to receive */
606 dmaRAMREG->PCP[(dmaChannel_t)dma_spiDmaChannels[SPI_GetSpiIndex(pSpiInterface->pNode)].rxChannel].ITCOUNT =
607 (frameLength << 16U) | 1U;
608
609 /* Re-enable channels; because auto-init is disabled */
610 /* Disable otherwise transmission is constantly on going */
611 dmaSetChEnable(
612 (dmaChannel_t)dma_spiDmaChannels[SPI_GetSpiIndex(pSpiInterface->pNode)].txChannel, (dmaTriggerType_t)DMA_HW);
613 dmaSetChEnable(
614 (dmaChannel_t)dma_spiDmaChannels[SPI_GetSpiIndex(pSpiInterface->pNode)].rxChannel, (dmaTriggerType_t)DMA_HW);
615
616 /* DMA config registers written, leave privilege mode */
617 FSYS_SwitchToUserMode();
618
619 OS_ExitTaskCritical();
620
621 /* Activate chip selects according to bitmask register CSNR */
622 for (uint8_t csNumber = 0u; csNumber < SPI_MAX_NUMBER_HW_CS; csNumber++) {
623 if (((pSpiInterface->pConfig->CSNR >> csNumber) & 0x1u) == 0u) {
624 /* Bitmask = 0 --> HW CS active --> set pin as SPI functional pin */
625 pSpiInterface->pNode->PC0 |= (uint32_t)1u << csNumber;
626 }
627 }
628
629 /* DMA_REQ Enable */
630 /* Starts DMA requests if SPIEN is also set to 1 */
631 pSpiInterface->pNode->GCR1 |= DMA_SPI_ENABLE_BIT;
632 pSpiInterface->pNode->INT0 |= DMA_REQUEST_ENABLE_BIT;
633
634 return retVal;
635}
