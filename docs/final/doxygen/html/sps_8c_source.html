94/** SPI buffer:
95 * Tx part used to write to contactor output control register
96 * Rx part used to read answer from former read register command
97 *
98 * When a read register command was issued on the MOSI line,
99 * the answers comes during the next command (read or write)
100 * on the MISO line
101 * (data sheet figure 6 page 9 in data sheet Rev. 2 - 11 September 2019)
102 *
103 * In the SPS driver, there are always two communications with the SPS IC
104 * each time the state machine is triggered:
105 * - first a register access (read or write)
106 * - then a write access to the channel output control register
107 * This way, the channel outputs are switched each time the state machine
108 * is triggered, and during the second access, if a read access was done
109 * before, the answer comes during the second access.
110 * @{
111 */
112static uint16_t sps_spiTxWriteToChannelChannelControlRegister[SPS_SPI_BUFFERSIZE] = {0};
113static uint16_t sps_spiRxReadAnswerDuringChannelControl[SPS_SPI_BUFFERSIZE] = {0};
114/**@}*/

149/**
150 * @brief Transmit all assembled messages for one timeframe
151 * @details This function leverages the fact that the SPS are connected in a
152 * "daisy-chain". This way the chain acts as a large shift register.
153 * When sending a message into the chain it is clocked in until every
154 * chip in the chain as its message. Then chip-select is released and
155 * the messages parsed. The answer is then assembled by the chip in
156 * its register and clocked out and received by the master with the
157 * next transmission.
158 *
159 * This function sends out a general command to each chip that has
160 * been assembled with #SPS_SetCommandTxBuffer(). The clocked out
161 * information that is received of this transaction is stored in a
162 * separate buffer and is the answer from the last transaction of the
163 * timeframe before the current one.
164 *
165 * After this transaction the answer to the general command is
166 * retrieved and the transmission is used to clock in an update of the
167 * state of the contactor channels. This update has been assembled
168 * before the call of this function with the function
169 * #SPS_SetContactorsTxBuffer().
170 * @return returns STD_OK if both transactions have been successful
171 */
172static STD_RETURN_TYPE_e SPS_Transmit(void);

273static void SPS_GlobalRegisterRead(
274 const uint16_t address,
275 const SPS_READ_TYPE_e controlOrDiagnostic,
276 uint16_t *pSpiTxBuffer) {
277 FAS_ASSERT(pSpiTxBuffer != NULL_PTR);
278 if (controlOrDiagnostic == SPS_READ_DIAGNOSTIC_REGISTER) {
279 for (uint8_t i = 0u; i < SPS_SPI_BUFFERSIZE; i++) {
280 /* R/W bit = 0 to read */
281 pSpiTxBuffer[i] = (address << SPS_ADDRESS_BIT_START); /* Register address */
282 }
283 } else if (controlOrDiagnostic == SPS_READ_CONTROL_REGISTER) {
284 for (uint8_t i = 0u; i < SPS_SPI_BUFFERSIZE; i++) {
285 pSpiTxBuffer[i] = (address << SPS_ADDRESS_BIT_START) | /* Register address */
286 ((uint16_t)1u << SPS_DIAG_CTRL_BIT_POSITION); /* Select diagnostic register */
287 }
288 } else {
289 /* Invalid access type */
290 FAS_ASSERT(FAS_TRAP);
291 }
292}

294static void SPS_GlobalRegisterWrite(const uint16_t address, uint8_t writeData, uint16_t *pSpiTxBuffer) {
295 FAS_ASSERT(pSpiTxBuffer != NULL_PTR);
296 for (uint8_t i = 0u; i < SPS_SPI_BUFFERSIZE; i++) {
297 pSpiTxBuffer[i] = ((uint16_t)1u << SPS_RW_BIT_POSITION) | /* R/W bit = 1 to write */
298 (address << SPS_ADDRESS_BIT_START) | /* Register address */
299 (writeData); /* Data to write */
300 }
301}

303static void SPS_SingleDeviceRegisterWrite(
304 uint8_t device,
305 const uint16_t address,
306 uint8_t writeData,
307 SPS_WRITE_TYPE_e writeType,
308 uint16_t *pSpiTxBuffer) {
309 FAS_ASSERT(pSpiTxBuffer != NULL_PTR);
310 /* The index (SPS_SPI_BUFFERSIZE - 1 - device) is used because the devices
311 in the daisy-chain are accessed like a shift register. The first message
312 sent to the daisy-chain will arrive to the last device and the last
313 message will arrive to the first device. To communicate with
314 devices 0-1-2 in the daisy-chain, the sequence 2-1-0 has to be sent.
315 */
316 /* Keep the previous data which lies in the lower 8 bits */
317 uint8_t preceedingWriteData = (uint8_t)((pSpiTxBuffer[(SPS_SPI_BUFFERSIZE - 1u - device)]) & 0xFFu);
318 /* Clear write data which will be replaced */
319 pSpiTxBuffer[(SPS_SPI_BUFFERSIZE - 1u - device)] &= 0xFF00u;
320 /* Write R/W bit and address in the higher 8 bits */
321 pSpiTxBuffer[(SPS_SPI_BUFFERSIZE - 1u - device)] =
322 (uint16_t)(((uint16_t)1u << SPS_RW_BIT_POSITION) | /* R/W bit = 1 to write */
323 (address << SPS_ADDRESS_BIT_START)); /* Register address */
324
325 if (writeType == SPS_replaceCurrentValue) {
326 pSpiTxBuffer[(SPS_SPI_BUFFERSIZE - 1u - device)] |= (uint16_t)writeData; /* Data to write */
327 } else if (writeType == SPS_orWithCurrentValue) {
328 pSpiTxBuffer[(SPS_SPI_BUFFERSIZE - 1u - device)] |=
329 ((uint16_t)preceedingWriteData | (uint16_t)writeData); /* Data to write */
330 } else if (writeType == SPS_andWithCurrentValue) {
331 pSpiTxBuffer[(SPS_SPI_BUFFERSIZE - 1u - device)] |=
332 ((uint16_t)preceedingWriteData & (uint16_t)writeData); /* Data to write */
333 } else {
334 /* Invalid write type */
335 FAS_ASSERT(FAS_TRAP);
336 }
337}

339static void SPS_SetCommandTxBuffer(const SPS_ACTION_e action) {
340 switch (action) {
341 case SPS_ACTION_CONFIGURE_CONTROL_REGISTER:
342 SPS_GlobalRegisterWrite(
343 SPS_GLOBAL_CONTROL_REGISTER_ADDRESS,
344 /* transition to Normal mode and set drive strength to strong (This is required for high speed!) */
345 (SPS_NORMAL_MODE << SPS_MODE_BIT_START) | (SPS_STRONG_DRIVE << SPS_DRIVE_STRENGTH_BIT_START),
346 sps_spiTxRegisterBuffer);
347 break;
348
349 case SPS_ACTION_TRIGGER_CURRENT_MEASUREMENT:
350 SPS_GlobalRegisterWrite(
351 SPS_C_CONTROL_REGISTER_ADDRESS,
352 0x000F, /* Measure all four outputs for current and voltage on demand*/
353 sps_spiTxRegisterBuffer);
354 break;
355
356 case SPS_ACTION_READ_CURRENT_MEASUREMENT1:
357 SPS_GlobalRegisterRead(
358 SPS_OD_IOUT1_DIAG_REGISTER_ADDRESS, SPS_READ_DIAGNOSTIC_REGISTER, sps_spiTxRegisterBuffer);
359 break;
360
361 case SPS_ACTION_READ_CURRENT_MEASUREMENT2:
362 SPS_GlobalRegisterRead(
363 SPS_OD_IOUT2_DIAG_REGISTER_ADDRESS, SPS_READ_DIAGNOSTIC_REGISTER, sps_spiTxRegisterBuffer);
364 break;
365
366 case SPS_ACTION_READ_CURRENT_MEASUREMENT3:
367 SPS_GlobalRegisterRead(
368 SPS_OD_IOUT3_DIAG_REGISTER_ADDRESS, SPS_READ_DIAGNOSTIC_REGISTER, sps_spiTxRegisterBuffer);
369 break;
370
371 case SPS_ACTION_READ_CURRENT_MEASUREMENT4:
372 SPS_GlobalRegisterRead(
373 SPS_OD_IOUT4_DIAG_REGISTER_ADDRESS, SPS_READ_DIAGNOSTIC_REGISTER, sps_spiTxRegisterBuffer);
374 break;
375
376 case SPS_ACTION_READ_EN_IRQ_PIN:
377 SPS_GlobalRegisterRead(
378 SPS_ISR_IRQ_DIAG_REGISTER_ADDRESS, SPS_READ_DIAGNOSTIC_REGISTER, sps_spiTxRegisterBuffer);
379 break;
380
381 default:
382 for (uint8_t i = 0u; i < SPS_SPI_BUFFERSIZE; i++) {
383 sps_spiTxRegisterBuffer[i] = 0x00;
384 }
385 break;
386 }
387}

389static void SPS_InitializeBuffers(void) {
390 for (uint8_t i = 0u; i < SPS_SPI_BUFFERSIZE; i++) {
391 sps_spiTxRegisterBuffer[i] = 0u;
392 SPS_SingleDeviceRegisterWrite(
393 i,
394 SPS_OUTPUT_CONTROL_REGISTER_ADDRESS,
395 0u, /* data to write, output control */
396 SPS_replaceCurrentValue, /* replace because we want to initialize with 0 */
397 sps_spiTxWriteToChannelChannelControlRegister);
398 }
399}

401static void SPS_SetContactorsTxBuffer(void) {
402 for (SPS_CHANNEL_INDEX channel = 0u; channel < SPS_NR_OF_AVAILABLE_SPS_CHANNELS; channel++) {
403 SPS_CHANNEL_STATE_s *pChannel = &sps_channelStatus[channel];
404 /* If channel state is changed by request, process request */
405 if (pChannel->channelRequested != pChannel->channel) {
406 pChannel->channel = pChannel->channelRequested;
407
408 /**
409 * With this computation, the contactor number is linked to the SPS IC
410 * position in the daisy-chain which is driving this contactor.
411 * Contactor 0-3: SPS IC 0
412 * Contactor 4-7: SPS IC 1
413 * Contactor 8-11: SPS IC 2
414 * ...
415 */
416 uint8_t spsDevicePositionInDaisyChain = channel / SPS_NR_CONTACTOR_PER_IC;
417
418 if (pChannel->channelRequested == SPS_CHANNEL_ON) {
419 SPS_SingleDeviceRegisterWrite(
420 spsDevicePositionInDaisyChain,
421 SPS_OUTPUT_CONTROL_REGISTER_ADDRESS,
422 1u << (channel % SPS_NR_CONTACTOR_PER_IC), /* data to write, output control */
423 SPS_orWithCurrentValue, /* OR because we want to set additional bits to 1 */
424 sps_spiTxWriteToChannelChannelControlRegister);
425 } else if (pChannel->channelRequested == SPS_CHANNEL_OFF) {
426 uint8_t writeData = (~(1u << (channel % SPS_NR_CONTACTOR_PER_IC)) & 0xFFu);
427 SPS_SingleDeviceRegisterWrite(
428 spsDevicePositionInDaisyChain,
429 SPS_OUTPUT_CONTROL_REGISTER_ADDRESS,
430 writeData, /* data to write, output control */
431 SPS_andWithCurrentValue, /* AND because we want to set additional bits to 0 */
432 sps_spiTxWriteToChannelChannelControlRegister);
433 } else {
434 /* invalid entry in sps_channelStatus! */
435 FAS_ASSERT(FAS_TRAP);
436 }
437 }
438 }
439}

441static STD_RETURN_TYPE_e SPS_Transmit(void) {
442 STD_RETURN_TYPE_e retVal1 = SPI_TransmitReceiveData(
443 &spi_spsInterface, sps_spiTxRegisterBuffer, sps_spiRxRegisterBuffer, SPS_SPI_BUFFERSIZE);
444
445 /* The chip select has to be high for at least 300ns according to data sheet. This code delays
446 for the smallest time available in #MCU_Delay_us() which is 1us. After this time we can
447 be sure that the SPI interface is able to receive again. */
448 MCU_Delay_us(1u);
449
450 STD_RETURN_TYPE_e retVal2 = SPI_TransmitReceiveData(
451 &spi_spsInterface,
452 sps_spiTxWriteToChannelChannelControlRegister,
453 sps_spiRxReadAnswerDuringChannelControl,
454 SPS_SPI_BUFFERSIZE);
455
456 STD_RETURN_TYPE_e retVal = STD_NOT_OK;
457 if ((retVal1 == STD_OK) && (retVal2 == STD_OK)) {
458 retVal = STD_OK;
459 }
460 return retVal;
461}

463static void SPS_GlobalReadCurrent(const uint8_t outputAllDevices) {
464 /* outputAllDevices has to be 1, 2, 3 or 4 */
465 FAS_ASSERT((outputAllDevices > 0u) && (outputAllDevices <= SPS_NR_CONTACTOR_PER_IC));
466 for (SPS_CHANNEL_INDEX channel = 0u; channel < SPS_NR_OF_AVAILABLE_SPS_CHANNELS; channel++) {
467 if ((channel % SPS_NR_CONTACTOR_PER_IC) == (outputAllDevices - 1u)) {
468 uint8_t buffer_position = channel / SPS_NR_CONTACTOR_PER_IC;
469 uint16_t buffer = sps_spiRxReadAnswerDuringChannelControl[(SPS_SPI_BUFFERSIZE - 1u - buffer_position)] &
470 SPS_BITMASK_DIAGNOSTIC_ONDEMAND_OUTPUT_CURRENT;
471 sps_channelStatus[channel].current_mA = (float_t)buffer * SPS_I_MEASUREMENT_LSB_mA;
472 }
473 }
474}

485/*========== Extern Function Implementations ================================*/
486extern void SPS_Ctrl(void) {
487 if (sps_timer > 0u) {
488 sps_timer--;
489 }
490 STD_RETURN_TYPE_e transmitRetval = STD_NOT_OK;
491
492 if (sps_timer == 0u) {
493 switch (sps_state) {
494 case SPS_START:
495 /* set SPI speed to low speed in order to safely reach the all SPS */
496 SPI_SpsInterfaceSwitchToLowSpeed(&spi_spsInterface);
497 SPS_InitializeBuffers();
498 sps_state = SPS_RESET_LOW;
499 break;
500
501 case SPS_RESET_LOW:
502 IO_PinReset(&SPS_RESET_GIO_PORT, SPS_RESET_PIN);
503 sps_state = SPS_RESET_HIGH;
504 sps_timer = SPS_RESET_WAIT_TIME_TICKS;
505 break;
506
507 case SPS_RESET_HIGH:
508 IO_PinSet(&SPS_RESET_GIO_PORT, SPS_RESET_PIN);
509 sps_state = SPS_CONFIGURE_CONTROL_REGISTER;
510 sps_timer = SPS_RESET_WAIT_TIME_TICKS;
511 break;
512
513 case SPS_CONFIGURE_CONTROL_REGISTER:
514 SPS_SetCommandTxBuffer(SPS_ACTION_CONFIGURE_CONTROL_REGISTER);
515 SPS_SetContactorsTxBuffer();
516 transmitRetval = SPS_Transmit();
517 if (transmitRetval == STD_OK) {
518 sps_state = SPS_TRIGGER_CURRENT_MEASUREMENT;
519 /* we have successfully configured the control registers to
520 strong drive mode and can now go to high speed communication */
521 SPI_SpsInterfaceSwitchToHighSpeed(&spi_spsInterface);
522 } else {
523 sps_state = SPS_START;
524 }
525 break;
526
527 case SPS_TRIGGER_CURRENT_MEASUREMENT:
528 SPS_SetCommandTxBuffer(SPS_ACTION_TRIGGER_CURRENT_MEASUREMENT);
529 SPS_SetContactorsTxBuffer();
530 transmitRetval = SPS_Transmit();
531 if (transmitRetval == STD_OK) {
532 sps_state = SPS_READ_EN_IRQ_PIN;
533 } else {
534 sps_state = SPS_START;
535 }
536 break;
537
538 case SPS_READ_EN_IRQ_PIN:
539 SPS_SetCommandTxBuffer(SPS_ACTION_READ_EN_IRQ_PIN);
540 SPS_SetContactorsTxBuffer();
541 transmitRetval = SPS_Transmit();
542 if (transmitRetval == STD_OK) {
543 sps_state = SPS_READ_MEASURED_CURRENT1;
544 } else {
545 sps_state = SPS_START;
546 }
547 break;
548
549 case SPS_READ_MEASURED_CURRENT1:
550 SPS_SetCommandTxBuffer(SPS_ACTION_READ_CURRENT_MEASUREMENT1);
551 SPS_SetContactorsTxBuffer();
552 transmitRetval = SPS_Transmit();
553 if (transmitRetval == STD_OK) {
554 sps_state = SPS_READ_MEASURED_CURRENT2;
555 } else {
556 sps_state = SPS_START;
557 }
558 SPS_GlobalReadCurrent(SPS_IC_CHANNEL_1);
559 break;
560
561 case SPS_READ_MEASURED_CURRENT2:
562 SPS_SetCommandTxBuffer(SPS_ACTION_READ_CURRENT_MEASUREMENT2);
563 SPS_SetContactorsTxBuffer();
564 transmitRetval = SPS_Transmit();
565 if (transmitRetval == STD_OK) {
566 sps_state = SPS_READ_MEASURED_CURRENT3;
567 } else {
568 sps_state = SPS_START;
569 }
570 SPS_GlobalReadCurrent(SPS_IC_CHANNEL_2);
571 break;
572
573 case SPS_READ_MEASURED_CURRENT3:
574 SPS_SetCommandTxBuffer(SPS_ACTION_READ_CURRENT_MEASUREMENT3);
575 SPS_SetContactorsTxBuffer();
576 transmitRetval = SPS_Transmit();
577 if (transmitRetval == STD_OK) {
578 sps_state = SPS_READ_MEASURED_CURRENT4;
579 } else {
580 sps_state = SPS_START;
581 }

657 channelFeedback = CONT_SWITCH_ON;
658 }
