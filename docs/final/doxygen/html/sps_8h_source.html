89/**

99/**

103 * @param[in] channelIndex number of the channel that should be accessed; This has to be
104 * a SPS channel and it has to be affiliated with #SPS_AFF_GENERAL_IO

107extern void SPS_RequestGeneralIoState(SPS_CHANNEL_INDEX channelIndex, SPS_CHANNEL_FUNCTION_e channelFunction);

109/**
110 * @brief Get feedback value
111 * @details This function compares the measured current in the contactor
112 * channel with a fixed threshold and assumes the contactor feedback
113 * to be closed when this threshold is passed on the channel.
114 * @param[in] channelIndex index of the channel (contactor) that should be accessed
115 * @return State of the channel's feedback
116 */
117extern CONT_ELECTRICAL_STATE_TYPE_e SPS_GetChannelCurrentFeedback(const SPS_CHANNEL_INDEX channelIndex);

119/**
120 * @brief Get the feedback state of a channel
121 * @details Retrieves the feedback state of a sps channel by looking up the
122 * appropriate channel in #sps_kChannelFeedbackMapping and retrieving
123 * the value through the PEX API.
124 * @param[in] channelIndex number of the SPS channel that shall be checked
125 * @param[in] normallyOpen true if the feedback is normally open, false if not
126 * @return State of the channel as reported by PEX
127 */
128extern CONT_ELECTRICAL_STATE_TYPE_e SPS_GetChannelPexFeedback(const SPS_CHANNEL_INDEX channelIndex, bool normallyOpen);
