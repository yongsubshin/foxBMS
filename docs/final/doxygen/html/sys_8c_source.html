111 * @brief check for multiple calls of state machine trigger function
112 * @details The trigger function is not reentrant, which means it cannot
113 * be called multiple times. This functions increments the
114 * triggerEntry counter once and must be called each time the
115 * trigger function is called. If triggerEntry is greater than

121 * @return #SYS_MULTIPLE_CALLS_YES if there were multiple calls,
122 * #SYS_MULTIPLE_CALLS_NO otherwise

212static SYS_CHECK_MULTIPLE_CALLS_e SYS_CheckMultipleCalls(SYS_STATE_s *pSystemState) {
213 FAS_ASSERT(pSystemState != NULL_PTR);
214 SYS_CHECK_MULTIPLE_CALLS_e multipleCalls = SYS_MULTIPLE_CALLS_NO;
215 OS_EnterTaskCritical();
216 if (pSystemState->triggerEntry == 0u) {
217 pSystemState->triggerEntry++;
218 } else {
219 multipleCalls = SYS_MULTIPLE_CALLS_YES;
220 }
221 OS_ExitTaskCritical();
222 return multipleCalls;
223}

226 FAS_ASSERT(pSystemState != NULL_PTR);

232}

234static void SYS_SetState(
235 SYS_STATE_s *pSystemState,
236 SYS_FSM_STATES_e nextState,
237 SYS_FSM_SUBSTATES_e nextSubstate,
238 uint16_t idleTime) {
239 FAS_ASSERT(pSystemState != NULL_PTR);
240 /* AXIVION Routine Generic-MissingParameterAssert: idleTime: parameter accepts whole range */
241 bool earlyExit = false;
242
243 pSystemState->timer = idleTime;
244 pSystemState->previousState = pSystemState->currentState;
245 pSystemState->previousSubstate = pSystemState->currentSubstate;
246
247 if ((pSystemState->currentState == nextState) && (pSystemState->currentSubstate == nextSubstate)) {
248 /* Next state and next substate equal to current state and substate: nothing to do */
249 pSystemState->nextState = SYS_FSM_STATE_DUMMY; /* no state transition required -> reset */
250 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_DUMMY; /* no substate transition required -> reset */
251 earlyExit = true;
252 }
253
254 if (earlyExit == false) {
255 if (pSystemState->currentState != nextState) {
256 /* distinguish between just a state transfer to the error state and a normal state transfer */
257 if (nextState == SYS_FSM_STATE_ERROR) {
258 /* Error state gets treated differently since we dont need to enter it through the Entry sub state */
259 pSystemState->previousState = pSystemState->currentState;
260 pSystemState->currentState = nextState;
261 pSystemState->previousSubstate = pSystemState->currentSubstate;
262 pSystemState->currentSubstate = nextSubstate;
263 } else {
264 /* Next state is different than the current one: switch to it and set substate to entry value */
265 pSystemState->previousState = pSystemState->currentState;
266 pSystemState->currentState = nextState;
267 pSystemState->previousSubstate = pSystemState->currentSubstate;
268 pSystemState->currentSubstate = SYS_FSM_SUBSTATE_ENTRY; /* entry state after a top level state change */
269 pSystemState->nextState = SYS_FSM_STATE_DUMMY; /* no state transition required -> reset */
270 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_DUMMY; /* no substate transition required -> reset */
271 }
272 } else if (pSystemState->currentSubstate != nextSubstate) {
273 /* Only the next substate is different, switch to it */
274 SYS_SetSubstate(pSystemState, nextSubstate, idleTime);
275 } else {
276 ;
277 }
278 }
279}

281static SYS_FSM_STATES_e SYS_ProcessInitializationState(SYS_STATE_s *pSystemState) {
282 FAS_ASSERT(pSystemState != NULL_PTR);
283 SYS_FSM_STATES_e nextState = SYS_FSM_STATE_INITIALIZATION; /* default behavior: stay in state */
284 SBC_STATEMACHINE_e sbcState = SBC_STATEMACHINE_UNDEFINED;
285
286 switch (pSystemState->currentSubstate) {
287 /**************************** ENTRY STATE ****************************************/
288 case SYS_FSM_SUBSTATE_ENTRY:
289 SYS_SetSubstate(pSystemState, SYS_FSM_CHECK_DEEP_DISCHARGE, SYS_FSM_SHORT_TIME);
290 break;
291
292 /**************************** READ FRAM ******************************************/
293 case SYS_FSM_CHECK_DEEP_DISCHARGE:
294 (void)FRAM_ReadData(FRAM_BLOCK_ID_DEEP_DISCHARGE_FLAG);
295 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
296 if (fram_deepDischargeFlags.deepDischargeFlag[s] == true) {
297 (void)DIAG_Handler(DIAG_ID_DEEP_DISCHARGE_DETECTED, DIAG_EVENT_NOT_OK, DIAG_STRING, s);
298 }
299 }
300 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_START_INITIALIZATION_SBC, SYS_FSM_SHORT_TIME);
301 break;
302
303 /**************************** INITIALIZE SBC *************************************/
304 case SYS_FSM_SUBSTATE_START_INITIALIZATION_SBC:
305 if (SBC_SetStateRequest(&sbc_stateMcuSupervisor, SBC_STATE_INIT_REQUEST) == SBC_OK) {
306 pSystemState->initializationTimeout = 0;
307 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_WAIT_INITIALIZATION_SBC, SYS_FSM_SHORT_TIME);
308 }
309 break;
310
311 case SYS_FSM_SUBSTATE_WAIT_INITIALIZATION_SBC:
312 sbcState = SBC_GetState(&sbc_stateMcuSupervisor);
313 if (sbcState == SBC_STATEMACHINE_RUNNING) {
314 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_INITIALIZATION_CAN, SYS_FSM_SHORT_TIME);
315 } else {
316 if (pSystemState->initializationTimeout >
317 (SYS_STATE_MACHINE_SBC_INIT_TIMEOUT_MS / SYS_TASK_CYCLE_CONTEXT_MS)) {
318 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_SBC_INITIALIZATION_ERROR;
319 nextState = SYS_FSM_STATE_ERROR;
320 } else {
321 pSystemState->timer = SYS_FSM_SHORT_TIME;
322 pSystemState->initializationTimeout++;
323 }
324 }
325 break;
326
327 /**************************** INITIALIZE CAN TRANSCEIVER ****************************/
328 case SYS_FSM_SUBSTATE_INITIALIZATION_CAN:
329 CAN_Initialize();
330 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_INITIALIZATION_RTC, SYS_FSM_SHORT_TIME);
331 break;
332
333 /**************************** INITIALIZE RTC ****************************************/
334 case SYS_FSM_SUBSTATE_INITIALIZATION_RTC:
335 if (RTC_IsRtcModuleInitialized() == true) {
336 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_START_UP_BIST, SYS_FSM_SHORT_TIME);
337 }
338 break;
339
340 /**************************** EXECUTE STARTUP BIST **********************************/
341 case SYS_FSM_SUBSTATE_START_UP_BIST:
342 /* run BIST functions: There is no need in this substate to
343 * transfer to the error state, as all functions used here are
344 * asserting to an infinite loop. */
345 SYS_GeneralMacroBist();
346 DATA_ExecuteDataBist();
347 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_SEND_BOOT_MESSAGE, SYS_FSM_SHORT_TIME);
348 break;
349
350 /**************************** SEND BOOT MESSAGE *************************************/
351 case SYS_FSM_SUBSTATE_SEND_BOOT_MESSAGE:
352 /* Send CAN boot message directly on CAN */
353 SYS_SendBootMessage();
354 nextState = SYS_FSM_STATE_PRE_RUNNING;
355 break;
356
357 default: /* LCOV_EXCL_LINE */
358 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
359 break; /* LCOV_EXCL_LINE */
360 }
361 return nextState;
362}

364static SYS_FSM_STATES_e SYS_ProcessPreRunningState(SYS_STATE_s *pSystemState) {
365 FAS_ASSERT(pSystemState != NULL_PTR);
366 SYS_FSM_STATES_e nextState = SYS_FSM_STATE_PRE_RUNNING; /* default behavior: stay in state */
367 bool allSensorsPresent = true;
368 uint16_t nextWaitTime = SYS_FSM_SHORT_TIME;
369 STD_RETURN_TYPE_e balancingInitializationState = STD_OK;
370 BAL_RETURN_TYPE_e balancingGlobalEnableState = BAL_ERROR;
371 STD_RETURN_TYPE_e bmsState = STD_NOT_OK;
372
373 switch (pSystemState->currentSubstate) {
374 /**************************** ENTRY STATE ****************************************/
375 case SYS_FSM_SUBSTATE_ENTRY:
376 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_INITIALIZE_INTERLOCK, SYS_FSM_SHORT_TIME);
377 break;
378
379 /****************************INITIALIZE INTERLOCK*************************************/
380 case SYS_FSM_SUBSTATE_INITIALIZE_INTERLOCK:
381 (void)ILCK_SetStateRequest(ILCK_STATE_INITIALIZATION_REQUEST);
382 pSystemState->initializationTimeout = 0;
383 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_START_INITIALIZATION_BAL, SYS_FSM_SHORT_TIME);
384 break;
385
386 /****************************INITIALIZE BALANCING*************************************/
387 case SYS_FSM_SUBSTATE_START_INITIALIZATION_BAL:
388 (void)BAL_SetStateRequest(BAL_STATE_INIT_REQUEST);
389 pSystemState->initializationTimeout = 0;
390 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_WAIT_INITIALIZATION_BAL, SYS_FSM_SHORT_TIME);
391 break;
392
393 case SYS_FSM_SUBSTATE_WAIT_INITIALIZATION_BAL:
394 balancingInitializationState = BAL_GetInitializationState();
395 if (balancingInitializationState == STD_OK) {
396 SYS_SetSubstate(
397 pSystemState, SYS_FSM_SUBSTATE_WAIT_INITIALIZATION_BAL_GLOBAL_ENABLE, SYS_FSM_SHORT_TIME);
398 } else {
399 if (pSystemState->initializationTimeout >
400 (SYS_STATE_MACHINE_BAL_INITIALIZATION_TIMEOUT_MS / SYS_TASK_CYCLE_CONTEXT_MS)) {
401 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_BAL_INITIALIZATION_ERROR;
402 nextState = SYS_FSM_STATE_ERROR;
403 } else {
404 pSystemState->timer = SYS_FSM_SHORT_TIME;
405 pSystemState->initializationTimeout++;
406 }
407 }
408 break;
409
410 case SYS_FSM_SUBSTATE_WAIT_INITIALIZATION_BAL_GLOBAL_ENABLE:
411#if (BS_BALANCING_DEFAULT_INACTIVE == true)
412 balancingGlobalEnableState = BAL_SetStateRequest(BAL_STATE_GLOBAL_DISABLE_REQUEST);
413#else /* BS_BALANCING_DEFAULT_INACTIVE is true */
414 balancingGlobalEnableState = BAL_SetStateRequest(BAL_STATE_GLOBAL_ENABLE_REQUEST);
415#endif /* BS_BALANCING_DEFAULT_INACTIVE is true */
416 if (balancingGlobalEnableState == BAL_OK) {
417 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_START_FIRST_MEASUREMENT_CYCLE, SYS_FSM_SHORT_TIME);
418 } else {
419 if (pSystemState->initializationTimeout >
420 (SYS_STATE_MACHINE_INITIALIZATION_TIMEOUT_MS / SYS_TASK_CYCLE_CONTEXT_MS)) {
421 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_BAL_GLOBAL_INITIALIZATION_ERROR;
422 nextState = SYS_FSM_STATE_ERROR;
423 } else {
424 pSystemState->timer = SYS_FSM_SHORT_TIME;
425 pSystemState->initializationTimeout++;
426 }
427 }
428 break;
429
430 /****************************START FIRST MEAS CYCLE**************************/
431 case SYS_FSM_SUBSTATE_START_FIRST_MEASUREMENT_CYCLE:
432 (void)MEAS_StartMeasurement();
433 pSystemState->initializationTimeout = 0;
434 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_WAIT_FIRST_MEASUREMENT_CYCLE, SYS_FSM_SHORT_TIME);
435 break;
436
437 case SYS_FSM_SUBSTATE_WAIT_FIRST_MEASUREMENT_CYCLE:
438 if (MEAS_IsFirstMeasurementCycleFinished() == true) {
439 /* allow initialization of algorithm module */
440 ALGO_UnlockInitialization();
441 /* MEAS_RequestOpenWireCheck(); */ /*TODO: check with strings */
442#if (BS_CURRENT_SENSOR_PRESENT == true)
443 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_START_CURRENT_SENSOR_PRESENCE_CHECK, SYS_FSM_SHORT_TIME);
444#else /* BS_CURRENT_SENSOR_PRESENT is true */
445 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_INITIALIZATION_MISC, SYS_FSM_SHORT_TIME);
446#endif /* BS_CURRENT_SENSOR_PRESENT is true */
447 } else {
448 if (pSystemState->initializationTimeout >
449 (SYS_STATE_MACHINE_INITIALIZATION_TIMEOUT_MS / SYS_TASK_CYCLE_CONTEXT_MS)) {
450 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_FIRST_MEAS_INITIALIZATION_ERROR;
451 nextState = SYS_FSM_STATE_ERROR;
452 } else {
453 pSystemState->timer = SYS_FSM_MEDIUM_TIME;
454 pSystemState->initializationTimeout++;
455 }
456 }
457 break;
458
459 /****************************CHECK CURRENT SENSOR PRESENCE*************************************/
460 case SYS_FSM_SUBSTATE_START_CURRENT_SENSOR_PRESENCE_CHECK:
461 pSystemState->initializationTimeout = 0;
462 CAN_EnablePeriodic(true);
463#if defined(CURRENT_SENSOR_ISABELLENHUETTE_TRIGGERED)
464 /* If triggered mode is used, CAN trigger message needs to
465 * be transmitted and current sensor response has to be
466 * received afterwards. This may take some time, therefore
467 * delay has to be increased.
468 */
469 nextWaitTime = SYS_FSM_LONG_TIME_MS;
470#else /* defined(CURRENT_SENSOR_ISABELLENHUETTE_TRIGGERED) */
471 nextWaitTime = SYS_FSM_LONG_TIME;
472#endif /* defined(CURRENT_SENSOR_ISABELLENHUETTE_TRIGGERED) */
473 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_WAIT_CURRENT_SENSOR_PRESENCE_CHECK, nextWaitTime);
474 break;
475
476 case SYS_FSM_SUBSTATE_WAIT_CURRENT_SENSOR_PRESENCE_CHECK:
477 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
478 if (CAN_IsCurrentSensorPresent(s) == true) {
479 if (CAN_IsCurrentSensorCcPresent(s) == true) {
480 SE_InitializeSoc(true, s);
481 } else {
482 SE_InitializeSoc(false, s);
483 }
484 if (CAN_IsCurrentSensorEcPresent(s) == true) {
485 SE_InitializeSoe(true, s);
486 } else {
487 SE_InitializeSoe(false, s);
488 }
489 SE_InitializeSoh(s);
490 } else {
491 allSensorsPresent = false;
492 }
493 }
494 if (allSensorsPresent == true) {
495 SOF_Init();
496 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_INITIALIZATION_MISC, SYS_FSM_SHORT_TIME);
497 } else {
498 if (pSystemState->initializationTimeout >
499 (SYS_STATE_MACHINE_INITIALIZATION_TIMEOUT_MS / SYS_TASK_CYCLE_CONTEXT_MS)) {
500 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_CURRENT_SENSOR_PRESENCE_ERROR;
501 nextState = SYS_FSM_STATE_ERROR;
502 } else {
503 pSystemState->timer = SYS_FSM_MEDIUM_TIME;
504 pSystemState->initializationTimeout++;
505 }
506 }
507 break;
508
509 /****************************INITIALIZED_MISC*************************************/
510 case SYS_FSM_SUBSTATE_INITIALIZATION_MISC:
511#if (BS_CURRENT_SENSOR_PRESENT == false)
512 CAN_EnablePeriodic(true);
513 for (uint8_t s = 0u; s < BS_NR_OF_STRINGS; s++) {
514 SE_InitializeSoc(false, s);
515 SE_InitializeSoe(false, s);
516 SE_InitializeSoh(s);
517 }
518#endif /* BS_CURRENT_SENSOR_PRESENT is false */
519
520 pSystemState->initializationTimeout = 0u;
521 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_INITIALIZATION_IMD, SYS_FSM_SHORT_TIME);
522 break;
523
524 /****************************INITIALIZE CONTACTORS*************************************/
525 /* TODO: check if necessary and add */
526
527 /****************************INITIALIZED_IMD*************************************/
528 case SYS_FSM_SUBSTATE_INITIALIZATION_IMD:
529 if (IMD_REQUEST_OK == IMD_RequestInitialization()) {
530 /* Request inquired successfully */
531 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_START_INITIALIZATION_BMS, SYS_FSM_MEDIUM_TIME);
532 pSystemState->initializationTimeout = 0u;
533 } else {
534 /* Request declined -> retry max. 3 times */
535 pSystemState->initializationTimeout++;
536 pSystemState->timer = SYS_FSM_SHORT_TIME;
537 if (pSystemState->initializationTimeout >= SYS_STATE_MACHINE_INITIALIZATION_REQUEST_RETRY_COUNTER) {
538 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_IMD_INITIALIZATION_ERROR;
539 nextState = SYS_FSM_STATE_ERROR;
540 }
541 }
542 break;
543
544 /****************************INITIALIZE BMS*************************************/
545 case SYS_FSM_SUBSTATE_START_INITIALIZATION_BMS:
546 (void)BMS_SetStateRequest(BMS_STATE_INIT_REQUEST);
547 pSystemState->initializationTimeout = 0;
548 SYS_SetSubstate(pSystemState, SYS_FSM_SUBSTATE_WAIT_INITIALIZATION_BMS, SYS_FSM_SHORT_TIME);
549 break;
550
551 case SYS_FSM_SUBSTATE_WAIT_INITIALIZATION_BMS:
552 bmsState = BMS_GetInitializationState();
553 if (bmsState == STD_OK) {
554 nextState = SYS_FSM_STATE_RUNNING;
555 } else {
556 if (pSystemState->initializationTimeout >
557 (SYS_STATE_MACHINE_INITIALIZATION_TIMEOUT_MS / SYS_TASK_CYCLE_CONTEXT_MS)) {
558 pSystemState->nextSubstate = SYS_FSM_SUBSTATE_BMS_INITIALIZATION_ERROR;
559 nextState = SYS_FSM_STATE_ERROR;
560 } else {
561 pSystemState->timer = SYS_FSM_SHORT_TIME;
562 pSystemState->initializationTimeout++;
563 }
564 }
565 break;
566
567 default: /* LCOV_EXCL_LINE */
568 FAS_ASSERT(FAS_TRAP); /* LCOV_EXCL_LINE */
569 break; /* LCOV_EXCL_LINE */
570 }
571 return nextState;
572}

586static STD_RETURN_TYPE_e SYS_RunStateMachine(SYS_STATE_s *pSystemState) {
587 FAS_ASSERT(pSystemState != NULL_PTR);
588 STD_RETURN_TYPE_e ranStateMachine = STD_OK;
589 SYS_FSM_STATES_e nextState = SYS_FSM_STATE_DUMMY;
590
591 SYS_STATE_REQUEST_e stateRequest = SYS_STATE_NO_REQUEST;
592
593 switch (pSystemState->currentState) {
594 /****************************UNINITIALIZED***********************************/
595 case SYS_FSM_STATE_UNINITIALIZED:
596 /* waiting for Initialization Request */
597 stateRequest = SYS_TransferStateRequest();
598 if (stateRequest == SYS_STATE_INITIALIZATION_REQUEST) {
599 SYS_SetState(pSystemState, SYS_FSM_STATE_INITIALIZATION, SYS_FSM_SUBSTATE_ENTRY, SYS_FSM_SHORT_TIME);
600 } else if (stateRequest == SYS_STATE_NO_REQUEST) {
601 /* no actual request pending */
602 } else {
603 pSystemState->illegalRequestsCounter++; /* illegal request pending */
604 }
605 break;
606
607 /****************************INITIALIZATION**********************************/
608 case SYS_FSM_STATE_INITIALIZATION:
609 nextState = SYS_ProcessInitializationState(pSystemState);
610 if (nextState == SYS_FSM_STATE_INITIALIZATION) {
611 /* staying in state, processed by state function */
612 } else if (nextState == SYS_FSM_STATE_ERROR) {
613 SYS_SetState(pSystemState, SYS_FSM_STATE_ERROR, pSystemState->nextSubstate, SYS_FSM_SHORT_TIME);
614 } else if (nextState == SYS_FSM_STATE_PRE_RUNNING) {
615 SYS_SetState(pSystemState, SYS_FSM_STATE_PRE_RUNNING, SYS_FSM_SUBSTATE_ENTRY, SYS_FSM_SHORT_TIME);
616 } else {
617 FAS_ASSERT(FAS_TRAP); /* Something went wrong */
618 }
619 break;
620
621 /****************************PRE RUNNING ************************************/
622 case SYS_FSM_STATE_PRE_RUNNING:
623 nextState = SYS_ProcessPreRunningState(pSystemState);
624 if (nextState == SYS_FSM_STATE_PRE_RUNNING) {
625 /* staying in state, processed by state function */
626 } else if (nextState == SYS_FSM_STATE_ERROR) {
627 SYS_SetState(pSystemState, SYS_FSM_STATE_ERROR, pSystemState->nextSubstate, SYS_FSM_SHORT_TIME);
628 } else if (nextState == SYS_FSM_STATE_RUNNING) {
629 SYS_SetState(pSystemState, SYS_FSM_STATE_RUNNING, SYS_FSM_SUBSTATE_ENTRY, SYS_FSM_SHORT_TIME);
630 } else {
631 FAS_ASSERT(FAS_TRAP); /* Something went wrong */
632 }
633 break;
634
635 /****************************RUNNING*****************************************/
636 case SYS_FSM_STATE_RUNNING:
637 nextState = SYS_ProcessRunningState(pSystemState);
638 if (nextState == SYS_FSM_STATE_RUNNING) {
639 /* staying in state, processed by state function */
640 pSystemState->timer = SYS_FSM_LONG_TIME;
641 } else if (nextState == SYS_FSM_STATE_ERROR) {
642 SYS_SetState(pSystemState, SYS_FSM_STATE_ERROR, pSystemState->nextSubstate, SYS_FSM_SHORT_TIME);
643 } else {
644 FAS_ASSERT(FAS_TRAP); /* Something went wrong */
645 }
646 break;
647
648 /****************************ERROR*******************************************/
649 case SYS_FSM_STATE_ERROR:
650 nextState = SYS_ProcessErrorState(pSystemState);
651 if (nextState == SYS_FSM_STATE_ERROR) {
652 /* staying in state, processed by state function */
653 pSystemState->timer = SYS_FSM_LONG_TIME;
654 } else {
655 FAS_ASSERT(FAS_TRAP); /* Something went wrong */
656 }
657 break;
658 /***************************DEFAULT CASE*************************************/
659 default:
660 FAS_ASSERT(FAS_TRAP); /* invalid state */
661 break;
662 }
663 return ranStateMachine;
664}

677static SYS_RETURN_TYPE_e SYS_CheckStateRequest(SYS_STATE_REQUEST_e stateRequest) {
678 SYS_RETURN_TYPE_e retval = SYS_ILLEGAL_REQUEST;
679 if (stateRequest == SYS_STATE_ERROR_REQUEST) {
680 retval = SYS_OK;
681 } else {
682 if (sys_state.stateRequest == SYS_STATE_NO_REQUEST) {
683 /* initialization is only allowed from the uninitialized state */
684 if (stateRequest == SYS_STATE_INITIALIZATION_REQUEST) {
685 if (sys_state.currentState == SYS_FSM_STATE_UNINITIALIZED) {
686 retval = SYS_OK;
687 } else {
688 retval = SYS_ALREADY_INITIALIZED;
689 }
690 } else {
691 retval = SYS_ILLEGAL_REQUEST;
692 }
693 } else {
694 retval = SYS_REQUEST_PENDING;
695 }
696 }
697 return retval;
698}

725 FAS_ASSERT(pSystemState != NULL_PTR);
726 bool earlyExit = false;
727 STD_RETURN_TYPE_e returnValue = STD_OK;
728
729 /* Check multiple calls of function */
730 if (SYS_MULTIPLE_CALLS_YES == SYS_CheckMultipleCalls(pSystemState)) {
731 returnValue = STD_NOT_OK;
732 earlyExit = true;
733 }
734
735 if (earlyExit == false) {
736 if (pSystemState->timer > 0u) {
737 if ((--pSystemState->timer) > 0u) {
738 pSystemState->triggerEntry--;
739 returnValue = STD_OK;
740 earlyExit = true;
741 }
742 }
743 }
744
745 if (earlyExit == false) {
746 (void)SYS_RunStateMachine(pSystemState);
747 pSystemState->triggerEntry--;
748 }
749 return returnValue;
750}
