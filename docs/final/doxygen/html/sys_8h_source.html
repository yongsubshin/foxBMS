126/**

130typedef struct {

133 SYS_FSM_STATES_e nextState; /*!< next state of driver state machine */
134 SYS_FSM_SUBSTATES_e nextSubstate; /*!< next substate of the state machine */
135 SYS_FSM_STATES_e currentState; /*!< current state of driver state machine */
136 SYS_FSM_SUBSTATES_e currentSubstate; /*!< current substate of the state machine */
137 SYS_FSM_STATES_e previousState; /*!< previous state of the state machine */
138 SYS_FSM_SUBSTATES_e previousSubstate; /*!< previous substate of the state machine */
139 uint32_t illegalRequestsCounter; /*!< counts the number of illegal requests to the SYS state machine */
140 uint16_t initializationTimeout; /*!< Timeout to wait for initialization of state machine state machine */
141 uint8_t triggerEntry; /*!< counter for re-entrance protection (function running flag) */
142} SYS_STATE_s;

149/**
150 * @brief sets the current state request of the state variable sys_state.
151 * @details This function is used to make a state request to the state machine,
152 * e.g., start voltage measurement, read result of voltage
153 * measurement, re-initialization. It calls #SYS_CheckStateRequest()
154 * to check if the request is valid. The state request is rejected if
155 * is not valid. The result of the check is returned immediately, so
156 * that the requester can act in case it made a non-valid state
157 * request.
158 * @param stateRequest state requested to set
159 * @return If the request was successfully set, it returns the SYS_OK, else
160 * the current state of requests (type #SYS_STATE_REQUEST_e)
161 */
162extern SYS_RETURN_TYPE_e SYS_SetStateRequest(SYS_STATE_REQUEST_e stateRequest);
