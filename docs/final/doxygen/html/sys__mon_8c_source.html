106static void SYSM_RecordTimingViolation(SYSM_TASK_ID_e taskId, uint32_t taskDuration, uint32_t timestampEnter) {
107 FAS_ASSERT(taskId < SYSM_TASK_ID_MAX);
108
109 sysm_flagFramCopyHasChanges = true;
110 sysm_localFramCopy.anyTimingIssueOccurred = true;
111
112 switch (taskId) {
113 case SYSM_TASK_ID_ENGINE:
114 sysm_localFramCopy.taskEngineViolatingDuration = taskDuration;
115 sysm_localFramCopy.taskEngineEnterTimestamp = timestampEnter;
116 break;
117 case SYSM_TASK_ID_CYCLIC_1ms:
118 sysm_localFramCopy.task1msViolatingDuration = taskDuration;
119 sysm_localFramCopy.task1msEnterTimestamp = timestampEnter;
120 break;
121 case SYSM_TASK_ID_CYCLIC_10ms:
122 sysm_localFramCopy.task10msViolatingDuration = taskDuration;
123 sysm_localFramCopy.task10msEnterTimestamp = timestampEnter;
124 break;
125 case SYSM_TASK_ID_CYCLIC_100ms:
126 sysm_localFramCopy.task100msViolatingDuration = taskDuration;
127 sysm_localFramCopy.task100msEnterTimestamp = timestampEnter;
128 break;
129 case SYSM_TASK_ID_CYCLIC_ALGORITHM_100ms:
130 sysm_localFramCopy.task100msAlgorithmViolatingDuration = taskDuration;
131 sysm_localFramCopy.task100msAlgorithmEnterTimestamp = timestampEnter;
132 break;
133 default:
134 /* this should not be possible to happen */
135 FAS_ASSERT(FAS_TRAP);
136 break;
137 }
138}

161void SYSM_CheckNotifications(void) {
162 static uint32_t sysm_timestamp = 0u;
163
164 /** early exit if nothing to do */
165 uint32_t local_timer = OS_GetTickCount();
166 if (sysm_timestamp == local_timer) {
167 return;
168 }
169 sysm_timestamp = local_timer;
170
171 uint32_t time_since_last_call = 0u;
172 uint32_t max_allowed_jitter = 0u;
173
174 for (SYSM_TASK_ID_e taskId = (SYSM_TASK_ID_e)0; taskId < SYSM_TASK_ID_MAX; taskId++) {
175 if (sysm_ch_cfg[taskId].enable == SYSM_ENABLED) {
176 /* check that the task gets called within its timing threshold plus jitter and
177 then check that the tasks duration is shorter than tasks cycle time */
178 time_since_last_call = local_timer - sysm_notifications[taskId].timestampEnter;
179 max_allowed_jitter = sysm_ch_cfg[taskId].cycleTime + sysm_ch_cfg[taskId].maxJitter;
180 if ((time_since_last_call > max_allowed_jitter) &&
181 (sysm_notifications[taskId].duration > sysm_ch_cfg[taskId].cycleTime)) {
182 /* module not running within its timed limits */
183 (void)DIAG_Handler(DIAG_ID_SYSTEM_MONITORING, DIAG_EVENT_NOT_OK, DIAG_SYSTEM, (uint32_t)taskId);
184 if (sysm_ch_cfg[taskId].enableRecording == SYSM_RECORDING_ENABLED) {
185 SYSM_RecordTimingViolation(
186 taskId, sysm_notifications[taskId].duration, sysm_notifications[taskId].timestampEnter);
187 }
188 sysm_ch_cfg[taskId].callbackFunction(taskId);
189 }
190 }
191 }

194void SYSM_Notify(SYSM_TASK_ID_e taskId, SYSM_NOTIFY_TYPE_e state, uint32_t timestamp) {
195 FAS_ASSERT(taskId < SYSM_TASK_ID_MAX);
196 FAS_ASSERT((state == SYSM_NOTIFY_ENTER) || (state == SYSM_NOTIFY_EXIT));
197 OS_EnterTaskCritical();
198 sysm_notifications[taskId].state = state;
199 if (state == SYSM_NOTIFY_ENTER) {
200 sysm_notifications[taskId].timestampEnter = timestamp;
201 } else if (state == SYSM_NOTIFY_EXIT) {
202 sysm_notifications[taskId].timestampExit = timestamp;
203 sysm_notifications[taskId].duration = timestamp - sysm_notifications[taskId].timestampEnter;
204 } else {
205 /* state has an illegal value */
206 FAS_ASSERT(FAS_TRAP);
207 }
208 OS_ExitTaskCritical();
209}

211extern void SYSM_GetRecordedTimingViolations(SYSM_TIMING_VIOLATION_RESPONSE_s *pAnswer) {
212 FAS_ASSERT(pAnswer != NULL_PTR);
213
214 OS_EnterTaskCritical();
215 FRAM_SYS_MON_RECORD_s localSysMonRecord = {false, 0u, 0u, 0u, 0u, 0u, 0u, 0u, 0u, 0u, 0u};
216 SYSM_CopyFramStruct(&fram_sysMonViolationRecord, &localSysMonRecord);
217 OS_ExitTaskCritical();
218
219 if (localSysMonRecord.anyTimingIssueOccurred == false) {
220 /* no timing violations recorded, abort */
221 pAnswer->recordedViolationAny = false;
222 pAnswer->recordedViolationEngine = false;
223 pAnswer->recordedViolation1ms = false;
224 pAnswer->recordedViolation10ms = false;
225 pAnswer->recordedViolation100ms = false;
226 pAnswer->recordedViolation100msAlgo = false;
227 } else {
228 /* timing violation, find it */
229 pAnswer->recordedViolationAny = true;
230 pAnswer->recordedViolationEngine = SYSM_ConvertRecordedTimingsToViolation(
231 localSysMonRecord.taskEngineViolatingDuration, localSysMonRecord.taskEngineEnterTimestamp);
232 pAnswer->recordedViolation1ms = SYSM_ConvertRecordedTimingsToViolation(
233 localSysMonRecord.task1msViolatingDuration, localSysMonRecord.task1msEnterTimestamp);
234 pAnswer->recordedViolation10ms = SYSM_ConvertRecordedTimingsToViolation(
235 localSysMonRecord.task10msViolatingDuration, localSysMonRecord.task10msEnterTimestamp);
236 pAnswer->recordedViolation100ms = SYSM_ConvertRecordedTimingsToViolation(
237 localSysMonRecord.task100msViolatingDuration, localSysMonRecord.task100msEnterTimestamp);
238 pAnswer->recordedViolation100msAlgo = SYSM_ConvertRecordedTimingsToViolation(
239 localSysMonRecord.task100msAlgorithmViolatingDuration, localSysMonRecord.task100msAlgorithmEnterTimestamp);
240 }

243extern void SYSM_ClearAllTimingViolations(void) {
244 /* clear all diag entries */
245 for (SYSM_TASK_ID_e taskId = (SYSM_TASK_ID_e)0; taskId < SYSM_TASK_ID_MAX; taskId++) {
246 (void)DIAG_Handler(DIAG_ID_SYSTEM_MONITORING, DIAG_EVENT_OK, DIAG_SYSTEM, (uint32_t)taskId);
247 }
248 /* clear local FRAM copy */
249 OS_EnterTaskCritical();
250 sysm_localFramCopy.anyTimingIssueOccurred = false;
251 sysm_localFramCopy.taskEngineEnterTimestamp = 0u;
252 sysm_localFramCopy.taskEngineViolatingDuration = 0u;
253 sysm_localFramCopy.task1msEnterTimestamp = 0u;
254 sysm_localFramCopy.task1msViolatingDuration = 0u;
255 sysm_localFramCopy.task10msEnterTimestamp = 0u;
256 sysm_localFramCopy.task10msViolatingDuration = 0u;
257 sysm_localFramCopy.task100msEnterTimestamp = 0u;
258 sysm_localFramCopy.task100msViolatingDuration = 0u;
259 sysm_localFramCopy.task100msAlgorithmEnterTimestamp = 0u;
260 sysm_localFramCopy.task100msAlgorithmViolatingDuration = 0u;
261 sysm_flagFramCopyHasChanges = true;
262 OS_ExitTaskCritical();
263 /* commit to FRAM */
264 SYSM_UpdateFramData();
265}

267extern void SYSM_UpdateFramData(void) {
268 OS_EnterTaskCritical();
269 const bool updateNecessary = sysm_flagFramCopyHasChanges;
270 OS_ExitTaskCritical();
271 if (updateNecessary == true) {
272 OS_EnterTaskCritical();
273 SYSM_CopyFramStruct(&sysm_localFramCopy, &fram_sysMonViolationRecord);
274 sysm_flagFramCopyHasChanges = false;
275 OS_ExitTaskCritical();
276
277 (void)FRAM_WriteData(FRAM_BLOCK_ID_SYS_MON_RECORD);
278 }
