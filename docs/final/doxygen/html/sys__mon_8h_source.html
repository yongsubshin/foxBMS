66/*========== Macros and Definitions =========================================*/
67/** defines entry or exit */
68typedef enum {
69 SYSM_NOTIFY_ENTER, /**< entry into a task */
70 SYSM_NOTIFY_EXIT, /**< exit from a task */
71} SYSM_NOTIFY_TYPE_e;
72
73/** state (in summary) used for task or function notification */
74typedef struct {
75 SYSM_NOTIFY_TYPE_e state; /**< state of the task */
76 uint32_t timestampEnter; /**< timestamp of entry into state */
77 uint32_t timestampExit; /**< timestamp of exit from state */
78 uint32_t duration; /**< duration between last complete entry and exit cycle */
79} SYSM_NOTIFICATION_s;

81/** compact data format for reporting violation states */
82typedef struct {
83 bool recordedViolationAny; /*!< compound flag indicating if any violation occurred */
84 bool recordedViolationEngine; /*!< flag indicating if a engine violation is recorded */
85 bool recordedViolation1ms; /*!< flag indicating if a 1ms violation is recorded */
86 bool recordedViolation10ms; /*!< flag indicating if a 10ms violation is recorded */
87 bool recordedViolation100ms; /*!< flag indicating if a 100ms violation is recorded */
88 bool recordedViolation100msAlgo; /*!< flag indicating if a 100ms algorithm violation is recorded */
89} SYSM_TIMING_VIOLATION_RESPONSE_s;
