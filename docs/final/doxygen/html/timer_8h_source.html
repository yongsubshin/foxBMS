75/**
76 * @brief TIMER_create allows to create a new virtual timer
77 * @details This function calls intern xTimerCreateStatic to create a new timer
78 * instance with the given variables.
79 *
80 * @param cpxTimerName the name of the timer
81 * @param uxTimerPeriodInMS the amount of time it waits before calling the
82 * callback function pxCallbackFunction
83 * @param cuxAutoReload whether the timer only fires once or automatically
84 * restarts after firing
85 * @param cpxTimerID ID for our timer. Please use the defines in this header to create a new ID
86 * @param pxCallbackFunction the callback function that gets executed once the timer triggers it
87 * @param pxTimerBuffer a pointer to an StaticTimer_t type used as memory buffer for the timer
88 * @return a TimeHandle_t to interact with the timer
89 */
90extern TimerHandle_t TIMER_Create(
91 const char *cpxTimerName,
92 uint32_t uxTimerPeriodInMS,
93 const UBaseType_t cuxAutoReload,
94 void *const cpxTimerID,
95 TimerCallbackFunction_t pxCallbackFunction,
96 StaticTimer_t *pxTimerBuffer);

98/**
99 * @brief TIMER_start starts a given virtual timer
100 * @details This function calls intern xTimerStart and if the timer command
101 * queue is full waits for x ticks. If the timer has already been
102 * started it acts like TIMER_reset
103 *
104 * @param timerHandle the handle of the timer
105 * @param ticks2wait the amount of ticks to wait if the command queue is full
106 * @return #STD_OK if the timer was started #STD_NOT_OK if something went wrong
107 */
108extern STD_RETURN_TYPE_e TIMER_Start(TimerHandle_t timerHandle, uint32_t ticks2wait);
109

115 * @param timerHandle the handle of the timer

117 * @return #STD_OK if the timer was stopped #STD_NOT_OK if something went wrong

119extern STD_RETURN_TYPE_e TIMER_Stop(TimerHandle_t timerHandle, uint32_t ticks2wait);
