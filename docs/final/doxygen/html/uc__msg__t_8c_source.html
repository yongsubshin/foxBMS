104/* Calculate the CRC of a message */
105uint16_t crc_add_item(uint16_t remainder, uint16_t item) {
106 int i;
107
108 remainder ^= item;
109
110 for (i=0; i < 16; i++)
111 {
112 /*
113 * Try to divide the current data bit.
114 */
115 if (remainder & 0x8000) {
116 remainder = (remainder << 1) ^ ((0x9eb2 << 1) + 0x1);
117 } else {
118 remainder = (remainder << 1);
119 }
120 }
121
122 return (remainder);
123}

125uint16_t calc_crc(const uc_msg_t *msg) {
126 uint16_t remainder = 0;
127 int i;
128
129 remainder = crc_add_item(remainder, msg->head);
130 remainder = crc_add_item(remainder, msg->data.dhead);
131
132 for (i=0; i < (msg->message_length - 3); i++) {
133 remainder = crc_add_item(remainder, msg->data.data[i]);
134 }
135
136 return(remainder);
137}

139/* returns true if CRC matches argument */
140int check_crc(const uc_msg_t *msg, const uint16_t ref_crc) {
141 uint16_t msg_crc;
142
143 get_crc(msg, &msg_crc);
144
145 return (msg_crc == ref_crc);
146}
