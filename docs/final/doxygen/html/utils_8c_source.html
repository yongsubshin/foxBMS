74extern void UTIL_ExtractCharactersFromString(
75 uint64_t *pExtractedCharacters,
76 const char *pkString,
77 uint8_t stringLength,
78 uint8_t startChar,
79 uint8_t numberOfChars) {
80 FAS_ASSERT(pExtractedCharacters != NULL_PTR);
81 FAS_ASSERT(pkString != NULL_PTR);
82 FAS_ASSERT(startChar < stringLength);
83 FAS_ASSERT(numberOfChars <= stringLength);
84
85 /* Calculate last character to be extracted */
86 uint8_t lastChar = startChar + numberOfChars;
87 if (lastChar > stringLength) {
88 lastChar = stringLength;
89 }
90
91 /* Extract characters */
92 for (uint8_t i = startChar; i < lastChar; i++) {
93 *pExtractedCharacters = *pExtractedCharacters << UTIL_SHIFT_ONE_BYTE;
94 *pExtractedCharacters += (uint8_t)pkString[i];
95 }
96}
