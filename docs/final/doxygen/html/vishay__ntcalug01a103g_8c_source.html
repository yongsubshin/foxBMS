227/**
228 * Defines for calculating the ADC voltage on the ends of the operating range.
229 * The ADC voltage is calculated with the following formula:
230 *
231 * V_adc = ( ( V_supply * R_ntc ) / ( R + R_ntc ) )
232 *
233 * Depending on the position of the NTC in the voltage resistor (R_1/R_2),
234 * different R_ntc values are used for the calculation.
235 * @{
236 */
237#if defined(TS_VISHAY_NTCALUG01A103G_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && \
238 (TS_VISHAY_NTCALUG01A103G_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
239#define TS_VISHAY_NTCALUG01A103G_ADC_VOLTAGE_V_MAX_V \
240 (float_t)( \
241 (TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * \
242 ts_ntcalug01a103gLut[ts_ntcalug01a103gLutSize - 1].resistance_Ohm) / \
243 (ts_ntcalug01a103gLut[ts_ntcalug01a103gLutSize - 1].resistance_Ohm + \
244 TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm))
245#define TS_VISHAY_NTCALUG01A103G_ADC_VOLTAGE_V_MIN_V \
246 (float_t)( \
247 (TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * ts_ntcalug01a103gLut[0].resistance_Ohm) / \
248 (ts_ntcalug01a103gLut[0].resistance_Ohm + TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm))
249#else /* TS_VISHAY_NTCALUG01A103G_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == false */
250#define TS_VISHAY_NTCALUG01A103G_ADC_VOLTAGE_V_MIN_V \
251 ((float_t)((TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V * \
252 ts_ntcalug01a103gLut[ts_ntcalug01a103gLutSize - 1].resistance_Ohm) / \
253 (ts_ntcalug01a103gLut[ts_ntcalug01a103gLutSize - 1].resistance_Ohm + \
254 TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm)))

268extern int16_t TS_Vis00GetTemperatureFromLut(uint16_t adcVoltage_mV) {
269 int16_t temperature_ddegC = 0;
270 float_t resistance_Ohm = 0.0f;
271 float_t adcVoltage_V = (float_t)adcVoltage_mV / 1000.0f; /* Convert mV to V */
272
273 /* Check for valid ADC measurements to prevent undefined behavior */
274 if (adcVoltage_V > TS_VISHAY_NTCALUG01A103G_ADC_VOLTAGE_V_MAX_V) {
275 /* Invalid measured ADC voltage -> sensor out of operating range or disconnected/shorted */
276 temperature_ddegC = INT16_MIN;
277 } else if (adcVoltage_V < TS_VISHAY_NTCALUG01A103G_ADC_VOLTAGE_V_MIN_V) {
278 /* Invalid measured ADC voltage -> sensor out of operating range or shorted/disconnected */
279 temperature_ddegC = INT16_MAX;
280 } else {
281 /* Calculate NTC resistance based on measured ADC voltage */
282#if defined(TS_VISHAY_NTCALUG01A103G_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && \
283 (TS_VISHAY_NTCALUG01A103G_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
284 /* R_1 = R_2 * ( ( V_supply / V_adc ) - 1 ) */
285 resistance_Ohm = TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
286 ((TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V / adcVoltage_V) - 1);
287#else /* TS_VISHAY_NTCALUG01A103G_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == false */
288 /* R_2 = R_1 * ( V_2 / ( V_supply - V_adc ) ) */
289 resistance_Ohm = TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
290 (adcVoltage_V / (TS_VISHAY_NTCALUG01A103G_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V - adcVoltage_V));
291#endif /* TS_VISHAY_NTCALUG01A103G_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 */
292
293 /* Variables for interpolating LUT value */
294 uint16_t between_high = 0;
295 uint16_t between_low = 0;
296 for (uint16_t i = 1u; i < ts_ntcalug01a103gLutSize; i++) {
297 if (resistance_Ohm < ts_ntcalug01a103gLut[i].resistance_Ohm) {
298 between_low = i + 1u;
299 between_high = i;
300 }
301 }
302
303 /* Interpolate between LUT values, but do not extrapolate LUT! */
304 if (!(((between_high == 0u) && (between_low == 0u)) || /* measured resistance > maximum LUT resistance */
305 (between_low >= ts_ntcalug01a103gLutSize))) { /* measured resistance < minimum LUT resistance */
306 temperature_ddegC = (int16_t)MATH_LinearInterpolation(
307 ts_ntcalug01a103gLut[between_low].resistance_Ohm,
308 ts_ntcalug01a103gLut[between_low].temperature_ddegC,
309 ts_ntcalug01a103gLut[between_high].resistance_Ohm,
310 ts_ntcalug01a103gLut[between_high].temperature_ddegC,
311 resistance_Ohm);
312 }
313 }
314
315 /* Return temperature based on measured NTC resistance */
316 return temperature_ddegC;
317}
