93 * different R_ntc values are used for the calculation.

128extern int16_t TS_Vis02GetTemperatureFromLut(uint16_t adcVoltage_mV) {
129 int16_t temperature_ddegC = INT16_MIN;
130 float_t adcVoltage_V = (float_t)adcVoltage_mV / TS_SCALING_FACTOR_1V_IN_MV_FLOAT; /* Convert mV to V */
131
132 /* Check for valid ADC measurements to prevent undefined behavior */
133 if (adcVoltage_V > TS_VISHAY_NTCLE413E2103F102L_ADC_VOLTAGE_V_MAX_V) {
134 /* Invalid measured ADC voltage -> sensor out of operating range or disconnected/shorted */
135 temperature_ddegC = INT16_MIN;
136 } else if (adcVoltage_V < TS_VISHAY_NTCLE413E2103F102L_ADC_VOLTAGE_V_MIN_V) {
137 /* Invalid measured ADC voltage -> sensor out of operating range or shorted/disconnected */
138 temperature_ddegC = INT16_MAX;
139 } else {
140 /* Calculate NTC resistance based on measured ADC voltage */
141#if defined(TS_VISHAY_NTCLE413E2103F102L_POSITION_IN_RESISTOR_DIVIDER_IS_R_1) && \
142 (TS_VISHAY_NTCLE413E2103F102L_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 == true)
143 /* formula: R_1 = R_2 * ( ( V_supply / V_adc ) - 1 ) */
144 const float_t resistance_Ohm =
145 TS_VISHAY_NTCLE413E2103F102L_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
146 ((TS_VISHAY_NTCLE413E2103F102L_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V / adcVoltage_V) - 1);
147#else /* TS_VISHAY_NTCLE413E2103F102L_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 is false */
148 /* formula: R_2 = R_1 * ( V_2 / ( V_supply - V_adc ) ) */
149 const float_t resistance_Ohm =
150 TS_VISHAY_NTCLE413E2103F102L_RESISTOR_DIVIDER_RESISTANCE_R_1_R_2_Ohm *
151 (adcVoltage_V / (TS_VISHAY_NTCLE413E2103F102L_RESISTOR_DIVIDER_SUPPLY_VOLTAGE_V - adcVoltage_V));
152#endif /* TS_VISHAY_NTCLE413E2103F102L_POSITION_IN_RESISTOR_DIVIDER_IS_R_1 */
153
154 /* Variables for interpolating LUT value */
155 uint16_t between_high = 0;
156 uint16_t between_low = 0;
157 for (uint16_t i = 0u; i < ts_ntcle413e2103f102lLutSize; i++) {
158 if (resistance_Ohm < ts_ntcle413e2103f102lLut[i].resistance_Ohm) {
159 between_low = i + 1u;
160 between_high = i;
161 }
162 }
163
164 /* Interpolate between LUT values, but do not extrapolate LUT! */
165 if (!(((between_high == 0u) && (between_low == 0u)) || /* measured resistance > maximum LUT resistance */
166 (between_low >= ts_ntcle413e2103f102lLutSize))) { /* measured resistance < minimum LUT resistance */
167 temperature_ddegC = (int16_t)MATH_LinearInterpolation(
168 ts_ntcle413e2103f102lLut[between_low].resistance_Ohm,
169 (float_t)ts_ntcle413e2103f102lLut[between_low].temperature_ddegC,
170 ts_ntcle413e2103f102lLut[between_high].resistance_Ohm,
171 (float_t)ts_ntcle413e2103f102lLut[between_high].temperature_ddegC,
172 resistance_Ohm);
173 }
174 }
175
176 /* Return temperature based on measured NTC resistance */
177 return temperature_ddegC;
178}
